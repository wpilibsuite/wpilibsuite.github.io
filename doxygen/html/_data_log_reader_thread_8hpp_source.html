45class DataLogReaderThread {
46 public:
47 using EntryAddedCallback =
48 std::function<void(DataLogReaderThread&, const DataLogReaderEntry&)>;
49
50 explicit DataLogReaderThread(wpi::log::DataLogReader reader)
51 : DataLogReaderThread{std::move(reader), EntryAddedCallback{}} {}
52
53 /**
54 * Connects the callback before starting the reader thread. The callback runs
55 * on the reader thread and receives this object so it can query entries.
56 */
57 DataLogReaderThread(wpi::log::DataLogReader reader,
58 EntryAddedCallback entryAddedCallback)
59 : m_reader{std::move(reader)} {
60 if (entryAddedCallback) {
61 sigEntryAdded.connect(
62 [this, callback = std::move(entryAddedCallback)](
63 const DataLogReaderEntry& entry) { callback(*this, entry); });
64 }
65 m_thread = std::thread{[this] { ReadMain(); }};
66 }
67 ~DataLogReaderThread();
68
69 bool IsDone() const { return m_done; }
70 std::string_view GetBufferIdentifier() const {
71 return m_reader.GetBufferIdentifier();
72 }
73 unsigned int GetNumRecords() const { return m_numRecords; }
74 unsigned int GetNumEntries() const {
75 std::scoped_lock lock{m_mutex};
76 return m_entriesByName.size();
77 }
78
79 // Passes Entry& to func
80 template <typename T>
81 void ForEachEntryName(T&& func) {
82 std::scoped_lock lock{m_mutex};
83 for (auto&& kv : m_entriesByName) {
84 func(kv.second);
85 }
86 }
87
88 const DataLogReaderEntry* GetEntry(std::string_view name) const {
89 std::scoped_lock lock{m_mutex};
90 auto it = m_entriesByName.find(name);
91 if (it == m_entriesByName.end()) {
92 return nullptr;
93 }
94 return &it->second;
95 }
96
97 const DataLogReaderEntry* GetEntry(int entry) const {
98 std::scoped_lock lock{m_mutex};
99 auto it = m_entriesById.find(entry);
100 if (it == m_entriesById.end()) {
101 return nullptr;
102 }
103 return it->second;
104 }
105
106 wpi::util::StructDescriptorDatabase& GetStructDatabase() {

111
112 const wpi::log::DataLogReader& GetReader() const { return m_reader; }
113
114 // note: these are called on separate thread
115 wpi::util::sig::Signal_mt<const DataLogReaderEntry&> sigEntryAdded;

117
118 private:
119 void ReadMain();
120
121 wpi::log::DataLogReader m_reader;
