25class Logger {

30 Logger() = default;

35 void SetLogger(LogFunc func) { m_func = func; }

37 void set_min_level(unsigned int level) { m_min_level = level; }
38 unsigned int min_level() const { return m_min_level; }
39
40 void DoLog(unsigned int level, const char* file, unsigned int line,
41 const char* msg);
42
43 void LogV(unsigned int level, const char* file, unsigned int line,
44 std::string_view format, std::format_args args);
45
46 template <typename... Args>
47 void Log(unsigned int level, const char* file, unsigned int line,
48 std::string_view format, Args&&... args) {
49 if (m_func && level >= m_min_level) {
50 LogV(level, file, line, format, std::make_format_args(args...));
51 }
52 }
53
54 bool HasLogger() const { return m_func != nullptr; }
55
56 private:
57 LogFunc m_func;
