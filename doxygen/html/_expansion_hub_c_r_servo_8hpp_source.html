16/** This class controls a specific servo in continuous rotation mode hooked up
17 * to an ExpansionHub. */
18class ExpansionHubCRServo {
19 public:
20 /**
21 * Constructs a continuous rotation servo at the requested channel on a
22 * specific USB port.
23 *
24 * @sa ExpansionHubServo for a servo mode, or non-continuous rotation servo
25 * @param usbId The USB port ID the hub is connected to
26 * @param channel The servo channel
27 */
28 ExpansionHubCRServo(int usbId, int channel);
29 ~ExpansionHubCRServo() noexcept;
30
31 /**
32 * Set the servo throttle.
33 *
34 * Throttle values range from -1.0 to 1.0 corresponding to full reverse to
35 * full forward.
36 *
37 * @param value Throttle from -1.0 to 1.0.
38 * @Common This is one of the commonly used methods for this class
39 */
40 void SetThrottle(double value);
41
42 /**
43 * Sets the raw pulse width output on the servo.
44 *
45 * @param pulseWidth Pulse width
46 */
47 void SetPulseWidth(wpi::units::microsecond_t pulseWidth);
48
49 /**
50 * Sets if the servo output is enabled or not. Defaults to false.
51 *
52 * @param enabled True to enable, false to disable
53 */
54 void SetEnabled(bool enabled);
55
56 /**
57 * Sets the frame period for the servo. Defaults to 20ms.
58 *
59 * @param framePeriod The frame period
60 */
61 void SetFramePeriod(wpi::units::microsecond_t framePeriod);
62
63 /**
64 * Gets if the underlying ExpansionHub is connected.
65 *
66 * @return True if hub is connected, otherwise false
67 */
68 bool IsHubConnected() const { return m_hub.IsHubConnected(); }
69
70 /**
71 * Sets the PWM range for the servo.
72 * By default, this is 600 to 2400 microseconds.
73 *
74 * Maximum must be greater than minimum.
75 *
76 * @param minPwm Minimum PWM
77 * @param maxPwm Maximum PWM
78 */
79 void SetPWMRange(wpi::units::microsecond_t minPwm,
80 wpi::units::microsecond_t maxPwm);
81
82 /**
83 * Sets whether the servo is reversed.
84 *
85 * This will reverse SetThrottle.
86 *
87 * @param reversed True to reverse, false for normal
88 */
89 void SetReversed(bool reversed);
90
91 private:
92 wpi::units::microsecond_t GetFullRangeScaleFactor() const;
93
94 ExpansionHub m_hub;
95 int m_channel;
96
97 wpi::units::microsecond_t m_minPwm = 600_us;
98 wpi::units::microsecond_t m_maxPwm = 2400_us;
99
100 bool m_reversed = false;
101
102 wpi::nt::IntegerPublisher m_pulseWidthPublisher;
103 wpi::nt::IntegerPublisher m_framePeriodPublisher;
104 wpi::nt::BooleanPublisher m_enabledPublisher;
105};
