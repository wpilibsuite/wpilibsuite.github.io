18/**
19 * HTTP protocol parser. Performs incremental parsing with callbacks for each
20 * part of the HTTP protocol. As this is incremental, it's suitable for use
21 * with event based frameworks that provide arbitrary chunks of data.
22 */
23class HttpParser {
24 public:
25 enum class Type {
26 REQUEST = HTTP_REQUEST,
27 RESPONSE = HTTP_RESPONSE,
28 BOTH = HTTP_BOTH
29 };
30
31 /**
32 * Constructor.
33 * @param type Type of parser (request or response or both)
34 */
35 explicit HttpParser(Type type);
36
37 /**
38 * Reset the parser to initial state.
39 * This allows reusing the same parser object from request to request.
40 * @param type Type of parser (request or response or both)
41 */
42 void Reset(Type type);
43
44 /**
45 * Set the maximum accepted length for URLs, field names, and field values.
46 * The default is 1024.
47 * @param len maximum length
48 */
49 void SetMaxLength(size_t len) { m_maxLength = len; }
50
51 /**
52 * Executes the parser. An empty input is treated as EOF.
53 * @param in input data
54 * @return Trailing input data after the parse.
55 */
56 std::string_view Execute(std::string_view in);
57
58 /**
59 * Get HTTP major version.
60 */
61 unsigned int GetMajor() const { return m_parser.http_major; }
62
63 /**
64 * Get HTTP minor version.
65 */
66 unsigned int GetMinor() const { return m_parser.http_minor; }
67
68 /**
69 * Get HTTP status code. Valid only on responses. Valid in and after
70 * the OnStatus() callback has been called.
71 */
72 unsigned int GetStatusCode() const { return m_parser.status_code; }
73
74 /**
75 * Get HTTP method. Valid only on requests.
76 */
77 llhttp_method GetMethod() const {
78 return static_cast<llhttp_method>(m_parser.method);
79 }
80
81 /**
82 * Determine if an error occurred.
83 * @return False if no error.
84 */
85 bool HasError() const {
86 return (llhttp_get_errno(&m_parser) != HPE_OK &&
87 llhttp_get_errno(&m_parser) != HPE_PAUSED_UPGRADE) ||
88 (m_finishErr != HPE_OK && m_finishErr != HPE_PAUSED);
89 }
90
91 /**
92 * Get error number.
93 */
94 llhttp_errno GetError() const {
95 if (m_finishErr) {
96 return m_finishErr;
97 }
98 return llhttp_get_errno(&m_parser);
99 }
100
101 /**
102 * Abort the parse. Call this from a callback handler to indicate an error.
103 * This will result in GetError() returning one of the callback-related
104 * errors (e.g. HPE_CB_message_begin).
105 */
106 void Abort() { m_aborted = true; }
107
108 /**
109 * Determine if an upgrade header was present and the parser has exited
110 * because of that. Should be checked when Execute() returns in addition to
111 * checking GetError().
112 * @return True if upgrade header, false otherwise.
113 */
114 bool IsUpgrade() const { return m_parser.upgrade; }
115
116 /**
117 * If this returns false in the headersComplete or messageComplete
118 * callback, then this should be the last message on the connection.
119 * If you are the server, respond with the "Connection: close" header.
120 * If you are the client, close the connection.
121 */
122 bool ShouldKeepAlive() const { return llhttp_should_keep_alive(&m_parser); }
123
124 /**
125 * Pause the parser.
126 * @param paused True to pause, false to unpause.
127 */
128 void Pause(bool paused) {
129 if (paused) {
130 llhttp_pause(&m_parser);
131 } else {
132 llhttp_resume(&m_parser);
133 }
134 }
135
136 /**
137 * Get URL. Valid in and after the url callback has been called.
138 */
139 std::string_view GetUrl() const { return m_urlBuf.str(); }
140
141 /**
142 * Message begin callback.
143 */
144 wpi::util::sig::Signal<> messageBegin;
145
146 /**
147 * URL callback.
148 *
149 * The parameter to the callback is the complete URL string.
150 */
151 wpi::util::sig::Signal<std::string_view> url;
152
153 /**
154 * Status callback.
155 *
156 * The parameter to the callback is the complete status string.
157 * GetStatusCode() can be used to get the numeric status code.
158 */
159 wpi::util::sig::Signal<std::string_view> status;
160
161 /**
162 * Header field callback.
163 *
164 * The parameters to the callback are the field name and field value.
165 */
166 wpi::util::sig::Signal<std::string_view, std::string_view> header;
167
168 /**
169 * Headers complete callback.
170 *
171 * The parameter to the callback is whether the connection should be kept
172 * alive. If this is false, then this should be the last message on the
173 * connection. If you are the server, respond with the "Connection: close"
174 * header. If you are the client, close the connection.
175 */
176 wpi::util::sig::Signal<bool> headersComplete;
177
178 /**
179 * Body data callback.
180 *
181 * The parameter to the callback is the data chunk. Note this callback will be
182 * called multiple times arbitrarily (e.g. it's possible that it may be called
183 * with just a few characters at a time).
184 */
185 wpi::util::sig::Signal<std::string_view> body;
186
187 /**
188 * Headers complete callback.
189 *
190 * The parameter to the callback is whether the connection should be kept
191 * alive. If this is false, then this should be the last message on the
192 * connection. If you are the server, respond with the "Connection: close"
193 * header. If you are the client, close the connection.
194 */
195 wpi::util::sig::Signal<bool> messageComplete;
196
197 /**
198 * Chunk header callback.
199 *
200 * The parameter to the callback is the chunk size.
201 */
202 wpi::util::sig::Signal<uint64_t> chunkHeader;
203
204 /**
205 * Chunk complete callback.
206 */
207 wpi::util::sig::Signal<> chunkComplete;
208
209 private:
210 llhttp_t m_parser;
211 llhttp_settings_t m_settings;
212
213 size_t m_maxLength = 1024;
214 enum class State { START, URL, STATUS, FIELD, VALUE };
215 State m_state = State::START;
216 wpi::util::SmallString<128> m_urlBuf;
217 wpi::util::SmallString<32> m_fieldBuf;
218 wpi::util::SmallString<128> m_valueBuf;
219
220 bool m_aborted = false;
221 llhttp_errno m_finishErr = HPE_OK;
222};
