50[[nodiscard]] ada_really_inline ada::url_components url::get_components()
51 const {
52 url_components out{};
53
54 // protocol ends with ':'. for example: "https:"
55 out.protocol_end = uint32_t(get_protocol().size());
56
57 // Trailing index is always the next character of the current one.
58 // NOLINTNEXTLINE(clang-analyzer-deadcode.DeadStores)
59 size_t running_index = out.protocol_end;
60
61 if (host.has_value()) {
62 // 2 characters for "//" and 1 character for starting index
63 out.host_start = out.protocol_end + 2;
64
65 if (has_credentials()) {
66 out.username_end = uint32_t(out.host_start + username.size());
67
68 out.host_start += uint32_t(username.size());
69
70 if (!password.empty()) {
71 out.host_start += uint32_t(password.size() + 1);
72 }
73
74 out.host_end = uint32_t(out.host_start + host->size());
75 } else {
76 out.username_end = out.host_start;
77
78 // Host does not start with "@" if it does not include credentials.
79 out.host_end = uint32_t(out.host_start + host->size()) - 1;
80 }
81
82 running_index = out.host_end + 1;
83 } else {
84 // Update host start and end date to the same index, since it does not
85 // exist.
86 out.host_start = out.protocol_end;
87 out.host_end = out.host_start;
88
89 if (!has_opaque_path && path.starts_with("//")) {
90 // If url's host is null, url does not have an opaque path, url's path's
91 // size is greater than 1, and url's path[0] is the empty string, then
92 // append U+002F (/) followed by U+002E (.) to output.
93 running_index = out.protocol_end + 2;
94 } else {
95 running_index = out.protocol_end;
96 }
97 }
98
99 if (port.has_value()) {
100 out.port = *port;
101 running_index += helpers::fast_digit_count(*port) + 1; // Port omits ':'
102 }
103
104 out.pathname_start = uint32_t(running_index);
105
106 running_index += path.size();
107
108 if (query.has_value()) {
109 out.search_start = uint32_t(running_index);
110 running_index += get_search().size();
111 if (get_search().empty()) {
112 running_index++;
113 }
114 }
115
116 if (hash.has_value()) {
117 out.hash_start = uint32_t(running_index);
118 }
119
120 return out;
121}

188[[nodiscard]] ada_really_inline std::string url::get_href() const {
189 if (is_special() && host.has_value() && username.empty() &&
190 password.empty() && !port.has_value()) [[likely]] {
191 const std::string_view scheme = ada::scheme::details::is_special_list[type];
192 const size_t host_size = host->size();
193 const size_t path_size = path.size();
194 const size_t query_size = query.has_value() ? query->size() : 0;
195 const size_t hash_size = hash.has_value() ? hash->size() : 0;
196 const size_t total = scheme.size() + 3 + host_size + path_size +
197 (query.has_value() ? query_size + 1 : 0) +
198 (hash.has_value() ? hash_size + 1 : 0);
199 std::string output(total, '\0');
200 char* p = output.data();
201 std::memcpy(p, scheme.data(), scheme.size());
202 p += scheme.size();
203 p[0] = ':';
204 p[1] = '/';
205 p[2] = '/';
206 p += 3;
207 // NOLINTNEXTLINE(bugprone-not-null-terminated-result)
208 std::memcpy(p, host->data(), host_size);
209 p += host_size;
210 std::memcpy(p, path.data(), path_size);
211 p += path_size;
212 if (query.has_value()) {
213 *p++ = '?';
214 // NOLINTNEXTLINE(bugprone-not-null-terminated-result)
215 std::memcpy(p, query->data(), query_size);
216 p += query_size;
217 }
218 if (hash.has_value()) {
219 *p++ = '#';
220 // NOLINTNEXTLINE(bugprone-not-null-terminated-result)
221 std::memcpy(p, hash->data(), hash_size);
222 }
223 return output;
224 }
225
226 std::string output;
227 output.reserve(get_href_size());
228
229 if (is_special()) {
230 output.append(ada::scheme::details::is_special_list[type]);
231 output += ':';
232 } else {
233 output.append(non_special_scheme);
234 output += ':';
235 }
236
237 if (host.has_value()) {
238 output += '/';
239 output += '/';
240 if (has_credentials()) {
241 output.append(username);
242 if (!password.empty()) {
243 output += ':';
244 output.append(password);
245 }
246 output += '@';
247 }
248 output.append(*host);
249 if (port.has_value()) {
250 output += ':';
251 char port_buf[5];
252 auto [ptr, ec] = std::to_chars(port_buf, port_buf + 5, *port);
253 (void)ec;
254 output.append(port_buf, static_cast<size_t>(ptr - port_buf));
255 }
256 } else if (!has_opaque_path && path.starts_with("//")) {
257 // If url's host is null, url does not have an opaque path, url's path's
258 // size is greater than 1, and url's path[0] is the empty string, then
259 // append U+002F (/) followed by U+002E (.) to output.
260 output += '/';
261 output += '.';
262 }
263 output.append(path);
264 if (query.has_value()) {
265 output += '?';
266 output.append(*query);
267 }
268 if (hash.has_value()) {
269 output += '#';
270 output.append(*hash);
271 }
272 return output;
273}

275[[nodiscard]] inline size_t url::get_href_size() const noexcept {
276 size_t size = 0;
277 if (is_special()) {
278 size += ada::scheme::details::is_special_list[type].size() + 1;
279 } else {
280 size += non_special_scheme.size() + 1;
281 }
282 if (host.has_value()) {
283 size += host->size();
284 size += 2;
285 if (has_credentials()) {
286 size += username.size();
287 if (!password.empty()) {
288 size += 1 + password.size();
289 }
290 size += 1;
291 }
292 if (port.has_value()) {
293 size += 1;
294 uint16_t p = *port;
295 size += (p >= 10000) ? 5
296 : (p >= 1000) ? 4
297 : (p >= 100) ? 3
298 : (p >= 10) ? 2
299 : 1;
300 }
301 } else if (!has_opaque_path && path.starts_with("//")) {
302 size += 2;
303 }
304 size += path.size();
305 if (query.has_value()) {
306 size += 1 + query->size();
307 }
308 if (hash.has_value()) {
309 size += 1 + hash->size();
310 }
311 return size;
312}

314ada_really_inline size_t url::parse_port(std::string_view view,
315 bool check_trailing_content) noexcept {
316 ada_log("parse_port('", view, "') ", view.size());
317 if (!view.empty() && view[0] == '-') {
318 ada_log("parse_port: view[0] == '0' && view.size() > 1");
319 is_valid = false;
320 return 0;
321 }
322 uint16_t parsed_port{};
323 auto r = std::from_chars(view.data(), view.data() + view.size(), parsed_port);
324 if (r.ec == std::errc::result_out_of_range) {
325 ada_log("parse_port: r.ec == std::errc::result_out_of_range");
326 is_valid = false;
327 return 0;
328 }
329 ada_log("parse_port: ", parsed_port);
330 const auto consumed = size_t(r.ptr - view.data());
331 ada_log("parse_port: consumed ", consumed);
332 if (check_trailing_content) {
333 is_valid &=
334 (consumed == view.size() || view[consumed] == '/' ||
335 view[consumed] == '?' || (is_special() && view[consumed] == '\\'));
336 }
337 ada_log("parse_port: is_valid = ", is_valid);
338 if (is_valid) {
339 // scheme_default_port can return 0, and we should allow 0 as a base port.
340 auto default_port = scheme_default_port();
341 bool is_port_valid = (default_port == 0 && parsed_port == 0) ||
342 (default_port != parsed_port);
343 port = (r.ec == std::errc() && is_port_valid) ? std::optional(parsed_port)
344 : std::nullopt;
345 }
346 return consumed;
347}
