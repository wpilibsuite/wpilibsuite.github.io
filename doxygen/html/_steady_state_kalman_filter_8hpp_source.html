27/**
28 * A Kalman filter combines predictions from a model and measurements to give an
29 * estimate of the true system state. This is useful because many states cannot
30 * be measured directly as a result of sensor noise, or because the state is
31 * "hidden".
32 *
33 * Kalman filters use a K gain matrix to determine whether to trust the model or
34 * measurements more. Kalman filter theory uses statistics to compute an optimal
35 * K gain which minimizes the sum of squares error in the state estimate. This K
36 * gain is used to correct the state estimate by some amount of the difference
37 * between the actual measurements and the measurements predicted by the model.
38 *
39 * This class assumes predict() and correct() are called in pairs, so the Kalman
40 * gain converges to a steady-state value. If they aren't, use KalmanFilter
41 * instead.
42 *
43 * For more on the underlying math, read
44 * https://file.tavsys.net/control/controls-engineering-in-frc.pdf chapter 9
45 * "Stochastic control theory".
46 *
47 * @tparam States Number of states.
48 * @tparam Inputs Number of inputs.
49 * @tparam Outputs Number of outputs.
50 */
51template <int States, int Inputs, int Outputs>
52class SteadyStateKalmanFilter {
53 public:
54 using StateVector = Vectord<States>;
55 using InputVector = Vectord<Inputs>;
56 using OutputVector = Vectord<Outputs>;
57
58 using StateArray = wpi::util::array<double, States>;
59 using OutputArray = wpi::util::array<double, Outputs>;
60
61 /**
62 * Constructs a steady-state Kalman filter with the given plant.
63 *
64 * See
65 * https://docs.wpilib.org/en/stable/docs/software/advanced-controls/state-space/state-space-observers.html#process-and-measurement-noise-covariance-matrices
66 * for how to select the standard deviations.
67 *
68 * @param plant The plant used for the prediction step.
69 * @param stateStdDevs Standard deviations of model states.
70 * @param measurementStdDevs Standard deviations of measurements.
71 * @param dt Nominal discretization timestep.
72 * @throws std::invalid_argument If the system is undetectable.
73 */
74 SteadyStateKalmanFilter(LinearSystem<States, Inputs, Outputs>& plant,
75 const StateArray& stateStdDevs,
76 const OutputArray& measurementStdDevs,
77 wpi::units::second_t dt) {
78 m_plant = &plant;
79
80 auto contQ = CovarianceMatrix(stateStdDevs);
81 auto contR = CovarianceMatrix(measurementStdDevs);
82
83 Matrixd<States, States> discA;
84 Matrixd<States, States> discQ;
85 DiscretizeAQ<States>(plant.A(), contQ, dt, &discA, &discQ);
86
87 auto discR = DiscretizeR<Outputs>(contR, dt);
88
89 const auto& C = plant.C();
90
91 if (!IsDetectable<States, Outputs>(discA, C)) {
92 std::string msg = std::format(
93 "The system passed to the Kalman filter is undetectable!\n\n"
94 "A =\n{}\nC =\n{}\n",
95 discA, C);
96
97 wpi::math::MathSharedStore::ReportError(msg);
98 throw std::invalid_argument(msg);
99 }
100
101 if (auto P = DARE<States, Outputs>(discA.transpose(), C.transpose(), discQ,
102 discR)) {
103 // S = CPCᵀ + R
104 Matrixd<Outputs, Outputs> S = C * P.value() * C.transpose() + discR;
105
106 // K = PCᵀS⁻¹
107 // K = PCᵀ / S
108 // K = (Sᵀ \ CPᵀ)ᵀ
109 // K = (S \ CP)ᵀ because S and P are symmetric
110 //
111 // [1] wpimath/docs/LinalgIdentities.md
112 m_K = S.ldlt().solve(C * P.value()).transpose();
113 } else if (P.error() == DAREError::QNotSymmetric ||
114 P.error() == DAREError::QNotPositiveSemidefinite) {
115 std::string msg =
116 std::format("{}\n\nQ =\n{}\n", to_string(P.error()), discQ);
117
118 wpi::math::MathSharedStore::ReportError(msg);
119 throw std::invalid_argument(msg);
120 } else if (P.error() == DAREError::RNotSymmetric ||
121 P.error() == DAREError::RNotPositiveDefinite) {
122 std::string msg =
123 std::format("{}\n\nR =\n{}\n", to_string(P.error()), discR);
124
125 wpi::math::MathSharedStore::ReportError(msg);
126 throw std::invalid_argument(msg);
127 } else if (P.error() == DAREError::ABNotStabilizable) {
128 std::string msg = std::format(
129 "The (A, C) pair is not detectable.\n\nA =\n{}\nC =\n{}\n",
130 to_string(P.error()), discA, C);
131
132 wpi::math::MathSharedStore::ReportError(msg);
133 throw std::invalid_argument(msg);
134 } else if (P.error() == DAREError::ACNotDetectable) {
135 std::string msg = std::format("{}\n\nA =\n{}\nQ =\n{}\n",
136 to_string(P.error()), discA, discQ);
137
138 wpi::math::MathSharedStore::ReportError(msg);
139 throw std::invalid_argument(msg);
140 }
141
142 Reset();
143 wpi::util::ReportUsage("KalmanFilter", "");
144 }
145
146 SteadyStateKalmanFilter(SteadyStateKalmanFilter&&) = default;
147 SteadyStateKalmanFilter& operator=(SteadyStateKalmanFilter&&) = default;
148
149 /**
150 * Returns the steady-state Kalman gain matrix K.
151 */
152 const Matrixd<States, Outputs>& K() const { return m_K; }
153
154 /**
155 * Returns an element of the steady-state Kalman gain matrix K.
156 *
157 * @param i Row of K.
158 * @param j Column of K.
159 */
160 double K(int i, int j) const { return m_K(i, j); }
161
162 /**
163 * Returns the state estimate x-hat.
164 */
165 const StateVector& Xhat() const { return m_xHat; }
166
167 /**
168 * Returns an element of the state estimate x-hat.
169 *
170 * @param i Row of x-hat.
171 */
172 double Xhat(int i) const { return m_xHat(i); }
173
174 /**
175 * Set initial state estimate x-hat.
176 *
177 * @param xHat The state estimate x-hat.
178 */
179 void SetXhat(const StateVector& xHat) { m_xHat = xHat; }
180
181 /**
182 * Set an element of the initial state estimate x-hat.
183 *
184 * @param i Row of x-hat.
185 * @param value Value for element of x-hat.
186 */
187 void SetXhat(int i, double value) { m_xHat(i) = value; }
188
189 /**
190 * Resets the observer.
191 */
192 void Reset() { m_xHat.setZero(); }
193
194 /**
195 * Project the model into the future with a new control input u.
196 *
197 * @param u New control input from controller.
198 * @param dt Timestep for prediction.
199 */
200 void Predict(const InputVector& u, wpi::units::second_t dt) {
201 m_xHat = m_plant->CalculateX(m_xHat, u, dt);
202 }
203
204 /**
205 * Correct the state estimate x-hat using the measurements in y.
206 *
207 * @param u Same control input used in the last predict step.
208 * @param y Measurement vector.
209 */
210 void Correct(const InputVector& u, const OutputVector& y) {
211 const auto& C = m_plant->C();
212 const auto& D = m_plant->D();
213
214 // x̂ₖ₊₁⁺ = x̂ₖ₊₁⁻ + K(y − (Cx̂ₖ₊₁⁻ + Duₖ₊₁))
215 m_xHat += m_K * (y - (C * m_xHat + D * u));
216 }
217
218 private:
219 LinearSystem<States, Inputs, Outputs>* m_plant;
220
221 /**
222 * The steady-state Kalman gain matrix.
223 */
224 Matrixd<States, Outputs> m_K;
225
226 /**
227 * The state estimate.
228 */
229 StateVector m_xHat;
230};
