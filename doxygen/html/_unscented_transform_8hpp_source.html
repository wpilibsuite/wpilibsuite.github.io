19/**
20 * Computes unscented transform of a set of sigma points and weights. CovDim
21 * returns the mean and square-root covariance of the sigma points in a tuple.
22 *
23 * This works in conjunction with the UnscentedKalmanFilter class. For use with
24 * square-root form UKFs.
25 *
26 * @tparam CovDim Dimension of covariance of sigma points after passing
27 * through the transform.
28 * @tparam NumSigmas Number of sigma points.
29 * @param sigmas List of sigma points.
30 * @param Wm Weights for the mean.
31 * @param Wc Weights for the covariance.
32 * @param meanFunc A function that computes the mean of NumSigmas state
33 * vectors using a given set of weights.
34 * @param residualFunc A function that computes the residual of two state
35 * vectors (i.e. it subtracts them.)
36 * @param squareRootR Square-root of the noise covariance of the sigma points.
37 *
38 * @return Tuple of x, mean of sigma points; S, square-root covariance of
39 * sigmas.
40 */
41template <int CovDim, int NumSigmas>
42std::tuple<Vectord<CovDim>, Matrixd<CovDim, CovDim>>
43SquareRootUnscentedTransform(
44 const Matrixd<CovDim, NumSigmas>& sigmas, const Vectord<NumSigmas>& Wm,
45 const Vectord<NumSigmas>& Wc,
46 std::function<Vectord<CovDim>(const Matrixd<CovDim, NumSigmas>&,
47 const Vectord<NumSigmas>&)>
48 meanFunc,
49 std::function<Vectord<CovDim>(const Vectord<CovDim>&,
50 const Vectord<CovDim>&)>
51 residualFunc,
52 const Matrixd<CovDim, CovDim>& squareRootR) {
53 // New mean is usually just the sum of the sigmas * weights:
54 //
55 // 2n
56 // x̂ = Σ Wᵢ⁽ᵐ⁾𝒳ᵢ
57 // i=0
58 //
59 // equations (19) and (23) in the paper show this,
60 // but we allow a custom function, usually for angle wrapping
61 Vectord<CovDim> x = meanFunc(sigmas, Wm);
62
63 // Form an intermediate matrix S⁻ as:
64 //
65 // [√{W₁⁽ᶜ⁾}(𝒳_{1:2L} - x̂) √{Rᵛ}]
66 //
67 // the part of equations (20) and (24) within the "qr{}"
68 //
69 // Note that we allow a custom function instead of the difference to allow
70 // angle wrapping. Furthermore, we allow an arbitrary number of sigma points
71 // to support similar methods such as the Scaled Spherical Simplex Filter
72 // (S3F).
73 Matrixd<CovDim, NumSigmas - 1 + CovDim> Sbar;
74 for (int i = 0; i < NumSigmas - 1; i++) {
75 Sbar.template block<CovDim, 1>(0, i) =
76 std::sqrt(Wc[1]) *
77 residualFunc(sigmas.template block<CovDim, 1>(0, 1 + i), x);
78 }
79 Sbar.template block<CovDim, CovDim>(0, NumSigmas - 1) = squareRootR;
80
81 // Compute the square-root covariance of the sigma points.
82 //
83 // We transpose S⁻ first because we formed it by horizontally
84 // concatenating each part; it should be vertical so we can take
85 // the QR decomposition as defined in the "QR Decomposition" passage
86 // of section 3. "EFFICIENT SQUARE-ROOT IMPLEMENTATION"
87 //
88 // The resulting matrix R is the square-root covariance S, but it
89 // is upper triangular, so we need to transpose it.
90 //
91 // equations (20) and (24)
92 Matrixd<CovDim, CovDim> S = Sbar.transpose()
93 .householderQr()
94 .matrixQR()
95 .template block<CovDim, CovDim>(0, 0)
96 .template triangularView<Eigen::Upper>()
97 .transpose();
98
99 // Update or downdate the square-root covariance with (𝒳₀-x̂)
100 // depending on whether its weight (W₀⁽ᶜ⁾) is positive or negative.
101 //
102 // equations (21) and (25)
103 Eigen::internal::llt_inplace<double, Eigen::Lower>::rankUpdate(
104 S, residualFunc(sigmas.template block<CovDim, 1>(0, 0), x), Wc[0]);
105
106 return std::make_tuple(x, S);
107}
