17/// Automatic problem scaling factors.
18///
19/// @tparam Scalar Scalar type.
20template <typename Scalar>
21struct ProblemScaling {
22 /// Type alias for dense vector.
23 using DenseVector = Eigen::Vector<Scalar, Eigen::Dynamic>;
24 /// Type alias for sparse matrix.
25 using SparseMatrix = Eigen::SparseMatrix<Scalar>;
26 /// Type alias for sparse vector.
27 using SparseVector = Eigen::SparseVector<Scalar>;
28
29 /// Cost scaling factor d_f.
30 Scalar f = Scalar(1);
31
32 /// Equality constraint scaling factors d_cₑ.
33 DenseVector c_e;
34
35 /// Inequality constraint scaling factors d_cᵢ.
36 DenseVector c_i;
37
38 /// Constructs identity problem scaling.
39 ProblemScaling() = default;
40
41 /// Constructs problem scaling from explicit factors.
42 ///
43 /// @param f Cost scaling factor d_f.
44 /// @param c_e Equality constraint scaling factors d_cₑ.
45 /// @param c_i Inequality constraint scaling factors d_cᵢ.
46 ProblemScaling(Scalar f, DenseVector c_e, DenseVector c_i)
47 : f{f}, c_e{std::move(c_e)}, c_i{std::move(c_i)} {}
48
49 /// Computes Newton problem scaling.
50 ///
51 /// Scales the cost so the largest gradient component at the starting point
52 /// is at most gₘₐₓ:
53 ///
54 /// d_f = min(1, gₘₐₓ / ‖∇f(x₀)‖_∞)
55 ///
56 /// See §3.8 Automatic Scaling of the Problem Statement in [2].
57 ///
58 /// @param g Cost gradient ∇f, evaluated at the starting point.
59 explicit ProblemScaling(const DenseVector& g) {
60 constexpr Scalar g_max(100);
61
62 f = std::min(Scalar(1), g_max / g.template lpNorm<Eigen::Infinity>());
63 }
64
65 /// Computes SQP problem scaling.
66 ///
67 /// Scales the cost and each equality constraint so the largest gradient
68 /// component at the starting point is at most gₘₐₓ:
69 ///
70 /// d_f = min(1, gₘₐₓ / ‖∇f(x₀)‖_∞)
71 /// d_cₑ[j] = min(1, gₘₐₓ / ‖∇cₑⱼ(x₀)‖_∞)
72 ///
73 /// See §3.8 Automatic Scaling of the Problem Statement in [2].
74 ///
75 /// @param g Cost gradient ∇f, evaluated at the starting point.
76 /// @param A_e Equality constraint Jacobian Aₑ, evaluated at the starting
77 /// point.
78 ProblemScaling(const DenseVector& g, const SparseMatrix& A_e) {
79 constexpr Scalar g_max(100);
80
81 f = std::min(Scalar(1), g_max / g.template lpNorm<Eigen::Infinity>());
82 c_e = (g_max / sparse_inf_norms(A_e).array()).min(Scalar(1)).matrix();
83 }
84
85 /// Computes interior-point problem scaling.
86 ///
87 /// Scales the cost and each constraint so the largest gradient
88 /// component at the starting point is at most gₘₐₓ:
89 ///
90 /// d_f = min(1, gₘₐₓ / ‖∇f(x₀)‖_∞)
91 /// d_c[j] = min(1, gₘₐₓ / ‖∇cⱼ(x₀)‖_∞)
92 ///
93 /// See §3.8 Automatic Scaling of the Problem Statement in [2].
94 ///
95 /// @param g Cost gradient ∇f, evaluated at the starting point.
96 /// @param A_e Equality constraint Jacobian Aₑ, evaluated at the starting
97 /// point.
98 /// @param A_i Inequality constraint Jacobian Aᵢ, evaluated at the starting
99 /// point.
100 ProblemScaling(const DenseVector& g, const SparseMatrix& A_e,
101 const SparseMatrix& A_i) {
102 constexpr Scalar g_max(100);
103
104 f = std::min(Scalar(1), g_max / g.template lpNorm<Eigen::Infinity>());
105 c_e = (g_max / sparse_inf_norms(A_e).array()).min(Scalar(1)).matrix();
106 c_i = (g_max / sparse_inf_norms(A_i).array()).min(Scalar(1)).matrix();
107 }
108
109 /// Whether the problem scaling is identity.
110 ///
111 /// @return True if the problem scaling is identity.
112 bool is_identity() const {
113 return f == Scalar(1) && c_e.size() == 0 && c_i.size() == 0;
114 }
115};
