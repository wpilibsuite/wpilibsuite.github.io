15/**
16 * Handle input from standard Joysticks connected to the Driver Station.
17 *
18 * This class handles standard input that comes from the Driver Station. Each
19 * time a value is requested the most recent value is returned. There is a
20 * single class instance for each joystick and the mapping of ports to hardware
21 * buttons depends on the code in the Driver Station.
22 */
23class Joystick : public HIDDevice {
24 public:
25 /// Default X axis channel.
26 static constexpr int DEFAULT_X_CHANNEL = 0;
27 /// Default Y axis channel.
28 static constexpr int DEFAULT_Y_CHANNEL = 1;
29 /// Default Z axis channel.
30 static constexpr int DEFAULT_Z_CHANNEL = 2;
31 /// Default twist axis channel.
32 static constexpr int DEFAULT_TWIST_CHANNEL = 2;
33 /// Default throttle axis channel.
34 static constexpr int DEFAULT_THROTTLE_CHANNEL = 3;

39 enum AxisType {

43 Y_AXIS,
44 /// Z axis.
45 Z_AXIS,
46 /// Twist axis.
47 TWIST_AXIS,
48 /// Throttle axis.
49 THROTTLE_AXIS
50 };
51
52 /**
53 * Represents a digital button on a joystick.
54 */
55 enum ButtonType {
56 /// TRIGGER.
57 TRIGGER_BUTTON,
58 /// TOP.
59 TOP_BUTTON
60 };

168 /**
169 * Get the button value (starting at button 1).
170 *
171 * The buttons are returned in a single 16 bit value with one bit representing
172 * the state of each button. The appropriate button is returned as a boolean
173 * value.
174 *
175 * @param button The button number to be read (starting at 1)
176 * @return The state of the button
177 */
178 bool GetRawButton(int button) const;
179
180 /**
181 * Whether the button was pressed since the last check. Button indexes begin
182 * at 1.
183 *
184 * @param button The button index, beginning at 1.
185 * @return Whether the button was pressed since the last check.
186 */
187 bool GetRawButtonPressed(int button);
188
189 /**
190 * Whether the button was released since the last check. Button indexes begin
191 * at 1.
192 *
193 * @param button The button index, beginning at 1.
194 * @return Whether the button was released since the last check.
195 */
196 bool GetRawButtonReleased(int button);

198 /**

204 double GetRawAxis(int axis) const;

206 /**
207 * Get the angle in degrees of a POV on the HID.
208 *
209 * The POV angles start at 0 in the up direction, and increase clockwise (e.g.
210 * right is 90, upper-left is 315).
211 *
212 * @param pov The index of the POV to read, starting at 0.

307 bool GetTopReleased();

310 * Constructs an event instance around the top button's digital signal.

312 * @param loop the event loop instance to attach the event to.
313 * @return an event instance representing the top button's digital signal
314 * attached to the given loop.
315 */
316 BooleanEvent Top(EventLoop* loop) const;
317
318 /**
319 * Get the magnitude of the vector formed by the joystick's
320 * current position relative to its origin.
321 *
322 * @return The magnitude of the direction vector
323 */
324 double GetMagnitude() const;
325
326 /**
327 * Get the direction of the vector formed by the joystick and its origin. 0 is
328 * forward and clockwise is positive. (Straight right is π/2 radians or 90
329 * degrees.)
330 *
331 * @return The direction of the vector.
332 */
333 wpi::units::radian_t GetDirection() const;
334
335 private:
336 enum Axis { X, Y, Z, TWIST, THROTTLE, NUM_AXES };
337 enum Button { TRIGGER = 1, TOP = 2 };
338
339 std::array<int, Axis::NUM_AXES> m_axes;
340 GenericHID* m_hid;
341};
