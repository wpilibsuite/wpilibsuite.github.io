107 /**
108 * Get the last action for a particular path.
109 *
110 * @param path path
111 * @return Action value, or empty if no update to that path
112 */
113 template <typename T>
114 std::optional<T> GetLastValue(std::string_view path) const {
115 std::scoped_lock lock{m_mutex};
116 auto* action = GetLastActionImpl(path);
117 if (!action || !std::holds_alternative<T>(action->value)) {
118 return std::nullopt;
119 }
120 return std::get<T>(action->value);
121 }

129 const Schema* GetSchema(std::string_view schemaName) {
130 std::scoped_lock lock{m_mutex};
