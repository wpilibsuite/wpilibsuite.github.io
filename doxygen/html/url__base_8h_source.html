20/**
21 * @brief Enum representing the type of host in a URL.
22 *
23 * Used to distinguish between regular domain names, IPv4 addresses,
24 * and IPv6 addresses for proper parsing and serialization.
25 */
26enum url_host_type : uint8_t {
27 /** Regular domain name (e.g., "www.example.com") */
28 DEFAULT = 0,
29 /** IPv4 address (e.g., "127.0.0.1") */
30 IPV4 = 1,
31 /** IPv6 address (e.g., "[::1]" or "[2001:db8::1]") */
32 IPV6 = 2,
33};

35/**
36 * @brief Abstract base class for URL representations.
37 *
38 * The `url_base` class provides the common interface and state shared by
39 * both `ada::url` and `ada::url_aggregator`. It contains basic URL attributes
40 * like validity status and scheme type, but delegates component storage and
41 * access to derived classes.
42 *
43 * @note This is an abstract class and cannot be instantiated directly.
44 * Use `ada::url` or `ada::url_aggregator` instead.
45 *
46 * @see url
47 * @see url_aggregator
48 */
49struct url_base {

64 /**

80 [[nodiscard]] ada_really_inline constexpr bool is_special() const noexcept;
