15/**
16 * Generates sigma points and weights according to Papakonstantinou's paper[1]
17 * for the UnscentedKalmanFilter class.
18 *
19 * It parameterizes the sigma points using alpha and beta terms. Unless you know
20 * better, this should be your default choice due to its high accuracy and
21 * performance.
22 *
23 * [1] K. Papakonstantinou "A Scaled Spherical Simplex Filter (S3F) with a
24 * decreased n + 2 sigma points set size and equivalent 2n + 1 Unscented Kalman
25 * Filter (UKF) accuracy"
26 *
27 * @tparam States The dimenstionality of the state. States + 2 weights will be
28 * generated.
29 */
30template <int States>
31class S3SigmaPoints {
32 public:
33 static constexpr int NumSigmas = States + 2;
34
35 /**
36 * Constructs a generator for Papakonstantinou sigma points.
37 *
38 * @param alpha Determines the spread of the sigma points around the mean.
39 * Usually a small positive value (1e-3).
40 * @param beta Incorporates prior knowledge of the distribution of the mean.
41 * For Gaussian distributions, beta = 2 is optimal.
42 */
43 explicit S3SigmaPoints(double alpha = 1e-3, double beta = 2)
44 : m_alpha{alpha} {
45 ComputeWeights(beta);
46 }
47
48 /**
49 * Computes the sigma points for an unscented Kalman filter given the mean (x)
50 * and square-root covariance (S) of the filter.
51 *
52 * @param x An array of the means.
53 * @param S Square-root covariance of the filter.
54 *
55 * @return Two dimensional array of sigma points. Each column contains all of
56 * the sigmas for one dimension in the problem space. Ordered by Xi_0,
57 * Xi_{1..n+1}.
58 */
59 Matrixd<States, NumSigmas> SquareRootSigmaPoints(
60 const Vectord<States>& x, const Matrixd<States, States>& S) const {
61 // table (1), equation (12)
62 wpi::util::array<double, States> q{wpi::util::empty_array};
63 for (size_t t = 1; t <= States; ++t) {
64 q[t - 1] = m_alpha * std::sqrt(static_cast<double>(t * (States + 1)) /
65 static_cast<double>(t + 1));
66 }
67
68 Matrixd<States, NumSigmas> C;
69 C.template block<States, 1>(0, 0) = Vectord<States>::Constant(0.0);
70 for (int col = 1; col < NumSigmas; ++col) {
71 for (int row = 0; row < States; ++row) {
72 if (row < col - 2) {
73 C(row, col) = 0.0;
74 } else if (row == col - 2) {
75 C(row, col) = q[row];
76 } else {
77 C(row, col) = -q[row] / (row + 1);
78 }
79 }
80 }
81
82 Matrixd<States, NumSigmas> sigmas;
83 for (int col = 0; col < NumSigmas; ++col) {
84 sigmas.col(col) = x + S * C.col(col);
85 }
86
87 return sigmas;
88 }
89
90 /**
91 * Returns the weight for each sigma point for the mean.
92 */
93 const Vectord<NumSigmas>& Wm() const { return m_Wm; }
94
95 /**
96 * Returns an element of the weight for each sigma point for the mean.
97 *
98 * @param i Element of vector to return.
99 */
100 double Wm(int i) const { return m_Wm(i, 0); }
101
102 /**
103 * Returns the weight for each sigma point for the covariance.
104 */
105 const Vectord<NumSigmas>& Wc() const { return m_Wc; }
106
107 /**
108 * Returns an element of the weight for each sigma point for the covariance.
109 *
110 * @param i Element of vector to return.
111 */
112 double Wc(int i) const { return m_Wc(i, 0); }
113
114 private:
115 Vectord<NumSigmas> m_Wm;
116 Vectord<NumSigmas> m_Wc;
117 double m_alpha;
118
119 /**
120 * Computes the weights for the scaled unscented Kalman filter.
121 *
122 * @param beta Incorporates prior knowledge of the distribution of the mean.
123 */
124 void ComputeWeights(double beta) {
125 double alpha_sq = m_alpha * m_alpha;
126
127 double c = 1.0 / (alpha_sq * (States + 1));
128 m_Wm = Vectord<NumSigmas>::Constant(c);
129 m_Wc = Vectord<NumSigmas>::Constant(c);
130
131 m_Wm(0) = 1.0 - 1.0 / alpha_sq;
132 m_Wc(0) = 1.0 - 1.0 / alpha_sq + (1 - alpha_sq + beta);
133 }
134};
