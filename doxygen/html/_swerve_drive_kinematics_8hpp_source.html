40/**
41 * Helper class that converts a chassis velocity (dx, dy, and dtheta components)
42 * into individual module states (velocity and angle).
43 *
44 * The inverse kinematics (converting from a desired chassis velocity to
45 * individual module states) uses the relative locations of the modules with
46 * respect to the center of rotation. The center of rotation for inverse
47 * kinematics is also variable. This means that you can set your set your center
48 * of rotation in a corner of the robot to perform special evasion maneuvers.
49 *
50 * Forward kinematics (converting an array of module states into the overall
51 * chassis motion) is performs the exact opposite of what inverse kinematics
52 * does. Since this is an overdetermined system (more equations than variables),
53 * we use a least-squares approximation.
54 *
55 * The inverse kinematics: [moduleVelocities] = [moduleLocations] *
56 * [chassisVelocities] We take the Moore-Penrose pseudoinverse of
57 * [moduleLocations] and then multiply by [moduleVelocities] to get our chassis
58 * velocities.
59 *
60 * Forward kinematics is also used for odometry -- determining the position of
61 * the robot on the field using encoders and a gyro.
62 */
63template <size_t NumModules>
64class SwerveDriveKinematics
65 : public Kinematics<
66 wpi::util::array<SwerveModulePosition, NumModules>,
67 wpi::util::array<SwerveModuleVelocity, NumModules>,
68 wpi::util::array<SwerveModuleAcceleration, NumModules>> {
69 public:
70 /**
71 * Constructs a swerve drive kinematics object. This takes in a variable
72 * number of module locations as Translation2ds. The order in which you pass
73 * in the module locations is the same order that you will receive the module
74 * states when performing inverse kinematics. It is also expected that you
75 * pass in the module states in the same order when calling the forward
76 * kinematics methods.
77 *
78 * @param moduleTranslations The locations of the modules relative to the
79 * physical center of the robot.
80 */
81 template <std::convertible_to<Translation2d>... ModuleTranslations>
82 requires(sizeof...(ModuleTranslations) == NumModules)
83 explicit SwerveDriveKinematics(ModuleTranslations&&... moduleTranslations)
84 : m_modules{moduleTranslations...},
85 m_moduleHeadings(wpi::util::empty_array) {
86 for (size_t i = 0; i < NumModules; i++) {
87 // clang-format off
88 m_firstOrderInverseKinematics.template block<2, 3>(i * 2, 0) <<
89 1, 0, (-m_modules[i].Y()).value(),
90 0, 1, (+m_modules[i].X()).value();
91
92 m_secondOrderInverseKinematics.template block<2, 4>(i * 2, 0) <<
93 1, 0, (-m_modules[i].X()).value(), (-m_modules[i].Y()).value(),
94 0, 1, (-m_modules[i].Y()).value(), (+m_modules[i].X()).value();
95 // clang-format on
96 }
97
98 m_firstOrderForwardKinematics =
99 m_firstOrderInverseKinematics.householderQr();
100 m_secondOrderForwardKinematics =
101 m_secondOrderInverseKinematics.householderQr();
102
103 wpi::util::ReportUsage("SwerveDriveKinematics", "");
104 }

156 /**
157 * Performs inverse kinematics to return the module states from a desired
158 * chassis velocity. This method is often used to convert joystick values into
159 * module velocities and angles.
160 *
161 * This function also supports variable centers of rotation. During normal
162 * operations, the center of rotation is usually the same as the physical
163 * center of the robot; therefore, the argument is defaulted to that use case.
164 * However, if you wish to change the center of rotation for evasive
165 * maneuvers, vision alignment, or for any other use case, you can do so.
166 *
167 * In the case that the desired chassis velocities are zero (i.e. the robot
168 * will be stationary), the previously calculated module angle will be
169 * maintained.
170 *
171 * @param chassisVelocities The desired chassis velocity.
172 * @param centerOfRotation The center of rotation. For example, if you set the
173 * center of rotation at one corner of the robot and provide a chassis
174 * velocity that only has a dtheta component, the robot will rotate around
175 * that corner.
176 * @return An array containing the module states. Use caution because these
177 * module states are not normalized. Sometimes, a user input may cause one
178 * of the module velocities to go above the attainable max velocity. Use
179 * the DesaturateWheelVelocities(wpi::util::array<SwerveModuleVelocity,
180 * NumModules>, wpi::units::meters_per_second_t) function to rectify this
181 * issue. In addition, you can leverage the power of C++17 to directly
182 * assign the module states to variables:
183 *
184 * @code{.cpp}
185 * auto [fl, fr, bl, br] =
186 * kinematics.ToSwerveModuleVelocities(chassisVelocities);
187 * @endcode
188 */
189 wpi::util::array<SwerveModuleVelocity, NumModules> ToSwerveModuleVelocities(
190 const ChassisVelocities& chassisVelocities,
191 const Translation2d& centerOfRotation = Translation2d{}) const {
192 wpi::util::array<SwerveModuleVelocity, NumModules> moduleVelocities(
193 wpi::util::empty_array);
194
195 if (chassisVelocities.vx == 0_mps && chassisVelocities.vy == 0_mps &&
196 chassisVelocities.omega == 0_rad_per_s) {
197 for (size_t i = 0; i < NumModules; i++) {
198 moduleVelocities[i] = {0_mps, m_moduleHeadings[i]};
199 }
200
201 return moduleVelocities;
202 }
203
204 // We have a new center of rotation. We need to compute the matrix again.
205 if (centerOfRotation != m_previousCoR) {
206 SetInverseKinematics(centerOfRotation);
207 }
208
209 Eigen::Vector3d chassisVelocitiesVector{chassisVelocities.vx.value(),
210 chassisVelocities.vy.value(),
211 chassisVelocities.omega.value()};
212
213 Matrixd<NumModules * 2, 1> moduleVelocityMatrix =
214 m_firstOrderInverseKinematics * chassisVelocitiesVector;
215
216 for (size_t i = 0; i < NumModules; i++) {
217 wpi::units::meters_per_second_t x{moduleVelocityMatrix(i * 2, 0)};
218 wpi::units::meters_per_second_t y{moduleVelocityMatrix(i * 2 + 1, 0)};
219
220 auto velocity = wpi::units::math::hypot(x, y);
221 auto rotation = velocity > 1e-6_mps ? Rotation2d{x.value(), y.value()}
222 : m_moduleHeadings[i];
223
224 moduleVelocities[i] = {velocity, rotation};
225 m_moduleHeadings[i] = rotation;
226 }
227
228 return moduleVelocities;
229 }

257 /**
258 * Performs forward kinematics to return the resulting chassis state from the
259 * given module states. This method is often used for odometry -- determining
260 * the robot's position on the field using data from the real-world velocity
261 * and angle of each module on the robot.
262 *
263 * @param moduleVelocities The state of the modules as an wpi::util::array of
264 * type SwerveModuleVelocity, NumModules long as measured from respective
265 * encoders and gyros. The order of the swerve module states should be
266 * same as passed into the constructor of this class.
267 * @return The resulting chassis velocity.
268 */
269 ChassisVelocities ToChassisVelocities(
270 const wpi::util::array<SwerveModuleVelocity, NumModules>&
271 moduleVelocities) const override {
272 Matrixd<NumModules * 2, 1> moduleVelocityMatrix;
273
274 for (size_t i = 0; i < NumModules; ++i) {
275 SwerveModuleVelocity module = moduleVelocities[i];
276 moduleVelocityMatrix(i * 2, 0) =
277 module.velocity.value() * module.angle.Cos();
278 moduleVelocityMatrix(i * 2 + 1, 0) =
279 module.velocity.value() * module.angle.Sin();
280 }
281
282 Eigen::Vector3d chassisVelocitiesVector =
283 m_firstOrderForwardKinematics.solve(moduleVelocityMatrix);
284
285 return {wpi::units::meters_per_second_t{chassisVelocitiesVector(0)},
286 wpi::units::meters_per_second_t{chassisVelocitiesVector(1)},
287 wpi::units::radians_per_second_t{chassisVelocitiesVector(2)}};
288 }

309 /**
310 * Performs forward kinematics to return the resulting Twist2d from the
311 * given module position deltas. This method is often used for odometry --
312 * determining the robot's position on the field using data from the
313 * real-world position delta and angle of each module on the robot.
314 *
315 * @param moduleDeltas The latest change in position of the modules (as a
316 * SwerveModulePosition type) as measured from respective encoders and
317 * gyros. The order of the swerve module states should be same as passed
318 * into the constructor of this class.
319 * @return The resulting Twist2d.
320 */
321 Twist2d ToTwist2d(
322 wpi::util::array<SwerveModulePosition, NumModules> moduleDeltas) const {
323 Matrixd<NumModules * 2, 1> moduleDeltaMatrix;
324
325 for (size_t i = 0; i < NumModules; ++i) {
326 SwerveModulePosition module = moduleDeltas[i];
327 moduleDeltaMatrix(i * 2, 0) =
328 module.distance.value() * module.angle.Cos();
329 moduleDeltaMatrix(i * 2 + 1, 0) =
330 module.distance.value() * module.angle.Sin();
331 }
332
333 Eigen::Vector3d chassisDeltaVector =
334 m_firstOrderForwardKinematics.solve(moduleDeltaMatrix);
335
336 return {wpi::units::meter_t{chassisDeltaVector(0)},
337 wpi::units::meter_t{chassisDeltaVector(1)},
338 wpi::units::radian_t{chassisDeltaVector(2)}};
339 }

353 /**
354 * Renormalizes the wheel velocities if any individual velocity is above the
355 * specified maximum.
356 *
357 * Sometimes, after inverse kinematics, the requested velocity
358 * from one or more modules may be above the max attainable velocity for the
359 * driving motor on that module. To fix this issue, one can reduce all the
360 * wheel velocities to make sure that all requested module velocities are
361 * at-or-below the absolute threshold, while maintaining the ratio of
362 * velocities between modules.
363 *
364 * Scaling down the module velocities rotates the direction of net motion in
365 * the opposite direction of rotational velocity, which makes discretizing the
366 * chassis velocities inaccurate because the discretization did not account
367 * for this translational skew.
368 *
369 * @param moduleVelocities The array of module velocities.
370 * @param attainableMaxVelocity The absolute max velocity that a module can
371 * reach.
372 * @return The array of desaturated module velocities.
373 */
374 [[nodiscard]]
375 static wpi::util::array<SwerveModuleVelocity, NumModules>
376 DesaturateWheelVelocities(
377 wpi::util::array<SwerveModuleVelocity, NumModules> moduleVelocities,
378 wpi::units::meters_per_second_t attainableMaxVelocity) {
379 auto realMaxVelocity = wpi::units::math::abs(
380 std::max_element(moduleVelocities.begin(), moduleVelocities.end(),
381 [](const auto& a, const auto& b) {
382 return wpi::units::math::abs(a.velocity) <
383 wpi::units::math::abs(b.velocity);
384 })
385 ->velocity);
386
387 if (realMaxVelocity > attainableMaxVelocity) {
388 wpi::util::array<SwerveModuleVelocity, NumModules> velocities(
389 wpi::util::empty_array);
390 for (size_t i = 0; i < NumModules; ++i) {
391 velocities[i] = {moduleVelocities[i].velocity / realMaxVelocity *
392 attainableMaxVelocity,
393 moduleVelocities[i].angle};
394 }
395 return velocities;
396 }
397 return moduleVelocities;
398 }

400 /**
401 * Renormalizes the wheel velocities if any individual velocity is above the
402 * specified maximum, as well as getting rid of joystick saturation at edges
403 * of joystick.
404 *
405 * Sometimes, after inverse kinematics, the requested velocity
406 * from one or more modules may be above the max attainable velocity for the
407 * driving motor on that module. To fix this issue, one can reduce all the
408 * wheel velocities to make sure that all requested module velocities are
409 * at-or-below the absolute threshold, while maintaining the ratio of
410 * velocities between modules.
411 *
412 * Scaling down the module velocities rotates the direction of net motion in
413 * the opposite direction of rotational velocity, which makes discretizing the
414 * chassis velocities inaccurate because the discretization did not account
415 * for this translational skew.
416 *
417 * @param moduleVelocities The array of module velocities.
418 * @param desiredChassisVelocity The desired velocity of the robot
419 * @param attainableMaxModuleVelocity The absolute max velocity a module can
420 * reach
421 * @param attainableMaxRobotTranslationVelocity The absolute max velocity the
422 * robot can reach while translating
423 * @param attainableMaxRobotRotationVelocity The absolute max velocity the
424 * robot can reach while rotating
425 * @return The array of desaturated module velocities
426 */
427 [[nodiscard]]
428 static wpi::util::array<SwerveModuleVelocity, NumModules>
429 DesaturateWheelVelocities(
430 wpi::util::array<SwerveModuleVelocity, NumModules> moduleVelocities,
431 ChassisVelocities desiredChassisVelocity,
432 wpi::units::meters_per_second_t attainableMaxModuleVelocity,
433 wpi::units::meters_per_second_t attainableMaxRobotTranslationVelocity,
434 wpi::units::radians_per_second_t attainableMaxRobotRotationVelocity) {
435 auto realMaxVelocity = wpi::units::math::abs(
436 std::max_element(moduleVelocities.begin(), moduleVelocities.end(),
437 [](const auto& a, const auto& b) {
438 return wpi::units::math::abs(a.velocity) <
439 wpi::units::math::abs(b.velocity);
440 })
441 ->velocity);
442
443 if (attainableMaxRobotTranslationVelocity == 0_mps ||
444 attainableMaxRobotRotationVelocity == 0_rad_per_s ||
445 realMaxVelocity == 0_mps) {
446 return moduleVelocities;
447 }
448
449 auto translationalK = wpi::units::math::hypot(desiredChassisVelocity.vx,
450 desiredChassisVelocity.vy) /
451 attainableMaxRobotTranslationVelocity;
452
453 auto rotationalK = wpi::units::math::abs(desiredChassisVelocity.omega) /
454 attainableMaxRobotRotationVelocity;
455
456 auto k = wpi::units::math::max(translationalK, rotationalK);
457
458 auto scale =
459 wpi::units::math::min(k * attainableMaxModuleVelocity / realMaxVelocity,
460 wpi::units::scalar_t{1});
461 wpi::util::array<SwerveModuleVelocity, NumModules> velocities(
462 wpi::util::empty_array);
463 for (size_t i = 0; i < NumModules; ++i) {
464 velocities[i] = {moduleVelocities[i].velocity * scale,
465 moduleVelocities[i].angle};
466 }
467 return velocities;
468 }

486 /**
487 * Performs inverse kinematics to return the module accelerations from a
488 * desired chassis acceleration. This method is often used for dynamics
489 * calculations -- converting desired robot accelerations into individual
490 * module accelerations.
491 *
492 * This function also supports variable centers of rotation. During normal
493 * operations, the center of rotation is usually the same as the physical
494 * center of the robot; therefore, the argument is defaulted to that use case.
495 * However, if you wish to change the center of rotation for evasive
496 * maneuvers, vision alignment, or for any other use case, you can do so.
497 *
498 * @param chassisAccelerations The desired chassis accelerations.
499 * @param angularVelocity The desired robot angular velocity.
500 * @param centerOfRotation The center of rotation. For example, if you set the
501 * center of rotation at one corner of the robot and provide a chassis
502 * acceleration that only has a dtheta component, the robot will rotate
503 * around that corner.
504 * @return An array containing the module accelerations.
505 */
506 wpi::util::array<SwerveModuleAcceleration, NumModules>
507 ToSwerveModuleAccelerations(
508 const ChassisAccelerations& chassisAccelerations,
509 const units::radians_per_second_t angularVelocity = 0.0_rad_per_s,
510 const Translation2d& centerOfRotation = Translation2d{}) const {
511 // Derivation for second-order kinematics from "Swerve Drive Second Order
512 // Kinematics" by FRC Team 449 - The Blair Robot Project, Rafi Pedersen
513 // https://www.chiefdelphi.com/uploads/short-url/qzj4k2LyBs7rLxAem0YajNIlStH.pdf
514
515 wpi::util::array<SwerveModuleAcceleration, NumModules> moduleAccelerations(
516 wpi::util::empty_array);
517
518 if (chassisAccelerations.ax == 0.0_mps_sq &&
519 chassisAccelerations.ay == 0.0_mps_sq &&
520 chassisAccelerations.alpha == 0.0_rad_per_s_sq) {
521 for (size_t i = 0; i < NumModules; i++) {
522 moduleAccelerations[i] = {0.0_mps_sq, Rotation2d{0.0_rad}};
523 }
524 return moduleAccelerations;
525 }
526
527 if (centerOfRotation != m_previousCoR) {
528 SetInverseKinematics(centerOfRotation);
529 }
530
531 Eigen::Vector4d chassisAccelerationsVector{
532 chassisAccelerations.ax.value(), chassisAccelerations.ay.value(),
533 angularVelocity.value() * angularVelocity.value(),
534 chassisAccelerations.alpha.value()};
535
536 Matrixd<NumModules * 2, 1> moduleAccelerationsMatrix =
537 m_secondOrderInverseKinematics * chassisAccelerationsVector;
538
539 for (size_t i = 0; i < NumModules; i++) {
540 units::meters_per_second_squared_t x{moduleAccelerationsMatrix(i * 2, 0)};
541 units::meters_per_second_squared_t y{
542 moduleAccelerationsMatrix(i * 2 + 1, 0)};
543
544 // For swerve modules, we need to compute both linear acceleration and
545 // angular acceleration The linear acceleration is the magnitude of the
546 // acceleration vector
547 units::meters_per_second_squared_t linearAcceleration =
548 units::math::hypot(x, y);
549
550 if (linearAcceleration.value() < 1e-6) {
551 moduleAccelerations[i] = {linearAcceleration, {}};
552 } else {
553 moduleAccelerations[i] = {linearAcceleration,
554 Rotation2d{x.value(), y.value()}};
555 }
556 }
557
558 return moduleAccelerations;
559 }

597 /**
598 * Performs forward kinematics to return the resulting chassis accelerations
599 * from the given module accelerations. This method is often used for dynamics
600 * calculations -- determining the robot's acceleration on the field using
601 * data from the real-world acceleration of each module on the robot.
602 *
603 * @param moduleAccelerations The accelerations of the modules as measured
604 * from respective encoders and gyros. The order of the swerve module
605 * accelerations should be same as passed into the constructor of this
606 * class.
607 * @return The resulting chassis accelerations.
608 */
609 ChassisAccelerations ToChassisAccelerations(
610 const wpi::util::array<SwerveModuleAcceleration, NumModules>&
611 moduleAccelerations) const override {
612 // Derivation for second-order kinematics from "Swerve Drive Second Order
613 // Kinematics" by FRC Team 449 - The Blair Robot Project, Rafi Pedersen
614 // https://www.chiefdelphi.com/uploads/short-url/qzj4k2LyBs7rLxAem0YajNIlStH.pdf
615
616 Matrixd<NumModules * 2, 1> moduleAccelerationsMatrix;
617
618 for (size_t i = 0; i < NumModules; i++) {
619 SwerveModuleAcceleration module = moduleAccelerations[i];
620
621 moduleAccelerationsMatrix(i * 2 + 0, 0) =
622 module.acceleration.value() * module.angle.Cos();
623 moduleAccelerationsMatrix(i * 2 + 1, 0) =
624 module.acceleration.value() * module.angle.Sin();
625 }
626
627 Eigen::Vector4d chassisAccelerationsVector =
628 m_secondOrderForwardKinematics.solve(moduleAccelerationsMatrix);
629
630 // the second order kinematics equation for swerve drive yields a state
631 // vector [aₓ, a_y, ω², α]
632 return {units::meters_per_second_squared_t{chassisAccelerationsVector(0)},
633 units::meters_per_second_squared_t{chassisAccelerationsVector(1)},
634 units::radians_per_second_squared_t{chassisAccelerationsVector(3)}};
635 }
636
637 private:
638 wpi::util::array<Translation2d, NumModules> m_modules;
639 mutable Matrixd<NumModules * 2, 3> m_firstOrderInverseKinematics;
640 Eigen::HouseholderQR<Matrixd<NumModules * 2, 3>>
641 m_firstOrderForwardKinematics;
642 mutable Matrixd<NumModules * 2, 4> m_secondOrderInverseKinematics;
643 Eigen::HouseholderQR<Matrixd<NumModules * 2, 4>>
644 m_secondOrderForwardKinematics;
645 mutable wpi::util::array<Rotation2d, NumModules> m_moduleHeadings;
646
647 mutable Translation2d m_previousCoR;
648
649 /**
650 * Sets both inverse kinematics matrices based on the new center of rotation.
651 * This does not check if the new center of rotation is different from the
652 * previous one, so a check should be included before the call to this
653 * function.
654 *
655 * @param centerOfRotation new center of rotation
656 */
657 void SetInverseKinematics(const Translation2d& centerOfRotation) const {
658 for (size_t i = 0; i < NumModules; i++) {
659 const double rx = m_modules[i].X().value() - centerOfRotation.X().value();
660 const double ry = m_modules[i].Y().value() - centerOfRotation.Y().value();
661
662 m_firstOrderInverseKinematics.row(i * 2 + 0) << 1, 0, -ry;
663 m_firstOrderInverseKinematics.row(i * 2 + 1) << 0, 1, rx;
664
665 m_secondOrderInverseKinematics.row(i * 2 + 0) << 1, 0, -rx, -ry;
666 m_secondOrderInverseKinematics.row(i * 2 + 1) << 0, 1, -ry, +rx;
667 }
668 m_previousCoR = centerOfRotation;
669 }
670};
