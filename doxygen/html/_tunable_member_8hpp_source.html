31template <typename T>
32class TunableMemberPointer {
33 public:
34 template <std::derived_from<ComplexTunable> Class>
35 explicit TunableMemberPointer(T Class::* member) {
36 static_assert(alignof(Storage<Class>) == alignof(Storage<StorageSizer>));
37 static_assert(sizeof(Storage<Class>) <=
38 (2 * sizeof(Storage<StorageSizer>)));
39 ::new (m_storage) Storage<Class>{member};
40 }
41
42 const T& Get(const ComplexTunable* obj) const {
43 const StorageBase* storage =
44 std::launder(reinterpret_cast<const StorageBase*>(m_storage));
45 return storage->Get(obj);
46 }
47
48 T& Get(ComplexTunable* obj) const {
49 const StorageBase* storage =
50 std::launder(reinterpret_cast<const StorageBase*>(m_storage));
51 return storage->Get(obj);
52 }
53
54 private:
55 struct StorageBase {
56 virtual const T& Get(const ComplexTunable* obj) const = 0;
57 virtual T& Get(ComplexTunable* obj) const = 0;
58 };
59 template <typename Class>
60 struct Storage : public StorageBase {
61 explicit Storage(T Class::* member) : m_ptr{member} {}
62 const T& Get(const ComplexTunable* obj) const override {
63 return static_cast<const Class*>(obj)->*m_ptr;
64 }
65 T& Get(ComplexTunable* obj) const override {
66 return static_cast<Class*>(obj)->*m_ptr;
67 }
68 T Class::* m_ptr;
69 };
70 class StorageSizer : public ComplexTunable {
71 public:
72 void PublishTunable(TunableTable&) override {}
73 };
74
75 // 2x in case of virtual base classes or similar
76 alignas(Storage<StorageSizer>)
77 std::byte m_storage[2 * sizeof(Storage<StorageSizer>)];
78};

179template <typename T, typename... I>
180 requires wpi::util::StructSerializable<T, I...>
181class TunableMemberStruct : public detail::TunableMemberStructBase {
182 public:
183 template <std::derived_from<ComplexTunable> Class>
184 explicit TunableMemberStruct(T Class::* member, I... info)
185 : m_ptr{member}, m_info{std::move(info)...} {}
186
187 template <std::derived_from<ComplexTunable> Class>
188 explicit TunableMemberStruct(T Class::* member, const TunableConfig& config,
189 I... info)
190 : detail::TunableMemberStructBase{config},
191 m_ptr{member},
192 m_info{std::move(info)...} {}
193
194 const T& Get(ComplexTunable* obj) const { return m_ptr.Get(obj); }
195
196 void Set(ComplexTunable* obj, T value) {
197 m_ptr.Get(obj) = std::move(value);
198 SetTunableChanged();
199 }
200
201 T& Mutate(ComplexTunable* obj) {
202 SetTunableChanged();
203 return m_ptr.Get(obj);
204 }
205
206 protected:
207 std::string GetStructTypeName() const override {
208 return std::apply(
209 [](const I&... info) {
210 return std::string{wpi::util::GetStructTypeName<T>(info...)};
211 },
212 m_info);
213 }
214
215 size_t GetStructSize(ComplexTunable*) const override {
216 return std::apply(
217 [](const I&... info) { return wpi::util::GetStructSize<T>(info...); },
218 m_info);
219 }
220
221 void ForEachStructSchema(
222 wpi::util::function_ref<void(std::string_view typeName,
223 std::string_view schema)>
224 fn) const override {
225 std::apply(
226 [&](const I&... info) {
227 wpi::util::ForEachStructSchema<T>(fn, info...);
228 },
229 m_info);
230 }
231
232 bool UnpackStruct(ComplexTunable* obj,
233 std::span<const uint8_t> data) override {
234 if (data.size() != GetStructSize(obj)) {
235 return false;
236 }
237 std::apply(
238 [&](const I&... info) {
239 wpi::util::UnpackStructInto(&m_ptr.Get(obj), data, info...);
240 },
241 m_info);
242 SetTunableChanged();
243 return true;
244 }
245
246 void PackStruct(ComplexTunable* obj, std::span<uint8_t> buf) const override {
247 std::apply(
248 [&](const I&... info) {
249 wpi::util::PackStruct(buf, m_ptr.Get(obj), info...);
250 },
251 m_info);
252 }
253
254 private:
255 detail::TunableMemberPointer<T> m_ptr;
256 [[no_unique_address]]
257 std::tuple<I...> m_info;
258};

260template <typename T, typename... I>
261 requires wpi::util::StructSerializable<T, I...>
262class TunableMemberStructVector : public detail::TunableMemberStructBase {
263 public:
264 template <std::derived_from<ComplexTunable> Class>
265 explicit TunableMemberStructVector(T Class::* member, I... info)
266 : m_ptr{member}, m_info{std::move(info)...} {}
267
268 template <std::derived_from<ComplexTunable> Class>
269 explicit TunableMemberStructVector(T Class::* member,
270 const TunableConfig& config, I... info)
271 : detail::TunableMemberStructBase{config},
272 m_ptr{member},
273 m_info{std::move(info)...} {}
274
275 const std::vector<T>& Get(ComplexTunable* obj) const {
276 return m_ptr.Get(obj);
277 }
278
279 void Set(ComplexTunable* obj, std::vector<T> value) {
280 m_ptr.Get(obj) = std::move(value);
281 SetTunableChanged();
282 }
283 void Set(ComplexTunable* obj, std::span<const T> value) {
284 m_ptr.Get(obj).assign(value.begin(), value.end());
285 SetTunableChanged();
286 }
287
288 std::vector<T>& Mutate(ComplexTunable* obj) {
289 SetTunableChanged();
290 return m_ptr.Get(obj);
291 }
292
293 protected:
294 std::string GetStructTypeName() const override {
295 return std::apply(
296 [](const I&... info) {
297 return std::string{
298 wpi::util::MakeStructArrayTypeName<T, std::dynamic_extent>(
299 info...)};
300 },
301 m_info);
302 }
303
304 size_t GetStructSize(ComplexTunable* obj) const override {
305 return m_ptr.Get(obj).size() *
306 std::apply(
307 [](const I&... info) {
308 return wpi::util::GetStructSize<T>(info...);
309 },
310 m_info);
311 }
312
313 void ForEachStructSchema(
314 wpi::util::function_ref<void(std::string_view typeName,
315 std::string_view schema)>
316 fn) const override {
317 std::apply(
318 [&](const I&... info) {
319 wpi::util::ForEachStructSchema<T>(fn, info...);
320 },
321 m_info);
322 }
323
324 bool UnpackStruct(ComplexTunable* obj,
325 std::span<const uint8_t> data) override {
326 auto& value = m_ptr.Get(obj);
327 bool unpacked = std::apply(
328 [&](const I&... info) {
329 size_t size = wpi::util::GetStructSize<T>(info...);
330 if (size == 0 || data.size() % size != 0) {
331 return false;
332 }
333 value.clear();
334 value.reserve(data.size() / size);
335 for (size_t pos = 0, end = data.size(); pos < end; pos += size) {
336 value.emplace_back(
337 wpi::util::UnpackStruct<T>(data.subspan(pos, size), info...));
338 }
339 return true;
340 },
341 m_info);
342 if (unpacked) {
343 SetTunableChanged();
344 }
345 return unpacked;
346 }
347
348 void PackStruct(ComplexTunable* obj, std::span<uint8_t> buf) const override {
349 std::apply(
350 [&](const I&... info) {
351 size_t size = wpi::util::GetStructSize<T>(info...);
352 for (auto&& val : m_ptr.Get(obj)) {
353 if (buf.size() < size) {
354 break; // shouldn't happen, but avoid clobbering memory
355 }
356 wpi::util::PackStruct(buf.first(size), val, info...);
357 buf = buf.subspan(size);
358 }
359 },
360 m_info);
361 }
362
363 private:
364 detail::TunableMemberPointer<std::vector<T>> m_ptr;
365 [[no_unique_address]]
366 std::tuple<I...> m_info;
367};

379

381 explicit TunableMemberProtobuf(T Class::* member, const TunableConfig& config)
382 : detail::TunableMemberProtobufBase{config}, m_ptr{member} {}
383
384 const T& Get(ComplexTunable* obj) const { return m_ptr.Get(obj); }
385
386 void Set(ComplexTunable* obj, T value) {
387 m_ptr.Get(obj) = std::move(value);
388 SetTunableChanged();
389 }
390
391 T& Mutate(ComplexTunable* obj) {
392 SetTunableChanged();
393 return m_ptr.Get(obj);
394 }
395
396 protected:
397 std::string GetProtobufTypeString() const override {
398 return m_message.GetTypeString();
399 }
400
401 bool UnpackProtobuf(ComplexTunable* obj,
402 std::span<const uint8_t> data) override {
403 auto value = m_message.Unpack(data);
404 if (!value) {
405 return false;
406 }
407 m_ptr.Get(obj) = std::move(*value);
408 SetTunableChanged();
409 return true;
410 }
411
412 bool PackProtobuf(ComplexTunable* obj,
413 std::vector<uint8_t>& buf) const override {
414 return m_message.Pack(buf, m_ptr.Get(obj));
415 }
416
417 void ForEachProtobufDescriptor(
418 wpi::util::function_ref<bool(std::string_view filename)> exists,
419 wpi::util::function_ref<void(std::string_view filename,
420 std::string_view descriptor)>
421 fn) const override {
422 m_message.ForEachProtobufDescriptor(
423 exists,
424 [&](std::string_view filename, std::span<const uint8_t> descriptor) {
425 auto data = reinterpret_cast<const char*>(descriptor.data());
426 fn(filename, {data, descriptor.size()});
427 });
428 }
429
430 private:
431 detail::TunableMemberPointer<T> m_ptr;
432 [[no_unique_address]]
433 mutable wpi::util::ProtobufMessage<T> m_message;
434};
