17/**
18 * Class for operating a compressor connected to a pneumatics module.
19 *
20 * The module will automatically run in closed loop mode by default whenever a
21 * Solenoid object is created. For most cases, a Compressor object does not need
22 * to be instantiated or used in a robot program. This class is only required in
23 * cases where the robot program needs a more detailed status of the compressor
24 * or to enable/disable closed loop control.
25 *
26 * Note: you cannot operate the compressor directly from this class as doing so
27 * would circumvent the safety provided by using the pressure switch and closed
28 * loop control. You can only turn off closed loop control, thereby stopping
29 * the compressor from operating.
30 */
31class Compressor : public wpi::telemetry::TelemetryLoggable {
32 public:
33 /**
34 * Constructs a compressor for a specified module and type.
35 *
36 * @param busId The bus ID.
37 * @param module The module ID to use.
38 * @param moduleType The module type to use.
39 */
40 Compressor(CANPort busId, int module, PneumaticsModuleType moduleType);
41
42 /**
43 * Constructs a compressor for a default module and specified type.
44 *
45 * @param busId The bus ID.
46 * @param moduleType The module type to use.
47 */
48 Compressor(CANPort busId, PneumaticsModuleType moduleType);
49
50 ~Compressor() override;
51
52 Compressor(const Compressor&) = delete;
53 Compressor& operator=(const Compressor&) = delete;
54
55 Compressor(Compressor&&) = default;
56 Compressor& operator=(Compressor&&) = default;
57
58 /**
59 * Returns whether the compressor is active or not.
60 *
61 * @return true if the compressor is on - otherwise false.
62 */
63 bool IsEnabled() const;
64
65 /**
66 * Returns the state of the pressure switch.
67 *
68 * @return True if pressure switch indicates that the system is not full,
69 * otherwise false.
70 */
71 bool GetPressureSwitchValue() const;
72
73 /**
74 * Get the current drawn by the compressor.
75 *
76 * @return Current drawn by the compressor.
77 */
78 wpi::units::ampere_t GetCurrent() const;
79
80 /**
81 * If supported by the device, returns the analog input voltage (on channel
82 * 0).
83 *
84 * This function is only supported by the REV PH. On CTRE PCM, this will
85 * return 0.
86 *
87 * @return The analog input voltage, in volts.
88 */
89 wpi::units::volt_t GetAnalogVoltage() const;
90
91 /**
92 * If supported by the device, returns the pressure read by the analog
93 * pressure sensor (on channel 0).
94 *
95 * This function is only supported by the REV PH with the REV Analog Pressure
96 * Sensor. On CTRE PCM, this will return 0.
97 *
98 * @return The pressure read by the analog pressure sensor.
99 */
100 wpi::units::pounds_per_square_inch_t GetPressure() const;
101
102 /**
103 * Disable the compressor.
104 */
105 void Disable();
106
107 /**
108 * Enables the compressor in digital mode using the digital pressure switch.
109 * The compressor will turn on when the pressure switch indicates that the
110 * system is not full, and will turn off when the pressure switch indicates
111 * that the system is full.
112 */
113 void EnableDigital();
114
115 /**
116 * If supported by the device, enables the compressor in analog mode. This
117 * mode uses an analog pressure sensor connected to analog channel 0 to cycle
118 * the compressor. The compressor will turn on when the pressure drops below
119 * {@code minPressure} and will turn off when the pressure reaches {@code
120 * maxPressure}. This mode is only supported by the REV PH with the REV Analog
121 * Pressure Sensor connected to analog channel 0.
122 *
123 * On CTRE PCM, this will enable digital control.
124 *
125 * @param minPressure The minimum pressure. The compressor will turn on when
126 * the pressure drops below this value.
127 * @param maxPressure The maximum pressure. The compressor will turn off when
128 * the pressure reaches this value.
129 */
130 void EnableAnalog(wpi::units::pounds_per_square_inch_t minPressure,
131 wpi::units::pounds_per_square_inch_t maxPressure);
132
133 /**
134 * If supported by the device, enables the compressor in hybrid mode. This
135 * mode uses both a digital pressure switch and an analog pressure sensor
136 * connected to analog channel 0 to cycle the compressor. This mode is only
137 * supported by the REV PH with the REV Analog Pressure Sensor connected to
138 * analog channel 0.
139 *
140 * The compressor will turn on when \a both:
141 *
142 * - The digital pressure switch indicates the system is not full AND
143 * - The analog pressure sensor indicates that the pressure in the system
144 * is below the specified minimum pressure.
145 *
146 * The compressor will turn off when \a either:
147 *
148 * - The digital pressure switch is disconnected or indicates that the system
149 * is full OR
150 * - The pressure detected by the analog sensor is greater than the specified
151 * maximum pressure.
152 *
153 * On CTRE PCM, this will enable digital control.
154 *
155 * @param minPressure The minimum pressure. The compressor will turn on
156 * when the pressure drops below this value and the pressure switch indicates
157 * that the system is not full.
158 * @param maxPressure The maximum pressure. The compressor will turn
159 * off when the pressure reaches this value or the pressure switch is
160 * disconnected or indicates that the system is full.
161 */
162 void EnableHybrid(wpi::units::pounds_per_square_inch_t minPressure,
163 wpi::units::pounds_per_square_inch_t maxPressure);
164
165 /**
166 * Returns the active compressor configuration.
167 *
168 * @return The active compressor configuration.
169 */
170 CompressorConfigType GetConfigType() const;
171
172 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
173
174 std::string_view GetTelemetryType() const override;
175
176 private:
177 std::shared_ptr<PneumaticsBase> m_module;
178 PneumaticsModuleType m_moduleType;
179};
