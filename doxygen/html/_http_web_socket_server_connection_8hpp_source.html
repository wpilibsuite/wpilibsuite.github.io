21/**
22 * A server-side HTTP connection that also accepts WebSocket upgrades.
23 *
24 * @tparam Derived derived class for std::enable_shared_from_this.
25 */
26template <typename Derived>
27class HttpWebSocketServerConnection
28 : public HttpServerConnection,
29 public std::enable_shared_from_this<Derived> {
30 public:
31 /**
32 * Constructor.
33 *
34 * @param stream network stream
35 * @param protocols Acceptable subprotocols
36 * @param requestTimeout Timeout for server handshake to be established
37 */
38 HttpWebSocketServerConnection(std::shared_ptr<uv::Stream> stream,
39 std::span<const std::string_view> protocols,
40 uv::Timer::Time requestTimeout = {})
41 : HttpServerConnection{stream, requestTimeout},
42 m_helper{m_request},
43 m_protocols{protocols.begin(), protocols.end()} {
44 // Handle upgrade event
45 m_helper.upgrade.connect([this] {
46 // Negotiate sub-protocol
47 wpi::util::SmallVector<std::string_view, 2> protocols{m_protocols.begin(),
48 m_protocols.end()};
49 std::string_view protocol = m_helper.MatchProtocol(protocols).second;
50
51 // Check that the upgrade is valid
52 if (!IsValidWsUpgrade(protocol)) {
53 return;
54 }
55
56 // Disconnect HttpServerConnection header reader
57 CancelRequestTimeout();
58 m_dataConn.disconnect();
59 m_messageCompleteConn.disconnect();
60
61 // Accepting the stream may destroy this (as it replaces the stream user
62 // data), so grab a shared pointer first.
63 auto self = this->shared_from_this();
64
65 // Accept the upgrade
66 auto ws = m_helper.Accept(m_stream, protocol);
67
68 // Set this as the websocket user data to keep it around
69 ws->SetData(self);
70
71 // Store in member
72 m_websocket = ws.get();
73
74 // Call derived class function
75 ProcessWsUpgrade();
76 });
77 }
78
79 /**
80 * Constructor.
81 *
82 * @param stream network stream
83 * @param protocols Acceptable subprotocols
84 * @param requestTimeout Timeout for server handshake to be established
85 */
86 HttpWebSocketServerConnection(
87 std::shared_ptr<uv::Stream> stream,
88 std::initializer_list<std::string_view> protocols,
89 uv::Timer::Time requestTimeout = {})
90 : HttpWebSocketServerConnection(
91 stream, {protocols.begin(), protocols.end()}, requestTimeout) {}
92
93 protected:
94 /**
95 * Check that an incoming WebSocket upgrade is okay. This is called prior
96 * to accepting the upgrade (so prior to ProcessWsUpgrade()).
97 *
98 * The implementation should check other headers and return true if the
99 * WebSocket connection should be accepted.
100 *
101 * @param protocol negotiated subprotocol
102 */
103 virtual bool IsValidWsUpgrade(std::string_view protocol) { return true; }
104
105 /**
106 * Process an incoming WebSocket upgrade. This is called after the header
107 * reader has been disconnected and the websocket has been accepted.
108 *
109 * The implementation should set up appropriate callbacks on the websocket
110 * object to continue communication.
111 *
112 * @note When a WebSocket upgrade occurs, the stream user data is replaced
113 * with the websocket, and the websocket user data points to "this".
114 * Replace the websocket user data with caution!
115 */
116 virtual void ProcessWsUpgrade() = 0;
117
118 /**
119 * WebSocket connection; not valid until ProcessWsUpgrade is called.
120 */
121 WebSocket* m_websocket = nullptr;
122
123 private:
124 WebSocketServerHelper m_helper;
125 wpi::util::SmallVector<std::string, 2> m_protocols;
126};
127
128} // namespace wpi::net
