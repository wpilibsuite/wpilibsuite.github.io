18/**
19 * A detection of an AprilTag tag.
20 */
21class WPILIB_DLLEXPORT AprilTagDetection final {
22 public:
23 AprilTagDetection() = delete;
24 AprilTagDetection(const AprilTagDetection&) = delete;
25 AprilTagDetection& operator=(const AprilTagDetection&) = delete;
26
27 /** A point. Used for center and corner points. */
28 struct Point {
29 double x;
30 double y;
31 };
32
33 /**
34 * Gets the decoded tag's family name.
35 *
36 * @return Decoded family name
37 */
38 std::string_view GetFamily() const;
39
40 /**
41 * Gets the decoded ID of the tag.
42 *
43 * @return Decoded ID
44 */
45 int GetId() const { return id; }
46
47 /**
48 * Gets how many error bits were corrected. Note: accepting large numbers of
49 * corrected errors leads to greatly increased false positive rates.
50 * NOTE: As of this implementation, the detector cannot detect tags with
51 * a hamming distance greater than 2.
52 *
53 * @return Hamming distance (number of corrected error bits)
54 */
55 int GetHamming() const { return hamming; }
56
57 /**
58 * Gets a measure of the quality of the binary decoding process: the
59 * average difference between the intensity of a data bit versus
60 * the decision threshold. Higher numbers roughly indicate better
61 * decodes. This is a reasonable measure of detection accuracy
62 * only for very small tags-- not effective for larger tags (where
63 * we could have sampled anywhere within a bit cell and still
64 * gotten a good detection.)
65 *
66 * @return Decision margin
67 */
68 float GetDecisionMargin() const { return decision_margin; }
69
70 /**
71 * Gets the 3x3 homography matrix describing the projection from an
72 * "ideal" tag (with corners at (-1,1), (1,1), (1,-1), and (-1,
73 * -1)) to pixels in the image.
74 *
75 * @return Homography matrix data
76 */
77 std::span<const double, 9> GetHomography() const;
78
79 /**
80 * Gets the 3x3 homography matrix describing the projection from an
81 * "ideal" tag (with corners at (-1,1), (1,1), (1,-1), and (-1,
82 * -1)) to pixels in the image.
83 *
84 * @return Homography matrix
85 */
86 Eigen::Matrix3d GetHomographyMatrix() const;
87
88 /**
89 * Gets the center of the detection in image pixel coordinates.
90 *
91 * @return Center point
92 */
93 const Point& GetCenter() const { return *reinterpret_cast<const Point*>(c); }
94
95 /**
96 * Gets a corner of the tag in image pixel coordinates. These always
97 * wrap counter-clock wise around the tag. Index 0 is the bottom left corner.
98 *
99 * @param ndx Corner index (range is 0-3, inclusive)
100 * @return Corner point
101 */
102 const Point& GetCorner(int ndx) const {
103 return *reinterpret_cast<const Point*>(p[ndx]);
104 }
105
106 /**
107 * Gets the corners of the tag in image pixel coordinates. These always
108 * wrap counter-clock wise around the tag. The first set of corner coordinates
109 * are the coordinates for the bottom left corner.
110 *
111 * @param cornersBuf Corner point array (X and Y for each corner in order)
112 * @return Corner point array (copy of cornersBuf span)
113 */
114 std::span<double, 8> GetCorners(std::span<double, 8> cornersBuf) const {
115 for (int i = 0; i < 4; i++) {
116 cornersBuf[i * 2] = p[i][0];
117 cornersBuf[i * 2 + 1] = p[i][1];
118 }
119 return cornersBuf;
120 }
121
122 private:
123 // This class *must* be standard-layout-compatible with apriltag_detection
124 // as we use reinterpret_cast from that structure. This means the below
125 // members must exactly match the contents of the apriltag_detection struct.
126
127 // The tag family.
128 void* family;
129
130 // The decoded ID of the tag.
131 int id;
132
133 // How many error bits were corrected? Note: accepting large numbers of
134 // corrected errors leads to greatly increased false positive rates.
135 // NOTE: As of this implementation, the detector cannot detect tags with
136 // a hamming distance greater than 2.
137 int hamming;
138
139 // A measure of the quality of the binary decoding process: the
140 // average difference between the intensity of a data bit versus
141 // the decision threshold. Higher numbers roughly indicate better
142 // decodes. This is a reasonable measure of detection accuracy
143 // only for very small tags-- not effective for larger tags (where
144 // we could have sampled anywhere within a bit cell and still
145 // gotten a good detection.)
146 float decision_margin;
147
148 // The 3x3 homography matrix describing the projection from an
149 // "ideal" tag (with corners at (-1,1), (1,1), (1,-1), and (-1,
150 // -1)) to pixels in the image.
151 void* H;
152
153 // The center of the detection in image pixel coordinates.
154 double c[2];
155
156 // The corners of the tag in image pixel coordinates. These always
157 // wrap counter-clock wise around the tag.
158 double p[4][2];
159};
