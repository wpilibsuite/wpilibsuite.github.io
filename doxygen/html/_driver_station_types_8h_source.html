192inline HAL_ControlWord HAL_MakeControlWord(int64_t opModeHash,
193 HAL_RobotMode robotMode,
194 HAL_Bool enabled, HAL_Bool eStop,
195 HAL_Bool fmsAttached,
196 HAL_Bool dsAttached) {
197 HAL_ControlWord word;
198 word.value =
199 (opModeHash & HAL_CONTROLWORD_OPMODE_HASH_MASK) |
200 (((uint64_t)(robotMode) << HAL_CONTROLWORD_ROBOT_MODE_SHIFT) & // NOLINT
201 HAL_CONTROLWORD_ROBOT_MODE_MASK) |
202 (enabled ? HAL_CONTROLWORD_ENABLED_MASK : 0) |
203 (eStop ? HAL_CONTROLWORD_ESTOP_MASK : 0) |
204 (fmsAttached ? HAL_CONTROLWORD_FMS_ATTACHED_MASK : 0) |
205 (dsAttached ? HAL_CONTROLWORD_DS_ATTACHED_MASK : 0);
206 return word;
207}
208
209inline int64_t HAL_ControlWord_GetOpModeHash(HAL_ControlWord word) {
210 return word.value & HAL_CONTROLWORD_OPMODE_HASH_MASK;
211}
212
213inline int64_t HAL_ControlWord_GetOpModeId(HAL_ControlWord word) {
214 // if the hash portion is zero, return 0
215 if ((word.value & HAL_CONTROLWORD_OPMODE_HASH_MASK) == 0) {
216 return 0;
217 }
218 // otherwise return the full ID (which includes the robot mode)
219 return word.value &
220 (HAL_CONTROLWORD_OPMODE_HASH_MASK | HAL_CONTROLWORD_ROBOT_MODE_MASK);
221}
222
223inline void HAL_ControlWord_SetOpModeId(HAL_ControlWord* word, int64_t id) {
224 // clear out the old hash and robot mode
225 word->value &=
226 ~(HAL_CONTROLWORD_OPMODE_HASH_MASK | HAL_CONTROLWORD_ROBOT_MODE_MASK);
227 // set the new id
228 word->value |=
229 id & (HAL_CONTROLWORD_OPMODE_HASH_MASK | HAL_CONTROLWORD_ROBOT_MODE_MASK);
230}

232inline HAL_RobotMode HAL_ControlWord_GetRobotMode(HAL_ControlWord word) {
233 // NOLINTBEGIN
234 return (HAL_RobotMode)((word.value & HAL_CONTROLWORD_ROBOT_MODE_MASK) >>
235 HAL_CONTROLWORD_ROBOT_MODE_SHIFT);
236 // NOLINTEND
237}
