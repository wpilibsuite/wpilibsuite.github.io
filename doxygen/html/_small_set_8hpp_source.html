30/// SmallSetIterator - This class implements a const_iterator for SmallSet by
31/// delegating to the underlying SmallVector or Set iterators.
32template <typename T, unsigned N, typename C>
33class SmallSetIterator
34 : public iterator_facade_base<SmallSetIterator<T, N, C>,
35 std::forward_iterator_tag, T> {
36private:
37 using SetIterTy = typename std::set<T, C>::const_iterator;
38 using VecIterTy = typename SmallVector<T, N>::const_iterator;
39
40 /// Iterators to the parts of the SmallSet containing the data. They are set
41 /// depending on isSmall.
42 union {
43 SetIterTy SetIter;
44 VecIterTy VecIter;
45 };
46
47 bool IsSmall;
48
49public:
50 SmallSetIterator(SetIterTy SetIter) : SetIter(SetIter), IsSmall(false) {}
51
52 SmallSetIterator(VecIterTy VecIter) : VecIter(VecIter), IsSmall(true) {}
53
54 // Spell out destructor, copy/move constructor and assignment operators for
55 // MSVC STL, where set<T>::const_iterator is not trivially copy constructible.
56 ~SmallSetIterator() {
57 if (IsSmall)
58 VecIter.~VecIterTy();
59 else
60 SetIter.~SetIterTy();
61 }
62
63 SmallSetIterator(const SmallSetIterator &Other) : IsSmall(Other.IsSmall) {
64 if (IsSmall)
65 VecIter = Other.VecIter;
66 else
67 // Use placement new, to make sure SetIter is properly constructed, even
68 // if it is not trivially copy-able (e.g. in MSVC).
69 new (&SetIter) SetIterTy(Other.SetIter);
70 }
71
72 SmallSetIterator(SmallSetIterator &&Other) : IsSmall(Other.IsSmall) {
73 if (IsSmall)
74 VecIter = std::move(Other.VecIter);
75 else
76 // Use placement new, to make sure SetIter is properly constructed, even
77 // if it is not trivially copy-able (e.g. in MSVC).
78 new (&SetIter) SetIterTy(std::move(Other.SetIter));
79 }
80
81 SmallSetIterator& operator=(const SmallSetIterator& Other) {
82 // Call destructor for SetIter, so it gets properly destroyed if it is
83 // not trivially destructible in case we are setting VecIter.
84 if (!IsSmall)
85 SetIter.~SetIterTy();
86
87 IsSmall = Other.IsSmall;
88 if (IsSmall)
89 VecIter = Other.VecIter;
90 else
91 new (&SetIter) SetIterTy(Other.SetIter);
92 return *this;
93 }
94
95 SmallSetIterator& operator=(SmallSetIterator&& Other) {
96 // Call destructor for SetIter, so it gets properly destroyed if it is
97 // not trivially destructible in case we are setting VecIter.
98 if (!IsSmall)
99 SetIter.~SetIterTy();
100
101 IsSmall = Other.IsSmall;
102 if (IsSmall)
103 VecIter = std::move(Other.VecIter);
104 else
105 new (&SetIter) SetIterTy(std::move(Other.SetIter));
106 return *this;
107 }
108
109 bool operator==(const SmallSetIterator &RHS) const {
110 if (IsSmall != RHS.IsSmall)
111 return false;
112 if (IsSmall)
113 return VecIter == RHS.VecIter;
114 return SetIter == RHS.SetIter;
115 }
116
117 SmallSetIterator &operator++() { // Preincrement
118 if (IsSmall)
119 ++VecIter;
120 else
121 ++SetIter;
122 return *this;
123 }
124
125 SmallSetIterator operator++(int) { // Postincrement
126 SmallSetIterator Copy = *this;
127 ++(*this);
128 return Copy;
129 }
130
131 const T &operator*() const { return IsSmall ? *VecIter : *SetIter; }
132};

134/// SmallSet - This maintains a set of unique values, optimizing for the case
135/// when the set is small (less than N). In this case, the set can be
136/// maintained with no mallocs. If the set gets large, we expand to using an
137/// std::set to maintain reasonable lookup times.
138template <typename T, unsigned N, typename C = std::less<T>>
139class SmallSet {
140 /// Use a SmallVector to hold the elements here (even though it will never
141 /// reach its 'large' stage) to avoid calling the default ctors of elements
142 /// we will never use.
143 SmallVector<T, N> Vector;
144 std::set<T, C> Set;
145
146 // In small mode SmallPtrSet uses linear search for the elements, so it is
147 // not a good idea to choose this value too high. You may consider using a
148 // DenseSet<> instead if you expect many elements in the set.
149 static_assert(N <= 32, "N should be small");
150
151public:
152 using key_type = T;
153 using size_type = size_t;
154 using value_type = T;
155 using const_iterator = SmallSetIterator<T, N, C>;
156
157 SmallSet() = default;
158 SmallSet(const SmallSet &) = default;
159 SmallSet(SmallSet &&) = default;
160
161 template <typename IterT> SmallSet(IterT Begin, IterT End) {
162 insert(Begin, End);
163 }
164
165 template <typename Range>
166 SmallSet(wpi::util::from_range_t, Range &&R)
167 : SmallSet(adl_begin(R), adl_end(R)) {}
168
169 SmallSet(std::initializer_list<T> L) { insert(L.begin(), L.end()); }
170
171 SmallSet &operator=(const SmallSet &) = default;
172 SmallSet &operator=(SmallSet &&) = default;
173
174 [[nodiscard]] bool empty() const { return Vector.empty() && Set.empty(); }
175
176 [[nodiscard]] size_type size() const {
177 return isSmall() ? Vector.size() : Set.size();
178 }
179
180 /// count - Return 1 if the element is in the set, 0 otherwise.
181 [[nodiscard]] size_type count(const T &V) const {
182 return contains(V) ? 1 : 0;
183 }
184
185 /// insert - Insert an element into the set if it isn't already there.
186 /// Returns a pair. The first value of it is an iterator to the inserted
187 /// element or the existing element in the set. The second value is true
188 /// if the element is inserted (it was not in the set before).
189 std::pair<const_iterator, bool> insert(const T &V) { return insertImpl(V); }
190
191 std::pair<const_iterator, bool> insert(T &&V) {
192 return insertImpl(std::move(V));
193 }
194
195 template <typename IterT>
196 void insert(IterT I, IterT E) {
197 for (; I != E; ++I)
198 insert(*I);
199 }
200
201 template <typename Range> void insert_range(Range &&R) {
202 insert(adl_begin(R), adl_end(R));
203 }
204
205 bool erase(const T &V) {
206 if (!isSmall())
207 return Set.erase(V);
208 auto I = vfind(V);
209 if (I != Vector.end()) {
210 Vector.erase(I);
211 return true;
212 }
213 return false;
214 }
215
216 void clear() {
217 Vector.clear();
218 Set.clear();
219 }
220
221 [[nodiscard]] const_iterator begin() const {
222 if (isSmall())
223 return {Vector.begin()};
224 return {Set.begin()};
225 }
226
227 [[nodiscard]] const_iterator end() const {
228 if (isSmall())
229 return {Vector.end()};
230 return {Set.end()};
231 }
232
233 /// Check if the SmallSet contains the given element.
234 [[nodiscard]] bool contains(const T &V) const {
235 if (isSmall())
236 return vfind(V) != Vector.end();
237 return Set.find(V) != Set.end();
238 }
239
240private:
241 bool isSmall() const { return Set.empty(); }
242
243 template <typename ArgType>
244 std::pair<const_iterator, bool> insertImpl(ArgType &&V) {
245 static_assert(std::is_convertible_v<ArgType, T>,
246 "ArgType must be convertible to T!");
247 if (!isSmall()) {
248 auto [I, Inserted] = Set.insert(std::forward<ArgType>(V));
249 return {const_iterator(I), Inserted};
250 }
251
252 auto I = vfind(V);
253 if (I != Vector.end()) // Don't reinsert if it already exists.
254 return {const_iterator(I), false};
255 if (Vector.size() < N) {
256 Vector.push_back(std::forward<ArgType>(V));
257 return {const_iterator(std::prev(Vector.end())), true};
258 }
259 // Otherwise, grow from vector to set.
260 Set.insert(std::make_move_iterator(Vector.begin()),
261 std::make_move_iterator(Vector.end()));
262 Vector.clear();
263 return {const_iterator(Set.insert(std::forward<ArgType>(V)).first), true};
264 }
265
266 // Handwritten linear search. The use of std::find might hurt performance as
267 // its implementation may be optimized for larger containers.
268 typename SmallVector<T, N>::const_iterator vfind(const T &V) const {
269 for (auto I = Vector.begin(), E = Vector.end(); I != E; ++I)
270 if (*I == V)
271 return I;
272 return Vector.end();
273 }
274};

276/// If this set is of pointer values, transparently switch over to using
277/// SmallPtrSet for performance.
278template <typename PointeeType, unsigned N>
279class SmallSet<PointeeType *, N> : public SmallPtrSet<PointeeType *, N> {};

281/// Equality comparison for SmallSet.
282///
283/// Iterates over elements of LHS confirming that each element is also a member
284/// of RHS, and that RHS contains no additional values.
285/// Equivalent to N calls to RHS.count.
286/// For small-set mode amortized complexity is O(N^2)
287/// For large-set mode amortized complexity is linear, worst case is O(N^2) (if
288/// every hash collides).
289template <typename T, unsigned LN, unsigned RN, typename C>
290[[nodiscard]] bool operator==(const SmallSet<T, LN, C> &LHS,
291 const SmallSet<T, RN, C> &RHS) {
292 if (LHS.size() != RHS.size())
293 return false;
294
295 // All elements in LHS must also be in RHS
296 return std::all_of(LHS.begin(), LHS.end(), [&RHS](const T &E) { return RHS.count(E); });
297}
298
299/// Inequality comparison for SmallSet.
300///
301/// Equivalent to !(LHS == RHS). See operator== for performance notes.
302template <typename T, unsigned LN, unsigned RN, typename C>
303[[nodiscard]] bool operator!=(const SmallSet<T, LN, C> &LHS,
304 const SmallSet<T, RN, C> &RHS) {
305 return !(LHS == RHS);
306}
