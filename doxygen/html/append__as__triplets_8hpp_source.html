14/// Appends sparse matrices to list of triplets at the given offset.
15///
16/// The triplets are appended in column-major order (e.g., first column of mat1,
17/// first column of mat2 underneath first column of mat1, second column of mat1,
18/// second column of mat2 underneath second column of mat1).
19///
20/// @tparam Scalar Scalar type.
21/// @param triplets List of triplets.
22/// @param row_offset Row offset for first matrix.
23/// @param col_offset Column offset for first matrix.
24/// @param mats Sparse matrices to append.
25template <typename Scalar>
26void append_as_triplets(
27 gch::small_vector<Eigen::Triplet<Scalar>>& triplets, int row_offset,
28 int col_offset, std::initializer_list<Eigen::SparseMatrix<Scalar>> mats) {
29 // Compute row offset for each matrix
30 gch::small_vector<int> mat_row_offsets;
31 int mat_row_offset = 0;
32 for (const auto& mat : mats) {
33 mat_row_offsets.emplace_back(mat_row_offset);
34 mat_row_offset += mat.rows();
35 }
36
37 // Append elements in column-major order
38 for (int col = 0; col < mats.begin()[0].cols(); ++col) {
39 for (size_t i = 0; i < mats.size(); ++i) {
40 for (typename Eigen::SparseMatrix<Scalar>::InnerIterator it{
41 mats.begin()[i], col};
42 it; ++it) {
43 triplets.emplace_back(row_offset + mat_row_offsets[i] + it.row(),
44 col_offset + it.col(), it.value());
45 }
46 }
47 }
48}

50/// Append diagonal matrix to list of triplets at the given offset.
51///
52/// @tparam Scalar Scalar type.
53/// @param triplets List of triplets.
54/// @param row_offset Row offset for first matrix.
55/// @param col_offset Column offset for first matrix.
56/// @param diag Diagonal of matrix.
57template <typename Scalar>
58void append_diagonal_as_triplets(
59 gch::small_vector<Eigen::Triplet<Scalar>>& triplets, int row_offset,
60 int col_offset, const Eigen::Vector<Scalar, Eigen::Dynamic>& diag) {
61 for (int row = 0; row < diag.rows(); ++row) {
62 triplets.emplace_back(row_offset + row, col_offset + row, diag[row]);
63 }
64}
