19/**
20 * This class incorporates time-delayed measurements into a Kalman filter's
21 * state estimate.
22 *
23 * @tparam States The number of states.
24 * @tparam Inputs The number of inputs.
25 * @tparam Outputs The number of outputs.
26 */
27template <int States, int Inputs, int Outputs, typename KalmanFilterType>
28class KalmanFilterLatencyCompensator {
29 public:
30 /**
31 * This class contains all the information about our observer at a given time.
32 */
33 struct ObserverSnapshot {
34 /// The state estimate.
35 Vectord<States> xHat;
36 /// The square root error covariance.
37 Matrixd<States, States> squareRootErrorCovariances;
38 /// The inputs.
39 Vectord<Inputs> inputs;
40 /// The local measurements.
41 Vectord<Outputs> localMeasurements;
42
43 ObserverSnapshot(const KalmanFilterType& observer, const Vectord<Inputs>& u,
44 const Vectord<Outputs>& localY)
45 : xHat(observer.Xhat()),
46 squareRootErrorCovariances(observer.S()),
47 inputs(u),
48 localMeasurements(localY) {}
49 };
50
51 /**
52 * Clears the observer snapshot buffer.
53 */
54 void Reset() { m_pastObserverSnapshots.clear(); }
55
56 /**
57 * Add past observer states to the observer snapshots list.
58 *
59 * @param observer The observer.
60 * @param u The input at the timestamp.
61 * @param localY The local output at the timestamp
62 * @param timestamp The timestamp of the state.
63 */
64 void AddObserverState(const KalmanFilterType& observer, Vectord<Inputs> u,
65 Vectord<Outputs> localY,
66 wpi::units::second_t timestamp) {
67 // Add the new state into the vector.
68 m_pastObserverSnapshots.emplace_back(timestamp,
69 ObserverSnapshot{observer, u, localY});
70
71 // Remove the oldest snapshot if the vector exceeds our maximum size.
72 if (m_pastObserverSnapshots.size() > MAX_PAST_OBSERVER_STATES) {
73 m_pastObserverSnapshots.erase(m_pastObserverSnapshots.begin());
74 }
75 }
76
77 /**
78 * Add past global measurements (such as from vision)to the estimator.
79 *
80 * @param observer The observer to apply the past global
81 * measurement.
82 * @param nominalDt The nominal timestep.
83 * @param y The measurement.
84 * @param globalMeasurementCorrect The function take calls correct() on the
85 * observer.
86 * @param timestamp The timestamp of the measurement.
87 */
88 template <int Rows>
89 void ApplyPastGlobalMeasurement(
90 KalmanFilterType* observer, wpi::units::second_t nominalDt,
91 Vectord<Rows> y,
92 std::function<void(const Vectord<Inputs>& u, const Vectord<Rows>& y)>
93 globalMeasurementCorrect,
94 wpi::units::second_t timestamp) {
95 if (m_pastObserverSnapshots.size() == 0) {
96 // State map was empty, which means that we got a measurement right at
97 // startup. The only thing we can do is ignore the measurement.
98 return;
99 }
100
101 // Perform a binary search to find the index of first snapshot whose
102 // timestamp is greater than or equal to the global measurement timestamp
103 auto it = std::lower_bound(
104 m_pastObserverSnapshots.cbegin(), m_pastObserverSnapshots.cend(),
105 timestamp,
106 [](const auto& entry, const auto& ts) { return entry.first < ts; });
107
108 size_t indexOfClosestEntry;
109
110 if (it == m_pastObserverSnapshots.cbegin()) {
111 // If the global measurement is older than any snapshot, throw out the
112 // measurement because there's no state estimate into which to incorporate
113 // the measurement
114 if (timestamp < it->first) {
115 return;
116 }
117
118 // If the first snapshot has same timestamp as the global measurement, use
119 // that snapshot
120 indexOfClosestEntry = 0;
121 } else if (it == m_pastObserverSnapshots.cend()) {
122 // If all snapshots are older than the global measurement, use the newest
123 // snapshot
124 indexOfClosestEntry = m_pastObserverSnapshots.size() - 1;
125 } else {
126 // Index of snapshot taken after the global measurement
127 int nextIdx = std::distance(m_pastObserverSnapshots.cbegin(), it);
128
129 // Index of snapshot taken before the global measurement. Since we already
130 // handled the case where the index points to the first snapshot, this
131 // computation is guaranteed to be nonnegative.
132 int prevIdx = nextIdx - 1;
133
134 // Find the snapshot closest in time to global measurement
135 wpi::units::second_t prevTimeDiff = wpi::units::math::abs(
136 timestamp - m_pastObserverSnapshots[prevIdx].first);
137 wpi::units::second_t nextTimeDiff = wpi::units::math::abs(
138 timestamp - m_pastObserverSnapshots[nextIdx].first);
139 indexOfClosestEntry = prevTimeDiff < nextTimeDiff ? prevIdx : nextIdx;
140 }
141
142 wpi::units::second_t lastTimestamp =
143 m_pastObserverSnapshots[indexOfClosestEntry].first - nominalDt;
144
145 // We will now go back in time to the state of the system at the time when
146 // the measurement was captured. We will reset the observer to that state,
147 // and apply correction based on the measurement. Then, we will go back
148 // through all observer states until the present and apply past inputs to
149 // get the present estimated state.
150 for (size_t i = indexOfClosestEntry; i < m_pastObserverSnapshots.size();
151 ++i) {
152 auto& [key, snapshot] = m_pastObserverSnapshots[i];
153
154 if (i == indexOfClosestEntry) {
155 observer->SetS(snapshot.squareRootErrorCovariances);
156 observer->SetXhat(snapshot.xHat);
157 }
158
159 observer->Predict(snapshot.inputs, key - lastTimestamp);
160 observer->Correct(snapshot.inputs, snapshot.localMeasurements);
161
162 if (i == indexOfClosestEntry) {
163 // Note that the measurement is at a timestep close but probably not
164 // exactly equal to the timestep for which we called predict. This makes
165 // the assumption that the dt is small enough that the difference
166 // between the measurement time and the time that the inputs were
167 // captured at is very small.
168 globalMeasurementCorrect(snapshot.inputs, y);
169 }
170
171 lastTimestamp = key;
172 snapshot = ObserverSnapshot{*observer, snapshot.inputs,
173 snapshot.localMeasurements};
174 }
175 }
176
177 private:
178 static constexpr size_t MAX_PAST_OBSERVER_STATES = 300;
179 std::vector<std::pair<wpi::units::second_t, ObserverSnapshot>>
180 m_pastObserverSnapshots;
181};
182} // namespace wpi::math
