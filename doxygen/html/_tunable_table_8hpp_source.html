29/**
30 * Tunables are used to allow values in the robot program to be changed from
31 * dashboards or debug tools.
32 *
33 * <p>For more advanced use cases, use NetworkTables directly.
34 */
35class TunableTable final {
36 public:
37 /**
38 * Constructs a tunable table.
39 *
40 * @param path path with trailing "/".
41 */
42 explicit TunableTable(std::string path) : m_path{std::move(path)} {}
43
44 /**
45 * Gets the table path.
46 *
47 * @return path with trailing "/"
48 */
49 const std::string& GetPath() const { return m_path; }
50
51 /**
52 * Gets a child tunable table.
53 *
54 * @param name table name
55 * @return table
56 */
57 TunableTable GetTable(std::string_view name);
58
59 /**
60 * Publishes a tunable.
61 *
62 * @param name the name
63 * @param tunable the tunable
64 * @return true if the backend accepted the tunable
65 */
66 bool Publish(std::string_view name, detail::TunableBase& tunable);
67
68 /**
69 * Publishes a complex tunable.
70 *
71 * @param name the name
72 * @param tunable the complex tunable object
73 * @return true if the backend accepted the tunable
74 */
75 bool Publish(std::string_view name, ComplexTunable& tunable);
76
77 /**
78 * Publishes a custom tunable.
79 *
80 * @param name the name
81 * @param tunable the custom tunable
82 * @return true if the backend accepted the tunable
83 */
84 template <typename T, typename... I>
85 requires detail::IsCustomTunable<T, I...>
86 bool Publish(std::string_view name, Tunable<T, I...>& tunable) {
87 return Publish(name, tunable.GetInnerTunable());
88 }
89
90 /**
91 * Publishes a tunable for a member variable of a complex tunable.
92 *
93 * @param name the name
94 * @param tunable the complex tunable object
95 * @param member the member tunable for the member variable
96 * @return true if the backend accepted the tunable
97 */
98 bool Publish(std::string_view name, ComplexTunable* tunable,
99 std::unique_ptr<detail::TunableMemberBase> member);
100
101 /**
102 * Publishes a tunable for a member variable of a complex tunable.
103 *
104 * @param name the name
105 * @param tunable the complex tunable object
106 * @param member the member variable pointer
107 * @param info additional information forwarded to the member tunable wrapper
108 * @return true if the backend accepted the tunable
109 */
110 template <typename T, std::derived_from<ComplexTunable> Class, typename... I>
111 requires std::convertible_to<
112 std::unique_ptr<detail::TunableMemberBase>,
113 decltype(MakeTunableMember(std::declval<T Class::*>()))>
114 bool Publish(std::string_view name, Class* tunable, T Class::* member,
115 I&&... info) {
116 return Publish(name, tunable,
117 MakeTunableMember(member, std::forward<I>(info)...));
118 }
119
120 /**
121 * Publishes a tunable for a member variable of a complex tunable.
122 *
123 * @param name the name
124 * @param tunable the complex tunable object
125 * @param member the member variable pointer
126 * @param config the tunable config
127 * @param info additional information forwarded to the member tunable wrapper
128 * @return true if the backend accepted the tunable
129 */
130 template <typename T, std::derived_from<ComplexTunable> Class, typename... I>
131 requires std::convertible_to<
132 std::unique_ptr<detail::TunableMemberBase>,
133 decltype(MakeTunableMember(std::declval<T Class::*>(),
134 std::declval<TunableConfig>()))>
135 bool Publish(std::string_view name, Class* tunable, T Class::* member,
136 const TunableConfig& config, I&&... info) {
137 return Publish(name, tunable,
138 MakeTunableMember(member, config, std::forward<I>(info)...));
139 }
140
141 /**
142 * Removes a tunable and any descendant tunables.
143 * @param name name
144 */
145 void Remove(std::string_view name);
146
147 private:
148 std::string m_path;
149};
