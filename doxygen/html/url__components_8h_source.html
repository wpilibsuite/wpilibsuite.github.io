14/**
15 * @brief Stores byte offsets for URL components within a buffer.
16 *
17 * The `url_components` struct uses 32-bit offsets to track the boundaries
18 * of each URL component within a single string buffer. This enables efficient
19 * component extraction without additional memory allocations.
20 *
21 * Component layout in a URL:
22 * ```
23 * https://user:pass@example.com:1234/foo/bar?baz#quux
24 * | | | | ^^^^| | |
25 * | | | | | | | `----- hash_start
26 * | | | | | | `--------- search_start
27 * | | | | | `----------------- pathname_start
28 * | | | | `--------------------- port
29 * | | | `----------------------- host_end
30 * | | `---------------------------------- host_start
31 * | `--------------------------------------- username_end
32 * `--------------------------------------------- protocol_end
33 * ```
34 *
35 * @note The 32-bit offsets limit URLs to 4GB in length.
36 * @note A value of `omitted` (UINT32_MAX) indicates the component is not
37 * present.
38 */
39struct url_components {
40 /** Sentinel value indicating a component is not present. */
41 constexpr static uint32_t omitted = uint32_t(-1);
42
43 url_components() = default;
44 url_components(const url_components& u) = default;
45 url_components(url_components&& u) noexcept = default;
46 url_components& operator=(url_components&& u) noexcept = default;
47 url_components& operator=(const url_components& u) = default;
48 ~url_components() = default;
49
50 /** Offset of the end of the protocol/scheme (position of ':'). */
51 uint32_t protocol_end{0};
52
53 /**
54 * Offset of the end of the username.
55 * Initialized to 0 (not `omitted`) to simplify username/password getters.
56 */
57 uint32_t username_end{0};
58
59 /** Offset of the start of the host. */
60 uint32_t host_start{0};
61
62 /** Offset of the end of the host. */
63 uint32_t host_end{0};
64
65 /** Port number, or `omitted` if no port is specified. */
66 uint32_t port{omitted};
67
68 /** Offset of the start of the pathname. */
69 uint32_t pathname_start{0};
70
71 /** Offset of the '?' starting the query, or `omitted` if no query. */
72 uint32_t search_start{omitted};
73
74 /** Offset of the '#' starting the fragment, or `omitted` if no fragment. */
75 uint32_t hash_start{omitted};
76
77 /**
78 * Validates that offsets are in ascending order and consistent.
79 * Useful for debugging to detect internal corruption.
80 * @return `true` if offsets are consistent, `false` otherwise.
81 */
82 [[nodiscard]] constexpr bool check_offset_consistency() const noexcept;
83
84 /**

87 */
88 [[nodiscard]] std::string to_string() const;

90}; // struct url_components
