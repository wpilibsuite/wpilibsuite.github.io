20/**
21 * This class estimates the input bias required to make the expected motor
22 * velocity match the actual velocity over one timestep.
23 *
24 * Subtracting this input bias from the next control input will cancel it and
25 * eliminate steady-state error. This method of fixing steady-state error is
26 * superior to a pure integral controller because it only integrates for
27 * unmodeled disturbances, not during expected transient behavior while
28 * converging to a reference. This means there won't be overshoot during
29 * transients.
30 *
31 * The feedforward gains passed to this class should match the ones used by the
32 * mechanism's feedforward. The estimator estimates the error in that
33 * feedforward model; if the gains differ, the difference shows up as input
34 * error. In particular, errors in kV appear as an input bias proportional to
35 * velocity, and errors in kA appear as an input bias during acceleration.
36 * Logging the input bias estimate against velocity is a useful way to find
37 * these: a slope indicates kV error, and a sign flip on direction reversal
38 * indicates kS error.
39 */
40template <class Distance>
41 requires wpi::units::length_unit<Distance> ||
42 wpi::units::angle_unit<Distance> ||
43 wpi::units::dimensionless_unit<Distance>
44class DCMotorBiasEstimator {
45 public:
46 using Velocity =
47 wpi::units::compound_unit<Distance, wpi::units::inverse<units::seconds>>;
48 using Velocity_t = wpi::units::unit_t<Velocity>;
49 using Acceleration =
50 wpi::units::compound_unit<Velocity, wpi::units::inverse<units::seconds>>;
51
52 using kV_t = wpi::units::unit_t<
53 wpi::units::compound_unit<units::volt, wpi::units::inverse<Velocity>>>;
54 using kA_t = wpi::units::unit_t<wpi::units::compound_unit<
55 units::volt, wpi::units::inverse<Acceleration>>>;
56 using VelocityPerVolt_t = wpi::units::unit_t<
57 wpi::units::compound_unit<Velocity, wpi::units::inverse<units::volt>>>;
58
59 /**
60 * Constructs a DC motor bias estimator.
61 *
62 * The model is
63 *
64 * <pre>
65 * a = −kV/kA·v + 1/kA·(u + d) − kS/kA·sgn(v) − kG/kA
66 * </pre>
67 *
68 * where a is the acceleration, v is the velocity, u is the applied voltage,
69 * and d is the estimated input bias.
70 *
71 * For an arm, whose gravity term depends on angle, pass zero for kG and
72 * subtract the gravity feedforward voltage from the voltage passed to
73 * Calculate() instead.
74 *
75 * @param kS The static gain. This should match the feedforward's kS, or be
76 * zero if the feedforward doesn't use one.
77 * @param kG The gravity gain. This should match the feedforward's kG, or be
78 * zero if the feedforward doesn't use one.
79 * @param kV The velocity gain. Must be non-negative.
80 * @param kA The acceleration gain. Must be non-negative. kV and kA can't
81 * both be zero.
82 * @param timeConstant The approximate time for the input bias estimate to
83 * converge to a new constant input bias. Larger values reject
84 * disturbances more slowly but react less to model error and
85 * measurement noise. Must be positive.
86 * @param dt Nominal discretization timestep. Must be positive.
87 * @throws std::invalid_argument if any of the parameter constraints are
88 * violated.
89 */
90 constexpr DCMotorBiasEstimator(wpi::units::volt_t kS, wpi::units::volt_t kG,
91 kV_t kV, kA_t kA,
92 wpi::units::second_t timeConstant,
93 wpi::units::second_t dt = 20_ms)
94 : m_kS{kS}, m_kG{kG} {
95 if (kV < kV_t{0}) {
96 throw std::invalid_argument("kV must be non-negative");
97 }
98 if (kA < kA_t{0}) {
99 throw std::invalid_argument("kA must be non-negative");
100 }
101 if (kV == kV_t{0} && kA == kA_t{0}) {
102 throw std::invalid_argument("kV and kA can't both be zero");
103 }
104 if (timeConstant <= 0_s) {
105 throw std::invalid_argument("Time constant must be positive");
106 }
107 if (dt <= 0_s) {
108 throw std::invalid_argument("dt must be positive");
109 }
110
111 // Discretize the velocity dynamics v' = −kV/kA v + 1/kA (u + d) with a
112 // zero-order hold. See wpimath/docs/SimpleMotorFeedforward.md for the
113 // derivation, including the kA = 0 and kV = 0 cases.
114 if (kA == kA_t{0}) {
115 m_A = 0.0;
116 m_B = 1.0 / kV;
117 } else if (kV == kV_t{0}) {
118 m_A = 1.0;
119 m_B = dt / kA;
120 } else {
121 // B_d = A⁻¹(eᴬᵀ − 1)B simplifies to (1 − A_d)/kV
122 auto exponent = -kV / kA * dt;
123 m_A = wpi::units::math::exp(exponent);
124 m_B = -wpi::units::math::expm1(exponent) / kV;
125 }
126
127 // The augmented system is
128 //
129 // xₖ₊₁ = Φxₖ + Γuₖ
130 // yₖ = Cxₖ
131 //
132 // where
133 //
134 // x = [v] Φ = [A_d B_d] Γ = [B_d] C = [1 0]
135 // [d] [ 0 1 ] [ 0 ]
136 //
137 // The predictor-form observer is
138 //
139 // x̂ₖ₊₁ = Φx̂ₖ + Γuₖ + L(yₖ − Cx̂ₖ)
140 //
141 // where L = [l₁ l₂]ᵀ. Subtracting it from the augmented system gives the
142 // estimation error dynamics
143 //
144 // x̃ₖ₊₁ = (Φ − LC)x̃ₖ
145 //
146 // where x̃ = x − x̂ and
147 //
148 // Φ − LC = [A_d − l₁ B_d]
149 // [ −l₂ 1 ]
150 //
151 // The error dynamics' poles are the roots of the characteristic polynomial
152 //
153 // det(zI − (Φ − LC)) = det([z − A_d + l₁ −B_d ])
154 // ([ l₂ z − 1 ])
155 // = (z − A_d + l₁)(z − 1) + B_d l₂
156 // = z² − (1 + A_d − l₁)z + (A_d − l₁ + B_d l₂)
157 //
158 // Matching coefficients with the desired characteristic polynomial
159 //
160 // (z − p_v)(z − p_d) = z² − (p_v + p_d)z + p_v p_d
161 //
162 // gives
163 //
164 // 1 + A_d − l₁ = p_v + p_d
165 // A_d − l₁ + B_d l₂ = p_v p_d
166 //
167 // The first equation gives l₁. Substituting A_d − l₁ = p_v + p_d − 1 from
168 // the first equation into the second gives l₂.
169 //
170 // l₁ = 1 + A_d − p_v − p_d
171 // B_d l₂ = p_v p_d − p_v − p_d + 1
172 // l₂ = (1 − p_v)(1 − p_d)/B_d
173 //
174 // The input bias pole p_d uses the given time constant. The velocity pole
175 // p_v is placed 5x faster so the input bias estimate's convergence time is
176 // dominated by its own time constant.
177 wpi::units::dimensionless_t p_d = wpi::units::math::exp(-dt / timeConstant);
178 wpi::units::dimensionless_t p_v =
179 wpi::units::math::exp(-dt / (timeConstant / 5.0));
180 m_l1 = 1.0 + m_A - p_v - p_d;
181 m_l2 = (1.0 - p_v) * (1.0 - p_d) / m_B;
182 }
183
184 /**
185 * Constructs a DC motor bias estimator for a mechanism without a gravity
186 * term, such as a flywheel or drivetrain.
187 *
188 * The model is
189 *
190 * <pre>
191 * a = −kV/kA·v + 1/kA·(u + d) − kS/kA·sgn(v)
192 * </pre>
193 *
194 * where a is the acceleration, v is the velocity, u is the applied voltage,
195 * and d is the estimated input bias.
196 *
197 * @param kS The static gain. This should match the feedforward's kS, or be
198 * zero if the feedforward doesn't use one.
199 * @param kV The velocity gain. Must be non-negative.
200 * @param kA The acceleration gain. Must be non-negative. kV and kA can't
201 * both be zero.
202 * @param timeConstant The approximate time for the input bias estimate to
203 * converge to a new constant input bias. Larger values reject
204 * disturbances more slowly but react less to model error and
205 * measurement noise. Must be positive.
206 * @param dt Nominal discretization timestep. Must be positive.
207 * @throws std::invalid_argument if any of the parameter constraints are
208 * violated.
209 */
210 constexpr DCMotorBiasEstimator(wpi::units::volt_t kS, kV_t kV, kA_t kA,
211 wpi::units::second_t timeConstant,
212 wpi::units::second_t dt = 20_ms)
213 : DCMotorBiasEstimator{kS, 0_V, kV, kA, timeConstant, dt} {}
214
215 /**
216 * Returns the input bias estimate.
217 *
218 * @param velocity The velocity from the current timestep.
219 * @param voltage The voltage applied for the current timestep.
220 */
221 constexpr wpi::units::volt_t Calculate(Velocity_t velocity,
222 units::volt_t voltage) {
223 // Voltage left over after the static friction and gravity terms
224 wpi::units::volt_t u = voltage - m_kS * wpi::util::sgn(velocity) - m_kG;
225
226 Velocity_t residual = velocity - m_velocity;
227 m_velocity = m_A * m_velocity + m_B * (u + m_inputBias) + m_l1 * residual;
228 m_inputBias += m_l2 * residual;
229
230 return m_inputBias;
231 }
232
233 /**
234 * Resets the estimator to the given velocity with no input bias.
235 *
236 * @param velocity The current velocity.
237 */
238 constexpr void Reset(Velocity_t velocity) {
239 m_velocity = velocity;
240 m_inputBias = 0_V;
241 }
242
243 private:
244 wpi::units::volt_t m_kS;
245 wpi::units::volt_t m_kG;
246
247 // Discrete velocity model
248 wpi::units::dimensionless_t m_A{0};
249 VelocityPerVolt_t m_B{0};
250
251 // Observer gains
252 wpi::units::dimensionless_t m_l1{0};
253 kV_t m_l2{0};
254
255 // State estimate
256 Velocity_t m_velocity{0};
257 wpi::units::volt_t m_inputBias{0};
258};
