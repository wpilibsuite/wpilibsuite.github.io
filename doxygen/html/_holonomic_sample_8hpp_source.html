26/**
27 * Represents a single sample in a drivetrain trajectory.
28 */
29class HolonomicSample : public TrajectorySample {
30 public:
31 /** The robot pose at this sample (in the field reference frame). */
32 Pose2d pose;
33 /** The robot velocity at this sample (in the field reference frame). */
34 ChassisVelocities velocity;
35 /** The robot acceleration at this sample (in the field reference frame). */
36 ChassisAccelerations acceleration;
37
38 /** Constructs a default HolonomicSample with all zero values. */
39 constexpr HolonomicSample() = default;
40
41 ~HolonomicSample() = default;
42
43 /**
44 * Constructs a HolonomicSample.
45 *
46 * @param time The time of the sample relative to the trajectory start.
47 * @param p The robot pose at this sample (in the field reference frame).
48 * @param v The robot velocity at this sample (in the field reference frame).
49 * @param a The robot acceleration at this sample (in the field reference
50 * frame).
51 */
52 constexpr HolonomicSample(wpi::units::second_t time, const Pose2d& p,
53 const ChassisVelocities& v,
54 const ChassisAccelerations& a)
55 : TrajectorySample{time}, pose{p}, velocity{v}, acceleration{a} {}
56
57 /**
58 * Checks equality between this HolonomicSample and another.
59 *
60 * @return True if the samples are equal.
61 */
62 constexpr bool operator==(const HolonomicSample& other) const = default;
63
64 /**
65 * Transforms the pose of this sample by the given transform.
66 *
67 * @param transform The transform to apply to the pose.
68 * @return A new sample with the transformed pose.
69 */
70 constexpr HolonomicSample Transform(const Transform2d& transform) const {
71 return {time, pose.TransformBy(transform),
72 velocity.ToFieldRelative(transform.Rotation()),
73 acceleration.ToFieldRelative(transform.Rotation())};
74 }
75
76 /**
77 * Transforms this sample to be relative to the given pose.
78 *
79 * @param other The pose to make this sample relative to.
80 * @return A new sample with the relative pose.
81 */
82 constexpr HolonomicSample RelativeTo(const Pose2d& other) const {
83 return {time, pose.RelativeTo(other),
84 velocity.ToRobotRelative(other.Rotation()),
85 acceleration.ToRobotRelative(other.Rotation())};
86 }
87};
88
89/**
90 * Static kinematic interpolation between two samples (constant accel).
91 *
92 * @param start The start sample.
93 * @param end The end sample.
94 * @param t The interpolation parameter [0, 1].
95 * @return A new interpolated base sample.
96 */
97constexpr HolonomicSample KinematicInterpolate(const HolonomicSample& start,
98 const HolonomicSample& end,
99 double t) {
100 if (t <= 0.0) {
101 return {start.time, start.pose, start.velocity, start.acceleration};
102 } else if (t >= 1.0) {
103 return {end.time, end.pose, end.velocity, end.acceleration};
104 }
105
106 // Absolute time of the interpolated sample
107 const auto interpTime = wpi::util::Lerp(start.time, end.time, t);
108
109 // Elapsed time from the start sample, used for integration
110 const auto deltaT = interpTime - start.time;
111
112 // Constant acceleration
113 ChassisAccelerations accel{start.acceleration.ax, start.acceleration.ay,
114 start.acceleration.alpha};
115
116 // vₖ₊₁ = vₖ + aₖΔt
117 ChassisVelocities newVel{
118 start.velocity.vx + start.acceleration.ax * deltaT,
119 start.velocity.vy + start.acceleration.ay * deltaT,
120 start.velocity.omega + start.acceleration.alpha * deltaT};
121
122 // xₖ₊₁ = xₖ + vₖΔt + ½aₖ(Δt)²
123 Pose2d newPose{start.pose.Translation().X() + start.velocity.vx * deltaT +
124 0.5 * start.acceleration.ax * deltaT * deltaT,
125 start.pose.Translation().Y() + start.velocity.vy * deltaT +
126 0.5 * start.acceleration.ay * deltaT * deltaT,
127 Rotation2d{start.pose.Rotation().Radians() +
128 start.velocity.omega * deltaT +
129 0.5 * start.acceleration.alpha * deltaT * deltaT}};
130
131 return HolonomicSample{interpTime, newPose, newVel, accel};
132}
