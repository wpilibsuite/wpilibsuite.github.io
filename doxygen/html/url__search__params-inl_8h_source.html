35inline void url_search_params::initialize(std::string_view input) {
36 if (!input.empty() && input.front() == '?') {
37 input.remove_prefix(1);
38 }
39 if (input.empty()) {
40 return;
41 }
42 // Refuse overlong query strings (same process-wide cap as URL parsing).
43 if (input.size() > get_max_input_length()) {
44 return;
45 }
46
47 params.reserve(size_t(std::count(input.begin(), input.end(), '&')) + 1);
48
49 auto process_key_value = [&](const std::string_view current) {
50 const auto equal = current.find('=');
51 if (equal == std::string_view::npos) {
52 params.emplace_back(unicode::form_urlencoded_decode(current), "");
53 } else {
54 params.emplace_back(
55 unicode::form_urlencoded_decode(current.substr(0, equal)),
56 unicode::form_urlencoded_decode(current.substr(equal + 1)));
57 }
58 };
59
60 while (!input.empty()) {
61 const auto ampersand_index = input.find('&');
62
63 if (ampersand_index == std::string_view::npos) {
64 if (!input.empty()) {
65 process_key_value(input);
66 }
67 break;
68 } else if (ampersand_index != 0) {
69 process_key_value(input.substr(0, ampersand_index));
70 }
71
72 input.remove_prefix(ampersand_index + 1);
73 }
74}

122inline std::string url_search_params::to_string() const {
123 auto character_set = ada::character_sets::WWW_FORM_URLENCODED_PERCENT_ENCODE;
124 std::string out{};
125 for (size_t i = 0; i < params.size(); i++) {
126 auto key = ada::unicode::percent_encode(params[i].first, character_set);
127 auto value = ada::unicode::percent_encode(params[i].second, character_set);
128
129 // Performance optimization: Move this inside percent_encode.
130 std::ranges::replace(key, ' ', '+');
131 std::ranges::replace(value, ' ', '+');
132
133 if (i != 0) {
134 out += "&";
135 }
136 out.append(key);
137 out += "=";
138 out.append(value);
139 }
140 return out;
141}

143inline void url_search_params::set(const std::string_view key,
144 const std::string_view value) {
145 const auto find = [&key](const auto& param) { return param.first == key; };
146
147 auto it = std::ranges::find_if(params, find);
148
149 if (it == params.end()) {
150 params.emplace_back(key, value);
151 } else {
152 it->second = value;
153 params.erase(std::remove_if(std::next(it), params.end(), find),
154 params.end());
155 }
156}

170inline void url_search_params::sort() {
171 // Keys are expected to be valid UTF-8, but percent_decode can produce
172 // arbitrary byte sequences. Handle truncated/invalid sequences gracefully.
173 std::ranges::stable_sort(params, [](const key_value_pair& lhs,
174 const key_value_pair& rhs) {
175 size_t i = 0, j = 0;
176 uint32_t low_surrogate1 = 0, low_surrogate2 = 0;
177 while ((i < lhs.first.size() || low_surrogate1 != 0) &&
178 (j < rhs.first.size() || low_surrogate2 != 0)) {
179 uint32_t codePoint1 = 0, codePoint2 = 0;
180
181 if (low_surrogate1 != 0) {
182 codePoint1 = low_surrogate1;
183 low_surrogate1 = 0;
184 } else {
185 uint8_t c1 = uint8_t(lhs.first[i]);
186 if (c1 > 0x7F && c1 <= 0xDF && i + 1 < lhs.first.size()) {
187 codePoint1 = ((c1 & 0x1F) << 6) | (uint8_t(lhs.first[i + 1]) & 0x3F);
188 i += 2;
189 } else if (c1 > 0xDF && c1 <= 0xEF && i + 2 < lhs.first.size()) {
190 codePoint1 = ((c1 & 0x0F) << 12) |
191 ((uint8_t(lhs.first[i + 1]) & 0x3F) << 6) |
192 (uint8_t(lhs.first[i + 2]) & 0x3F);
193 i += 3;
194 } else if (c1 > 0xEF && c1 <= 0xF7 && i + 3 < lhs.first.size()) {
195 codePoint1 = ((c1 & 0x07) << 18) |
196 ((uint8_t(lhs.first[i + 1]) & 0x3F) << 12) |
197 ((uint8_t(lhs.first[i + 2]) & 0x3F) << 6) |
198 (uint8_t(lhs.first[i + 3]) & 0x3F);
199 i += 4;
200
201 codePoint1 -= 0x10000;
202 uint16_t high_surrogate = uint16_t(0xD800 + (codePoint1 >> 10));
203 low_surrogate1 = uint16_t(0xDC00 + (codePoint1 & 0x3FF));
204 codePoint1 = high_surrogate;
205 } else {
206 // ASCII (c1 <= 0x7F) or truncated/invalid UTF-8: treat as raw byte
207 codePoint1 = c1;
208 i++;
209 }
210 }
211
212 if (low_surrogate2 != 0) {
213 codePoint2 = low_surrogate2;
214 low_surrogate2 = 0;
215 } else {
216 uint8_t c2 = uint8_t(rhs.first[j]);
217 if (c2 > 0x7F && c2 <= 0xDF && j + 1 < rhs.first.size()) {
218 codePoint2 = ((c2 & 0x1F) << 6) | (uint8_t(rhs.first[j + 1]) & 0x3F);
219 j += 2;
220 } else if (c2 > 0xDF && c2 <= 0xEF && j + 2 < rhs.first.size()) {
221 codePoint2 = ((c2 & 0x0F) << 12) |
222 ((uint8_t(rhs.first[j + 1]) & 0x3F) << 6) |
223 (uint8_t(rhs.first[j + 2]) & 0x3F);
224 j += 3;
225 } else if (c2 > 0xEF && c2 <= 0xF7 && j + 3 < rhs.first.size()) {
226 codePoint2 = ((c2 & 0x07) << 18) |
227 ((uint8_t(rhs.first[j + 1]) & 0x3F) << 12) |
228 ((uint8_t(rhs.first[j + 2]) & 0x3F) << 6) |
229 (uint8_t(rhs.first[j + 3]) & 0x3F);
230 j += 4;
231 codePoint2 -= 0x10000;
232 uint16_t high_surrogate = uint16_t(0xD800 + (codePoint2 >> 10));
233 low_surrogate2 = uint16_t(0xDC00 + (codePoint2 & 0x3FF));
234 codePoint2 = high_surrogate;
235 } else {
236 // ASCII (c2 <= 0x7F) or truncated/invalid UTF-8: treat as raw byte
237 codePoint2 = c2;
238 j++;
239 }
240 }
241
242 if (codePoint1 != codePoint2) {
243 return (codePoint1 < codePoint2);
244 }
245 }
246 return (j < rhs.first.size() || low_surrogate2 != 0);
247 });
248}
