23/**
24 * A plant defined using state-space notation.
25 *
26 * A plant is a mathematical model of a system's dynamics.
27 *
28 * For more on the underlying math, read
29 * https://file.tavsys.net/control/controls-engineering-in-frc.pdf.
30 *
31 * @tparam States Number of states.
32 * @tparam Inputs Number of inputs.
33 * @tparam Outputs Number of outputs.
34 */
35template <int States, int Inputs, int Outputs>
36class LinearSystem {
37 public:
38 using StateVector = Vectord<States>;
39 using InputVector = Vectord<Inputs>;
40 using OutputVector = Vectord<Outputs>;
41
42 /**
43 * Constructs a discrete plant with the given continuous system coefficients.
44 *
45 * @param A System matrix.
46 * @param B Input matrix.
47 * @param C Output matrix.
48 * @param D Feedthrough matrix.
49 * @throws std::domain_error if any matrix element isn't finite.
50 */
51 constexpr LinearSystem(const Matrixd<States, States>& A,
52 const Matrixd<States, Inputs>& B,
53 const Matrixd<Outputs, States>& C,
54 const Matrixd<Outputs, Inputs>& D) {
55 auto allFinite = [](const auto& mat) {
56 if consteval {
57 for (int row = 0; row < mat.rows(); ++row) {
58 for (int col = 0; col < mat.cols(); ++col) {
59 if (!gcem::internal::is_finite(mat(row, col))) {
60 return false;
61 }
62 }
63 }
64 return true;
65 } else {
66 return mat.allFinite();
67 }
68 };
69
70 if (!allFinite(A)) {
71 throw std::domain_error(
72 "Elements of A aren't finite. This is usually due to model "
73 "implementation errors.");
74 }
75 if (!allFinite(B)) {
76 throw std::domain_error(
77 "Elements of B aren't finite. This is usually due to model "
78 "implementation errors.");
79 }
80 if (!allFinite(C)) {
81 throw std::domain_error(
82 "Elements of C aren't finite. This is usually due to model "
83 "implementation errors.");
84 }
85 if (!allFinite(D)) {
86 throw std::domain_error(
87 "Elements of D aren't finite. This is usually due to model "
88 "implementation errors.");
89 }
90
91 m_A = A;
92 m_B = B;
93 m_C = C;
94 m_D = D;
95 }
96
97 constexpr LinearSystem(const LinearSystem&) = default;
98 constexpr LinearSystem& operator=(const LinearSystem&) = default;
99 constexpr LinearSystem(LinearSystem&&) = default;
100 constexpr LinearSystem& operator=(LinearSystem&&) = default;
101
102 /**
103 * Returns the system matrix A.
104 */
105 constexpr const Matrixd<States, States>& A() const { return m_A; }
106
107 /**
108 * Returns an element of the system matrix A.
109 *
110 * @param i Row of A.
111 * @param j Column of A.
112 */
113 constexpr double A(int i, int j) const { return m_A(i, j); }
114
115 /**
116 * Returns the input matrix B.
117 */
118 constexpr const Matrixd<States, Inputs>& B() const { return m_B; }
119
120 /**
121 * Returns an element of the input matrix B.
122 *
123 * @param i Row of B.
124 * @param j Column of B.
125 */
126 constexpr double B(int i, int j) const { return m_B(i, j); }
127
128 /**
129 * Returns the output matrix C.
130 */
131 constexpr const Matrixd<Outputs, States>& C() const { return m_C; }
132
133 /**
134 * Returns an element of the output matrix C.
135 *
136 * @param i Row of C.
137 * @param j Column of C.
138 */
139 constexpr double C(int i, int j) const { return m_C(i, j); }
140
141 /**
142 * Returns the feedthrough matrix D.
143 */
144 constexpr const Matrixd<Outputs, Inputs>& D() const { return m_D; }
145
146 /**
147 * Returns an element of the feedthrough matrix D.
148 *
149 * @param i Row of D.
150 * @param j Column of D.
151 */
152 constexpr double D(int i, int j) const { return m_D(i, j); }
153
154 /**
155 * Computes the new x given the old x and the control input.
156 *
157 * This is used by state observers directly to run updates based on state
158 * estimate.
159 *
160 * @param x The current state.
161 * @param clampedU The control input.
162 * @param dt Timestep for model update.
163 */
164 StateVector CalculateX(const StateVector& x, const InputVector& clampedU,
165 wpi::units::second_t dt) const {
166 Matrixd<States, States> discA;
167 Matrixd<States, Inputs> discB;
168 DiscretizeAB<States, Inputs>(m_A, m_B, dt, &discA, &discB);
169
170 return discA * x + discB * clampedU;
171 }
172
173 /**
174 * Computes the new y given the control input.
175 *
176 * This is used by state observers directly to run updates based on state
177 * estimate.
178 *
179 * @param x The current state.
180 * @param clampedU The control input.
181 */
182 OutputVector CalculateY(const StateVector& x,
183 const InputVector& clampedU) const {
184 return m_C * x + m_D * clampedU;
185 }
186
187 /**
188 * Returns the LinearSystem with the outputs listed in outputIndices.
189 *
190 * <p>This is used by state observers such as the Kalman Filter.
191 *
192 * @param outputIndices the list of output indices to include in the sliced
193 * system.
194 * @return the sliced LinearSystem with outputs set to row vectors of
195 * LinearSystem.
196 * @throws std::domain_error if any outputIndices are outside the range of
197 * system outputs.
198 * @throws std::domain_error if number of outputIndices exceeds the system
199 * outputs.
200 * @throws std::domain_error if duplication exists in outputIndices.
201 */
202 template <std::same_as<int>... OutputIndices>
203 LinearSystem<States, Inputs, sizeof...(OutputIndices)> Slice(
204 OutputIndices... outputIndices) {
205 static_assert(sizeof...(OutputIndices) <= Outputs,
206 "More outputs requested than available. This is usually due "
207 "to model implementation errors.");
208
209 wpi::util::for_each(
210 [](size_t i, const auto& elem) {
211 if (elem < 0 || elem >= Outputs) {
212 throw std::domain_error(
213 "Slice indices out of range. This is usually due to model "
214 "implementation errors.");
215 }
216 },
217 outputIndices...);
218
219 // Sort and deduplicate output indices
220 wpi::util::SmallVector<int> outputIndicesArray{outputIndices...};
221 std::sort(outputIndicesArray.begin(), outputIndicesArray.end());
222 auto last =
223 std::unique(outputIndicesArray.begin(), outputIndicesArray.end());
224 outputIndicesArray.erase(last, outputIndicesArray.end());
225
226 if (outputIndicesArray.size() != sizeof...(outputIndices)) {
227 throw std::domain_error(
228 "Duplicate indices exist. This is usually due to model "
229 "implementation errors.");
230 }
231
232 return LinearSystem<States, Inputs, sizeof...(OutputIndices)>{
233 m_A, m_B, m_C(outputIndicesArray, Eigen::placeholders::all),
234 m_D(outputIndicesArray, Eigen::placeholders::all)};
235 }
236
237 private:
238 /**
239 * Continuous system matrix.
240 */
241 Matrixd<States, States> m_A;
242
243 /**
244 * Continuous input matrix.
245 */
246 Matrixd<States, Inputs> m_B;
247
248 /**
249 * Output matrix.
250 */
251 Matrixd<Outputs, States> m_C;
252
253 /**
254 * Feedthrough matrix.
255 */
256 Matrixd<Outputs, Inputs> m_D;
257};
