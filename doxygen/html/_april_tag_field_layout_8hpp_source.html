23/**
24 * Class for representing a layout of AprilTags on a field and reading them from
25 * a JSON format.
26 *
27 * The JSON format contains two top-level objects, "tags" and "field".
28 * The "tags" object is a list of all AprilTags contained within a layout. Each
29 * AprilTag serializes to a JSON object containing an ID and a
30 * wpi::math::Pose3d. The "field" object is a descriptor of the size of the
31 * field in meters with "width" and "length" values. This is to account for
32 * arbitrary field sizes when transforming the poses.
33 *
34 * Pose3ds in the JSON are measured using the normal FRC coordinate system, NWU
35 * with the origin at the bottom-right corner of the blue alliance wall.
36 * SetOrigin(OriginPosition) can be used to change the poses returned from
37 * GetTagPose(int) to be from the perspective of a specific alliance.
38 *
39 * Tag poses represent the center of the tag, with a zero rotation representing
40 * a tag that is upright and facing away from the (blue) alliance wall (that is,
41 * towards the opposing alliance). */
42class WPILIB_DLLEXPORT AprilTagFieldLayout {
43 public:
44 /**
45 * Common origin positions for the AprilTag coordinate system.
46 */
47 enum class OriginPosition {
48 /// Blue alliance wall, right side.
49 kBlueAllianceWallRightSide,
50 /// Red alliance wall, right side.
51 kRedAllianceWallRightSide,
52 };
