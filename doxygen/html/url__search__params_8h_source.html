48/**
49 * @brief Class for parsing and manipulating URL query strings.
50 *
51 * The `url_search_params` class provides methods to parse, modify, and
52 * serialize URL query parameters (the part after '?' in a URL). It handles
53 * percent-encoding and decoding automatically.
54 *
55 * All string inputs must be valid UTF-8. The caller is responsible for
56 * ensuring UTF-8 validity.
57 *
58 * Construction and `reset` refuse query strings longer than
59 * `get_max_input_length()` (the object is left empty). Individual `append` /
60 * `set` calls are not length-capped.
61 *
62 * @see https://url.spec.whatwg.org/#interface-urlsearchparams
63 */
64struct url_search_params {
65 url_search_params() = default;
66
67 /**
68 * Constructs url_search_params by parsing a query string.
69 * @param input A query string (with or without leading '?'). Must be UTF-8.
70 * If longer than `get_max_input_length()`, the object stays empty.
71 */
72 explicit url_search_params(const std::string_view input) {
73 initialize(input);
74 }

126 /**

132 inline bool has(std::string_view key) noexcept;
