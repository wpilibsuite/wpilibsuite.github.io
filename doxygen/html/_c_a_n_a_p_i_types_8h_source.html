125HAL_ENUM(HAL_CANFlags) {
126 /** Placeholder for no flags */
127 HAL_CAN_NO_FLAGS = 0x0,
128 /**
129 * Mask for if frame will do FD bit rate switching.

134 * Mask for is frame will contain an FD length.

136 HAL_CAN_FD_DATALENGTH = 0x2,
137};

139struct HAL_CANMessage {

141 int32_t flags;

145 uint8_t data[64];
146};
