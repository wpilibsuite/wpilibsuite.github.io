17/**
18 * Solenoid class for running high voltage Digital Output on a pneumatics
19 * module.
20 *
21 * The Solenoid class is typically used for pneumatics solenoids, but could be
22 * used for any device within the current spec of the module.
23 */
24class Solenoid : public wpi::telemetry::TelemetryLoggable {
25 public:
26 /**
27 * Constructs a solenoid for a specified module and type.
28 *
29 * @param busId The bus ID.
30 * @param module The module ID to use.
31 * @param moduleType The module type to use.
32 * @param channel The channel the solenoid is on.
33 */
34 Solenoid(CANPort busId, int module, PneumaticsModuleType moduleType,
35 int channel);
36
37 /**
38 * Constructs a solenoid for a default module and specified type.
39 *
40 * @param busId The bus ID.
41 * @param moduleType The module type to use.
42 * @param channel The channel the solenoid is on.
43 */
44 Solenoid(CANPort busId, PneumaticsModuleType moduleType, int channel);
45
46 ~Solenoid() override;
47
48 Solenoid(Solenoid&&) = default;
49 Solenoid& operator=(Solenoid&&) = default;
50
51 /**
52 * Set the value of a solenoid.
53 *
54 * @param on Turn the solenoid output off or on.
55 */
56 void Set(bool on);
57
58 /**
59 * Read the current value of the solenoid.
60 *
61 * @return The current value of the solenoid.
62 */
63 bool Get() const;
64
65 /**
66 * Toggle the value of the solenoid.
67 *
68 * If the solenoid is set to on, it'll be turned off. If the solenoid is set
69 * to off, it'll be turned on.
70 */
71 void Toggle();
72
73 /**
74 * Get the channel this solenoid is connected to.
75 */
76 int GetChannel() const;
77
78 /**
79 * Check if solenoid is Disabled.
80 *
81 * If a solenoid is shorted, it is added to the DisabledList and
82 * disabled until power cycle, or until faults are cleared.
83 *
84 * @see ClearAllPCMStickyFaults()
85 *
86 * @return If solenoid is disabled due to short.
87 */
88 bool IsDisabled() const;
89
90 /**
91 * Set the pulse duration in the pneumatics module. This is used in
92 * conjunction with the startPulse method to allow the pneumatics module to
93 * control the timing of a pulse.
94 *
95 * On the PCM, the timing can be controlled in 0.01 second increments, with a
96 * maximum of 2.55 seconds. On the PH, the timing can be controlled in 0.001
97 * second increments, with a maximum of 65.534 seconds.
98 *
99 * @param duration The duration of the pulse.
100 *
101 * @see startPulse()
102 */
103 void SetPulseDuration(wpi::units::second_t duration);
104
105 /**
106 * %Trigger the pneumatics module to generate a pulse of the duration set in
107 * setPulseDuration.
108 *
109 * @see setPulseDuration()
110 */
111 void StartPulse();
112
113 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
114
115 std::string_view GetTelemetryType() const override;
116
117 private:
118 std::shared_ptr<PneumaticsBase> m_module;
119 int m_mask;
120 int m_channel;
121};
122
123} // namespace wpi
