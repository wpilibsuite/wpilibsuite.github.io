8/// \file
9/// This file provides a collection of function (or more generally, callable)
10/// type erasure utilities supplementing those provided by the standard library
11/// in `<function>`.
12///
13/// It provides `unique_function`, which works like `std::function` but supports
14/// move-only callable objects and const-qualification.
15///
16/// Future plans:
17/// - Add a `function` that provides ref-qualified support, which doesn't work
18/// with `std::function`.
19/// - Provide support for specifying multiple signatures to type erase callable
20/// objects with an overload set, such as those produced by generic lambdas.
21/// - Expand to include a copyable utility that directly replaces std::function
22/// but brings the above improvements.
23///
24/// Note that LLVM's utilities are greatly simplified by not supporting
25/// allocators.
26///
27/// If the standard library ever begins to provide comparable facilities we can
28/// consider switching to those.
29///
30//===----------------------------------------------------------------------===//

45/// unique_function is a type-erasing functor similar to std::function.
46///
47/// It can hold move-only function objects, like lambdas capturing unique_ptrs.
48/// Accordingly, it is movable but not copyable.
49///
50/// It supports const-qualification:
51/// - unique_function<int() const> has a const operator().
52/// It can only hold functions which themselves have a const operator().
53/// - unique_function<int()> has a non-const operator().
54/// It can hold functions with a non-const operator(), like mutable lambdas.
55template <typename FunctionT> class unique_function;

67template <typename CallableT, typename ThisT>
68using EnableUnlessSameType =
69 std::enable_if_t<!std::is_same<remove_cvref_t<CallableT>, ThisT>::value>;
70template <typename CallableT, typename Ret, typename... Params>
71using EnableIfCallable = std::enable_if_t<std::disjunction<
72 std::is_void<Ret>,
73 std::is_same<decltype(std::declval<CallableT>()(std::declval<Params>()...)),
74 Ret>,
75 std::is_same<const decltype(std::declval<CallableT>()(
76 std::declval<Params>()...)),
77 Ret>,
78 std::is_convertible<decltype(std::declval<CallableT>()(
79 std::declval<Params>()...)),
80 Ret>>::value>;

82template <typename ReturnT, typename... ParamTs> class UniqueFunctionBase {
83protected:
84 static constexpr size_t InlineStorageSize = sizeof(void *) * 4;
85 static constexpr size_t InlineStorageAlign = alignof(void *);
86
87 // Provide a type function to map parameters that won't observe extra copies
88 // or moves and which are small enough to likely pass in register to values
89 // and all other types to l-value reference types. We use this to compute the
90 // types used in our erased call utility to minimize copies and moves unless
91 // doing so would force things unnecessarily into memory.
92 //
93 // The heuristic used is related to common ABI register passing conventions.
94 // It doesn't have to be exact though, and in one way it is more strict
95 // because we want to still be able to observe either moves *or* copies.
96 template <typename T> struct AdjustedParamTBase {
97 static_assert(!std::is_reference<T>::value,
98 "references should be handled by template specialization");
99 static constexpr bool IsSizeLessThanThreshold =
100 sizeof(T) <= 2 * sizeof(void *);
101 using type =
102 std::conditional_t<std::is_trivially_copy_constructible<T>::value &&
103 std::is_trivially_move_constructible<T>::value &&
104 IsSizeLessThanThreshold,
105 T, T &>;
106 };
107
108 // This specialization ensures that 'AdjustedParam<V<T>&>' or
109 // 'AdjustedParam<V<T>&&>' does not trigger a compile-time error when 'T' is
110 // an incomplete type and V a templated type.
111 template <typename T> struct AdjustedParamTBase<T &> { using type = T &; };
112 template <typename T> struct AdjustedParamTBase<T &&> { using type = T &; };
113
114 template <typename T>
115 using AdjustedParamT = typename AdjustedParamTBase<T>::type;
116
117 // The type of the erased function pointer we use as a callback to dispatch to
118 // the stored callable when it is trivial to move and destroy.
119 using CallPtrT = ReturnT (*)(const UniqueFunctionBase *Self,
120 AdjustedParamT<ParamTs>... Params);
121 using DestroyMovePtrT = void (*)(UniqueFunctionBase *LHS,
122 UniqueFunctionBase *RHS);
123
124 // The main storage buffer. This will either have a pointer to out-of-line
125 // storage or an inline buffer storing the callable.
126 union StorageT {
127 // For out-of-line storage we keep a pointer to the underlying storage.
128 void *OutOfLine;
129 static_assert(
130 sizeof(OutOfLine) <= InlineStorageSize,
131 "Should always use all of the out-of-line storage for inline storage!");
132
133 // For in-line storage, we just provide an aligned character buffer. We
134 // provide four pointers worth of storage here.
135 // This is mutable as an inlined `const unique_function<void() const>` may
136 // still modify its own mutable members.
137 alignas(InlineStorageAlign) mutable std::byte Inline[InlineStorageSize];
138 } Storage;
139
140 CallPtrT CallPtr = nullptr;
141 DestroyMovePtrT DestroyMovePtr = nullptr;
142
143 // A simple tag type so the call-as type to be passed to the constructor.
144 template <typename T> struct CalledAs {};
145
146 // Essentially the "main" unique_function constructor, but subclasses
147 // provide the qualified type to be used for the call.
148 // (We always store a T, even if the call will use a pointer to const T).
149 template <typename CallableT, typename CalledAsT>
150 UniqueFunctionBase(CallableT Callable, CalledAs<CalledAsT>) {
151 // static as workaround an MSVC bug which treats constexpr uses as odr-uses.
152 static constexpr bool UsesInlineStorage =
153 sizeof(CallableT) <= InlineStorageSize &&
154 alignof(CallableT) <= InlineStorageAlign;
155 void *CallableAddr = &Storage.Inline;
156 if constexpr (!UsesInlineStorage) {
157 CallableAddr = allocate_buffer(sizeof(CallableT), alignof(CallableT));
158 Storage.OutOfLine = CallableAddr;
159 }
160
161 // Now move into the storage.
162 new (CallableAddr) CallableT(std::move(Callable));
163
164 CallPtr = [](const UniqueFunctionBase *Self,
165 AdjustedParamT<ParamTs>... Params) -> ReturnT {
166 void *CallableAddr =
167 UsesInlineStorage ? &Self->Storage.Inline : Self->Storage.OutOfLine;
168 auto &Func = *reinterpret_cast<CalledAsT *>(CallableAddr);
169 return Func(std::forward<ParamTs>(Params)...);
170 };
171
172 // For trivial inline storage, we don't need to do anything on move/destroy.
173 if constexpr (!std::is_trivially_move_constructible_v<CallableT> ||
174 !std::is_trivially_destructible_v<CallableT> ||
175 !UsesInlineStorage) {
176 // If LHS is set, move LHS callable to RHS, then destroy RHS callable.
177 DestroyMovePtr = [](UniqueFunctionBase *LHS, UniqueFunctionBase *RHS) {
178 if constexpr (!UsesInlineStorage) {
179 if (LHS) {
180 // Out-of-line move: just move the pointer.
181 LHS->Storage.OutOfLine = RHS->Storage.OutOfLine;
182 } else {
183 // Out-of-line destroy.
184 void *RHSCallableAddr = RHS->Storage.OutOfLine;
185 reinterpret_cast<CallableT *>(RHSCallableAddr)->~CallableT();
186 deallocate_buffer(RHSCallableAddr, sizeof(CallableT),
187 alignof(CallableT));
188 }
189 } else {
190 auto *RHSCallable =
191 reinterpret_cast<CallableT *>(&RHS->Storage.Inline);
192 if (LHS) {
193 // Inline move: move-construct first...
194 auto *LHSCallable =
195 reinterpret_cast<CallableT *>(&LHS->Storage.Inline);
196 new (LHSCallable) CallableT(std::move(*RHSCallable));
197 }
198 // ... destroy RHS in any case.
199 RHSCallable->~CallableT();
200 }
201 };
202 }
203 }
204
205 ~UniqueFunctionBase() {
206 if (DestroyMovePtr)
207 DestroyMovePtr(nullptr, this);
208 }
209
210 UniqueFunctionBase(UniqueFunctionBase &&RHS) noexcept {
211 CallPtr = RHS.CallPtr;
212 DestroyMovePtr = RHS.DestroyMovePtr;
213 if (DestroyMovePtr)
214 DestroyMovePtr(this, &RHS);
215 else // Trivial callable stored inline => memcpy.
216 memcpy(&Storage.Inline, &RHS.Storage.Inline, InlineStorageSize);
217
218 RHS.CallPtr = nullptr;
219 RHS.DestroyMovePtr = nullptr; // Moved everything out of RHS.
220#ifndef NDEBUG
221 // In debug builds, we also scribble across the rest of the storage.
222 memset(RHS.Storage.Inline, 0xAD, InlineStorageSize);
223#endif
224 }
225
226 UniqueFunctionBase &operator=(UniqueFunctionBase &&RHS) noexcept {
227 if (this == &RHS)
228 return *this;
229
230 // Because we don't try to provide any exception safety guarantees we can
231 // implement move assignment very simply by first destroying the current
232 // object and then move-constructing over top of it.
233 this->~UniqueFunctionBase();
234 new (this) UniqueFunctionBase(std::move(RHS));
235 return *this;
236 }
237
238 UniqueFunctionBase() = default;
239
240public:
241 explicit operator bool() const { return (bool)CallPtr; }
242};

246template <typename R, typename... P>
247class unique_function<R(P...)> : public detail::UniqueFunctionBase<R, P...> {
248 using Base = detail::UniqueFunctionBase<R, P...>;
249
250public:
251 unique_function() = default;
252 unique_function(std::nullptr_t) {}
253 unique_function(unique_function &&) = default;
254 unique_function(const unique_function &) = delete;
255 unique_function &operator=(unique_function &&) = default;
256 unique_function &operator=(const unique_function &) = delete;
257
258 template <typename CallableT>
259 unique_function(
260 CallableT Callable,
261 detail::EnableUnlessSameType<CallableT, unique_function> * = nullptr,
262 detail::EnableIfCallable<CallableT, R, P...> * = nullptr)
263 : Base(std::forward<CallableT>(Callable),
264 typename Base::template CalledAs<CallableT>{}) {}
265
266 R operator()(P... Params) { return this->CallPtr(this, Params...); }
267};
