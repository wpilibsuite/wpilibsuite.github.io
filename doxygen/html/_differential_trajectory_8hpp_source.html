21/**
22 * A trajectory for differential drive robots with drivetrain-specific
23 * interpolation.
24 *
25 * This trajectory uses numerical integration of the differential drive
26 * differential equation for accurate interpolation.
27 */
28class WPILIB_DLLEXPORT DifferentialTrajectory
29 : public Trajectory<DifferentialSample> {
30 public:
31 /**
32 * Constructs a DifferentialTrajectory from a vector of samples.
33 *
34 * @param samples The samples of the trajectory. Order does not matter as
35 * they will be sorted internally.
36 * @throws std::invalid_argument if the vector of samples is empty.
37 */
38 explicit DifferentialTrajectory(std::vector<DifferentialSample> samples)
39 : Trajectory<DifferentialSample>(std::move(samples)) {}
40
41 /**
42 * Interpolates between two samples using numerical integration of the
43 * differential drive differential equation.
44 *
45 * @param start The starting sample.
46 * @param end The ending sample.
47 * @param t The interpolation parameter between 0 and 1.
48 * @return The interpolated sample.
49 */
50 DifferentialSample Interpolate(const DifferentialSample& start,
51 const DifferentialSample& end,
52 double t) const final;
53
54 /**
55 * Transforms all poses in the trajectory by the given transform.
56 *
57 * @param transform The transform to apply to the poses.
58 * @return The transformed trajectory.
59 */
60 DifferentialTrajectory TransformBy(const Transform2d& transform) const;
61
62 /**
63 * Transforms all poses so they are relative to the given pose.
64 *
65 * @param pose The pose to make the trajectory relative to.
66 * @return The transformed trajectory.
67 */
68 DifferentialTrajectory RelativeTo(const Pose2d& pose) const;
69
70 /**
71 * Concatenates another trajectory to this trajectory.
72 *
73 * @param other The trajectory to concatenate.
74 * @return The concatenated trajectory.
75 */
76 DifferentialTrajectory Concatenate(const DifferentialTrajectory& other) const;
77
78 /**
79 * Concatenates another trajectory to this trajectory.
80 *
81 * @param other The trajectory to concatenate.
82 * @return The concatenated trajectory.
83 */
84 DifferentialTrajectory operator+(const DifferentialTrajectory& other) const {
85 return Concatenate(other);
86 }
87};
