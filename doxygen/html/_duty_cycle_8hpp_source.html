14namespace wpi {
15/**
16 * Class to read a duty cycle PWM input.
17 *
18 * <p>PWM input signals are specified with a frequency and a ratio of high to
19 * low in that frequency. These can be attached to any SmartIO.
20 */
21class DutyCycle : public wpi::util::Sendable,
22 public wpi::util::SendableHelper<DutyCycle> {
23 public:
24 /**
25 * Constructs a DutyCycle input from a smartio channel.
26 *
27 * @param source The channel to use.
28 */
29 explicit DutyCycle(int source);
30
31 DutyCycle(DutyCycle&&) = default;
32 DutyCycle& operator=(DutyCycle&&) = default;
33
34 /**
35 * Close the DutyCycle and free all resources.
36 */
37 ~DutyCycle() override = default;
38
39 /**
40 * Get the frequency of the duty cycle signal.
41 *
42 * @return frequency
43 */
44 wpi::units::hertz_t GetFrequency() const;
45
46 /**
47 * Get the output ratio of the duty cycle signal.
48 *
49 * <p> 0 means always low, 1 means always high.
50 *
51 * @return output ratio between 0 and 1
52 */
53 double GetOutput() const;
54
55 /**
56 * Get the raw high time of the duty cycle signal.
57 *
58 * @return high time of last pulse
59 */
60 wpi::units::second_t GetHighTime() const;
61
62 /**
63 * Get the channel of the source.
64 *
65 * @return the source channel
66 */
67 int GetSourceChannel() const;
68
69 protected:
70 void InitSendable(wpi::util::SendableBuilder& builder) override;
71
72 private:
73 void InitDutyCycle();
74 int m_channel;
75 wpi::hal::Handle<HAL_DutyCycleHandle, HAL_FreeDutyCycle> m_handle;
76};
77} // namespace wpi
