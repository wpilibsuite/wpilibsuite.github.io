17/**
18 * Interface for telemetry backends.
19 *
20 * Implementations registered with TelemetryRegistry must be thread-safe.
21 * Backend and entry methods may be called concurrently, and entry removal may
22 * overlap a call already in progress on a previously returned entry.
23 */
24class TelemetryBackend {
25 public:
26 TelemetryBackend() = default;
27 virtual ~TelemetryBackend() = default;
28 TelemetryBackend(const TelemetryBackend&) = delete;
29 TelemetryBackend& operator=(const TelemetryBackend&) = delete;
30
31 /**
32 * Create an entry for the given path.
33 *
34 * Backend implementations must not throw from this method. Recoverable
35 * failures should be reported through TelemetryRegistry::ReportWarning() and
36 * represented with a discard entry.
37 *
38 * @param path full name
39 * @return telemetry entry
40 */
41 virtual std::shared_ptr<TelemetryEntry> GetEntry(std::string_view path) = 0;
42
43 /**
44 * Removes an entry for the given path.
45 *
46 * Backend implementations must not throw from this method.
47 *
48 * @param path normalized full name
49 */
50 virtual void RemoveEntry(std::string_view path) { (void)path; }
51
52 /**
53 * Returns whether there is a data schema already registered with the given
54 * name. This does NOT perform a check as to whether the schema has already
55 * been published by another node on the network.
56 *
57 * Backend implementations must not throw from this method.
58 *
59 * @param schemaName Name (the string passed as the data type for topics using
60 * this schema)
61 * @return True if schema already registered
62 */
63 virtual bool HasSchema(std::string_view schemaName) const = 0;
64
65 /**
66 * Registers a data schema. Data schemas provide information for how a
67 * certain data type string can be decoded. The type string of a data schema
68 * indicates the type of the schema itself (e.g. "protobuf" for protobuf
69 * schemas, "struct" for struct schemas, etc). In NetworkTables, schemas are
70 * published just like normal topics, with the name being generated from the
71 * provided name: "/.schema/<name>". Duplicate calls to this function with
72 * the same name are silently ignored.
73 *
74 * @param schemaName Name (the string passed as the data type for topics using
75 * this schema)
76 * @param type Type of schema (e.g. "protobuf", "struct", etc)
77 * @param schema Schema data
78 *
79 * Backend implementations must not throw from this method. Recoverable schema
80 * publication failures should be reported through
81 * TelemetryRegistry::ReportWarning() and skipped.
82 */
83 virtual void AddSchema(std::string_view schemaName, std::string_view type,
84 std::span<const uint8_t> schema) = 0;
85
86 /**
87 * Registers a data schema. Data schemas provide information for how a
88 * certain data type string can be decoded. The type string of a data schema
89 * indicates the type of the schema itself (e.g. "protobuf" for protobuf
90 * schemas, "struct" for struct schemas, etc). In NetworkTables, schemas are
91 * published just like normal topics, with the name being generated from the
92 * provided name: "/.schema/<name>". Duplicate calls to this function with
93 * the same name are silently ignored.
94 *
95 * @param schemaName Name (the string passed as the data type for topics using
96 * this schema)
97 * @param type Type of schema (e.g. "protobuf", "struct", etc)
98 * @param schema Schema data
99 *
100 * Backend implementations must not throw from this method. Recoverable schema
101 * publication failures should be reported through
102 * TelemetryRegistry::ReportWarning() and skipped.
103 */
104 virtual void AddSchema(std::string_view schemaName, std::string_view type,
105 std::string_view schema) = 0;
106};
