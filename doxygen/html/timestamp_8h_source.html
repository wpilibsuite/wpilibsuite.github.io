13/**
14 * The default implementation used for Now().
15 * In general this is the time returned by the operating system.
16 * @return Time in nanoseconds.
17 */
18int64_t WPI_NowDefault(void);

20/**
21 * Set the implementation used by WPI_Now().
22 * The implementation must return monotonic time in nanoseconds to maintain
23 * the contract of WPI_Now().
24 * The program start time is reset to the current time when this is called by
25 * calling the new implementation.
26 * Passing the WPI_NowDefault function explicitly sets that implementation and
27 * resets the program start time to the current time. To restore the default
28 * implementation and original program start time, pass NULL.
29 * @param func Function called by WPI_Now() to return the time.
30 */
31void WPI_SetNowImpl(int64_t (*func)(void));
32
33/**
34 * Return a value representing the current time in nanoseconds.
35 * The epoch is not defined.
36 * @return Time in nanoseconds.
37 */
38int64_t WPI_Now(void);
39
40/**
41 * Return the timestamp at which the program started.
42 * This uses the same time base as WPI_Now(), in nanoseconds.
43 * This is reset to the current time when WPI_SetNowImpl() is called with a
44 * non-NULL implementation; if WPI_SetNowImpl(NULL) is called, it is reset to
45 * the original program start time.
46 *
47 * @return Program start time in nanoseconds.
48 */
49int64_t WPI_GetProgramStartTime(void);
50
51/**
52 * Return the current system time in nanoseconds since the Unix epoch
53 * (January 1st, 1970 00:00 UTC).
54 *
55 * @return Time in nanoseconds.
56 */
57int64_t WPI_GetSystemTime(void);
