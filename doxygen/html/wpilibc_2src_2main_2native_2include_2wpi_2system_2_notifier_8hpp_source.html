23/**
24 * Notifiers run a user-provided callback function on a separate thread.
25 *
26 * If StartSingle() is used, the callback will run once. If StartPeriodic() is
27 * used, the callback will run repeatedly with the given period until stop() is
28 * called.
29 */
30class Notifier {
31 public:
32 /**
33 * Create a Notifier with the given callback.
34 *
35 * Configure when the callback runs with StartSingle() or StartPeriodic().
36 *
37 * @param callback The callback to run.
38 */
39 explicit Notifier(std::function<void()> callback)
40 : Notifier(0, std::move(callback)) {}
41
42 template <typename Arg, typename... Args>
43 Notifier(std::invocable<Arg, Args...> auto&& callback, Arg&& arg,
44 Args&&... args)
45 : Notifier(std::bind(std::forward<decltype(callback)>(callback),
46 std::forward<Arg>(arg),
47 std::forward<Args>(args)...)) {}
48
49 /**
50 * Create a Notifier with the given callback.
51 *
52 * Configure when the callback runs with StartSingle() or StartPeriodic().
53 *
54 * This overload makes the underlying thread run with a real-time priority.
55 * This is useful for reducing scheduling jitter on processes which are
56 * sensitive to timing variance, like model-based control.
57 *
58 * @param priority The FIFO real-time scheduler priority ([1..99] where a
59 * higher number represents higher priority). See "man 7
60 * sched" for more details.
61 * @param callback The callback to run.
62 */
63 explicit Notifier(int priority, std::function<void()> callback);
64
65 template <typename Arg, typename... Args>
66 Notifier(int priority, std::invocable<Arg, Args...> auto&& callback,
67 Arg&& arg, Args&&... args)
68 : Notifier(priority, std::bind(std::forward<decltype(callback)>(callback),
69 std::forward<Arg>(arg),
70 std::forward<Args>(args)...)) {}
71
72 /**
73 * Free the resources for a timer event.
74 */
75 ~Notifier();
76
77 Notifier(Notifier&& rhs);
78 Notifier& operator=(Notifier&& rhs);
79
80 /**
81 * Sets the name of the notifier. Used for debugging purposes only.
82 *
83 * @param name Name
84 */
85 void SetName(std::string_view name);
86
87 /**
88 * Change the callback function.
89 *
90 * @param callback The callback function.
91 */
92 void SetCallback(std::function<void()> callback);
93
94 /**
95 * Run the callback once after the given delay.
96 *
97 * @param delay Time to wait before the callback is called.
98 */
99 void StartSingle(wpi::units::second_t delay);
100
101 /**
102 * Run the callback periodically with the given period.
103 *
104 * The user-provided callback should be written so that it completes before
105 * the next time it's scheduled to run.
106 *
107 * @param period Period after which to call the callback starting one
108 * period after the call to this method.
109 */
110 void StartPeriodic(wpi::units::second_t period);
111
112 /**
113 * Run the callback periodically with the given frequency.
114 *
115 * The user-provided callback should be written so that it completes before
116 * the next time it's scheduled to run.
117 *
118 * @param frequency Frequency after which to call the callback starting one
119 * period after the call to this method.
120 */
121 void StartPeriodic(wpi::units::hertz_t frequency);
122
123 /**
124 * Stop further callback invocations.
125 *
126 * No further periodic callbacks will occur. Single invocations will also be
127 * cancelled if they haven't yet occurred.
128 *
129 * If a callback invocation is in progress, this function will block until the
130 * callback is complete.
131 */
132 void Stop();
133
134 /**
135 * Gets the overrun count.
136 *
137 * An overrun occurs when a notifier's alarm is not serviced before the next
138 * scheduled alarm time.
139 *
140 * @return overrun count
141 */
142 int32_t GetOverrun() const;
143
144 private:
145 // The thread waiting on the HAL alarm
146 std::thread m_thread;
147
148 // The mutex held while updating process information
149 wpi::util::mutex m_processMutex;
150
151 // HAL handle (atomic for proper destruction)
152 std::atomic<HAL_NotifierHandle> m_notifier{0};
153
154 // The user-provided callback
155 std::function<void()> m_callback;
156};
157
158} // namespace wpi
