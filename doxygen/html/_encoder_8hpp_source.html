14/**
15 * Class to read quad encoders.
16 *
17 * Quadrature encoders are devices that count shaft rotation and can sense
18 * direction. The output of the QuadEncoder class is an integer that can count
19 * either up or down, and can go negative for reverse direction counting. When
20 * creating QuadEncoders, a direction is supplied that changes the sense of the
21 * output to make code more readable if the encoder is mounted such that forward
22 * movement generates negative values. Quadrature encoders have two digital
23 * outputs, an A Channel and a B Channel that are out of phase with each other
24 * to allow the FPGA to do direction sensing.
25 *
26 * All encoders will immediately start counting - Reset() them if you need them
27 * to be zeroed before use.
28 */
29class Encoder : public CounterBase,
30 public wpi::util::Sendable,
31 public wpi::util::SendableHelper<Encoder> {
32 public:
33 /**
34 * Encoder constructor.
35 *
36 * Construct a Encoder given a and b channels.
37 *
38 * The counter will start counting immediately.
39 *
40 * @param aChannel The 'a' SmartIO channel.
41 * @param bChannel The 'b' SmartIO channel.
42 * @param reverseDirection represents the orientation of the encoder and
43 * inverts the output values if necessary so forward
44 * represents positive values.
45 * @param encodingType either X1, X2, or X4 to indicate 1X, 2X or 4X
46 * decoding. If 4X is selected, then an encoder FPGA
47 * object is used and the returned counts will be 4x
48 * the encoder spec'd value since all rising and
49 * falling edges are counted. If 1X or 2X are selected
50 * then a counter object will be used and the returned
51 * value will either exactly match the spec'd count or
52 * be double (2x) the spec'd count.
53 */
54 Encoder(int aChannel, int bChannel, bool reverseDirection = false,
55 EncodingType encodingType = EncodingType::X4);
56
57 Encoder(Encoder&&) = default;
58 Encoder& operator=(Encoder&&) = default;
59
60 ~Encoder() override = default;
61
62 // CounterBase interface
63 /**
64 * Gets the current count.
65 *
66 * Returns the current count on the Encoder. This method compensates for the
67 * decoding type.
68 *
69 * @return Current count from the Encoder adjusted for the 1x, 2x, or 4x scale
70 * factor.
71 */
72 int Get() const override;
73
74 /**
75 * Reset the Encoder distance to zero.
76 *
77 * Resets the current count to zero on the encoder.
78 */
79 void Reset() override;
80
81 /**
82 * Returns the period of the most recent pulse.
83 *
84 * Returns the period of the most recent Encoder pulse in seconds. This method
85 * compensates for the decoding type.
86 *
87 * Warning: This returns unscaled periods. Use GetRate() for rates that are
88 * scaled using the value from SetDistancePerPulse().
89 *
90 * @return Period in seconds of the most recent pulse.
91 * @deprecated Use getRate() in favor of this method.
92 */
93 [[deprecated("Use GetRate() in favor of this method")]]
94 wpi::units::second_t GetPeriod() const override;
95
96 /**
97 * Sets the maximum period for stopped detection.
98 *
99 * Sets the value that represents the maximum period of the Encoder before it
100 * will assume that the attached device is stopped. This timeout allows users
101 * to determine if the wheels or other shaft has stopped rotating.
102 * This method compensates for the decoding type.
103 *
104 * @param maxPeriod The maximum time between rising and falling edges before
105 * the FPGA will report the device stopped. This is expressed
106 * in seconds.
107 * @deprecated Use SetMinRate() in favor of this method. This takes unscaled
108 * periods and SetMinRate() scales using value from
109 * SetDistancePerPulse().
110 */
111 [[deprecated(
112 "Use SetMinRate() in favor of this method. This takes unscaled periods "
113 "and SetMinRate() scales using value from SetDistancePerPulse().")]]
114 void SetMaxPeriod(wpi::units::second_t maxPeriod) override;
115
116 /**
117 * Determine if the encoder is stopped.
118 *
119 * Using the MaxPeriod value, a boolean is returned that is true if the
120 * encoder is considered stopped and false if it is still moving. A stopped
121 * encoder is one where the most recent pulse width exceeds the MaxPeriod.
122 *
123 * @return True if the encoder is considered stopped.
124 */
125 bool GetStopped() const override;
126
127 /**
128 * The last direction the encoder value changed.
129 *
130 * @return The last direction the encoder value changed.
131 */
132 bool GetDirection() const override;
133
134 /**
135 * Gets the raw value from the encoder.
136 *
137 * The raw value is the actual count unscaled by the 1x, 2x, or 4x scale
138 * factor.
139 *
140 * @return Current raw count from the encoder
141 */
142 int GetRaw() const;
143
144 /**
145 * The encoding scale factor 1x, 2x, or 4x, per the requested encodingType.
146 *
147 * Used to divide raw edge counts down to spec'd counts.
148 */
149 int GetEncodingScale() const;
150
151 /**
152 * Get the distance the robot has driven since the last reset.
153 *
154 * @return The distance driven since the last reset as scaled by the value
155 * from SetDistancePerPulse().
156 */
157 double GetDistance() const;
158
159 /**
160 * Get the current rate of the encoder.
161 *
162 * Units are distance per second as scaled by the value from
163 * SetDistancePerPulse().
164 *
165 * @return The current rate of the encoder.
166 */
167 double GetRate() const;
168
169 /**
170 * Set the minimum rate of the device before the hardware reports it stopped.
171 *
172 * @param minRate The minimum rate. The units are in distance per second as
173 * scaled by the value from SetDistancePerPulse().
174 */
175 void SetMinRate(double minRate);
176
177 /**
178 * Set the distance per pulse for this encoder.
179 *
180 * This sets the multiplier used to determine the distance driven based on the
181 * count value from the encoder.
182 *
183 * Do not include the decoding type in this scale. The library already
184 * compensates for the decoding type.
185 *
186 * Set this value based on the encoder's rated Pulses per Revolution and
187 * factor in gearing reductions following the encoder shaft.
188 *
189 * This distance can be in any units you like, linear or angular.
190 *
191 * @param distancePerPulse The scale factor that will be used to convert
192 * pulses to useful units.
193 */
194 void SetDistancePerPulse(double distancePerPulse);
195
196 /**
197 * Get the distance per pulse for this encoder.
198 *
199 * @return The scale factor that will be used to convert pulses to useful
200 * units.
201 */
202 double GetDistancePerPulse() const;
203
204 /**
205 * Set the direction sensing for this encoder.
206 *
207 * This sets the direction sensing on the encoder so that it could count in
208 * the correct software direction regardless of the mounting.
209 *
210 * @param reverseDirection true if the encoder direction should be reversed
211 */
212 void SetReverseDirection(bool reverseDirection);
213
214 /**
215 * Set the Samples to Average which specifies the number of samples of the
216 * timer to average when calculating the period.
217 *
218 * Perform averaging to account for mechanical imperfections or as
219 * oversampling to increase resolution.
220 *
221 * @param samplesToAverage The number of samples to average from 1 to 127.
222 */
223 void SetSamplesToAverage(int samplesToAverage);
224
225 /**
226 * Get the Samples to Average which specifies the number of samples of the
227 * timer to average when calculating the period.
228 *
229 * Perform averaging to account for mechanical imperfections or as
230 * oversampling to increase resolution.
231 *
232 * @return The number of samples being averaged (from 1 to 127)
233 */
234 int GetSamplesToAverage() const;
235
236 /**
237 * Indicates this encoder is used by a simulated device.
238 *
239 * @param device simulated device handle
240 */
241 void SetSimDevice(HAL_SimDeviceHandle device);
242
243 int GetFPGAIndex() const;
244
245 void InitSendable(wpi::util::SendableBuilder& builder) override;
246
247 private:
248 /**
249 * Common initialization code for Encoders.
250 *
251 * This code allocates resources for Encoders and is common to all
252 * constructors. The counter will start counting immediately.
253 * @param aChannel The a channel.
254 * @param bChannel The b channel.
255 * @param reverseDirection If true, counts down instead of up (this is all
256 * relative)
257 * @param encodingType either X1, X2, or X4 to indicate 1X, 2X or 4X
258 * decoding. If 4X is selected, then an encoder FPGA
259 * object is used and the returned counts will be 4x
260 * the encoder spec'd value since all rising and
261 * falling edges are counted. If 1X or 2X are selected
262 * then a counter object will be used and the returned
263 * value will either exactly match the spec'd count or
264 * be double (2x) the spec'd count.
265 */
266 void InitEncoder(int aChannel, int bChannel, bool reverseDirection,
267 EncodingType encodingType);
268
269 /**
270 * The scale needed to convert a raw counter value into a number of encoder
271 * pulses.
272 */
273 double DecodingScaleFactor() const;
274
275 wpi::hal::Handle<HAL_EncoderHandle, HAL_FreeEncoder> m_encoder;
276};
