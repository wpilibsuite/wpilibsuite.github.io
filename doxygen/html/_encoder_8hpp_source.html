14/**
15 * Class to read quad encoders.
16 *
17 * Quadrature encoders are devices that count shaft rotation and can sense
18 * direction. The output of the QuadEncoder class is an integer that can count
19 * either up or down, and can go negative for reverse direction counting. When
20 * creating QuadEncoders, a direction is supplied that changes the sense of the
21 * output to make code more readable if the encoder is mounted such that forward
22 * movement generates negative values. Quadrature encoders have two digital
23 * outputs, an A Channel and a B Channel that are out of phase with each other
24 * to allow the FPGA to do direction sensing.
25 *
26 * All encoders will immediately start counting - Reset() them if you need them
27 * to be zeroed before use.
28 */
29class Encoder : public CounterBase, public wpi::telemetry::TelemetryLoggable {
30 public:
31 /**
32 * Encoder constructor.
33 *
34 * Construct a Encoder given a and b channels.
35 *
36 * The counter will start counting immediately.
37 *
38 * @param aChannel The 'a' SmartIO channel.
39 * @param bChannel The 'b' SmartIO channel.
40 * @param reverseDirection represents the orientation of the encoder and
41 * inverts the output values if necessary so forward
42 * represents positive values.
43 * @param encodingType either X1, X2, or X4 to indicate 1X, 2X or 4X
44 * decoding. If 4X is selected, then an encoder FPGA
45 * object is used and the returned counts will be 4x
46 * the encoder spec'd value since all rising and
47 * falling edges are counted. If 1X or 2X are selected
48 * then a counter object will be used and the returned
49 * value will either exactly match the spec'd count or
50 * be double (2x) the spec'd count.
51 */
52 Encoder(int aChannel, int bChannel, bool reverseDirection = false,
53 EncodingType encodingType = EncodingType::X4);
54
55 Encoder(Encoder&&) = default;
56 Encoder& operator=(Encoder&&) = default;
57
58 ~Encoder() override = default;
59
60 // CounterBase interface
61 /**
62 * Gets the current count.
63 *
64 * Returns the current count on the Encoder. This method compensates for the
65 * decoding type.
66 *
67 * @return Current count from the Encoder adjusted for the 1x, 2x, or 4x scale
68 * factor.
69 */
70 int Get() const override;
71
72 /**
73 * Reset the Encoder distance to zero.
74 *
75 * Resets the current count to zero on the encoder.
76 */
77 void Reset() override;
78
79 /**
80 * Determine if the encoder is stopped.

85 */
86 bool GetStopped() const override;
87
88 /**
89 * The last direction the encoder value changed.
90 *
91 * @return The last direction the encoder value changed.
92 */
93 bool GetDirection() const override;
94
95 /**
96 * Gets the raw value from the encoder.
97 *
98 * The raw value is the actual count unscaled by the 1x, 2x, or 4x scale
99 * factor.
100 *
101 * @return Current raw count from the encoder
102 */
103 int GetRaw() const;
104
105 /**
106 * The scale factor used to convert raw encoder counts to scaled count values.
107 *
108 * Used to divide raw encoder counts down to the values returned by Get().
109 */
110 int GetEncodingScale() const;
111
112 /**
113 * Get the distance the robot has driven since the last reset.
114 *
115 * @return The distance driven since the last reset as scaled by the value
116 * from SetDistancePerPulse().
117 */
118 double GetDistance() const;
119
120 /**
121 * Get the current rate of the encoder.
122 *
123 * Units are distance per second as scaled by the value from
124 * SetDistancePerPulse().
125 *
126 * @return The current rate of the encoder.
127 */
128 double GetRate() const;
129
130 /**
131 * Sets the time window used to calculate the encoder rate.
132 *
133 * @param window The rate calculation window. Valid values are 5 ms through
134 * 255 ms. The default is 50 ms.
135 */
136 void SetRateWindow(wpi::units::millisecond_t window);
137
138 /**
139 * Set the distance per pulse for this encoder.
140 *
141 * This sets the multiplier used to determine the distance driven based on the
142 * count value from the encoder.
143 *
144 * Do not include the decoding type in this scale. The library already
145 * compensates for the decoding type.
146 *
147 * Set this value based on the encoder's rated Pulses per Revolution and
148 * factor in gearing reductions following the encoder shaft.
149 *
150 * This distance can be in any units you like, linear or angular.
151 *
152 * @param distancePerPulse The scale factor that will be used to convert
153 * pulses to useful units.
154 */
155 void SetDistancePerPulse(double distancePerPulse);
156
157 /**
158 * Get the distance per pulse for this encoder.
159 *
160 * @return The scale factor that will be used to convert pulses to useful
161 * units.
162 */
163 double GetDistancePerPulse() const;
164
165 /**
166 * Set the direction sensing for this encoder.
167 *
168 * This sets the direction sensing on the encoder so that it could count in
169 * the correct software direction regardless of the mounting.
170 *
171 * @param reverseDirection true if the encoder direction should be reversed
172 */
173 void SetReverseDirection(bool reverseDirection);
174
175 /**
176 * Indicates this encoder is used by a simulated device.
177 *
178 * @param device simulated device handle
179 */
180 void SetSimDevice(HAL_SimDeviceHandle device);
181
182 int GetFPGAIndex() const;
183
184 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
185
186 std::string_view GetTelemetryType() const override;
187
188 private:
189 /**
190 * Common initialization code for Encoders.
191 *
192 * This code allocates resources for Encoders and is common to all
193 * constructors. The counter will start counting immediately.
194 * @param aChannel The a channel.
195 * @param bChannel The b channel.
196 * @param reverseDirection If true, counts down instead of up (this is all
197 * relative)
198 * @param encodingType either X1, X2, or X4 to indicate 1X, 2X or 4X
199 * decoding. If 4X is selected, then an encoder FPGA
200 * object is used and the returned counts will be 4x
201 * the encoder spec'd value since all rising and
202 * falling edges are counted. If 1X or 2X are selected
203 * then a counter object will be used and the returned
204 * value will either exactly match the spec'd count or
205 * be double (2x) the spec'd count.
206 */
207 void InitEncoder(int aChannel, int bChannel, bool reverseDirection,
208 EncodingType encodingType);
209
210 /**
211 * The scale needed to convert a raw counter value into a number of encoder
212 * pulses.
213 */
214 double DecodingScaleFactor() const;
215
216 wpi::util::Handle<HAL_EncoderHandle, HAL_FreeEncoder> m_encoder;
217 EncodingType m_type;
218};
