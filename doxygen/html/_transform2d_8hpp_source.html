24/**
25 * Represents a transformation for a Pose2d in the pose's frame.
26 *
27 * Transforms are applied intrinsically, i.e. relative to the pose's own frame
28 * rather than the global frame. This is in contrast to the rotation classes,
29 * which apply rotations extrinsically.
30 */
31class WPILIB_DLLEXPORT Transform2d final {
32 public:
33 /**
34 * Constructs the transform that maps the initial pose to the final pose.
35 *
36 * @param initial The initial pose for the transformation.
37 * @param final The final pose for the transformation.
38 */
39 constexpr Transform2d(const Pose2d& initial, const Pose2d& final);
40
41 /**
42 * Constructs a transform with the given translation and rotation components.
43 *
44 * @param translation Translational component of the transform.
45 * @param rotation Rotational component of the transform.
46 */
47 constexpr Transform2d(Translation2d translation, Rotation2d rotation)
48 : m_translation{std::move(translation)},
49 m_rotation{std::move(rotation)} {}
50
51 /**
52 * Constructs a transform with x and y translations instead of a separate
53 * Translation2d.
54 *
55 * @param x The x component of the translational component of the transform.
56 * @param y The y component of the translational component of the transform.
57 * @param rotation The rotational component of the transform.
58 */
59 constexpr Transform2d(wpi::units::meter_t x, wpi::units::meter_t y,
60 Rotation2d rotation)
61 : m_translation{x, y}, m_rotation{std::move(rotation)} {}
62
63 /**
64 * Constructs a pose with the specified affine transformation matrix.
65 *
66 * @param matrix The affine transformation matrix.
67 * @throws std::domain_error if the affine transformation matrix is invalid.
68 */
69 constexpr explicit Transform2d(const Eigen::Matrix3d& matrix)
70 : m_translation{Eigen::Vector2d{{matrix(0, 2)}, {matrix(1, 2)}}},
71 m_rotation{Eigen::Matrix2d{{matrix(0, 0), matrix(0, 1)},
72 {matrix(1, 0), matrix(1, 1)}}} {
73 if (matrix(2, 0) != 0.0 || matrix(2, 1) != 0.0 || matrix(2, 2) != 1.0) {
74 throw std::domain_error("Affine transformation matrix is invalid");
75 }
76 }
77
78 /**
79 * Constructs the identity transform -- maps an initial pose to itself.
80 */
81 constexpr Transform2d() = default;
82
83 /**
84 * Returns the translation component of the transformation.
85 *
86 * @return Reference to the translational component of the transform.
87 */
88 constexpr const Translation2d& Translation() const { return m_translation; }
89
90 /**
91 * Returns the X component of the transformation's translation.
92 *
93 * @return The x component of the transformation's translation.
94 */
95 constexpr wpi::units::meter_t X() const { return m_translation.X(); }
96
97 /**
98 * Returns the Y component of the transformation's translation.
99 *
100 * @return The y component of the transformation's translation.
101 */
102 constexpr wpi::units::meter_t Y() const { return m_translation.Y(); }
103
104 /**
105 * Returns an affine transformation matrix representation of this
106 * transformation.
107 */
108 constexpr Eigen::Matrix3d ToMatrix() const {
109 auto vec = m_translation.ToVector();
110 auto mat = m_rotation.ToMatrix();
111 return Eigen::Matrix3d{{mat(0, 0), mat(0, 1), vec(0)},
112 {mat(1, 0), mat(1, 1), vec(1)},
113 {0.0, 0.0, 1.0}};
114 }
115
116 /**
117 * Returns the rotational component of the transformation.
118 *
119 * @return Reference to the rotational component of the transform.
120 */
121 constexpr const Rotation2d& Rotation() const { return m_rotation; }
122
123 /**
124 * Returns a Twist2d of the current transform (pose delta). If b is the output
125 * of {@code a.Log()}, then {@code b.Exp()} would yield a.
126 *
127 * @return The twist that maps the current transform.
128 */
129 constexpr Twist2d Log() const;
130
131 /**
132 * Invert the transformation. This is useful for undoing a transformation.
133 *
134 * @return The inverted transformation.
135 */
136 constexpr Transform2d Inverse() const {
137 // We are rotating the difference between the translations
138 // using a clockwise rotation matrix. This transforms the global
139 // delta into a local delta (relative to the initial pose).
140 return Transform2d{(-Translation()).RotateBy(-Rotation()), -Rotation()};
141 }
142
143 /**
144 * Multiplies the transform by the scalar.
145 *
146 * @param scalar The scalar.
147 * @return The scaled Transform2d.
148 */
149 constexpr Transform2d operator*(double scalar) const {
150 return Transform2d(m_translation * scalar, m_rotation * scalar);
151 }
152
153 /**
154 * Divides the transform by the scalar.
155 *
156 * @param scalar The scalar.
157 * @return The scaled Transform2d.
158 */
159 constexpr Transform2d operator/(double scalar) const {
160 return *this * (1.0 / scalar);
161 }
162
163 /**
164 * Composes two transformations. The second transform is applied relative to
165 * the orientation of the first.
166 *
167 * @param other The transform to compose with this one.
168 * @return The composition of the two transformations.
169 */
170 constexpr Transform2d operator+(const Transform2d& other) const;
171
172 /**
173 * Checks equality between this Transform2d and another object.
174 */
175 constexpr bool operator==(const Transform2d&) const = default;
176
177 private:
178 Translation2d m_translation;
179 Rotation2d m_rotation;
180};

189constexpr Transform2d::Transform2d(const Pose2d& initial, const Pose2d& final) {
190 // To transform the global translation delta to be relative to the initial
191 // pose, rotate by the inverse of the initial pose's orientation.
192 m_translation = (final.Translation() - initial.Translation())
193 .RotateBy(-initial.Rotation());
194
195 m_rotation = final.Rotation().RelativeTo(initial.Rotation());
196}
197
198constexpr Transform2d Transform2d::operator+(const Transform2d& other) const {
199 return Transform2d{Pose2d{}, Pose2d{}.TransformBy(*this).TransformBy(other)};
200}
201
202constexpr Twist2d Transform2d::Log() const {
203 const auto dtheta = m_rotation.Radians().value();
204 const auto halfDtheta = dtheta / 2.0;
205
206 const auto cosMinusOne = m_rotation.Cos() - 1;
207
208 double halfThetaByTanOfHalfDtheta;
209
210 if (gcem::abs(cosMinusOne) < 1E-9) {
211 halfThetaByTanOfHalfDtheta = 1.0 - 1.0 / 12.0 * dtheta * dtheta;
212 } else {
213 halfThetaByTanOfHalfDtheta = -(halfDtheta * m_rotation.Sin()) / cosMinusOne;
214 }
215
216 const Translation2d translationPart =
217 m_translation.RotateBy({halfThetaByTanOfHalfDtheta, -halfDtheta}) *
218 gcem::hypot(halfThetaByTanOfHalfDtheta, halfDtheta);
219
220 return {translationPart.X(), translationPart.Y(),
221 wpi::units::radian_t{dtheta}};
222}
