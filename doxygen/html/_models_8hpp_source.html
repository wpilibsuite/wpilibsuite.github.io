44 /**
45 * Creates a flywheel state-space model from physical constants.
46 *
47 * The states are [angular velocity], the inputs are [voltage], and the
48 * outputs are [angular velocity].
49 *
50 * @param motor The motor (or gearbox) attached to the flywheel.
51 * @param J The moment of inertia J of the flywheel.
52 * @param gearing Gear ratio from motor to flywheel (greater than 1 is a
53 * reduction).
54 * @throws std::domain_error if J <= 0 or gearing <= 0.
55 */
56 static constexpr LinearSystem<1, 1, 1> FlywheelFromPhysicalConstants(
57 DCMotor motor, wpi::units::kilogram_square_meter_t J, double gearing) {
58 if (J <= 0_kg_sq_m) {
59 throw std::domain_error("J must be greater than zero.");
60 }
61 if (gearing <= 0.0) {
62 throw std::domain_error("gearing must be greater than zero.");
63 }
64
65 Matrixd<1, 1> A{
66 {(-gcem::pow(gearing, 2) * motor.Kt / (motor.Kv * motor.R * J))
67 .value()}};
68 Matrixd<1, 1> B{{(gearing * motor.Kt / (motor.R * J)).value()}};
69 Matrixd<1, 1> C{{1.0}};
70 Matrixd<1, 1> D{{0.0}};
71
72 return LinearSystem<1, 1, 1>(A, B, C, D);
73 }

76 * Creates a flywheel state-space model from SysId constants kᵥ (V/(rad/s))
77 * and kₐ (V/(rad/s²)) from the feedforward model u = kᵥv + kₐa.
78 *
79 * The states are [velocity], the inputs are [voltage], and the outputs are
80 * [velocity].
81 *
82 * @param kV The velocity gain, in V/(rad/s).
83 * @param kA The acceleration gain, in V/(rad/s²).
84 * @throws std::domain_error if kV < 0 or kA <= 0.
85 * @see <a
86 * href="https://github.com/wpilibsuite/allwpilib/tree/main/sysid">https://github.com/wpilibsuite/allwpilib/tree/main/sysid</a>

88 static constexpr LinearSystem<1, 1, 1> FlywheelFromSysId(
89 decltype(1_V / 1_rad_per_s) kV, decltype(1_V / 1_rad_per_s_sq) kA) {
90 if (kV < decltype(kV){0}) {
91 throw std::domain_error("Kv must be greater than or equal to zero.");
92 }
93 if (kA <= decltype(kA){0}) {
94 throw std::domain_error("Ka must be greater than zero.");

96
97 Matrixd<1, 1> A{{-kV.value() / kA.value()}};
98 Matrixd<1, 1> B{{1.0 / kA.value()}};
99 Matrixd<1, 1> C{{1.0}};
100 Matrixd<1, 1> D{{0.0}};
101
102 return LinearSystem<1, 1, 1>(A, B, C, D);

105 /**
106 * Creates an elevator state-space model from physical constants.
107 *
108 * The states are [position, velocity], the inputs are [voltage], and the
109 * outputs are [position, velocity].
110 *
111 * @param motor The motor (or gearbox) attached to the carriage.
112 * @param mass The mass of the elevator carriage, in kilograms.
113 * @param radius The radius of the elevator's driving drum, in meters.
114 * @param gearing Gear ratio from motor to carriage (greater than 1 is a
115 * reduction).
116 * @throws std::domain_error if mass <= 0, radius <= 0, or gearing <= 0.
117 */
118 static constexpr LinearSystem<2, 1, 2> ElevatorFromPhysicalConstants(
119 DCMotor motor, wpi::units::kilogram_t mass, wpi::units::meter_t radius,
120 double gearing) {
121 if (mass <= 0_kg) {
122 throw std::domain_error("mass must be greater than zero.");
123 }
124 if (radius <= 0_m) {
125 throw std::domain_error("radius must be greater than zero.");
126 }
127 if (gearing <= 0.0) {
128 throw std::domain_error("gearing must be greater than zero.");
129 }
130
131 Matrixd<2, 2> A{
132 {0.0, 1.0},
133 {0.0, (-gcem::pow(gearing, 2) * motor.Kt /
134 (motor.R * wpi::units::math::pow<2>(radius) * mass * motor.Kv))
135 .value()}};
136 Matrixd<2, 1> B{{0.0},
137 {(gearing * motor.Kt / (motor.R * radius * mass)).value()}};
138 Matrixd<2, 2> C{{1.0, 0.0}, {0.0, 1.0}};
139 Matrixd<2, 1> D{{0.0}, {0.0}};
140
141 return LinearSystem<2, 1, 2>(A, B, C, D);
142 }

238 /**
239 * Creates a differential drive state-space model from physical constants.
240 *
241 * The states are [left velocity, right velocity], the inputs are [left
242 * voltage, right voltage], and the outputs are [left velocity, right
243 * velocity].
244 *
245 * @param motor The motor (or gearbox) driving the drivetrain.
246 * @param mass The mass of the robot in kilograms.
247 * @param r The radius of the wheels in meters.
248 * @param rb The radius of the base (half of the trackwidth), in meters.
249 * @param J The moment of inertia of the robot.
250 * @param gearing Gear ratio from motor to wheel (greater than 1 is a
251 * reduction).
252 * @throws std::domain_error if mass <= 0, r <= 0, rb <= 0, J <= 0, or
253 * gearing <= 0.
254 */
255 static constexpr LinearSystem<2, 2, 2> DifferentialDriveFromPhysicalConstants(
256 const DCMotor& motor, wpi::units::kilogram_t mass, wpi::units::meter_t r,
257 wpi::units::meter_t rb, wpi::units::kilogram_square_meter_t J,
258 double gearing) {
259 if (mass <= 0_kg) {
260 throw std::domain_error("mass must be greater than zero.");
261 }
262 if (r <= 0_m) {
263 throw std::domain_error("r must be greater than zero.");
264 }
265 if (rb <= 0_m) {
266 throw std::domain_error("rb must be greater than zero.");
267 }
268 if (J <= 0_kg_sq_m) {
269 throw std::domain_error("J must be greater than zero.");
270 }
271 if (gearing <= 0.0) {
272 throw std::domain_error("gearing must be greater than zero.");
273 }
274
275 auto C1 = -gcem::pow(gearing, 2) * motor.Kt /
276 (motor.Kv * motor.R * wpi::units::math::pow<2>(r));
277 auto C2 = gearing * motor.Kt / (motor.R * r);
278
279 Matrixd<2, 2> A{
280 {((1 / mass + wpi::units::math::pow<2>(rb) / J) * C1).value(),
281 ((1 / mass - wpi::units::math::pow<2>(rb) / J) * C1).value()},
282 {((1 / mass - wpi::units::math::pow<2>(rb) / J) * C1).value(),
283 ((1 / mass + wpi::units::math::pow<2>(rb) / J) * C1).value()}};
284 Matrixd<2, 2> B{
285 {((1 / mass + wpi::units::math::pow<2>(rb) / J) * C2).value(),
286 ((1 / mass - wpi::units::math::pow<2>(rb) / J) * C2).value()},
287 {((1 / mass - wpi::units::math::pow<2>(rb) / J) * C2).value(),
288 ((1 / mass + wpi::units::math::pow<2>(rb) / J) * C2).value()}};
289 Matrixd<2, 2> C{{1.0, 0.0}, {0.0, 1.0}};
290 Matrixd<2, 2> D{{0.0, 0.0}, {0.0, 0.0}};
291
292 return LinearSystem<2, 2, 2>(A, B, C, D);
293 }

295 /**
296 * Creates a differential drive state-space model from SysId constants kᵥ and
297 * kₐ in both linear {(V/(m/s), (V/(m/s²))} and angular {(V/(rad/s),
298 * (V/(rad/s²))} cases.
299 *
300 * The states are [left velocity, right velocity], the inputs are [left
301 * voltage, right voltage], and the outputs are [left velocity, right
302 * velocity].
303 *
304 * @param kvLinear The linear velocity gain in volts per (meters per
305 * second).
306 * @param kaLinear The linear acceleration gain in volts per (meters per
307 * second squared).
308 * @param kvAngular The angular velocity gain in volts per (meters per
309 * second).
310 * @param kaAngular The angular acceleration gain in volts per (meters per
311 * second squared).
312 * @throws domain_error if kvLinear <= 0, kaLinear <= 0, kvAngular <= 0,
313 * or kaAngular <= 0.
314 * @see <a
315 * href="https://github.com/wpilibsuite/allwpilib/tree/main/sysid">https://github.com/wpilibsuite/allwpilib/tree/main/sysid</a>
316 */
317 static constexpr LinearSystem<2, 2, 2> DifferentialDriveFromSysId(
318 decltype(1_V / 1_mps) kvLinear, decltype(1_V / 1_mps_sq) kaLinear,
319 decltype(1_V / 1_mps) kvAngular, decltype(1_V / 1_mps_sq) kaAngular) {
320 if (kvLinear <= decltype(kvLinear){0}) {
321 throw std::domain_error("Kv,linear must be greater than zero.");
322 }
323 if (kaLinear <= decltype(kaLinear){0}) {
324 throw std::domain_error("Ka,linear must be greater than zero.");
325 }
326 if (kvAngular <= decltype(kvAngular){0}) {
327 throw std::domain_error("Kv,angular must be greater than zero.");
328 }
329 if (kaAngular <= decltype(kaAngular){0}) {
330 throw std::domain_error("Ka,angular must be greater than zero.");
331 }
332
333 double A1 = -0.5 * (kvLinear.value() / kaLinear.value() +
334 kvAngular.value() / kaAngular.value());
335 double A2 = -0.5 * (kvLinear.value() / kaLinear.value() -
336 kvAngular.value() / kaAngular.value());
337 double B1 = 0.5 / kaLinear.value() + 0.5 / kaAngular.value();
338 double B2 = 0.5 / kaLinear.value() - 0.5 / kaAngular.value();
339
340 Matrixd<2, 2> A{{A1, A2}, {A2, A1}};
341 Matrixd<2, 2> B{{B1, B2}, {B2, B1}};
342 Matrixd<2, 2> C{{1.0, 0.0}, {0.0, 1.0}};
343 Matrixd<2, 2> D{{0.0, 0.0}, {0.0, 0.0}};
344
345 return LinearSystem<2, 2, 2>(A, B, C, D);
346 }

348 /**
349 * Creates a differential drive state-space model from SysId constants kᵥ and
350 * kₐ in both linear {(V/(m/s), (V/(m/s²))} and angular {(V/(rad/s),
351 * (V/(rad/s²))} cases.
352 *
353 * The states are [left velocity, right velocity], the inputs are [left
354 * voltage, right voltage], and the outputs are [left velocity, right
355 * velocity].
356 *
357 * @param kvLinear The linear velocity gain in volts per (meters per
358 * second).
359 * @param kaLinear The linear acceleration gain in volts per (meters per
360 * second squared).
361 * @param kvAngular The angular velocity gain in volts per (radians per
362 * second).
363 * @param kaAngular The angular acceleration gain in volts per (radians per
364 * second squared).
365 * @param trackwidth The distance between the differential drive's left and
366 * right wheels, in meters.
367 * @throws domain_error if kvLinear <= 0, kaLinear <= 0, kvAngular <= 0,
368 * kaAngular <= 0, or trackwidth <= 0.
369 * @see <a
370 * href="https://github.com/wpilibsuite/allwpilib/tree/main/sysid">https://github.com/wpilibsuite/allwpilib/tree/main/sysid</a>
371 */
372 static constexpr LinearSystem<2, 2, 2> DifferentialDriveFromSysId(
373 decltype(1_V / 1_mps) kvLinear, decltype(1_V / 1_mps_sq) kaLinear,
374 decltype(1_V / 1_rad_per_s) kvAngular,
375 decltype(1_V / 1_rad_per_s_sq) kaAngular,
376 wpi::units::meter_t trackwidth) {
377 if (kvLinear <= decltype(kvLinear){0}) {
378 throw std::domain_error("Kv,linear must be greater than zero.");
379 }
380 if (kaLinear <= decltype(kaLinear){0}) {
381 throw std::domain_error("Ka,linear must be greater than zero.");
382 }
383 if (kvAngular <= decltype(kvAngular){0}) {
384 throw std::domain_error("Kv,angular must be greater than zero.");
385 }
386 if (kaAngular <= decltype(kaAngular){0}) {
387 throw std::domain_error("Ka,angular must be greater than zero.");
388 }
389 if (trackwidth <= 0_m) {
390 throw std::domain_error("r must be greater than zero.");
391 }
392
393 // We want to find a factor to include in Kv,angular that will convert
394 // `u = Kv,angular omega` to `u = Kv,angular v`.
395 //
396 // v = omega r
397 // omega = v/r
398 // omega = 1/r v
399 // omega = 1/(trackwidth/2) v
400 // omega = 2/trackwidth v
401 //
402 // So multiplying by 2/trackwidth converts the angular gains from V/(rad/s)
403 // to V/(m/s).
404 return DifferentialDriveFromSysId(kvLinear, kaLinear,
405 kvAngular * 2.0 / trackwidth * 1_rad,
406 kaAngular * 2.0 / trackwidth * 1_rad);
407 }
