41 /**

51 /**
52 * Constructs a Color8Bit from a hex string.
53 *
54 * @param hexString a string of the format <tt>\#RRGGBB</tt>
55 * @throws std::invalid_argument if the hex string is invalid.
56 */
57 explicit constexpr Color8Bit(std::string_view hexString) {
58 if (hexString.length() != 7 || !hexString.starts_with("#") ||
59 !wpi::util::isHexDigit(hexString[1]) ||
60 !wpi::util::isHexDigit(hexString[2]) ||
61 !wpi::util::isHexDigit(hexString[3]) ||
62 !wpi::util::isHexDigit(hexString[4]) ||
63 !wpi::util::isHexDigit(hexString[5]) ||
64 !wpi::util::isHexDigit(hexString[6])) {
65 throw std::invalid_argument(
66 std::format("Invalid hex string for Color \"{}\"", hexString));
67 }
68
69 red = wpi::util::hexDigitValue(hexString[1]) * 16 +
70 wpi::util::hexDigitValue(hexString[2]);
71 green = wpi::util::hexDigitValue(hexString[3]) * 16 +
72 wpi::util::hexDigitValue(hexString[4]);
73 blue = wpi::util::hexDigitValue(hexString[5]) * 16 +
74 wpi::util::hexDigitValue(hexString[6]);
75 }
76
77 constexpr bool operator==(const Color8Bit&) const = default;
78
79 constexpr operator Color() const { // NOLINT
80 return Color(red / 255.0, green / 255.0, blue / 255.0);
81 }

83 /**
84 * Create a Color8Bit from a hex string.
85 *
86 * @param hexString a string of the format <tt>\#RRGGBB</tt>
87 * @return Color8Bit object from hex string.
88 * @throws std::invalid_argument if the hex string is invalid.
89 */
90 static constexpr Color8Bit FromHexString(std::string_view hexString) {
91 if (hexString.length() != 7 || !hexString.starts_with("#") ||
92 !wpi::util::isHexDigit(hexString[1]) ||
93 !wpi::util::isHexDigit(hexString[2]) ||
94 !wpi::util::isHexDigit(hexString[3]) ||
95 !wpi::util::isHexDigit(hexString[4]) ||
96 !wpi::util::isHexDigit(hexString[5]) ||
97 !wpi::util::isHexDigit(hexString[6])) {
98 throw std::invalid_argument(
99 std::format("Invalid hex string for Color \"{}\"", hexString));
100 }
101
102 int r = wpi::util::hexDigitValue(hexString[0]) * 16 +
103 wpi::util::hexDigitValue(hexString[1]);
104 int g = wpi::util::hexDigitValue(hexString[2]) * 16 +
105 wpi::util::hexDigitValue(hexString[3]);
106 int b = wpi::util::hexDigitValue(hexString[4]) * 16 +
107 wpi::util::hexDigitValue(hexString[5]);
108 return Color8Bit{r, g, b};
109 }
110
111 /**
112 * Return this color represented as a hex string.
113 *
114 * @return a string of the format <tt>\#RRGGBB</tt>
115 */
116 constexpr auto HexString() const {
117 return wpi::util::ct_string<char, std::char_traits<char>, 7>{
118 {'#', wpi::util::hexdigit(red / 16), wpi::util::hexdigit(red % 16),
119 wpi::util::hexdigit(green / 16), wpi::util::hexdigit(green % 16),
120 wpi::util::hexdigit(blue / 16), wpi::util::hexdigit(blue % 16)}};
121 }
