15namespace wpi {
16/**
17 * Tachometer for getting rotational velocity from a device.
18 *
19 * The Tachometer class measures the rate of digital pulses to determine the
20 * rotation velocity of a mechanism. Examples of devices that could be used
21 * with the tachometer class are a hall effect sensor, break beam sensor, or
22 * optical sensor detecting tape on a shooter wheel. Unlike encoders, this class
23 * only needs a single digital input.
24 */
25class Tachometer : public wpi::telemetry::TelemetryLoggable {
26 public:
27 /**
28 * Constructs a new tachometer.
29 *
30 * @param channel The DIO Channel.
31 * @param configuration Edge configuration
32 */
33 Tachometer(int channel, EdgeConfiguration configuration);
34
35 Tachometer(Tachometer&&) = default;
36 Tachometer& operator=(Tachometer&&) = default;
37
38 ~Tachometer() override = default;
39
40 /**
41 * Sets the configuration for the channel.
42 *
43 * @param configuration The channel configuration.
44 */
45 void SetEdgeConfiguration(EdgeConfiguration configuration);
46
47 /**
48 * Sets the time window used to calculate the tachometer rate.
49 *
50 * @param window The rate calculation window. Valid values are 5 ms through
51 * 255 ms. The default is 50 ms.
52 */
53 void SetRateWindow(wpi::units::millisecond_t window);
54
55 /**
56 * Gets the tachometer frequency.
57 *
58 * @return Current frequency.
59 */
60 wpi::units::hertz_t GetFrequency() const;
61
62 /**
63 * Gets the number of edges per revolution.
64 *
65 * @return Edges per revolution.
66 */
67 int GetEdgesPerRevolution() const;
68
69 /**
70 * Sets the number of edges per revolution.
71 *
72 * @param edges Edges per revolution.
73 */
74 void SetEdgesPerRevolution(int edges);
75
76 /**
77 * Gets the current tachometer revolutions per second.
78 *
79 * SetEdgesPerRevolution must be set with a non 0 value for this to work.
80 *
81 * @return Current RPS.
82 * @Common This is one of the commonly used methods for this class
83 */
84 wpi::units::turns_per_second_t GetRevolutionsPerSecond() const;
85
86 /**
87 * Gets the current tachometer revolutions per minute.
88 *
89 * SetEdgesPerRevolution must be set with a non 0 value for this to work.
90 *
91 * @return Current RPM.
92 * @Common This is one of the commonly used methods for this class
93 */
94 wpi::units::revolutions_per_minute_t GetRevolutionsPerMinute() const;
95
96 /**
97 * Gets if the tachometer is stopped.
98 *
99 * The tachometer is stopped when its current frequency is zero.
100 *
101 * @return True if the tachometer is stopped.
102 */
103 bool GetStopped() const;
104
105 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
106
107 std::string_view GetTelemetryType() const override;
108
109 private:
110 wpi::util::Handle<HAL_CounterHandle, HAL_FreeCounter> m_handle;
111 int m_edgesPerRevolution{1};
112 int32_t m_channel;
113};
114} // namespace wpi
