92 /**
93 * The state of one joystick button. Button indexes begin at 0.
94 *
95 * @param stick The joystick to read.

99 static bool GetStickButton(int stick, int button);

101 /**
102 * The state of one joystick button, only if available. Button indexes begin
103 * at 0.
104 *
105 * @param stick The joystick to read.
106 * @param button The button index, beginning at 0.
107 * @return The state of the joystick button, or empty if unavailable.
108 */
109 static std::optional<bool> GetStickButtonIfAvailable(int stick, int button);
110
111 /**
112 * Whether one joystick button was pressed since the last check. %Button
113 * indexes begin at 1.
114 *
115 * @param stick The joystick to read.
116 * @param button The button index, beginning at 0.
117 * @return Whether the joystick button was pressed since the last check.
118 */
119 static bool GetStickButtonPressed(int stick, int button);
120
121 /**
122 * Whether one joystick button was released since the last check. %Button
123 * indexes begin at 1.
124 *
125 * @param stick The joystick to read.
126 * @param button The button index, beginning at 0.
127 * @return Whether the joystick button was released since the last check.
128 */
129 static bool GetStickButtonReleased(int stick, int button);
130
131 /**
132 * Get the value of the axis on a joystick.
133 *
134 * This depends on the mapping of the joystick connected to the specified
135 * port.
136 *
137 * @param stick The joystick to read.
138 * @param axis The analog axis value to read from the joystick.
139 * @return The value of the axis on the joystick.
140 */
141 static double GetStickAxis(int stick, int axis);
142
143 /**
144 * Get the finger data of a touchpad on a joystick, if available.
145 *
146 * @param stick The joystick to read.
147 * @param touchpad The touchpad index to read from the joystick.
148 * @param finger The finger index to read from the touchpad.
149 * @return The finger data of the touchpad on the joystick.
150 */
151 static TouchpadFinger GetStickTouchpadFinger(int stick, int touchpad,
152 int finger);
153
154 /**
155 * Whether a finger on a touchpad is available.
156 *
157 * @param stick The joystick to read.
158 * @param touchpad The touchpad index to read from the joystick.
159 * @param finger The finger index to read from the touchpad.
160 * @return True if the finger data is available.
161 */
162 static bool GetStickTouchpadFingerAvailable(int stick, int touchpad,
163 int finger);
164
165 /**
166 * Get the value of the axis on a joystick, if available.
167 *
168 * This depends on the mapping of the joystick connected to the specified
169 * port.
170 *
171 * @param stick The joystick to read.
172 * @param axis The analog axis value to read from the joystick.
173 * @return The value of the axis on the joystick, or empty if not available.
174 */
175 static std::optional<double> GetStickAxisIfAvailable(int stick, int axis);
176
177 /**
178 * Get the state of a POV on the joystick.
179 *
180 * @return the angle of the POV.
181 */
182 static POVDirection GetStickPOV(int stick, int pov);
183
184 /**
185 * The state of the buttons on the joystick.
186 *
187 * @param stick The joystick to read.

190 static uint64_t GetStickButtons(int stick);

192 /**

198 static int GetStickAxesMaximumIndex(int stick);

200 /**

206 static int GetStickAxesAvailable(int stick);

208 /**

214 static int GetStickPOVsMaximumIndex(int stick);

216 /**
217 * Returns the mask of available POVs on a given joystick port.

221 */
222 static int GetStickPOVsAvailable(int stick);
223
224 /**
225 * Returns the maximum button index on a given joystick port.
226 *

230 static int GetStickButtonsMaximumIndex(int stick);
231
232 /**
233 * Returns the mask of available buttons on a given joystick port.
234 *
235 * @param stick The joystick port number
236 * @return The mask of available buttons on the indicated joystick

238 static uint64_t GetStickButtonsAvailable(int stick);

240 /**
241 * Returns a boolean indicating if the controller is an xbox controller.

248 /**

267 /**

273 static std::string GetJoystickName(int stick);

275 /**
276 * Returns if a joystick is connected to the Driver Station.
277 *
278 * This makes a best effort guess by looking at the reported number of axis,
279 * buttons, and POVs attached.
280 *
281 * @param stick The joystick port number
282 * @return true if a joystick is connected
283 */
284 static bool IsJoystickConnected(int stick);
285
286 /**
287 * Copy data from the DS task for the user. If no new data exists, it will

289 * loop.

291 static void RefreshData();

293 /**
294 * Registers the given handle for DS data refresh notifications.

296 * @param handle The event handle.

298 static void ProvideRefreshedDataEventHandle(WPI_EventHandle handle);

300 /**
301 * Unregisters the given handle from DS data refresh notifications.
302 *
303 * @param handle The event handle.
304 */
305 static void RemoveRefreshedDataEventHandle(WPI_EventHandle handle);
306
307 /**
308 * Allows the user to specify whether they want joystick connection alerts to
309 * be shown. This setting is ignored when the FMS is connected -- alerts will
310 * always be on in that scenario.
311 *
312 * @param silence Whether joystick connection alerts should be silenced.
313 */
314 static void SilenceJoystickConnectionAlert(bool silence);
315
316 /**
317 * Returns whether joystick connection alerts are silenced. This will
318 * always return false when connected to the FMS.
319 *
320 * @return Whether joystick connection alerts are silenced.
321 */
322 static bool IsJoystickConnectionAlertSilenced();
323
324 /**
325 * Starts logging DriverStation data to data log. Repeated calls are ignored.
326 *
327 * @param log data log
328 * @param logJoysticks if true, log joystick data
329 */
330 static void StartDataLog(wpi::log::DataLog& log, bool logJoysticks = true);
331
332 private:
333 DriverStationBackend() = default;
334};
