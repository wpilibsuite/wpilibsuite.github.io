29/**
30 * Driver for the goBILDA Pinpoint Odometry Computer.
31 *
32 * Aborted I2C transactions and invalid read data are reported through
33 * DeviceStatus::FAULT_BAD_READ instead of throwing an exception.
34 */
35class GoBildaPinpoint {
36 public:
37 /// Default 7-bit I2C address of the Pinpoint.
38 static constexpr int DEFAULT_ADDRESS = 0x31;
39
40 /** Error detection method used for I2C reads. */
41 enum class ErrorDetectionType {
42 /// Do not validate data read from the device.
43 NONE,
44 /// Validate the CRC-8 byte returned by v3 or newer firmware.
45 CRC,
46 /// Reject nonfinite values, invalid quaternions, and implausibly large
47 /// changes locally.
48 LOCAL_TEST
49 };
50
51 /** Reason the driver rejected data or an I2C transaction. */
52 enum class FailureReason {
53 /// The I2C controller aborted a read transaction.
54 I2C_READ_ABORTED,
55 /// The I2C controller aborted a write transaction.
56 I2C_WRITE_ABORTED,
57 /// Data returned by the device did not have the expected CRC-8 value.
58 CRC_MISMATCH,
59 /// A bulk read returned a nonpositive device loop time.
60 INVALID_LOOP_TIME,
61 /// A floating-point register returned a nonfinite value.
62 NONFINITE_VALUE,
63 /// A quaternion sample had a zero or near-zero norm.
64 INVALID_QUATERNION,
65 /// A position or orientation changed by more than the validation limit.
66 CHANGE_TOO_LARGE,
67 /// A velocity magnitude exceeded the local validation limit.
68 VELOCITY_TOO_LARGE
69 };
70
71 /** Register map for the Pinpoint. */
72 enum class Register {
73 /// Device identifier.
74 DEVICE_ID = 1,
75 /// Firmware version.
76 DEVICE_VERSION = 2,
77 /// Device status bit field.
78 DEVICE_STATUS = 3,
79 /// Device control command.
80 DEVICE_CONTROL = 4,
81 /// Device loop time.
82 LOOP_TIME = 5,
83 /// Raw X encoder value.
84 X_ENCODER_VALUE = 6,
85 /// Raw Y encoder value.
86 Y_ENCODER_VALUE = 7,
87 /// X position in millimeters.
88 X_POSITION = 8,
89 /// Y position in millimeters.
90 Y_POSITION = 9,
91 /// Heading in radians.
92 H_ORIENTATION = 10,
93 /// X velocity in millimeters per second.
94 X_VELOCITY = 11,
95 /// Y velocity in millimeters per second.
96 Y_VELOCITY = 12,
97 /// Heading velocity in radians per second.
98 H_VELOCITY = 13,
99 /// Encoder ticks per millimeter traveled.
100 MM_PER_TICK = 14,
101 /// X pod offset in millimeters.
102 X_POD_OFFSET = 15,
103 /// Y pod offset in millimeters.
104 Y_POD_OFFSET = 16,
105 /// IMU yaw scalar.
106 YAW_SCALAR = 17,
107 /// Bulk-read window.
108 BULK_READ = 18,
109 /// Quaternion W component.
110 QUATERNION_W = 19,
111 /// Quaternion X component.
112 QUATERNION_X = 20,
113 /// Quaternion Y component.
114 QUATERNION_Y = 21,
115 /// Quaternion Z component.
116 QUATERNION_Z = 22,
117 /// Pitch in radians.
118 PITCH = 23,
119 /// Roll in radians.
120 ROLL = 24,
121 /// Flexible bulk-read configuration.
122 SET_BULK_READ = 25
123 };
124
125 /** Pinpoint status, including device-reported faults and local read failures.
126 */
127 enum class DeviceStatus {
128 /// The device is starting up and has not initialized.
129 NOT_READY = 0,
130 /// The device is operating normally.
131 READY = 1,
132 /// The device is calibrating its IMU.
133 CALIBRATING = 1 << 1,
134 /// The X odometry pod was not detected.
135 FAULT_X_POD_NOT_DETECTED = 1 << 2,
136 /// The Y odometry pod was not detected.
137 FAULT_Y_POD_NOT_DETECTED = 1 << 3,
138 /// Neither odometry pod was detected.
139 FAULT_NO_PODS_DETECTED = (1 << 2) | (1 << 3),
140 /// The IMU reported a runaway condition.
141 FAULT_IMU_RUNAWAY = 1 << 4,
142 /// A read failed validation or the I2C transaction was aborted.
143 FAULT_BAD_READ = 1 << 5
144 };
145
146 /** Encoder direction relative to the Pinpoint coordinate system. */
147 enum class EncoderDirection {
148 /// Encoder count increases in the positive axis direction.
149 FORWARD,
150 /// Encoder count decreases in the positive axis direction.
151 REVERSED
152 };
153
154 /** goBILDA odometry pod with a predefined encoder resolution. */
155 enum class OdometryPod {
156 /// goBILDA Swingarm Odometry Pod.
157 SWINGARM,
158 /// goBILDA 4-Bar Odometry Pod.
159 FOUR_BAR
160 };

188 /**
189 * Reads all registers in the configured bulk-read scope and updates the
190 * cached measurements. Call this once per robot loop.
191 *
192 * An aborted or invalid I2C read does not throw; it sets the device status

195 * @throws std::runtime_error if CRC error detection is active with firmware
196 * older than v3.
