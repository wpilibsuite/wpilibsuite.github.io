18/**
19 * A trajectory for holonomic (omni-directional) drive robots.
20 *
21 * Uses constant-acceleration kinematic interpolation between samples.
22 */
23class WPILIB_DLLEXPORT HolonomicTrajectory
24 : public Trajectory<HolonomicSample> {
25 public:
26 /**
27 * Constructs a HolonomicTrajectory from a vector of samples.
28 *
29 * @param samples The samples of the trajectory. Order does not matter as
30 * they will be sorted internally.
31 * @throws std::invalid_argument if the vector of samples is empty.
32 */
33 explicit HolonomicTrajectory(std::vector<HolonomicSample> samples)
34 : Trajectory<HolonomicSample>(std::move(samples)) {}
35
36 /**
37 * Interpolates between two samples using kinematic interpolation.
38 *
39 * @param start The starting sample.
40 * @param end The ending sample.
41 * @param t The interpolation parameter between 0 and 1.
42 * @return The interpolated sample.
43 */
44 HolonomicSample Interpolate(const HolonomicSample& start,
45 const HolonomicSample& end, double t) const final;
46
47 /**
48 * Transforms all poses in the trajectory by the given transform.
49 *
50 * @param transform The transform to apply to the poses.
51 * @return The transformed trajectory.
52 */
53 HolonomicTrajectory TransformBy(const Transform2d& transform) const;
54
55 /**
56 * Transforms all poses so they are relative to the given pose.
57 *
58 * @param pose The pose to make the trajectory relative to.
59 * @return The transformed trajectory.
60 */
61 HolonomicTrajectory RelativeTo(const Pose2d& pose) const;
62
63 /**
64 * Concatenates another trajectory to this trajectory.
65 *
66 * @param other The trajectory to concatenate.
67 * @return The concatenated trajectory.
68 */
69 HolonomicTrajectory Concatenate(const HolonomicTrajectory& other) const;
70
71 /**
72 * Concatenates another trajectory to this trajectory.
73 *
74 * @param other The trajectory to concatenate.
75 * @return The concatenated trajectory.
76 */
77 HolonomicTrajectory operator+(const HolonomicTrajectory& other) const {
78 return Concatenate(other);
79 }
80};
