30/**
31 * A particle filter estimates the state of a system by representing the state
32 * distribution with a set of weighted samples called particles. Unlike Kalman
33 * filters, it doesn't assume the state distribution is Gaussian, so it can
34 * handle strongly nonlinear models and multimodal distributions.
35 *
36 * Each prediction step propagates every particle through the dynamics model and
37 * adds sampled process noise. Each correction step reweights the particles by
38 * the likelihood of the measurement given each particle, then resamples the
39 * particles if too few of them carry most of the weight.
40 *
41 * The state estimate x-hat is the weighted mean of the particles, and the error
42 * covariance P is their weighted sample covariance.
43 *
44 * The filter's computational cost is proportional to the number of particles.
45 * More particles give a more accurate approximation of the state distribution.
46 *
47 * This class implements the generic particle filter in
48 * [2, Algorithm 3, p. 180] with the dynamics model as the importance density
49 * [2, eq. (62), p. 179]. That importance density comes from the bootstrap
50 * filter [1, Section 3.1, p. 108], also known as the sampling importance
51 * resampling (SIR) filter [2, Section V-B1, p. 181]. Unlike those filters,
52 * this one only resamples when the effective sample size is low instead of
53 * every timestep.
54 *
55 * The N particles xᵢ with weights Wᵢ approximate the posterior state
56 * distribution as [2, eq. (49), p. 178]
57 *
58 * p(xₖ | y₁, …, yₖ) ≈ Σᵢ Wᵢ δ(xₖ − xᵢ)
59 *
60 * where δ is the Dirac delta and Σᵢ Wᵢ = 1.
61 *
62 * References:
63 *
64 * [1] N. J. Gordon, D. J. Salmond, and A. F. M. Smith, "Novel approach to
65 * nonlinear/non-Gaussian Bayesian state estimation," IEE Proceedings F,
66 * vol. 140, no. 2, pp. 107–113, 1993. DOI: 10.1049/ip-f-2.1993.0015
67 * https://www.ece.iastate.edu/~namrata/EE520/Gordonnovelapproach.pdf
68 *
69 * [2] M. S. Arulampalam, S. Maskell, N. Gordon, and T. Clapp, "A tutorial on
70 * particle filters for online nonlinear/non-Gaussian Bayesian tracking,"
71 * IEEE Transactions on Signal Processing, vol. 50, no. 2, pp. 174–188,
72 * 2002. DOI: 10.1109/78.978374
73 * https://people.eecs.berkeley.edu/~pabbeel/cs287-fa12/optreadings/Arulampalam_etal_2002.pdf
74 *
75 * [3] R. Douc, O. Cappé, and E. Moulines, "Comparison of resampling schemes for
76 * particle filtering," Proceedings of the 4th International Symposium on
77 * Image and Signal Processing and Analysis (ISPA), pp. 64–69, 2005.
78 * https://arxiv.org/abs/cs/0507025
79 *
80 * @tparam States Number of states.
81 * @tparam Inputs Number of inputs.
82 * @tparam Outputs Number of outputs.
83 */

103 /**
104 * Constructs a particle filter.
105 *
106 * All particles start at the zero state, so the initial error covariance is
107 * zero. Call SetXhat() and SetP() to set the initial state distribution.
108 *
109 * See
110 * https://docs.wpilib.org/en/stable/docs/software/advanced-controls/state-space/state-space-observers.html#process-and-measurement-noise-covariance-matrices
111 * for how to select the standard deviations.
112 *
113 * @param f A vector-valued function of x and u that returns
114 * the derivative of the state vector.
115 * @param h A vector-valued function of x and u that returns
116 * the measurement vector.
117 * @param stateStdDevs Standard deviations of model states.
118 * @param measurementStdDevs Standard deviations of measurements.
119 * @param numParticles Number of particles. Must be positive.
120 * @param dt Nominal discretization timestep.
121 * @throws std::invalid_argument if numParticles isn't positive.
122 */
123 ParticleFilter(
124 std::function<StateVector(const StateVector&, const InputVector&)> f,
125 std::function<OutputVector(const StateVector&, const InputVector&)> h,
126 const StateArray& stateStdDevs, const OutputArray& measurementStdDevs,
127 int numParticles, wpi::units::second_t dt)
128 : ParticleFilter(
129 std::move(f), std::move(h), stateStdDevs, measurementStdDevs,
130 [](const ParticleMatrix& particles, const WeightVector& weights)
131 -> StateVector { return particles * weights; },
132 [](const StateVector& a, const StateVector& b) -> StateVector {
133 return a - b;
134 },
135 [](const OutputVector& a, const OutputVector& b) -> OutputVector {
136 return a - b;
137 },
138 [](const StateVector& a, const StateVector& b) -> StateVector {
139 return a + b;
140 },
141 numParticles, dt) {}

143 /**
144 * Constructs a particle filter with custom mean, residual, and addition
145 * functions. Using custom functions for arithmetic can be useful if you have
146 * angles in the state or measurements, because they allow you to correctly
147 * account for the modular nature of angle arithmetic (see AngleMean(),
148 * AngleResidual(), and AngleAdd()).
149 *
150 * All particles start at the zero state, so the initial error covariance is
151 * zero. Call SetXhat() and SetP() to set the initial state distribution.
152 *
153 * See
154 * https://docs.wpilib.org/en/stable/docs/software/advanced-controls/state-space/state-space-observers.html#process-and-measurement-noise-covariance-matrices
155 * for how to select the standard deviations.
156 *
157 * @param f A vector-valued function of x and u that returns
158 * the derivative of the state vector.
159 * @param h A vector-valued function of x and u that returns
160 * the measurement vector.
161 * @param stateStdDevs Standard deviations of model states.
162 * @param measurementStdDevs Standard deviations of measurements.
163 * @param meanFuncX A function that computes the weighted mean of the
164 * particles (one particle per column) given a
165 * column vector of weights that sum to one.
166 * @param residualFuncX A function that computes the residual of two
167 * state vectors (i.e. it subtracts them.)
168 * @param residualFuncY A function that computes the residual of two
169 * measurement vectors (i.e. it subtracts them.)
170 * @param addFuncX A function that adds two state vectors.
171 * @param numParticles Number of particles. Must be positive.
172 * @param dt Nominal discretization timestep.
173 * @throws std::invalid_argument if numParticles isn't positive.
174 */
175 ParticleFilter(
176 std::function<StateVector(const StateVector&, const InputVector&)> f,
177 std::function<OutputVector(const StateVector&, const InputVector&)> h,
178 const StateArray& stateStdDevs, const OutputArray& measurementStdDevs,
179 std::function<StateVector(const ParticleMatrix&, const WeightVector&)>
180 meanFuncX,
181 std::function<StateVector(const StateVector&, const StateVector&)>
182 residualFuncX,
183 std::function<OutputVector(const OutputVector&, const OutputVector&)>
184 residualFuncY,
185 std::function<StateVector(const StateVector&, const StateVector&)>
186 addFuncX,
187 int numParticles, wpi::units::second_t dt)
188 : m_f(std::move(f)),
189 m_h(std::move(h)),
190 m_meanFuncX(std::move(meanFuncX)),
191 m_residualFuncX(std::move(residualFuncX)),
192 m_residualFuncY(std::move(residualFuncY)),
193 m_addFuncX(std::move(addFuncX)),
194 m_dt(dt) {
195 if (numParticles <= 0) {
196 std::string msg = "ParticleFilter numParticles must be positive";
197 wpi::math::MathSharedStore::ReportError(msg);
198 throw std::invalid_argument(msg);
199 }
200
201 for (int i = 0; i < States; ++i) {
202 m_contStateStdDevs(i) = stateStdDevs[i];
203 }
204 m_contR = CovarianceMatrix(measurementStdDevs);
205
206 m_particles = ParticleMatrix::Zero(States, numParticles);
207 m_weights = WeightVector::Constant(numParticles, 1.0 / numParticles);
208 m_resampled = ParticleMatrix{States, numParticles};
209
210 UpdateStatistics();
211
212 wpi::util::ReportUsage("ParticleFilter", "");
213 }

229 /**
230 * Sets the error covariance matrix P by redrawing the particles from a
231 * Gaussian distribution centered on the current state estimate x-hat with
232 * covariance P.
233 *
234 * @param P The error covariance matrix P. Must be symmetric positive
235 * semidefinite.
236 */
237 void SetP(const StateMatrix& P) {
238 // Draw each particle as
239 //
240 // xᵢ = x̂ + Sεᵢ
241 //
242 // where εᵢ ~ N(0, I) and SSᵀ = P. Then
243 //
244 // Cov(Sεᵢ) = S E[εᵢεᵢᵀ] Sᵀ = SISᵀ = P
245 //
246 // so xᵢ ~ N(x̂, P).
247 const StateVector xHat = m_xHat;
248 const StateMatrix S = CovarianceSqrt(P);
249 for (int i = 0; i < NumParticles(); ++i) {
250 m_particles.col(i) = m_addFuncX(xHat, S * SampleStandardNormal());
251 }
252 m_weights.setConstant(1.0 / NumParticles());
253 UpdateStatistics();
254 }

269 /**
270 * Sets the state estimate x-hat by shifting every particle by the same
271 * amount. This preserves the shape of the state distribution.
272 *
273 * @param xHat The state estimate x-hat.
274 */
275 void SetXhat(const StateVector& xHat) {
276 // Shift every particle by δ = x̂′ − x̂, where x̂′ is the new state estimate:
277 //
278 // xᵢ ← xᵢ + δ
279 //
280 // With the default arithmetic, the new mean is Σᵢ Wᵢ(xᵢ + δ) = x̂ + δ = x̂′,
281 // and each residual (xᵢ + δ) − (x̂ + δ) = xᵢ − x̂ is unchanged, so P is
282 // unchanged.
283 const StateVector delta = m_residualFuncX(xHat, m_xHat);
284 for (int i = 0; i < NumParticles(); ++i) {
285 m_particles.col(i) = m_addFuncX(m_particles.col(i), delta);
286 }
287 UpdateStatistics();
288 }

290 /**

294 * @param i Row of x-hat.

296 */
297 void SetXhat(int i, double value) {
298 StateVector xHat = m_xHat;

336 /**
337 * Project the model into the future with a new control input u.
338 *
339 * @param u New control input from controller.
340 * @param dt Timestep for prediction.
341 */
342 void Predict(const InputVector& u, wpi::units::second_t dt) {
343 // Sample each particle from the dynamics model [1, Section 3.1, p. 108]
344 // [2, Section V-B1, p. 181]:
345 //
346 // xᵢ ← RK4(f, xᵢ, u, dt) + wᵢ, wᵢ ~ N(0, Q_d)
347 //
348 // Because the importance density is the dynamics model itself, the weight
349 // update only depends on the likelihood [2, eq. (63), p. 179], so the
350 // weights don't change during prediction.
351 //
352 // For dynamics linearized as ẋ = Ax + Bu + w, where w has continuous
353 // process noise covariance Q, the discrete process noise covariance over a
354 // timestep T = dt is
355 //
356 // Q_d = ∫₀ᵀ e^(Aτ) Q e^(Aᵀτ) dτ
357 //
358 // which DiscretizeAQ() computes exactly. Each particle would need its own
359 // linearization A to use that, so the first-order term is used instead:
360 //
361 // Q_d = QT + O(T²)
362 //
363 // Q = diag(σ₁², …, σₙ²) is diagonal, so wᵢ = √T diag(σ) εᵢ with
364 // εᵢ ~ N(0, I) gives Cov(wᵢ) = T diag(σ)² = QT.
365 const StateVector discStdDevs = m_contStateStdDevs * std::sqrt(dt.value());
366
367 for (int i = 0; i < NumParticles(); ++i) {
368 StateVector x = RK4(m_f, StateVector{m_particles.col(i)}, u, dt);
369 m_particles.col(i) =
370 m_addFuncX(x, discStdDevs.cwiseProduct(SampleStandardNormal()));
371 }
372
373 m_dt = dt;
374
375 UpdateStatistics();
376 }

403 * Correct the state estimate x-hat using the measurements in y.

427 /**
428 * Correct the state estimate x-hat using the measurements in y.
429 *
430 * This is useful for when the measurements available during a timestep's
431 * Correct() call vary. The h(x, u) passed to the constructor is used if one
432 * is not provided (the two-argument version of this function).
433 *
434 * @param u Same control input used in the predict step.
435 * @param y Measurement vector.
436 * @param h A vector-valued function of x and u that returns
437 * the measurement vector.
438 * @param R Continuous measurement noise covariance matrix. Must
439 * be symmetric positive definite.
440 * @param residualFuncY A function that computes the residual of two
441 * measurement vectors (i.e. it subtracts them.)
442 */
443 template <int Rows>
444 void Correct(
445 const InputVector& u, const Vectord<Rows>& y,
446 std::function<Vectord<Rows>(const StateVector&, const InputVector&)> h,
447 const Matrixd<Rows, Rows>& R,
448 std::function<Vectord<Rows>(const Vectord<Rows>&, const Vectord<Rows>&)>
449 residualFuncY) {
450 // The general sequential importance sampling weight update is
451 // [2, eq. (48), p. 178]
452 //
453 // Wᵢ ∝ Wᵢ⁻ p(y | xᵢ) p(xᵢ | xᵢ⁻) / q(xᵢ | xᵢ⁻, y)
454 //
455 // where q is the importance density. Predict() samples from the dynamics
456 // model, so q(xᵢ | xᵢ⁻, y) = p(xᵢ | xᵢ⁻) and the update reduces to
457 // [2, eqs. (62)–(63), p. 179]
458 //
459 // Wᵢ ∝ Wᵢ⁻ p(y | xᵢ)
460 //
461 // With Gaussian measurement noise, the likelihood is
462 //
463 // p(y | xᵢ) = (2π)^(−m/2) |R_d|^(−½) exp(−½ rᵢᵀR_d⁻¹rᵢ)
464 //
465 // where rᵢ = y − h(xᵢ, u), m is the number of measurements, and
466 // R_d = R/T is the discrete measurement noise covariance from
467 // DiscretizeR(). The leading constant is the same for every particle, so it
468 // cancels during normalization and only exp(−½ rᵢᵀR_d⁻¹rᵢ) is needed.
469 //
470 // With the Cholesky decomposition R_d = LLᵀ,
471 //
472 // rᵢᵀR_d⁻¹rᵢ = rᵢᵀL⁻ᵀL⁻¹rᵢ = (L⁻¹rᵢ)ᵀ(L⁻¹rᵢ) = ‖L⁻¹rᵢ‖²
473 //
474 // so a triangular solve replaces the matrix inverse.
475 const Eigen::LLT<Matrixd<Rows, Rows>> discRllt{DiscretizeR<Rows>(R, m_dt)};
476
477 // Weights are updated in log space to avoid underflow when the likelihoods
478 // are tiny:
479 //
480 // ℓᵢ = ln Wᵢ⁻ − ½‖L⁻¹rᵢ‖²
481 WeightVector logWeights{NumParticles()};
482 for (int i = 0; i < NumParticles(); ++i) {
483 const Vectord<Rows> r = residualFuncY(y, h(m_particles.col(i), u));
484 const double logLikelihood =
485 -0.5 * discRllt.matrixL().solve(r).squaredNorm();
486 logWeights(i) = std::log(m_weights(i)) + logLikelihood;
487
488 // Treat particles with invalid measurements as impossible
489 if (std::isnan(logWeights(i))) {
490 logWeights(i) = -std::numeric_limits<double>::infinity();
491 }
492 }
493
494 const double maxLogWeight = logWeights.maxCoeff();
495 if (std::isfinite(maxLogWeight)) {
496 // Normalize with the log-sum-exp trick:
497 //
498 // Wᵢ = exp(ℓᵢ − ℓₘₐₓ) / Σⱼ exp(ℓⱼ − ℓₘₐₓ)
499 //
500 // The factor exp(−ℓₘₐₓ) cancels between the numerator and denominator,
501 // so this equals exp(ℓᵢ) / Σⱼ exp(ℓⱼ). Subtracting ℓₘₐₓ makes the
502 // largest term one, so the sum is at least one and normalization can't
503 // divide by zero.
504 m_weights = (logWeights.array() - maxLogWeight).exp();
505 m_weights /= m_weights.sum();
506 } else {
507 // No particle is consistent with the measurement, so the measurement
508 // carries no usable information. Fall back to equal weights rather than
509 // dividing by zero.
510 m_weights.setConstant(1.0 / NumParticles());
511 }
512
513 // Resample when the estimated effective sample size [2, eq. (51), p. 179]
514 //
515 // N̂ₑ = 1 / Σᵢ Wᵢ²
516 //
517 // drops below half the particle count [2, Algorithm 3, p. 180]. N̂ₑ ranges
518 // from 1 (all weight on one particle) to N (equal weights). [2] leaves the
519 // threshold as a design choice, and N/2 is a common heuristic.
520 const double effectiveSampleSize = 1.0 / m_weights.squaredNorm();
521 if (effectiveSampleSize < 0.5 * NumParticles()) {
522 Resample();
523 }
524
525 UpdateStatistics();
526 }

568 /**
569 * Returns S such that SSᵀ = P for a symmetric positive semidefinite P.
570 */
571 static StateMatrix CovarianceSqrt(const StateMatrix& P) {
572 // P = SSᵀ = ΠᵀLDLᵀΠ
573 // S = ΠᵀL√(D)
574 //
575 // where Π is the LDLT permutation (Eigen's P). D is clamped to be
576 // nonnegative to absorb roundoff in nearly singular covariances.
577 const Eigen::LDLT<StateMatrix> ldlt{P};
578 StateMatrix S = StateMatrix{ldlt.matrixL()} *
579 ldlt.vectorD().cwiseMax(0.0).cwiseSqrt().asDiagonal();
580 return ldlt.transpositionsP().transpose() * S;
581 }

583 /**
584 * Resamples the particles in proportion to their weights, then resets the
585 * weights to be equal.
586 */
587 void Resample() {
588 // Use systematic resampling [2, Algorithm 2, p. 180] [3, Section 2.4],
589 // which draws one uniform random offset, then selects particles at evenly
590 // spaced points along the cumulative weight distribution. It runs in O(N)
591 // time and needs only one random number.
592 //
593 // Residual and stratified resampling are proven to have lower conditional
594 // variance than multinomial resampling [3, Sections 3.2–3.3], but
595 // systematic resampling isn't in general [3, Section 3.4]. Systematic
596 // resampling is still widely used because it's simple and its empirical
597 // performance is generally close to residual and stratified resampling
598 // [3, Section 1].
599 //
600 // With the cumulative weights Cᵢ = W₀ + ⋯ + Wᵢ and C₋₁ = 0, draw
601 //
602 // U ~ Uniform[0, 1/N)
603 // uⱼ = U + j/N, j = 0, …, N − 1
604 //
605 // and select particle i for each uⱼ where C₍ᵢ₋₁₎ < uⱼ ≤ Cᵢ. Each uⱼ is
606 // uniform over a width-1/N interval, so particle i is selected NWᵢ times in
607 // expectation and the resampled set is an unbiased estimate of the weighted
608 // one.
609 const int N = NumParticles();
610 const double step = 1.0 / N;
611 double position = m_uniform(m_generator) * step;
612
613 int i = 0;
614 double cumulativeWeight = m_weights(0);
615 for (int j = 0; j < N; ++j) {
616 // Roundoff can leave the total cumulative weight slightly below one, so
617 // stop at the last particle
618 while (position > cumulativeWeight && i < N - 1) {
619 ++i;
620 cumulativeWeight += m_weights(i);
621 }
622 m_resampled.col(j) = m_particles.col(i);
623 position += step;
624 }
625
626 m_particles.swap(m_resampled);
627 m_weights.setConstant(step);
628 }
629
630 /**
631 * Recomputes the state estimate and error covariance from the particles.
632 */
633 void UpdateStatistics() {
634 // With the default arithmetic, these are the mean and covariance of the
635 // weighted particle approximation of the posterior:
636 //
637 // x̂ = Σᵢ Wᵢxᵢ
638 // P = Σᵢ Wᵢ(xᵢ − x̂)(xᵢ − x̂)ᵀ
639 //
640 // P has no small-sample bias correction. With equal weights, P is
641 // (N − 1)/N times the unbiased sample covariance, which is negligible for
642 // typical particle counts.
643 m_xHat = m_meanFuncX(m_particles, m_weights);
644
645 m_P.setZero();
646 for (int i = 0; i < NumParticles(); ++i) {
647 const StateVector r = m_residualFuncX(m_particles.col(i), m_xHat);
648 m_P += m_weights(i) * r * r.transpose();
649 }
650 }
651};
