18/**
19 * Class for double follower wheel odometry. Odometry allows you to track the
20 * robot's position on the field over the course of a match using readings from
21 * 2 encoders and a gyroscope.
22 *
23 * <p>Teams can use odometry during the autonomous period for complex tasks like
24 * path following. Furthermore, odometry can be used for latency compensation
25 * when using computer-vision systems.
26 */
27class WPILIB_DLLEXPORT TwoDeadWheelOdometry3d {
28 public:
29 /**
30 * Constructs a TwoDeadWheelOdometry3d object.
31 *
32 * @param xWheelYPos The y-position of the forward-facing wheel relative to
33 * the center of the robot in meters.
34 * @param yWheelXPos The x-position of the left-facing wheel relative to the
35 * center of the robot in meters.
36 * @param xWheelPos The distance traveled by the forward-facing wheel, in
37 * meters.
38 * @param yWheelPos The distance traveled by the left-facing wheel, in meters.
39 * @param gyroAngle The angle reported by the gyroscope. This does not need to
40 * be offset to match the robot's orientation on the field.
41 * @param initialPose The starting position of the robot on the field.
42 */
43 explicit TwoDeadWheelOdometry3d(wpi::units::meter_t xWheelYPos,
44 wpi::units::meter_t yWheelXPos,
45 wpi::units::meter_t xWheelPos,
46 wpi::units::meter_t yWheelPos,
47 const Rotation3d& gyroAngle,
48 const Pose3d& initialPose = Pose3d{});
49
50 /**
51 * Resets the robot's position on the field.
52 *
53 * @param xWheelPos The distance traveled by the forward-facing wheel, in
54 * meters.
55 * @param yWheelPos The distance traveled by the left-facing wheel, in meters.
56 * @param gyroAngle The angle reported by the gyroscope. This does not need to
57 * be offset to match the robot's orientation on the field.
58 * @param pose The new position of the robot on the field.
59 */
60 void ResetPosition(wpi::units::meter_t xWheelPos,
61 wpi::units::meter_t yWheelPos, const Rotation3d& gyroAngle,
62 const Pose3d& pose) {
63 m_previousGyroAngle = gyroAngle;
64 m_previousXWheelPos = xWheelPos;
65 m_previousYWheelPos = yWheelPos;
66 m_pose = pose;
67 }
68
69 /**
70 * Resets the pose.
71 *
72 * @param pose The pose to reset to.
73 */
74 void ResetPose(const Pose3d& pose) { m_pose = pose; }
75
76 /**
77 * Resets the translation of the pose.
78 *
79 * @param translation The translation to reset to.
80 */
81 void ResetTranslation(const Translation3d& translation) {
82 m_pose = Pose3d{translation, m_pose.Rotation()};
83 }
84
85 /**
86 * Resets the rotation of the pose.
87 *
88 * @param rotation The rotation to reset to.
89 */
90 void ResetRotation(const Rotation3d& rotation) {
91 m_pose = Pose3d{m_pose.Translation(), rotation};
92 }
93
94 /**
95 * Returns the position of the robot on the field.
96 * @return The pose of the robot.
97 */
98 const Pose3d& GetPose() const { return m_pose; }
99
100 /**
101 * Updates the robot's position.
102 *
103 * @param xWheelPos The distance traveled by the forward-facing wheel, in
104 * meters.
105 * @param yWheelPos The distance traveled by the left-facing wheel, in meters.
106 * @param gyroAngle The angle reported by the gyroscope. This does not need to
107 * be offset to match the robot's orientation on the field.
108 * @return The updated pose.
109 */
110 const Pose3d& Update(wpi::units::meter_t xWheelPos,
111 wpi::units::meter_t yWheelPos,
112 const Rotation3d& gyroAngle) {
113 const auto deltaAngle = gyroAngle.RelativeTo(m_previousGyroAngle);
114 const auto angleDifference = deltaAngle.ToVector();
115
116 const auto deltaTheta = deltaAngle.ToRotation2d().Radians();
117 const auto deltaX =
118 xWheelPos - m_previousXWheelPos + m_xWheelYPos * deltaTheta / 1_rad;
119 const auto deltaY =
120 yWheelPos - m_previousYWheelPos - m_yWheelXPos * deltaTheta / 1_rad;
121 Twist3d twist{deltaX,
122 deltaY,
123 0_m,
124 wpi::units::radian_t{angleDifference(0)},
125 wpi::units::radian_t{angleDifference(1)},
126 wpi::units::radian_t{angleDifference(2)}};
127
128 m_pose = m_pose + twist.Exp();
129
130 m_previousXWheelPos = xWheelPos;
131 m_previousYWheelPos = yWheelPos;
132 m_previousGyroAngle = gyroAngle;
133
134 return m_pose;
135 }
136
137 /**
138 * Converts measured wheel velocities to chassis velocities using inverse
139 * kinematics.
140 *
141 * @param vx The velocity of the forward-facing wheel.
142 * @param vy The velocity of the sideways-facing wheel.
143 * @param omega The angular velocity of the robot as reported by the gyro.
144 * @return The velocity of the chassis.
145 */
146 ChassisVelocities ToChassisVelocities(
147 wpi::units::meters_per_second_t vx, wpi::units::meters_per_second_t vy,
148 wpi::units::radians_per_second_t omega) const {
149 return {vx + m_xWheelYPos * omega / 1_rad,
150 vy - m_yWheelXPos * omega / 1_rad, omega};
151 }
152
153 private:
154 const wpi::units::meter_t m_xWheelYPos;
155 const wpi::units::meter_t m_yWheelXPos;
156 Pose3d m_pose;
157
158 wpi::units::meter_t m_previousXWheelPos;
159 wpi::units::meter_t m_previousYWheelPos;
160
161 Rotation3d m_previousGyroAngle;
162};
