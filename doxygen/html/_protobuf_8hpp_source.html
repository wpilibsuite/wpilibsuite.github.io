59 * Generally used internally for decoding submessages
60 *
61 * @param[in] stream the nanopb istream
62 */
63 explicit ProtoInputStream(pb_istream_t* stream)
64 : m_streamMsg{stream},
65 m_msgDesc{
66 Protobuf<std::remove_cvref_t<T>>::MessageStruct::msg_descriptor()} {
67 }
68
69 /**
70 * Constructs a nanopb istream from a buffer.
71 *
72 * @param[in] stream the stream buffer
73 */
74 explicit ProtoInputStream(std::span<const uint8_t> stream)
75 : m_streamLocal{pb_istream_from_buffer(
76 reinterpret_cast<const pb_byte_t*>(stream.data()), stream.size())},
77 m_msgDesc{
78 Protobuf<std::remove_cvref_t<T>>::MessageStruct::msg_descriptor()} {
79 }
80
81 /**
82 * Gets the backing nanopb stream object.
83 *
84 * @return nanopb stream
85 */
86 pb_istream_t* Stream() noexcept {
87 return m_streamMsg ? m_streamMsg : &m_streamLocal;
88 }
89
90 /**
91 * Gets the nanopb message descriptor
92 *
93 * @return the nanopb message descriptor
94 */
95 const pb_msgdesc_t* MsgDesc() const noexcept { return m_msgDesc; }
96
97 /**
98 * Decodes a protobuf. Flags are the same flags passed to pb_decode_ex.
99 *
100 * @param[in] msg The message to decode into
101 * @param[in] flags Flags to pass
102 * @return true if decoding was successful, false otherwise
103 */
104 bool Decode(typename Protobuf<std::remove_cvref_t<T>>::MessageStruct& msg,
105 unsigned int flags = 0) {
106 return pb_decode_ex(Stream(), m_msgDesc, &msg, flags);
107 }
108
109 private:
110 pb_istream_t m_streamLocal;
111 pb_istream_t* m_streamMsg{nullptr};
112 const pb_msgdesc_t* m_msgDesc;
113};

183 * Gets the backing nanopb stream object.

193 * pb_encode_submessage instead of pb_encode
194 *
195 * @return true if submessage, false otherwise
196 */
197 bool IsSubmessage() const noexcept { return m_streamMsg; }
198
199 /**
200 * Gets the nanopb message descriptor
201 *
202 * @return the nanopb message descriptor
203 */
204 const pb_msgdesc_t* MsgDesc() const noexcept { return m_msgDesc; }
205
206 /**
207 * Decodes a protobuf. Flags are the same flags passed to pb_decode_ex.
208 *
209 * @param[in] msg The message to encode from
210 * @return true if encoding was successful, false otherwise
211 */
212 bool Encode(
213 const typename Protobuf<std::remove_cvref_t<T>>::MessageStruct& msg) {
214 if (m_streamMsg) {
215 return detail::WriteSubmessage(m_streamMsg, m_msgDesc, &msg);
216 // return pb_encode_submessage(m_streamMsg, m_msgDesc, &msg);
217 }
218 return pb_encode(&m_streamLocal, m_msgDesc, &msg);
219 }
220
221 private:
222 pb_ostream_t m_streamLocal;
223 pb_ostream_t* m_streamMsg{nullptr};
224 const pb_msgdesc_t* m_msgDesc;
225};

227/**
228 * Specifies that a type is capable of protobuf serialization and
229 * deserialization.
230 *
231 * This is designed for serializing complex flexible data structures using
232 * code generated from a .proto file. Serialization consists of writing
233 * values into a nanopb Stream and deserialization consists of
234 * reading values from nanopb Stream.
235 *
236 * Implementations must define a template specialization for wpi::util::Protobuf
237 * with T being the type that is being serialized/deserialized, with the
238 * following static members (as enforced by this concept):
239 * - using MessageStruct = nanopb_message_struct_here: typedef to the wpilib
240 * modified nanopb message struct
241 * - std::optional<T> Unpack(wpi::util::ProtoInputStream<T>&): function
242 * for deserialization
243 * - bool Pack(wpi::util::ProtoOutputStream<T>&, T&& value): function
244 * for serialization
245 *
246 * As a suggestion, 2 extra type usings can be added to simplify the stream
247 * definitions, however these are not required.
248 * - using InputStream = wpi::util::ProtoInputStream<T>;
249 * - using OutputStream = wpi::util::ProtoOutputStream<T>;
250 */
251template <typename T>
252concept ProtobufSerializable = requires(
253 wpi::util::ProtoOutputStream<std::remove_cvref_t<T>>& ostream,
254 wpi::util::ProtoInputStream<std::remove_cvref_t<T>>& istream,
255 const T& value) {
256 typename Protobuf<typename std::remove_cvref_t<T>>;
257 {
258 Protobuf<typename std::remove_cvref_t<T>>::Unpack(istream)
259 } -> std::same_as<std::optional<typename std::remove_cvref_t<T>>>;
260 {
261 Protobuf<typename std::remove_cvref_t<T>>::Pack(ostream, value)
262 } -> std::same_as<bool>;
263 typename Protobuf<typename std::remove_cvref_t<T>>::MessageStruct;
264 {
265 Protobuf<typename std::remove_cvref_t<T>>::MessageStruct::msg_descriptor()
266 } -> std::same_as<const pb_msgdesc_t*>;
267 {
268 Protobuf<typename std::remove_cvref_t<T>>::MessageStruct::msg_name()
269 } -> std::same_as<std::string_view>;
270 {
271 Protobuf<typename std::remove_cvref_t<T>>::MessageStruct::file_descriptor()
272 } -> std::same_as<pb_filedesc_t>;
273};

275/**
276 * Specifies that a type is capable of in-place protobuf deserialization.
277 *
278 * In addition to meeting ProtobufSerializable, implementations must define a
279 * wpi::util::Protobuf<T> static member -
280 * `bool UnpackInto(T*, wpi::util::ProtoInputStream<T>&)` to update the
281 * pointed-to T with the contents of the message.
282 */
283template <typename T>
284concept MutableProtobufSerializable =
285 ProtobufSerializable<T> &&
286 requires(T* out, wpi::util::ProtoInputStream<T>& istream) {
287 {
288 Protobuf<typename std::remove_cvref_t<T>>::UnpackInto(out, istream)
289 } -> std::same_as<bool>;
290 };

302/**
303 * Ease of use wrapper to make nanopb streams more opaque to the user.
304 * This class is stateless and thread safe.
305 *
306 * @tparam T serialized object type
307 */
308template <ProtobufSerializable T>
309class ProtobufMessage {
310 public:
311 /**
312 * Unpacks from a byte array.
313 *
314 * @param data byte array
315 * @return Optional; empty if parsing failed
316 */
317 std::optional<std::remove_cvref_t<T>> Unpack(std::span<const uint8_t> data) {
318 ProtoInputStream<std::remove_cvref_t<T>> stream{data};
319 return Protobuf<std::remove_cvref_t<T>>::Unpack(stream);
320 }
321
322 /**
323 * Unpacks from a byte array into an existing object.
324 *
325 * @param[out] out output object
326 * @param[in] data byte array
327 * @return true if successful
328 */
329 bool UnpackInto(T* out, std::span<const uint8_t> data) {
330 if constexpr (MutableProtobufSerializable<T>) {
331 ProtoInputStream<std::remove_cvref_t<T>> stream{data};
332 return Protobuf<std::remove_cvref_t<T>>::UnpackInto(out, stream);
333 } else {
334 auto unpacked = Unpack(data);
335 if (!unpacked) {
336 return false;
337 }
338 *out = std::move(unpacked.value());
339 return true;
340 }
341 }
342
343 /**
344 * Packs object into a SmallVector.
345 *
346 * @param[out] out output bytes
347 * @param[in] value value
348 * @return true if successful
349 */
350 bool Pack(wpi::util::SmallVectorImpl<uint8_t>& out, const T& value) {
351 ProtoOutputStream<std::remove_cvref_t<T>> stream{out};
352 return Protobuf<std::remove_cvref_t<T>>::Pack(stream, value);
353 }
354
355 /**
356 * Packs object into a std::vector.
357 *
358 * @param[out] out output bytes
359 * @param[in] value value
360 * @return true if successful
361 */
362 bool Pack(std::vector<uint8_t>& out, const T& value) {
363 ProtoOutputStream<std::remove_cvref_t<T>> stream{out};
364 return Protobuf<std::remove_cvref_t<T>>::Pack(stream, value);
365 }

367 /**
368 * Gets the type string for the message.
369 *
370 * @return type string
371 */
372 std::string GetTypeString() const {
373 return detail::GetTypeString(
374 Protobuf<std::remove_cvref_t<T>>::MessageStruct::msg_descriptor());
375 }
