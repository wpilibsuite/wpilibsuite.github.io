59/// Read a value of a particular endianness from memory.
60template <typename value_type, std::size_t alignment = unaligned>
61[[nodiscard]] inline value_type read(const void *memory, std::endian endian) {
62 value_type ret;
63
64 memcpy(static_cast<void *>(&ret),
65 LLVM_ASSUME_ALIGNED(
66 memory, (detail::PickAlignment<value_type, alignment>::value)),
67 sizeof(value_type));
68 return byte_swap<value_type>(ret, endian);
69}

125template <typename value_type>
126using make_unsigned_t = std::make_unsigned_t<value_type>;

128/// Read a value of a particular endianness from memory, for a location
129/// that starts at the given bit offset within the first byte.
130template <typename value_type, std::endian endian, std::size_t alignment>
131[[nodiscard]] inline value_type readAtBitAlignment(const void *memory,
132 uint64_t startBit) {
133 assert(startBit < 8);
134 if (startBit == 0)
135 return read<value_type, alignment>(memory, endian);
136 else {
137 // Read two values and compose the result from them.
138 value_type val[2];
139 memcpy(&val[0],
140 LLVM_ASSUME_ALIGNED(
141 memory, (detail::PickAlignment<value_type, alignment>::value)),
142 sizeof(value_type) * 2);
143 val[0] = byte_swap<value_type>(val[0], endian);
144 val[1] = byte_swap<value_type>(val[1], endian);
145
146 // Shift bits from the lower value into place.
147 make_unsigned_t<value_type> lowerVal = val[0] >> startBit;
148 // Mask off upper bits after right shift in case of signed type.
149 make_unsigned_t<value_type> numBitsFirstVal =
150 (sizeof(value_type) * 8) - startBit;
151 lowerVal &= ((make_unsigned_t<value_type>)1 << numBitsFirstVal) - 1;
152
153 // Get the bits from the upper value.
154 make_unsigned_t<value_type> upperVal =
155 val[1] & (((make_unsigned_t<value_type>)1 << startBit) - 1);
156 // Shift them in to place.
157 upperVal <<= numBitsFirstVal;
158
159 return lowerVal | upperVal;
160 }
161}

163/// Write a value to memory with a particular endianness, for a location
164/// that starts at the given bit offset within the first byte.
165template <typename value_type, std::endian endian, std::size_t alignment>
166inline void writeAtBitAlignment(void *memory, value_type value,
167 uint64_t startBit) {
168 assert(startBit < 8);
169 if (startBit == 0)
170 write<value_type, alignment>(memory, value, endian);
171 else {
172 // Read two values and shift the result into them.
173 value_type val[2];
174 memcpy(&val[0],
175 LLVM_ASSUME_ALIGNED(
176 memory, (detail::PickAlignment<value_type, alignment>::value)),
177 sizeof(value_type) * 2);
178 val[0] = byte_swap<value_type>(val[0], endian);
179 val[1] = byte_swap<value_type>(val[1], endian);
180
181 // Mask off any existing bits in the upper part of the lower value that
182 // we want to replace.
183 val[0] &= ((make_unsigned_t<value_type>)1 << startBit) - 1;
184 make_unsigned_t<value_type> numBitsFirstVal =
185 (sizeof(value_type) * 8) - startBit;
186 make_unsigned_t<value_type> lowerVal = value;
187 if (startBit > 0) {
188 // Mask off the upper bits in the new value that are not going to go into
189 // the lower value. This avoids a left shift of a negative value, which
190 // is undefined behavior.
191 lowerVal &= (((make_unsigned_t<value_type>)1 << numBitsFirstVal) - 1);
192 // Now shift the new bits into place
193 lowerVal <<= startBit;
194 }
195 val[0] |= lowerVal;
196
197 // Mask off any existing bits in the lower part of the upper value that
198 // we want to replace.
199 val[1] &= ~(((make_unsigned_t<value_type>)1 << startBit) - 1);
200 // Next shift the bits that go into the upper value into position.
201 make_unsigned_t<value_type> upperVal = value >> numBitsFirstVal;
202 // Mask off upper bits after right shift in case of signed type.
203 upperVal &= ((make_unsigned_t<value_type>)1 << startBit) - 1;
204 val[1] |= upperVal;
205
206 // Finally, rewrite values.
207 val[0] = byte_swap<value_type>(val[0], endian);
208 val[1] = byte_swap<value_type>(val[1], endian);
209 memcpy(LLVM_ASSUME_ALIGNED(
210 memory, (detail::PickAlignment<value_type, alignment>::value)),
211 &val[0], sizeof(value_type) * 2);
212 }
213}

219template <typename ValueType, std::endian Endian, std::size_t Alignment,
220 std::size_t ALIGN = PickAlignment<ValueType, Alignment>::value>
221struct packed_endian_specific_integral {
222 using value_type = ValueType;
223 static constexpr std::endian endian = Endian;
224 static constexpr std::size_t alignment = Alignment;
225
226 packed_endian_specific_integral() = default;
227
228 explicit packed_endian_specific_integral(value_type val) { *this = val; }
229
230 value_type value() const {
231 return endian::read<value_type, alignment>((const void *)Value.buffer,
232 endian);
233 }
234 operator value_type() const { return value(); }
235
236 void operator=(value_type newValue) {
237 endian::write<value_type, alignment>((void *)Value.buffer, newValue,
238 endian);
239 }
240
241 packed_endian_specific_integral &operator+=(value_type newValue) {
242 *this = *this + newValue;
243 return *this;
244 }
245
246 packed_endian_specific_integral &operator-=(value_type newValue) {
247 *this = *this - newValue;
248 return *this;
249 }
250
251 packed_endian_specific_integral &operator|=(value_type newValue) {
252 *this = *this | newValue;
253 return *this;
254 }
255
256 packed_endian_specific_integral &operator&=(value_type newValue) {
257 *this = *this & newValue;
258 return *this;
259 }
260
261private:
262 struct {
263 alignas(ALIGN) char buffer[sizeof(value_type)];
264 } Value;
265
266public:
267 struct ref {
268 explicit ref(void *Ptr) : Ptr(Ptr) {}
269
270 operator value_type() const {
271 return endian::read<value_type, alignment>(Ptr, endian);
272 }
273
274 void operator=(value_type NewValue) {
275 endian::write<value_type, alignment>(Ptr, NewValue, endian);
276 }
277
278 private:
279 void *Ptr;
280 };
281};
