23/** The base class for tunables. */
24class WPILIB_DLLEXPORT TunableBase {

31 constexpr explicit TunableBase(TunableTypeValue type)

41 constexpr TunableBase(TunableBase&& other) : m_uid{other.m_uid} {
42 if (IsRegisteredUid(m_uid)) {

50 if (IsRegisteredUid(m_uid)) {
51 SetTunableChanged();
52 } else {
53 m_uid = GetUnregisteredTypeUid(other.m_uid);
54 }
55 }
56 return *this;
57 }
58
59 constexpr TunableBase& operator=(TunableBase&& other) {
60 if (this != &other) {
61 if (IsRegisteredUid(m_uid)) {
62 SetTunableChanged();
63 if (IsRegisteredUid(other.m_uid)) {
64 other.SetTunableChanged();
65 }
66 } else {
67 m_uid = other.m_uid;
68 if (IsRegisteredUid(m_uid)) {
69 TunableRegistry::MoveTunable(m_uid & UID_MASK, this);
70 other.m_uid = GetUnregisteredTypeUid(m_uid);
71 }
72 }
73 }
74 return *this;
75 }
76
77 constexpr ~TunableBase() {
78 if (IsRegisteredUid(m_uid)) {
79 TunableRegistry::UnregisterTunable(m_uid & UID_MASK);
80 }
81 }
82
83 uint32_t GetTunableUid() const { return m_uid & UID_MASK; }
84
85 constexpr void SetTunableChanged() {
86 bool wasChanged = m_uid & CHANGE_FLAG;
87 m_uid |= CHANGE_FLAG;
88 if (!std::is_constant_evaluated() && IsRegisteredUid(m_uid) &&
89 !wasChanged) {
90 TunableRegistry::NotifyChanged(m_uid & UID_MASK);
91 }
92 }
93
94 private:
95 constexpr static uint32_t TYPE_FLAG = 0x80000000;
96 constexpr static uint32_t CHANGE_FLAG = 0x40000000;
97 constexpr static uint32_t UID_MASK = 0x3FFFFFFF;
98
99 constexpr static bool IsRegisteredUid(uint32_t uid) {
100 return (uid & TYPE_FLAG) == 0;
101 }
102
103 constexpr static uint32_t GetUnregisteredTypeUid(uint32_t uid) {
104 if (IsRegisteredUid(uid)) {
105 return ((uid & UID_MASK) >> 24) | TYPE_FLAG;
106 } else {
107 return uid;
108 }
109 }
110
111 bool GetTunableChanged() const { return m_uid & CHANGE_FLAG; }
112 void ResetTunableChanged() { m_uid &= ~CHANGE_FLAG; }
113
114 uint32_t m_uid;
115};
