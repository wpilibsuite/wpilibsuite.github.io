17/**
18 * Holds the constants for a DC motor.
19 */
20class WPILIB_DLLEXPORT DCMotor {
21 public:
22 using radians_per_second_per_volt_t = wpi::units::unit_t<
23 wpi::units::compound_unit<wpi::units::radians_per_second,
24 wpi::units::inverse<wpi::units::volt>>>;
25 using newton_meters_per_ampere_t =
26 wpi::units::unit_t<wpi::units::compound_unit<
27 wpi::units::newton_meters, wpi::units::inverse<wpi::units::ampere>>>;
28
29 /// Voltage at which the motor constants were measured.
30 wpi::units::volt_t nominalVoltage;
31
32 /// Torque when stalled.
33 wpi::units::newton_meter_t stallTorque;
34
35 /// Current draw when stalled.
36 wpi::units::ampere_t stallCurrent;
37
38 /// Current draw under no load.
39 wpi::units::ampere_t freeCurrent;
40
41 /// Angular velocity under no load.
42 wpi::units::radians_per_second_t freeSpeed;
43
44 /// Motor internal resistance.
45 wpi::units::ohm_t R;
46
47 /// Motor velocity constant.
48 radians_per_second_per_volt_t Kv;
49
50 /// Motor torque constant.
51 newton_meters_per_ampere_t Kt;
52
53 /**
54 * Constructs a DC motor.
55 *
56 * @param nominalVoltage Voltage at which the motor constants were measured.
57 * @param stallTorque Torque when stalled.
58 * @param stallCurrent Current draw when stalled.
59 * @param freeCurrent Current draw under no load.
60 * @param freeSpeed Angular velocity under no load.
61 * @param numMotors Number of motors in a gearbox.
62 */
63 constexpr DCMotor(wpi::units::volt_t nominalVoltage,
64 wpi::units::newton_meter_t stallTorque,
65 wpi::units::ampere_t stallCurrent,
66 wpi::units::ampere_t freeCurrent,
67 wpi::units::radians_per_second_t freeSpeed,
68 int numMotors = 1)
69 : nominalVoltage(nominalVoltage),
70 stallTorque(stallTorque * numMotors),
71 stallCurrent(stallCurrent * numMotors),
72 freeCurrent(freeCurrent * numMotors),
73 freeSpeed(freeSpeed),
74 R(nominalVoltage / this->stallCurrent),
75 Kv(freeSpeed / (nominalVoltage - R * this->freeCurrent)),
76 Kt(this->stallTorque / this->stallCurrent) {}
77
78 /**
79 * Returns current drawn by motor with given velocity and input voltage.
80 *
81 * @param velocity The current angular velocity of the motor.
82 * @param inputVoltage The voltage being applied to the motor.
83 */
84 constexpr wpi::units::ampere_t Current(
85 wpi::units::radians_per_second_t velocity,
86 wpi::units::volt_t inputVoltage) const {
87 return -1.0 / Kv / R * velocity + 1.0 / R * inputVoltage;
88 }
89
90 /**
91 * Returns current drawn by motor for a given torque.
92 *
93 * @param torque The torque produced by the motor.
94 */
95 constexpr wpi::units::ampere_t Current(
96 wpi::units::newton_meter_t torque) const {
97 return torque / Kt;
98 }
99
100 /**
101 * Returns torque produced by the motor with a given current.
102 *
103 * @param current The current drawn by the motor.
104 */
105 constexpr wpi::units::newton_meter_t Torque(
106 wpi::units::ampere_t current) const {
107 return current * Kt;
108 }
109
110 /**
111 * Returns the voltage provided to the motor for a given torque and
112 * angular velocity.
113 *
114 * @param torque The torque produced by the motor.
115 * @param velocity The current angular velocity of the motor.
116 */
117 constexpr wpi::units::volt_t Voltage(
118 wpi::units::newton_meter_t torque,
119 wpi::units::radians_per_second_t velocity) const {
120 return 1.0 / Kv * velocity + 1.0 / Kt * R * torque;
121 }
122
123 /**
124 * Returns the angular velocity produced by the motor at a given torque and
125 * input voltage.
126 *
127 * @param torque The torque produced by the motor.
128 * @param inputVoltage The input voltage provided to the motor.
129 */
130 constexpr wpi::units::radians_per_second_t Velocity(
131 wpi::units::newton_meter_t torque,
132 wpi::units::volt_t inputVoltage) const {
133 return inputVoltage * Kv - 1.0 / Kt * torque * R * Kv;
134 }
135
136 /**
137 * Returns a copy of this motor with the given gearbox reduction applied.
138 *
139 * @param gearboxReduction The gearbox reduction.
140 */
141 constexpr DCMotor WithReduction(double gearboxReduction) {
142 return DCMotor(nominalVoltage, stallTorque * gearboxReduction, stallCurrent,
143 freeCurrent, freeSpeed / gearboxReduction);
144 }
