15/**
16 * Analog input class.
17 *
18 * Connected to each analog channel is an averaging and oversampling engine.
19 * This engine accumulates the specified ( by SetAverageBits() and
20 * SetOversampleBits() ) number of samples before returning a new value. This is
21 * not a sliding window average. The only difference between the oversampled
22 * samples and the averaged samples is that the oversampled samples are simply
23 * accumulated effectively increasing the resolution, while the averaged samples
24 * are divided by the number of samples to retain the resolution, but get more
25 * stable values.
26 */
27class AnalogInput : public wpi::telemetry::TelemetryLoggable {
28 public:
29 /**
30 * Construct an analog input.
31 *
32 * @param channel The SmartIO channel to use.
33 */
34 explicit AnalogInput(int channel);
35
36 AnalogInput(AnalogInput&&) = default;
37 AnalogInput& operator=(AnalogInput&&) = default;
38
39 ~AnalogInput() override = default;
40
41 /**
42 * Get a sample straight from this channel.
43 *
44 * The sample is a 12-bit value representing the 0V to 3.3V range of the A/D
45 * converter in the module. The units are in A/D converter codes. Use
46 * GetVoltage() to get the analog value in calibrated units.
47 *
48 * @return A sample straight from this channel.
49 * @Common This is one of the commonly used methods for this class
50 */
51 int GetValue() const;
52
53 /**
54 * Get a scaled sample straight from this channel.
55 *
56 * The value is scaled to units of Volts.
57 *
58 * @return A scaled sample straight from this channel.
59 */
60 double GetVoltage() const;
61
62 /**
63 * Get the channel number.
64 *
65 * @return The channel number.
66 */
67 int GetChannel() const;
68
69 /**
70 * Indicates this input is used by a simulated device.
71 *
72 * @param device simulated device handle
73 */
74 void SetSimDevice(HAL_SimDeviceHandle device);
75
76 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
77
78 std::string_view GetTelemetryType() const override;
79
80 private:
81 int m_channel;
82 wpi::util::Handle<HAL_AnalogInputHandle, HAL_FreeAnalogInputPort> m_port;
83};
