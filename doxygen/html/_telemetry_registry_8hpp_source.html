101 /**
102 * Returns whether there is a data schema already registered with the given
103 * name. This does NOT perform a check as to whether the schema has already
104 * been published by another node on the network.
105 *
106 * @param backend Backend
107 * @param schemaName Name (the string passed as the data type for topics using
108 * this schema)
109 * @return True if schema already registered

111 static bool HasSchema(TelemetryBackend& backend, std::string_view schemaName);

113 /**
114 * Registers a data schema. Data schemas provide information for how a
115 * certain data type string can be decoded. The type string of a data schema
116 * indicates the type of the schema itself (e.g. "protobuf" for protobuf
117 * schemas, "struct" for struct schemas, etc). In NetworkTables, schemas are
118 * published just like normal topics, with the name being generated from the
119 * provided name: "/.schema/<name>". Duplicate calls to this function with
120 * the same name are silently ignored.
121 *
122 * @param backend Backend
123 * @param schemaName Name (the string passed as the data type for topics using
124 * this schema)
125 * @param type Type of schema (e.g. "protobuf", "struct", etc)
126 * @param schema Schema data
127 */
128 static void AddSchema(TelemetryBackend& backend, std::string_view schemaName,
129 std::string_view type, std::span<const uint8_t> schema);
130
131 /**
132 * Registers a data schema. Data schemas provide information for how a
133 * certain data type string can be decoded. The type string of a data schema
134 * indicates the type of the schema itself (e.g. "protobuf" for protobuf
135 * schemas, "struct" for struct schemas, etc). In NetworkTables, schemas are
136 * published just like normal topics, with the name being generated from the
137 * provided name: "/.schema/<name>". Duplicate calls to this function with
138 * the same name are silently ignored.
139 *
140 * @param backend Backend
141 * @param schemaName Name (the string passed as the data type for topics using
142 * this schema)
143 * @param type Type of schema (e.g. "protobuf", "struct", etc)
144 * @param schema Schema data
145 */
146 static void AddSchema(TelemetryBackend& backend, std::string_view schemaName,
147 std::string_view type, std::string_view schema);
148
149 /**
150 * Registers a protobuf schema. Duplicate calls to this function with the same
151 * name are silently ignored.
152 *
153 * @tparam T protobuf serializable type
154 * @param backend Backend
155 * @param msg protobuf message
156 */
157 template <wpi::util::ProtobufSerializable T>
158 static void AddProtobufSchema(TelemetryBackend& backend,
159 wpi::util::ProtobufMessage<T>& msg) {
160 msg.ForEachProtobufDescriptor(
161 [&](auto typeString) { return HasSchema(backend, typeString); },
162 [&](auto typeString, auto schema) {
163 AddSchema(backend, typeString, "proto:FileDescriptorProto", schema);
164 });
165 }
166
167 /**
168 * Registers a struct schema. Duplicate calls to this function with the same
169 * name are silently ignored.
170 *
171 * @tparam T struct serializable type
172 * @param backend Backend
173 * @param info optional struct type info
174 */
175 template <typename T, typename... I>
176 requires wpi::util::StructSerializable<T, I...>
177 static void AddStructSchema(TelemetryBackend& backend, const I&... info) {
178 wpi::util::ForEachStructSchema<T>(
179 [&](auto typeString, auto schema) {
180 AddSchema(backend, typeString, "structschema", schema);
181 },
182 info...);
183 }
184
185 private:
186 struct EntryHandle {
187 std::shared_ptr<TelemetryEntry> entry;
188 std::shared_ptr<TelemetryBackend> backend;
189 };
190
191 static void RecordKeepDuplicates(std::string_view path);
192
193 static void RecordProperty(std::string_view path, std::string_view key,

198 static EntryHandle GetEntryHandle(std::string_view path);
199
200 static bool HasNonDiscardDescendant(std::string_view tablePath);
201};
