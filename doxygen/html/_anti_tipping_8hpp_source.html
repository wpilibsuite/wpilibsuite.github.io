22/**
23 * AntiTipping provides a proportional correction system to prevent the robot
24 * from tipping over during operation.
25 *
26 * <p>It uses pitch and roll measurements to detect excessive inclination and
27 * computes a correction velocity in the opposite direction of the tilt. The
28 * resulting correction can be added to the robot's translational velocity to
29 * help stabilize it.
30 *
31 * <h2>Usage</h2>
32 *
33 * <ol>
34 * <li>Instantiate with initial configuration parameters.
35 * <li>Call Calculate() periodically (e.g. once per control loop).
36 * <li>Add the resulting correction to your drive command.
37 * </ol>
38 *
39 * <h2>Tuning</h2>
40 *
41 * <p>All three parameters depend on the robot's center of gravity and
42 * drivetrain, so they are best found empirically:
43 *
44 * <ul>
45 * <li>tippingThreshold: Drive the robot normally, including hard
46 * acceleration, braking, and turning, and record the largest pitch/roll
47 * magnitude observed. Set the threshold a few degrees above that worst
48 * case so normal driving does not trigger a correction, but well below
49 * the angle at which the robot actually tips.
50 * <li>kp: The correction speed is kp * sin(θ), so kp controls how
51 * aggressively the robot drives out from under a tilt. Start small and
52 * increase until recovery is brisk without overshooting or oscillating.
53 * <li>maxCorrectionSpeed: Cap the correction at a fraction of the
54 * drivetrain's maximum speed so the anti-tip response stays controllable
55 * and never overpowers the driver.
56 * </ul>
57 */
58class WPILIB_DLLEXPORT AntiTipping {
59 public:
60 /// Proportional gain unit: meters per second per radian of inclination.
61 using kp_unit =
62 wpi::units::compound_unit<wpi::units::meters_per_second,
63 wpi::units::inverse<wpi::units::radians>>;
64
65 /**
66 * Creates a new AntiTipping instance.
67 *
68 * @param kp The proportional coefficient in meters per second per radian.
69 * The P controller input is the sine of the inclination angle, and the
70 * output is in meters per second.
71 * @param tippingThreshold Tipping detection threshold.
72 * @param maxCorrectionSpeed Maximum correction velocity.
73 */
74 constexpr AntiTipping(wpi::units::unit_t<kp_unit> kp,
75 wpi::units::radian_t tippingThreshold,
76 wpi::units::meters_per_second_t maxCorrectionSpeed)
77 : m_kp{kp},
78 m_tippingThreshold{tippingThreshold},
79 m_maxCorrectionSpeed{maxCorrectionSpeed} {}
80

82 * Sets the proportional coefficient.
83 *
84 * @param kp The proportional coefficient in meters per second per radian.
85 */
86 constexpr void SetP(wpi::units::unit_t<kp_unit> kp) { m_kp = kp; }

89 * Gets the proportional coefficient.

92 */
93 constexpr wpi::units::unit_t<kp_unit> GetP() const { return m_kp; }

102 }
103

112

114 * Sets the maximum correction velocity.

116 * @param speed The maximum correction speed.

119 m_maxCorrectionSpeed = speed;
120 }
121
122 /**
123 * Gets the maximum correction velocity.
124 *
125 * @return The maximum correction speed.
126 */
127 constexpr wpi::units::meters_per_second_t GetMaxCorrectionSpeed() const {
128 return m_maxCorrectionSpeed;
129 }
130
131 /**
132 * Updates tipping detection and computes the proportional correction.
133 *
134 * @param attitude Current robot attitude as a Rotation3d.
135 * @return Correction ChassisVelocities to counteract tipping. Returns zeros
136 * if below threshold.
137 */
138 constexpr ChassisVelocities Calculate(const Rotation3d& attitude) const {
139 // To find the correction, we rotate the z axis (scaled by the P gain) by
140 // the attitude, then project onto the x-y plane.
141 Translation2d correction =
142 Translation3d{0_m, 0_m, wpi::units::meter_t{m_kp.value()}}
143 .RotateBy(attitude)
144 .ToTranslation2d();
145 wpi::units::meters_per_second_t speed{correction.Norm().value()};
146
147 // Let inclination angle of 3D correction be θ.
148 //
149 // _o_ +z
150 // \ | ^
151 // h \θ| |
152 // \| +x <--
153 //
154 // where o is length of 2D correction and h is length of 3D correction.
155 //
156 // sinθ = o/h
157 // θ = asin(speed / m_kp)
158 wpi::units::radian_t inclinationAngle{
159 gcem::asin(speed.value() / m_kp.value())};
160
161 if (inclinationAngle < m_tippingThreshold) {
162 return {};
163 } else if (speed > m_maxCorrectionSpeed) {
164 correction = correction * (m_maxCorrectionSpeed.value() / speed.value());
165 }
166
167 return {wpi::units::meters_per_second_t{correction.X().value()},
168 wpi::units::meters_per_second_t{correction.Y().value()},
169 0_rad_per_s};
170 }
171
172 private:
173 wpi::units::unit_t<kp_unit> m_kp;
174 wpi::units::radian_t m_tippingThreshold;
175 wpi::units::meters_per_second_t m_maxCorrectionSpeed;
176};
