18/**
19 * Class for mecanum drive odometry. Odometry allows you to track the robot's
20 * position on the field over a course of a match using readings from your
21 * mecanum wheel encoders.
22 *
23 * Teams can use odometry during the autonomous period for complex tasks like
24 * path following. Furthermore, odometry can be used for latency compensation
25 * when using computer-vision systems.
26 */
27class WPILIB_DLLEXPORT MecanumDriveOdometry3d
28 : public Odometry3d<MecanumDriveKinematics, MecanumDriveWheelPositions,
29 MecanumDriveWheelVelocities,
30 MecanumDriveWheelAccelerations> {
31 public:
32 /**
33 * Constructs a MecanumDriveOdometry3d object.
34 *
35 * @param kinematics The mecanum drive kinematics for your drivetrain.
36 * @param gyroAngle The angle reported by the gyroscope. This does not need to
37 * be offset to match the robot's orientation on the field.
38 * @param wheelPositions The current distances measured by each wheel.
39 * @param initialPose The starting position of the robot on the field.
40 */
41 explicit MecanumDriveOdometry3d(
42 MecanumDriveKinematics kinematics, const Rotation3d& gyroAngle,
43 const MecanumDriveWheelPositions& wheelPositions,
44 const Pose3d& initialPose = Pose3d{});
45};
46
