29// Parse a URL into its components.
30// @param url URL to parse
31// @return Parsed URL components or ada::errors if parsing failed
32inline ada::result<ada::url_aggregator> ParseUrl(std::string_view urlStr) {
33 // Handle origin-form request targets (e.g., "/nt/...")
34 // by parsing with a dummy base URL
35 ada::result<ada::url_aggregator> url;
36 if (wpi::util::starts_with(urlStr, '/')) {
37 auto baseUrl = ada::parse("http://localhost").value();
38 // Origin-form: parse with a dummy base
39 url = ada::parse(urlStr, &baseUrl);
40 } else {
41 // Absolute-form: parse directly
42 url = ada::parse(urlStr);
43 }
44 return url;
45}

91/**
92 * Class for HTTP path matching. A root path is represented as a single
93 * empty element, otherwise the path consists of each non-empty element
94 * between the '/' characters:
95 * - "" -> []
96 * - "/" -> [""]
97 * - "/foo" -> ["foo"]
98 * - "/foo/bar" -> ["foo", "bar"]
99 * - "/foo//bar/" -> ["foo", "bar"]
100 *
101 * All path elements are unescaped.
102 */
103class HttpPath {
104 public:
105 /**
106 * Constructs an empty HTTP path.
107 */
108 HttpPath() = default;
109
110 /**
111 * Constructs a HTTP path from an escaped path string. Makes a copy of the
112 * path, so it's safe to be a temporary.
113 */
114 explicit HttpPath(std::string_view path);
115
116 /**
117 * Evaluates to true if the path is not empty.
118 */
119 explicit operator bool() const { return !empty(); }
120
121 /**
122 * Returns true if the path has no elements.
123 */
124 bool empty() const { return m_pathEnds.empty(); }
125
126 /**
127 * Returns number of elements in the path.
128 */
129 size_t size() const { return m_pathEnds.size(); }
130
131 /**
132 * Returns true if the path exactly matches the provided match list.
133 *
134 * @param match match list
135 * @return True if path equals match list
136 */
137 bool equals(std::initializer_list<std::string_view> match) const {
138 return equals(0, {match.begin(), match.end()});
139 }
140 bool equals(std::span<const std::string_view> match) const {
141 return equals(0, match);
142 }
143 bool equals(std::string_view match) const { return equals(0, {match}); }
144
145 /**
146 * Returns true if the elements of the path starting at the "start" element
147 * match the provided match list, and there are no additional elements.
148 *
149 * @param start element to start matching at
150 * @param match match list
151 * @return True if match
152 */
153 bool equals(size_t start,
154 std::initializer_list<std::string_view> match) const {
155 return equals(start, {match.begin(), match.end()});
156 }
157 bool equals(size_t start, std::span<const std::string_view> match) const {
158 if (m_pathEnds.size() != (start + match.size())) {
159 return false;
160 }
161 return startswith(start, match);
162 }
163 bool equals(size_t start, std::string_view match) const {
164 return equals(start, {match});
165 }
166
167 /**
168 * Returns true if the first elements of the path match the provided match
169 * list. The path may have additional elements.
170 *
171 * @param match match list
172 * @return True if path starts with match list
173 */
174 bool startswith(std::initializer_list<std::string_view> match) const {
175 return startswith(0, {match.begin(), match.end()});
176 }
177 bool startswith(std::span<const std::string_view> match) const {
178 return startswith(0, match);
179 }
180 bool startswith(std::string_view match) const {
181 return startswith(0, {match});
182 }
183
184 /**
185 * Returns true if the elements of the path starting at the "start" element
186 * match the provided match list. The path may have additional elements.
187 *
188 * @param start element to start matching at
189 * @param match match list
190 * @return True if path starting at the start element matches the match list
191 */
192 bool startswith(size_t start,
193 std::initializer_list<std::string_view> match) const {
194 return startswith(start, {match.begin(), match.end()});
195 }
196
197 bool startswith(size_t start, std::span<const std::string_view> match) const;
198
199 bool startswith(size_t start, std::string_view match) const {
200 return startswith(start, {match});
201 }
202
203 /**
204 * Gets a single element of the path.
205 */
206 std::string_view operator[](size_t n) const;
207
208 /**
209 * Returns a path reference with the first N elements of the path removed.
210 */
211 HttpPathRef drop_front(size_t n) const;
212
213 private:
214 wpi::util::SmallString<128> m_pathBuf;
215 wpi::util::SmallVector<size_t, 16> m_pathEnds;
216};

299class HttpRequest {
300 public:
301 HttpRequest() = default;

303 explicit HttpRequest(const HttpLocation& loc)
304 : host{loc.host}, port{loc.port} {
305 SetPath(loc.path, loc.params);

308
309 template <typename T>
310 HttpRequest(const HttpLocation& loc, const T& extraParams)
311 : host{loc.host}, port{loc.port} {
312 wpi::util::SmallVector<std::pair<std::string_view, std::string_view>, 4>
313 params;
314 for (const auto& p : loc.params) {
315 params.emplace_back(std::pair{GetFirst(p), GetSecond(p)});
316 }
317 for (const auto& p : extraParams) {
318 params.emplace_back(std::pair{GetFirst(p), GetSecond(p)});
319 }
320 SetPath(loc.path, params);
321 SetAuth(loc);
322 }
323
324 HttpRequest(const HttpLocation& loc, std::string_view path_)
325 : host{loc.host}, port{loc.port}, path{path_} {
326 SetAuth(loc);
327 }
328
329 template <typename T>
330 HttpRequest(const HttpLocation& loc, std::string_view path_, const T& params)
331 : host{loc.host}, port{loc.port} {
332 SetPath(path_, params);
333 SetAuth(loc);
334 }
335
336 wpi::util::SmallString<128> host;
337 int port;
338 std::string auth;
339 wpi::util::SmallString<128> path;
340
341 private:
342 void SetAuth(const HttpLocation& loc);
343
344 template <typename T>
345 void SetPath(std::string_view path_, const T& params) {
346 // Build location including query string
347 wpi::util::raw_svector_ostream pathOs{path};
348 pathOs << path_;
349 bool first = true;
350 for (const auto& param : params) {
351 if (first) {
352 pathOs << '?';
353 first = false;
354 } else {
355 pathOs << '&';
356 }
357 wpi::util::SmallString<64> escapeBuf;
358 pathOs << EscapeURI(GetFirst(param), escapeBuf, false);
359 if (!GetSecond(param).empty()) {
360 pathOs << '=' << EscapeURI(GetSecond(param), escapeBuf, false);
361 }
362 }
363 }
364
365 template <typename T>
366 static std::string_view GetFirst(const T& elem) {
367 return elem.first;
368 }
369 template <typename T>
370 static std::string_view GetFirst(const std::pair<std::string, T>& elem) {
371 return elem.first;
372 }
373 template <typename T>
374 static std::string_view GetSecond(const T& elem) {
375 return elem.second;
376 }
377};
