19/**
20 * Use a rate gyro to return the robots heading relative to a starting position.
21 *
22 * This class is for the Romi onboard gyro, and will only work in
23 * simulation/Romi mode. Only one instance of a RomiGyro is supported.
24 */
25class RomiGyro : public wpi::telemetry::TelemetryLoggable {
26 public:
27 RomiGyro();
28
29 /**
30 * Return the actual angle in radians that the robot is currently facing.
31 *
32 * The angle is based on integration of the returned rate form the gyro.
33 * The angle is continuous, that is, it will continue from 2π->3π radians.
34 * This allows algorithms that wouldn't want to see a discontinuity in the
35 * gyro output as it sweeps from 2π to 0 on the second time around.
36 *
37 * @return The current heading of the robot.
38 */
39 wpi::units::radian_t GetAngle() const;
40
41 /**
42 * Return the rate of rotation of the gyro
43 *
44 * The rate is based on the most recent reading of the gyro.
45 *
46 * @return The current rate.
47 */
48 wpi::units::radians_per_second_t GetRate() const;
49
50 /**
51 * Get the rate of turn in around the X-axis.
52 *
53 * @return Rate of turn.
54 */
55 wpi::units::radians_per_second_t GetRateX() const;
56
57 /**
58 * Get the rate of turn in around the Y-axis.
59 *
60 * @return Rate of turn.
61 */
62 wpi::units::radians_per_second_t GetRateY() const;
63
64 /**
65 * Get the rate of turn around the Z-axis.
66 *
67 * @return Rate of turn.
68 */
69 wpi::units::radians_per_second_t GetRateZ() const;
70
71 /**
72 * Get the currently reported angle around the X-axis.
73 *
74 * @return Current angle around X-axis.
75 */
76 wpi::units::radian_t GetAngleX() const;
77
78 /**
79 * Get the currently reported angle around the Y-axis.
80 *
81 * @return Current angle around Y-axis.
82 */
83 wpi::units::radian_t GetAngleY() const;
84
85 /**
86 * Get the currently reported angle around the Z-axis.
87 *
88 * @return Current angle around Z-axis.
89 */
90 wpi::units::radian_t GetAngleZ() const;
91
92 /**
93 * Resets the gyro
94 */
95 void Reset();
96
97 std::string_view GetTelemetryType() const override;
98 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
99
100 private:
101 hal::SimDevice m_simDevice;
102 hal::SimDouble m_simRateX;
103 hal::SimDouble m_simRateY;
104 hal::SimDouble m_simRateZ;
105 hal::SimDouble m_simAngleX;
106 hal::SimDouble m_simAngleY;
107 hal::SimDouble m_simAngleZ;
108
109 double m_angleXOffset = 0;
110 double m_angleYOffset = 0;
111 double m_angleZOffset = 0;
112};
