27/**
28 * @brief Enumeration of URL scheme types.
29 *
30 * Special schemes have specific parsing rules and default ports.
31 * Using an enum allows efficient scheme comparisons without string operations.
32 *
33 * Default ports:
34 * - HTTP: 80
35 * - HTTPS: 443
36 * - WS: 80
37 * - WSS: 443
38 * - FTP: 21
39 * - FILE: (none)
40 */
41enum type : uint8_t {
42 HTTP = 0, /**< http:// scheme (port 80) */
43 NOT_SPECIAL = 1, /**< Non-special scheme (no default port) */
44 HTTPS = 2, /**< https:// scheme (port 443) */
45 WS = 3, /**< ws:// WebSocket scheme (port 80) */
46 FTP = 4, /**< ftp:// scheme (port 21) */
47 WSS = 5, /**< wss:// secure WebSocket scheme (port 443) */
48 FILE = 6 /**< file:// scheme (no default port) */
49};
