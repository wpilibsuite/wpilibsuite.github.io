21class HttpServerConnection {
22 public:
23 explicit HttpServerConnection(std::shared_ptr<uv::Stream> stream,
24 uv::Timer::Time requestTimeout = {});
25 virtual ~HttpServerConnection();
26
27 protected:
28 /**
29 * Process an incoming HTTP request. This is called after the incoming
30 * message completes (e.g. from the HttpParser::messageComplete callback).
31 *
32 * The implementation should read request details from m_request and call the
33 * appropriate Send() functions to send a response back to the client.
34 */
35 virtual void ProcessRequest() = 0;
36
37 /** Cancel the timeout for receiving the initial request. */
38 void CancelRequestTimeout();
39
40 /**
41 * Build common response headers.
42 *
43 * Called by SendHeader() to send headers common to every response.
44 * Each line must be terminated with "\r\n".
45 *
46 * The default implementation sends the following:
47 * "Server: WebServer/1.0\r\n"
48 * "Cache-Control: no-store, no-cache, must-revalidate, pre-check=0, "
49 * "post-check=0, max-age=0\r\n"
50 * "Pragma: no-cache\r\n"
51 * "Expires: Mon, 3 Jan 2000 12:34:56 GMT\r\n"
52 *
53 * These parameters should ensure the browser does not cache the response.
54 * A browser should connect for each file and not serve files from its cache.
55 *
56 * @param os response stream
57 */
58 virtual void BuildCommonHeaders(wpi::util::raw_ostream& os);
59
60 /**
61 * Build HTTP response header, along with other header information like
62 * mimetype. Calls BuildCommonHeaders().
63 *
64 * @param os response stream
65 * @param code HTTP response code (e.g. 200)
66 * @param codeText HTTP response code text (e.g. "OK")
67 * @param contentType MIME content type (e.g. "text/plain")
68 * @param contentLength Length of content. If 0 is provided, m_keepAlive will
69 * be set to false.
70 * @param extra Extra HTTP headers to send, including final "\r\n"
71 */
72 virtual void BuildHeader(wpi::util::raw_ostream& os, int code,
73 std::string_view codeText,
74 std::string_view contentType, uint64_t contentLength,
75 std::string_view extra = {});
76
77 /**
78 * Send data to client.
79 *
80 * This is a convenience wrapper around m_stream.Write() to provide
81 * auto-close functionality.
82 *
83 * @param bufs Buffers to write. Deallocate() will be called on each
84 * buffer after the write completes. If different behavior
85 * is desired, call m_stream.Write() directly instead.
86 * @param closeAfter close the connection after the write completes
87 */
88 void SendData(std::span<const uv::Buffer> bufs, bool closeAfter = false);
89
90 /**
91 * Send HTTP response, along with other header information like mimetype.
92 * Calls BuildHeader().
93 *
94 * @param code HTTP response code (e.g. 200)
95 * @param codeText HTTP response code text (e.g. "OK")
96 * @param contentType MIME content type (e.g. "text/plain")
97 * @param content Response message content
98 * @param extraHeader Extra HTTP headers to send, including final "\r\n"
99 */
100 virtual void SendResponse(int code, std::string_view codeText,
101 std::string_view contentType,
102 std::string_view content,
103 std::string_view extraHeader = {});
104
105 /**
106 * Send HTTP response from static data, along with other header information
107 * like mimetype. Calls BuildHeader(). Supports gzip pre-compressed data
108 * (it will decompress if the client does not accept gzip encoded data).
109 * Unlike SendResponse(), content is not copied and its contents must remain
110 * valid for an unspecified lifetime.
111 *
112 * @param code HTTP response code (e.g. 200)
113 * @param codeText HTTP response code text (e.g. "OK")
114 * @param contentType MIME content type (e.g. "text/plain")
115 * @param content Response message content
116 * @param gzipped True if content is gzip compressed
117 * @param extraHeader Extra HTTP headers to send, including final "\r\n"
118 */
119 virtual void SendStaticResponse(int code, std::string_view codeText,
120 std::string_view contentType,
121 std::string_view content, bool gzipped,
122 std::string_view extraHeader = {});
123
124 /**
125 * Send error header and message.
126 * This provides standard code responses for 400, 401, 403, 404, 500, and 503.
127 * Other codes will be reported as 501. For arbitrary code handling, use
128 * SendResponse() instead.
129 *
130 * @param code HTTP error code (e.g. 404)
131 * @param message Additional message text
132 */
133 virtual void SendError(int code, std::string_view message = {});
134
135 /** The HTTP request. */
136 HttpParser m_request{HttpParser::Type::REQUEST};
137
138 /** Whether the connection should be kept alive. */
139 bool m_keepAlive = false;
140
141 /** If gzip is an acceptable encoding for responses. */
142 bool m_acceptGzip = false;
143
144 /** The underlying stream for the connection. */
145 uv::Stream& m_stream;
146
147 /** The header reader connection. */
148 wpi::util::sig::ScopedConnection m_dataConn;
149
150 /** The end stream connection. */
151 wpi::util::sig::ScopedConnection m_endConn;
152
153 /** The message complete connection. */
154 wpi::util::sig::Connection m_messageCompleteConn;
155
156 private:
157 std::shared_ptr<uv::Timer> m_requestTimer;
158};
159
160} // namespace wpi::net
