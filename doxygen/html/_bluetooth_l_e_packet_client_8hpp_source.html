102/**
103 * Packet-oriented Bluetooth LE client.
104 *
105 * Each Send() call writes one packet using the best available platform
106 * transport. Each packet callback receives one packet from a GATT notification
107 * value or L2CAP SDU.
108 */
109class BluetoothLEPacketClient {
110 public:
111 using PacketCallback = std::function<void(std::span<const uint8_t> packet)>;
112 using StatusCallback =
113 std::function<void(const BluetoothLEPacketConnectionStatus& status)>;
114
115 /**
116 * Creates a Bluetooth LE packet client.
117 *
118 * @param loop libuv loop used to marshal callbacks.
119 * @param packetCallback callback invoked for received packets.
120 * @param statusCallback callback invoked when connection status changes.
121 * @return client instance, or nullptr if the loop callback channel could not
122 * be created.
123 */
124 static std::shared_ptr<BluetoothLEPacketClient> Create(
125 wpi::net::uv::Loop& loop, PacketCallback packetCallback,
126 StatusCallback statusCallback = {});
127
128 /**
129 * Destroys the client and closes any active connection.
130 */
131 ~BluetoothLEPacketClient();
132
133 BluetoothLEPacketClient(const BluetoothLEPacketClient&) = delete;
134 BluetoothLEPacketClient& operator=(const BluetoothLEPacketClient&) = delete;
135
136 /**
137 * Returns whether this platform implementation supports explicit pairing.
138 *
139 * @return true if PairDevice() can pair devices on this platform.
140 */
141 static bool IsPairingSupported();
142
143 /**
144 * Scans for Bluetooth LE devices.
145 *
146 * @param timeout active scan duration. A zero duration returns cached devices
147 * on platforms that support cached discovery.
148 * @return scan result, including discovered connection targets.
149 */
150 static BluetoothLEDeviceScanResult ScanDevices(
151 std::chrono::milliseconds timeout);
152
153 /**
154 * Attempts to pair with a Bluetooth LE device.
155 *
156 * @param target platform-specific device target from ScanDevices().
157 * @return pairing result.
158 */
159 static BluetoothLEPairingResult PairDevice(std::string_view target);
160
161 /**
162 * Starts connecting to a Bluetooth LE packet transport.
163 *
164 * @param config connection configuration.
165 * @return true if the request was accepted.
166 */
167 bool Connect(BluetoothLEPacketClientConfig config);
168
169 /**
170 * Disconnects the active connection, if any.
171 *
172 * @param reason status text to publish after disconnecting.
173 */
174 void Disconnect(std::string_view reason = "Disconnected");
175
176 /**
177 * Sends one packet.
178 *
179 * @param packet packet payload to send.
180 * @param mode BEST_EFFORT may drop the packet if the transport is busy.
181 * QUEUED retains a single pending packet until it can be sent; newer
182 * packets cannot overtake it. Neither mode acknowledges peer receipt.
183 * @return true if the packet was accepted for sending; false if disconnected,
184 * the packet is too large, or the transport's send queue is full.
185 */
186 bool Send(
187 std::span<const uint8_t> packet,
188 BluetoothPacketSendMode mode = BluetoothPacketSendMode::BEST_EFFORT);
189
190 /**
191 * Gets the current connection status.
192 *
193 * @return current connection status.
194 */
195 BluetoothLEPacketConnectionStatus GetStatus() const;
196
197 private:
198 class Impl;
199
200 explicit BluetoothLEPacketClient(std::shared_ptr<Impl> impl);
201
202 std::shared_ptr<Impl> m_impl;
203};
