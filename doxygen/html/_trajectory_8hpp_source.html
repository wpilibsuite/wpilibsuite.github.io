26/**
27 * Represents a trajectory consisting of a list of samples,
28 * kinematically interpolating between them.
29 *
30 * @tparam SampleType The type of sample (e.g., SplineSample,
31 * DifferentialSample)
32 */
33template <typename SampleType>
34class Trajectory {
35 public:
36 /**
37 * Constructs a Trajectory from a vector of samples.
38 *
39 * @param samples The samples of the trajectory. Order does not matter as
40 * they will be sorted internally.
41 * @throws std::invalid_argument if the vector of samples is empty.
42 */
43 explicit Trajectory(std::vector<SampleType> samples) {
44 if (samples.empty()) {
45 throw std::invalid_argument(
46 "Trajectory manually initialized with no samples.");
47 }
48
49 // Sort samples by time
50 std::sort(samples.begin(), samples.end(),
51 [](const auto& a, const auto& b) { return a.time < b.time; });
52
53 m_samples = std::move(samples);
54
55 // Build interpolating map
56 for (const auto& sample : m_samples) {
57 m_sampleMap[sample.time] = sample;
58 }
59
60 m_duration = m_samples.back().time;
61 }
62
63 /**
64 * Returns the overall duration of the trajectory.
65 *
66 * @return The duration of the trajectory.
67 */
68 wpi::units::second_t Duration() const { return m_duration; }
69
70 /**
71 * Returns the samples of the trajectory.
72 *
73 * @return The samples of the trajectory.
74 */
75 const std::vector<SampleType>& Samples() const { return m_samples; }
76
77 /**
78 * Returns the first sample in the trajectory.
79 *
80 * @return The first sample.
81 */
82 const SampleType& Start() const { return m_samples.front(); }
83
84 /**
85 * Returns the last sample in the trajectory.
86 *
87 * @return The last sample.
88 */
89 const SampleType& End() const { return m_samples.back(); }
90
91 /**
92 * Sample the trajectory at a point in time.
93 *
94 * @param t The point in time since the beginning of the trajectory to sample.
95 * @return The sample at that point in time.
96 * @throws std::runtime_error if the trajectory has no samples.
97 */
98 SampleType SampleAt(wpi::units::second_t t) const {
99 if (m_samples.empty()) {
100 throw std::runtime_error(
101 "Trajectory cannot be sampled if it has no samples.");
102 }
103
104 if (t <= m_samples.front().time) {
105 return m_samples.front();
106 }
107 if (t >= m_duration) {
108 return m_samples.back();
109 }
110
111 // Find the two samples to interpolate between
112 auto upper = m_sampleMap.upper_bound(t);
113 if (upper == m_sampleMap.begin()) {
114 return upper->second;
115 }
116
117 auto lower = std::prev(upper);
118
119 // Calculate interpolation parameter
120 const double t_param = (t - lower->first) / (upper->first - lower->first);
121
122 // Use derived class's interpolation (runtime polymorphism)
123 return Interpolate(lower->second, upper->second, t_param);
124 }
125
126 /**
127 * Sample the trajectory at a point in time.
128 *
129 * @param t The point in time since the beginning of the trajectory to sample
130 * (in seconds).
131 * @return The sample at that point in time.
132 */
133 SampleType SampleAt(double t) const {
134 return SampleAt(wpi::units::second_t{t});
135 }
136
137 /**
138 * Interpolates between two samples. This method must be implemented by
139 * subclasses to provide drivetrain-specific interpolation logic.
140 *
141 * @param start The starting sample.
142 * @param end The ending sample.
143 * @param t The interpolation parameter between 0 and 1.
144 * @return The interpolated sample.
145 */
146 virtual SampleType Interpolate(const SampleType& start, const SampleType& end,
147 double t) const = 0;
148
149 /**
150 * Returns the initial pose of the trajectory.
151 *
152 * @return The initial pose of the trajectory.
153 */
154 Pose2d InitialPose() const { return m_samples.front().pose; }
155
156 /**
157 * Checks equality between this Trajectory and another object.
158 *
159 * @return True if the trajectories are equal.
160 */
161 bool operator==(const Trajectory& other) const {
162 return m_samples == other.m_samples;
163 }
164
165 virtual ~Trajectory() = default;
166
167 protected:
168 /**
169 * Returns all samples transformed to be relative to the given pose.
170 *
171 * @param pose The pose to make samples relative to.
172 * @return A vector of samples relative to the given pose.
173 */
174 std::vector<SampleType> RelativeSamples(const Pose2d& pose) const {
175 std::vector<SampleType> out;
176 out.reserve(m_samples.size());
177 for (const auto& s : m_samples) {
178 out.emplace_back(s.RelativeTo(pose));
179 }
180 return out;
181 }
182
183 /**
184 * Returns the concatenation of this trajectory's samples with another list,
185 * offsetting the other samples' times by this trajectory's duration.
186 *
187 * @param other The samples to concatenate.
188 * @return A vector containing all samples in order.
189 */
190 std::vector<SampleType> ConcatenateSamples(
191 const std::vector<SampleType>& other) const {
192 std::vector<SampleType> out;
193 out.reserve(m_samples.size() + other.size());
194 // copy existing
195 out.insert(out.end(), m_samples.begin(), m_samples.end());
196 // append other with time offset
197 for (auto s : other) {
198 s.time += m_duration;
199 out.emplace_back(std::move(s));
200 }
201 return out;
202 }
203
204 std::vector<SampleType> m_samples;
205 std::map<wpi::units::second_t, SampleType> m_sampleMap;
206 wpi::units::second_t m_duration{0};
207};
208
