25/**
26 * Represents a trajectory consisting of a list of samples,
27 * kinematically interpolating between them.
28 *
29 * @tparam SampleType The type of sample (e.g., SplineSample,
30 * DifferentialSample)
31 */
32template <TrajectorySample SampleType>
33class Trajectory {
34 public:
35 /**
36 * Constructs a Trajectory from a vector of samples.
37 *
38 * @param samples The samples of the trajectory. Order does not matter as
39 * they will be sorted internally.
40 * @throws std::invalid_argument if the vector of samples is empty or if the
41 * earliest sample's time is not zero.
42 */
43 explicit Trajectory(std::vector<SampleType> samples)
44 : m_samples(std::move(samples)) {
45 if (m_samples.empty()) {
46 throw std::invalid_argument(
47 "Trajectory manually initialized with no samples.");
48 }
49
50 // Sort samples by time
51 std::ranges::stable_sort(m_samples, {}, &SampleType::time);
52
53 if (Start().time != 0.0_s) {
54 throw std::invalid_argument(
55 "Trajectory sample times must be relative to the trajectory start "
56 "(the first sample must have time 0).");
57 }
58 }
59
60 /**
61 * Returns the overall duration of the trajectory.
62 *
63 * @return The duration of the trajectory.
64 */
65 wpi::units::second_t Duration() const { return End().time; }
66
67 /**
68 * Returns the samples of the trajectory.
69 *
70 * @return The samples of the trajectory.
71 */
72 const std::vector<SampleType>& Samples() const { return m_samples; }
73
74 /**
75 * Returns the first sample in the trajectory.
76 *
77 * @return The first sample.
78 */
79 const SampleType& Start() const { return m_samples.front(); }
80
81 /**
82 * Returns the last sample in the trajectory.
83 *
84 * @return The last sample.
85 */
86 const SampleType& End() const { return m_samples.back(); }
87
88 /**
89 * Sample the trajectory at a point in time.
90 *
91 * If several samples share a timestamp, such as at the join of a
92 * concatenated trajectory, the last one at that time is returned. Times at or
93 * before the start return the first sample, and times at or after the end
94 * return the last sample.
95 *
96 * @param t The point in time since the beginning of the trajectory to sample.
97 * @return The sample at that point in time.
98 */
99 SampleType SampleAt(wpi::units::second_t t) const {
100 if (t <= Start().time) {
101 return Start();
102 }
103 if (t >= End().time) {
104 return End();
105 }
106
107 // upper_bound is past every sample at time t, so the sample before it is
108 // the last one at or before t.
109 auto upper = std::ranges::upper_bound(m_samples, t, {}, &SampleType::time);
110 if (upper == m_samples.end()) {
111 // Only reachable if t is NaN, for which every comparison is false
112 return End();
113 }
114 auto lower = std::prev(upper);
115
116 // Return a stored sample as is instead of interpolating to the end of the
117 // previous segment, which isn't well defined for zero-length segments
118 if (lower->time == t) {
119 return *lower;
120 }
121
122 const double t_param = (t - lower->time) / (upper->time - lower->time);
123
124 return Interpolate(*lower, *upper, t_param);
125 }
126
127 /**
128 * Interpolates between two samples. This method must be implemented by
129 * subclasses to provide drivetrain-specific interpolation logic.
130 *
131 * @param start The starting sample.
132 * @param end The ending sample.
133 * @param t The interpolation parameter between 0 and 1.
134 * @return The interpolated sample.
135 */
136 virtual SampleType Interpolate(const SampleType& start, const SampleType& end,
137 double t) const = 0;
138
139 /**
140 * Checks equality between this Trajectory and another object.
141 *
142 * @return True if the trajectories are equal.
143 */
144 bool operator==(const Trajectory& other) const {
145 return m_samples == other.m_samples;
146 }
147
148 virtual ~Trajectory() = default;
149
150 protected:
151 std::vector<SampleType> m_samples;
152};
