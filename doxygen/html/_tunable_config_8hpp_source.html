21/** Additional optional configuration for tunables. */
22struct TunableConfig {
23 /** Defines when the backend should poll the tunable's local value. */
24 enum class Polling {
25 /** Use the framework default polling behavior. */
26 DEFAULT,
27
28 /** Poll the tunable's local value on every loop. */
29 ALWAYS_GET,
30
31 /** Poll the tunable's local value only when it is marked changed. */
32 GET_ON_CHANGE
33 };
34
35 /**
36 * Creates a config where the tunable gets its value only when it is changed.
37 *
38 * @return TunableConfig
39 */
40 static TunableConfig GetOnChange() {
41 TunableConfig config;
42 config.polling = Polling::GET_ON_CHANGE;
43 return config;
44 }
45
46 /**
47 * Creates a config where the tunable always gets its value on every loop.
48 *
49 * @return TunableConfig
50 */
51 static TunableConfig AlwaysGet() {
52 TunableConfig config;
53 config.polling = Polling::ALWAYS_GET;
54 return config;
55 }
56
57 /** Additional properties for the tunable. */
58 wpi::util::json properties = wpi::util::json::object();
59
60 /**
61 * Indicates the tunable should separately echo back its new value after being
62 * set.
63 */
64 bool robust = false;
65
66 /** The type string for the tunable. */
67 std::optional<std::string> typeString{};
68
69 /**
70 * Indicates whether the tunable is mutable. Setting this to
71 * false makes the tunable immutable, meaning that the Tunable's Set()
72 * function will not be called. This is useful for tunables that should only
73 * be changed locally, such as ones that represent a constant or a tunable
74 * that is only meant to be changed programmatically.
75 */
76 bool isMutable = true;
77
78 /**
79 * Callback to call when the tunable is tuned. This is called immediately
80 * after the tunable's Tune() function is called. The callback will be called
81 * by TunableRegistry::Update().
82 *
83 * Note that it is not safe to this-capture the tunable itself in the
84 * callback, as the tunable may be moved. Instead, use the callback parameter
85 * to access the tunable.
86 *
87 * Similarly, if the tunable is a member of a ComplexTunable, the
88 * ComplexTunable this pointer should not be captured by the
89 * callback--instead, set the parent value and it will be move-tracked and
90 * passed to the callback via the parent parameter.
91 *
92 * This callback must not throw. If an exception escapes, registry and
93 * backend state is not guaranteed to be restored.
94 */
95 std::function<void(detail::TunableBase& tunable, ComplexTunable* parent)>
96 onTune{};
97
98 /**
99 * Callback to call after a backend applies a remote value, but before the
100 * backend publishes the resulting value. This is intended for language
101 * bindings that need to reconcile getter/setter-backed values before a robust
102 * backend echoes the tune. User code should prefer onTune. This callback
103 * must not throw.
104 */
105 std::function<void(detail::TunableBase& tunable, ComplexTunable* parent)>
106 onRemoteSet{};
107
108 /**
109 * Parent object to pass to callback functions. Note since this is
110 * move-tracked by the framework, the pointer passed to a callback
111 * might not be the same pointer provided here.
112 */
113 ComplexTunable* parent = nullptr;
114
115 /**
116 * Defines when the backend should poll the tunable's local value.
117 */
118 Polling polling = Polling::DEFAULT;
119};
120
121} // namespace wpi::tunables
