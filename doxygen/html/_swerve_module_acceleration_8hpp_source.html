14/**
15 * Represents the accelerations of one swerve module.
16 */
17struct WPILIB_DLLEXPORT SwerveModuleAcceleration {
18 /**
19 * Acceleration of the wheel of the module.
20 */
21 units::meters_per_second_squared_t acceleration = 0_mps_sq;
22
23 /**
24 * Angle of the acceleration vector.
25 */
26 Rotation2d angle;
27
28 /**
29 * Checks equality between this SwerveModuleAccelerations and another object.
30 *
31 * @param other The other object.
32 * @return Whether the two objects are equal.
33 */
34 constexpr bool operator==(const SwerveModuleAcceleration& other) const {
35 return units::math::abs(acceleration - other.acceleration) < 1E-9_mps_sq &&
36 angle == other.angle;
37 }
38
39 /**
40 * Adds two SwerveModuleAccelerations and returns the sum.
41 * Note: This performs vector addition of the accelerations.
42 *
43 * @param other The SwerveModuleAccelerations to add.
44 * @return The sum of the SwerveModuleAccelerations.
45 */
46 SwerveModuleAcceleration operator+(
47 const SwerveModuleAcceleration& other) const {
48 // Convert to Cartesian coordinates, add, then convert back
49 auto thisX = acceleration * angle.Cos();
50 auto thisY = acceleration * angle.Sin();
51 auto otherX = other.acceleration * other.angle.Cos();
52 auto otherY = other.acceleration * other.angle.Sin();
53
54 auto sumX = thisX + otherX;
55 auto sumY = thisY + otherY;
56
57 auto resultAcceleration = units::math::hypot(sumX, sumY);
58 auto resultAngle = Rotation2d{sumX.value(), sumY.value()};
59
60 return {resultAcceleration, resultAngle};
61 }
62
63 /**
64 * Subtracts the other SwerveModuleAccelerations from the current
65 * SwerveModuleAccelerations and returns the difference.
66 *
67 * @param other The SwerveModuleAccelerations to subtract.
68 * @return The difference between the two SwerveModuleAccelerations.
69 */
70 SwerveModuleAcceleration operator-(
71 const SwerveModuleAcceleration& other) const {
72 return *this + (-other);
73 }
74
75 /**
76 * Returns the inverse of the current SwerveModuleAccelerations.
77 * This is equivalent to rotating the acceleration by pi (or 180 degrees).
78 *
79 * @return The inverse of the current SwerveModuleAccelerations.
80 */
81 constexpr SwerveModuleAcceleration operator-() const {
82 return {acceleration, angle + Rotation2d{180_deg}};
83 }
84
85 /**
86 * Multiplies the SwerveModuleAccelerations by a scalar and returns the new
87 * SwerveModuleAccelerations.
88 *
89 * @param scalar The scalar to multiply by.
90 * @return The scaled SwerveModuleAccelerations.
91 */
92 constexpr SwerveModuleAcceleration operator*(double scalar) const {
93 if (scalar < 0) {
94 return {-scalar * acceleration, angle + Rotation2d{180_deg}};
95 }
96 return {scalar * acceleration, angle};
97 }
98
99 /**
100 * Divides the SwerveModuleAccelerations by a scalar and returns the new
101 * SwerveModuleAccelerations.
102 *
103 * @param scalar The scalar to divide by.
104 * @return The scaled SwerveModuleAccelerations.
105 */
106 constexpr SwerveModuleAcceleration operator/(double scalar) const {
107 return operator*(1.0 / scalar);
108 }
109};
110} // namespace wpi::math
