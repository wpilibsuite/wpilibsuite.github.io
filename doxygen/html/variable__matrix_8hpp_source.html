1054 ///
1055 /// @return The matrix exponential.
1056 VariableMatrix<Scalar> exp() const {
1057 slp_assert(rows() == cols());
1058
1059 // Coefficients for (13, 13) Padé approximant of exp(A) are from the
1060 // following program:
1061 //
1062 // #!/usr/bin/env python
1063 //
1064 // import mpmath as mp
1065 //
1066 // # https://en.wikipedia.org/wiki/IEEE_754#Basic_and_interchange_formats
1067 // mp.mp.prec = 113 # quad precision
1068 //
1069 // L = 13
1070 // M = 13
1071 // p, q = mp.pade(mp.taylor(mp.exp, 0, L + M), L, M)
1072 //
1073 // print("constexpr std::array p{")
1074 // for k, p_k in enumerate(p):
1075 // print(f"Scalar({p_k}L){',' if k < len(p) - 1 else '};'}")
1076 // print("constexpr std::array q{")
1077 // for k, q_k in enumerate(q):
1078 // print(f"Scalar({q_k}L){',' if k < len(q) - 1 else '};'}")
1079 constexpr size_t NUM_COEFFS = 14;
1080 constexpr std::array p{Scalar(1.0L),
1081 Scalar(0.499999999999999999999987615564159L),
1082 Scalar(0.119999999999999999999993719796468L),
1083 Scalar(0.0183333333333333333333318078905823L),
1084 Scalar(0.00199275362318840579710121408775205L),
1085 Scalar(0.00016304347826086956521736560868541L),
1086 Scalar(1.03519668737060041407846479512981e-05L),
1087 Scalar(5.17598343685300207039205144279119e-07L),
1088 Scalar(2.04315135665250081725989398668247e-08L),
1089 Scalar(6.30602270571759511499920806570486e-10L),
1090 Scalar(1.48377004840414002705850442238441e-11L),
1091 Scalar(2.52915349159796595521307855480881e-13L),
1092 Scalar(2.81017054621996217245857568431558e-15L),
1093 Scalar(1.54404975067030888596595580141692e-17L)};
1094 constexpr std::array q{Scalar(1.0L),
1095 Scalar(-0.500000000000000000000012384435841L),
1096 Scalar(0.120000000000000000000006104232309L),
1097 Scalar(-0.0183333333333333333333347707904729L),
1098 Scalar(0.00199275362318840579710166350137732L),
1099 Scalar(-0.000163043478260869565217413851346892L),
1100 Scalar(1.03519668737060041407885187519331e-05L),
1101 Scalar(-5.17598343685300207039443859549262e-07L),
1102 Scalar(2.04315135665250081726103781506842e-08L),
1103 Scalar(-6.30602270571759511500345035969254e-10L),
1104 Scalar(1.48377004840414002705969744674248e-11L),
1105 Scalar(-2.52915349159796595521550648156349e-13L),
1106 Scalar(2.81017054621996217246180850207342e-15L),
1107 Scalar(-1.54404975067030888596810607146175e-17L)};
1108 static_assert(p.size() == NUM_COEFFS);
1109 static_assert(q.size() == NUM_COEFFS);

1111 // 13
1112 // P = Σ pₖAᵏ
1113 // k=0
1114 //
1115 // 13
1116 // Q = Σ qₖAᵏ
1117 // k=0
1118 VariableMatrix<Scalar> P{
1119 Eigen::Vector<Scalar, Eigen::Dynamic>::Constant(rows(), p[0])
1120 .asDiagonal()};
1121 VariableMatrix<Scalar> Q{
1122 Eigen::Vector<Scalar, Eigen::Dynamic>::Constant(rows(), q[0])
1123 .asDiagonal()};
1124 auto A_pow = *this;
1125 for (size_t k = 1; k < NUM_COEFFS; ++k) {
1126 P += p[k] * A_pow;
1127 Q += q[k] * A_pow;
1128 A_pow *= *this;
1129 }

1131 // https://mpmath.org/doc/current/calculus/approximation.html#mpmath.pade
1132 // defines the Padé approximant as exp(A)Q ≈ P, so:
1133 //
1134 // exp(A) ≈ P / Q
1135 // exp(A) ≈ (Qᵀ \ Pᵀ)ᵀ
1136 return VariableMatrix<Scalar>{
1137 Q.T().to_eigen().lu().solve(P.T().to_eigen()).transpose()};
1138 }

1757/// Returns the variable's gradient tree.
1758///
1759/// This function lazily allocates variables, so elements of the returned
1760/// VariableMatrix will be empty if the corresponding element of wrt had no
1761/// adjoint. Ensure Variable::expr isn't nullptr before calling member
1762/// functions.
1763///
1764/// @tparam Scalar Scalar type.
1765/// @param top_list Topologically sorted graph from parent to child.
1766/// @param wrt Variables with respect to which to compute the gradient.
1767/// @return The variable's gradient tree.
1768template <typename Scalar>
1769VariableMatrix<Scalar> gradient_tree(const ExpressionGraph<Scalar>& top_list,
1770 const VariableMatrix<Scalar>& wrt) {
1771 slp_assert(wrt.cols() == 1);
1772
1773 // Read docs/algorithms.md#Reverse_accumulation_automatic_differentiation
1774 // for background on reverse accumulation automatic differentiation.
1775
1776 if (top_list.empty()) {
1777 return VariableMatrix<Scalar>{detail::empty, wrt.rows(), 1};
1778 }
1779
1780 // Set root node's adjoint to 1 since df/df is 1
1781 top_list[0]->adjoint_expr = constant_ptr(Scalar(1));
1782
1783 // df/dx = (df/dy)(dy/dx). The adjoint of x is equal to the adjoint of y
1784 // multiplied by dy/dx. If there are multiple "paths" from the root node to
1785 // variable; the variable's adjoint is the sum of each path's adjoint
1786 // contribution.
1787 for (auto& node : top_list) {
1788 auto& lhs = node->args[0];
1789 auto& rhs = node->args[1];
1790
1791 if (lhs != nullptr) {
1792 if (rhs != nullptr) {
1793 // Binary operator
1794 lhs->adjoint_expr += node->grad_expr_l(lhs, rhs);
1795 rhs->adjoint_expr += node->grad_expr_r(lhs, rhs);
1796 } else {
1797 // Unary operator
1798 lhs->adjoint_expr += node->grad_expr_l(lhs, rhs);
1799 }
1800 }
1801 }
1802
1803 // Move gradient tree to return value
1804 VariableMatrix<Scalar> grad{detail::empty, wrt.rows(), 1};
1805 for (int row = 0; row < grad.rows(); ++row) {
1806 grad[row] = Variable{std::move(wrt[row].expr->adjoint_expr)};
1807 }
1808
1809 // Unlink adjoints to avoid circular references between them and their
1810 // parent expressions. This ensures all expressions are returned to the free
1811 // list.
1812 for (auto& node : top_list) {
1813 node->adjoint_expr = nullptr;
1814 }
1815
1816 return grad;
1817}
