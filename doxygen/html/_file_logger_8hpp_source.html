16/**
17 * A class version of `tail -f`, otherwise known as `tail -f` at home. Watches
18 * a file and puts the data somewhere else. Only works on Linux-based platforms.
19 */
20class FileLogger {
21 public:
22 FileLogger() = default;
23 /**
24 * Construct a FileLogger. When the specified file is modified, the callback
25 * will be called with the appended changes.
26 *
27 * @param file The path to the file.
28 * @param callback A callback that accepts the appended file data.
29 */
30 FileLogger(std::string_view file,
31 std::function<void(std::string_view)> callback);
32
33 /**
34 * Construct a FileLogger. When the specified file is modified, appended data
35 * will be appended to the specified data log.
36 *
37 * @param file The path to the file.
38 * @param log A data log.
39 * @param key The log key to append data to.
40 */
41 FileLogger(std::string_view file, log::DataLog& log, std::string_view key);
42 FileLogger(FileLogger&& other);
43 FileLogger& operator=(FileLogger&& rhs);
44 ~FileLogger();
45 /**
46 * Creates a function that chunks incoming data into blocks of whole lines and
47 * stores incomplete lines to add to the next block of data.
48 *
49 * @param callback A callback that accepts the blocks of whole lines.
50 * @return The function.
51 */
52 static std::function<void(std::string_view)> Buffer(
53 std::function<void(std::string_view)> callback);
54
55 private:
56#ifdef __linux__
57 void Stop();
58
59 int m_fileHandle = -1;
60 int m_inotifyHandle = -1;
61 int m_inotifyWatchHandle = -1;
62 std::shared_ptr<std::atomic<bool>> m_running =
63 std::make_shared<std::atomic<bool>>(true);
64 std::thread m_thread;
65#endif
66};
67} // namespace wpi::log
