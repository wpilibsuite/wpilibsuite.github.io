17namespace wpi::math {
18/**
19 * Class for differential drive odometry. Odometry allows you to track the
20 * robot's position on the field over the course of a match using readings from
21 * 2 encoders and a gyroscope.
22 *
23 * Teams can use odometry during the autonomous period for complex tasks like
24 * path following. Furthermore, odometry can be used for latency compensation
25 * when using computer-vision systems.
26 */
27class WPILIB_DLLEXPORT DifferentialDriveOdometry3d
28 : public Odometry3d<DifferentialDriveKinematics,
29 DifferentialDriveWheelPositions,
30 DifferentialDriveWheelVelocities,
31 DifferentialDriveWheelAccelerations> {
32 public:
33 /**
34 * Constructs a DifferentialDriveOdometry3d object.
35 *
36 * IF leftDistance and rightDistance are unspecified,
37 * You NEED to reset your encoders (to zero).
38 *
39 * @param gyroAngle The angle reported by the gyroscope. This does not need to
40 * be offset to match the robot's orientation on the field.
41 * @param leftDistance The distance traveled by the left encoder.
42 * @param rightDistance The distance traveled by the right encoder.
43 * @param initialPose The starting position of the robot on the field.
44 */
45 explicit DifferentialDriveOdometry3d(const Rotation3d& gyroAngle,
46 wpi::units::meter_t leftDistance,
47 wpi::units::meter_t rightDistance,
48 const Pose3d& initialPose = Pose3d{});
49
50 /**
51 * Resets the robot's position on the field.
52 *
53 * IF leftDistance and rightDistance are unspecified,
54 * You NEED to reset your encoders (to zero).
55 *
56 * The gyroscope angle does not need to be reset here on the user's robot
57 * code. The library automatically takes care of offsetting the gyro angle.
58 *
59 * @param pose The position on the field that your robot is at.
60 * @param gyroAngle The angle reported by the gyroscope. This does not need to
61 * be offset to match the robot's orientation on the field.
62 * @param leftDistance The distance traveled by the left encoder.
63 * @param rightDistance The distance traveled by the right encoder.
64 */
65 void ResetPosition(const Rotation3d& gyroAngle,
66 wpi::units::meter_t leftDistance,
67 wpi::units::meter_t rightDistance, const Pose3d& pose) {
68 Odometry3d::ResetPosition(gyroAngle, {leftDistance, rightDistance}, pose);
69 }
70
71 /**
72 * Updates the robot position on the field using distance measurements from
73 * encoders.
74 *
75 * @param gyroAngle The angle reported by the gyroscope. This does not need to
76 * be offset to match the robot's orientation on the field.
77 * @param leftDistance The distance traveled by the left encoder.
78 * @param rightDistance The distance traveled by the right encoder.
79 * @return The new pose of the robot.
80 */
81 const Pose3d& Update(const Rotation3d& gyroAngle,
82 wpi::units::meter_t leftDistance,
83 wpi::units::meter_t rightDistance) {
84 return Odometry3d::Update(gyroAngle, {leftDistance, rightDistance});
85 }
86};
87} // namespace wpi::math
