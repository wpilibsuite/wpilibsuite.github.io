19/**
20 * Represents robot chassis accelerations.
21 *
22 * A strictly non-holonomic drivetrain, such as a differential drive, should
23 * never have an ay component because it can never move sideways. Holonomic
24 * drivetrains such as swerve and mecanum will often have all three components.
25 */
26struct WPILIB_DLLEXPORT ChassisAccelerations {
27 /**
28 * Acceleration along the x-axis. (Fwd is +)
29 */
30 units::meters_per_second_squared_t ax = 0_mps_sq;
31
32 /**
33 * Acceleration along the y-axis. (Left is +)
34 */
35 units::meters_per_second_squared_t ay = 0_mps_sq;
36
37 /**
38 * Angular acceleration of the robot frame. (CCW is +)
39 */
40 units::radians_per_second_squared_t alpha = 0_rad_per_s_sq;
41
42 /**
43 * Converts this field-relative set of accelerations into a robot-relative
44 * ChassisAccelerations object.
45 *
46 * @param robotAngle The angle of the robot as measured by a gyroscope. The
47 * robot's angle is considered to be zero when it is facing directly away
48 * from your alliance station wall. Remember that this should be CCW
49 * positive.
50 * @return ChassisAccelerations object representing the accelerations in the
51 * robot's frame of reference.
52 */
53 constexpr ChassisAccelerations ToRobotRelative(
54 const Rotation2d& robotAngle) const {
55 // CW rotation into chassis frame
56 auto rotated =
57 Translation2d{units::meter_t{ax.value()}, units::meter_t{ay.value()}}
58 .RotateBy(-robotAngle);
59 return {units::meters_per_second_squared_t{rotated.X().value()},
60 units::meters_per_second_squared_t{rotated.Y().value()}, alpha};
61 }
62
63 /**
64 * Converts this robot-relative set of accelerations into a field-relative
65 * ChassisAccelerations object.
66 *
67 * @param robotAngle The angle of the robot as measured by a gyroscope. The
68 * robot's angle is considered to be zero when it is facing directly away
69 * from your alliance station wall. Remember that this should be CCW
70 * positive.
71 * @return ChassisAccelerations object representing the accelerations in the
72 * field's frame of reference.
73 */
74 constexpr ChassisAccelerations ToFieldRelative(
75 const Rotation2d& robotAngle) const {
76 // CCW rotation out of chassis frame
77 auto rotated =
78 Translation2d{units::meter_t{ax.value()}, units::meter_t{ay.value()}}
79 .RotateBy(robotAngle);
80 return {units::meters_per_second_squared_t{rotated.X().value()},
81 units::meters_per_second_squared_t{rotated.Y().value()}, alpha};
82 }
83
84 /**
85 * Adds two ChassisAccelerations and returns the sum.
86 *
87 * For example, ChassisAccelerations{1.0, 0.5, 1.5} +
88 * ChassisAccelerations{2.0, 1.5, 0.5} = ChassisAccelerations{3.0, 2.0, 2.0}
89 *
90 * @param other The ChassisAccelerations to add.
91 *
92 * @return The sum of the ChassisAccelerations.
93 */
94 constexpr ChassisAccelerations operator+(
95 const ChassisAccelerations& other) const {
96 return {ax + other.ax, ay + other.ay, alpha + other.alpha};
97 }
98
99 /**
100 * Subtracts the other ChassisAccelerations from the current
101 * ChassisAccelerations and returns the difference.
102 *
103 * For example, ChassisAccelerations{5.0, 4.0, 2.0} -
104 * ChassisAccelerations{1.0, 2.0, 1.0} = ChassisAccelerations{4.0, 2.0, 1.0}
105 *
106 * @param other The ChassisAccelerations to subtract.
107 *
108 * @return The difference between the two ChassisAccelerations.
109 */
110 constexpr ChassisAccelerations operator-(
111 const ChassisAccelerations& other) const {
112 return *this + -other;
113 }
114
115 /**
116 * Returns the inverse of the current ChassisAccelerations.
117 * This is equivalent to negating all components of the ChassisAccelerations.
118 *
119 * @return The inverse of the current ChassisAccelerations.
120 */
121 constexpr ChassisAccelerations operator-() const {
122 return {-ax, -ay, -alpha};
123 }
124
125 /**
126 * Multiplies the ChassisAccelerations by a scalar and returns the new
127 * ChassisAccelerations.
128 *
129 * For example, ChassisAccelerations{2.0, 2.5, 1.0} * 2
130 * = ChassisAccelerations{4.0, 5.0, 2.0}
131 *
132 * @param scalar The scalar to multiply by.
133 *
134 * @return The scaled ChassisAccelerations.
135 */
136 constexpr ChassisAccelerations operator*(double scalar) const {
137 return {scalar * ax, scalar * ay, scalar * alpha};
138 }
139
140 /**
141 * Divides the ChassisAccelerations by a scalar and returns the new
142 * ChassisAccelerations.
143 *
144 * For example, ChassisAccelerations{2.0, 2.5, 1.0} / 2
145 * = ChassisAccelerations{1.0, 1.25, 0.5}
146 *
147 * @param scalar The scalar to divide by.
148 *
149 * @return The scaled ChassisAccelerations.
150 */
151 constexpr ChassisAccelerations operator/(double scalar) const {
152 return operator*(1.0 / scalar);
153 }
154
155 /**
156 * Compares the ChassisAccelerations with another ChassisAccelerations.
157 *
158 * @param other The other ChassisAccelerations.
159 *
160 * @return The result of the comparison. Is true if they are the same.
161 */
162 constexpr bool operator==(const ChassisAccelerations& other) const = default;
163};
