22// 256-entry: 0xff = not hex, else nibble value.
23inline constexpr std::array<uint8_t, 256> make_hex_nibble_table() noexcept {
24 std::array<uint8_t, 256> t{};
25 for (size_t i = 0; i < 256; ++i) {
26 t[i] = 0xff;
27 }
28 for (size_t d = 0; d < 10; ++d) {
29 t[size_t{'0'} + d] = static_cast<uint8_t>(d);
30 }
31 for (size_t d = 0; d < 6; ++d) {
32 t[size_t{'a'} + d] = static_cast<uint8_t>(10 + d);
33 t[size_t{'A'} + d] = static_cast<uint8_t>(10 + d);
34 }
35 return t;
36}

40inline bool parse_ipv4_number(const char*& p, const char* end, uint64_t& value,
41 bool& is_pure_decimal) noexcept {
42 is_pure_decimal = false;
43 if (p >= end) [[unlikely]] {
44 return false;
45 }
46 if (end - p >= 2 && p[0] == '0' && ((p[1] | 0x20) == 'x')) {
47 p += 2;
48 if (p == end || *p == '.') {
49 value = 0;
50 return true;
51 }
52 uint64_t v = 0;
53 int digits = 0;
54 while (p < end && *p != '.') {
55 const uint8_t nib = hex_nibble[static_cast<unsigned char>(*p)];
56 if (nib == 0xff) [[unlikely]] {
57 return false;
58 }
59 if (v > (0xFFFFFFFFULL >> 4)) [[unlikely]] {
60 return false;
61 }
62 v = (v << 4) | nib;
63 ++p;
64 ++digits;
65 }
66 if (digits == 0) [[unlikely]] {
67 return false;
68 }
69 value = v;
70 return true;
71 }
72 if (end - p >= 2 && p[0] == '0' && p[1] >= '0' && p[1] <= '9') {
73 ++p;
74 uint64_t v = 0;
75 while (p < end && *p != '.') {
76 const char c = *p;
77 if (c < '0' || c > '7') [[unlikely]] {
78 return false;
79 }
80 if (v > (0xFFFFFFFFULL >> 3)) [[unlikely]] {
81 return false;
82 }
83 v = (v << 3) | static_cast<uint64_t>(c - '0');
84 ++p;
85 }
86 value = v;
87 return true;
88 }
89 if (*p < '0' || *p > '9') [[unlikely]] {
90 return false;
91 }
92 is_pure_decimal = true;
93 uint64_t v = static_cast<uint64_t>(*p - '0');
94 ++p;
95 while (p < end && *p != '.') {
96 const char c = *p;
97 if (c < '0' || c > '9') [[unlikely]] {
98 return false;
99 }
100 if (v > 429496729ULL) [[unlikely]] {
101 return false;
102 }
103 v = v * 10u + static_cast<uint64_t>(c - '0');
104 if (v > 0xFFFFFFFFULL) [[unlikely]] {
105 return false;
106 }
107 ++p;
108 }
109 value = v;
110 return true;
111}

113// Parse up to 4 hex digits. Returns digit count (0 if none).
114inline int parse_hex_piece(const char*& pointer, const char* end,
115 uint16_t& value) noexcept {
116 if (pointer == end) {
117 return 0;
118 }
119 const uint8_t n0 = hex_nibble[static_cast<unsigned char>(*pointer)];
120 if (n0 == 0xff) {
121 return 0;
122 }
123 uint32_t v = n0;
124 ++pointer;
125 int length = 1;
126 if (pointer != end) {
127 const uint8_t n1 = hex_nibble[static_cast<unsigned char>(*pointer)];
128 if (n1 != 0xff) {
129 v = (v << 4) | n1;
130 ++pointer;
131 ++length;
132 if (pointer != end) {
133 const uint8_t n2 = hex_nibble[static_cast<unsigned char>(*pointer)];
134 if (n2 != 0xff) {
135 v = (v << 4) | n2;
136 ++pointer;
137 ++length;
138 if (pointer != end) {
139 const uint8_t n3 = hex_nibble[static_cast<unsigned char>(*pointer)];
140 if (n3 != 0xff) {
141 v = (v << 4) | n3;
142 ++pointer;
143 ++length;
144 }
145 }
146 }
147 }
148 }
149 }
150 value = static_cast<uint16_t>(v);
151 return length;
152}

155// Classify an IPv6 host (no brackets) with one masked 512-bit load.
156// Returns false if the colon/dot shape is impossible (e.g. two "::", >8
157// colons, full form without 7 colons). Used as a cheap prefilter before the
158// full WHATWG piece parser.
159ada_really_inline bool ipv6_structure_plausible(const char* data,
160 size_t len) noexcept {
161 if (len < 2 || len > 45) {
162 return false;
163 }
164 const __mmask64 live = static_cast<__mmask64>((1ULL << len) - 1ULL);
165 const __m512i input =
166 _mm512_maskz_loadu_epi8(live, reinterpret_cast<const void*>(data));
167 const __mmask64 is_colon =
168 _mm512_mask_cmpeq_epi8_mask(live, input, _mm512_set1_epi8(':'));
169 const __mmask64 is_dot =
170 _mm512_mask_cmpeq_epi8_mask(live, input, _mm512_set1_epi8('.'));
171 const int colons =
172 static_cast<int>(_mm_popcnt_u64(static_cast<uint64_t>(is_colon)));
173 if (colons > 8) {
174 return false;
175 }
176 const uint64_t doubles =
177 static_cast<uint64_t>(is_colon) & (static_cast<uint64_t>(is_colon) << 1);
178 if (doubles != 0 && (doubles & (doubles - 1)) != 0) {
179 return false; // more than one "::" (or ":::...")
180 }
181 const bool has_double = doubles != 0;
182 const bool has_dot = is_dot != 0;
183 if (!has_double && !has_dot && colons != 7) {
184 return false;
185 }
186 return true;
187}
