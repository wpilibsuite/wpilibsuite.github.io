20namespace wpi::math {
21/**
22 * Helper class used to generate drivetrain spline trajectories with various
23 * constraints.
24 */
25class WPILIB_DLLEXPORT DrivetrainSplineTrajectoryGenerator {
26 public:
27 using PoseWithCurvature = std::pair<Pose2d, wpi::units::curvature_t>;
28
29 /**
30 * Generates a trajectory from the given control vectors and config. This
31 * method uses clamped cubic splines -- a method in which the exterior control
32 * vectors and interior waypoints are provided. The headings are automatically
33 * determined at the interior points to ensure continuous curvature.
34 *
35 * @param initial The initial control vector.
36 * @param interiorWaypoints The interior waypoints.
37 * @param end The ending control vector.
38 * @param config The configuration for the trajectory.
39 * @return The generated spline trajectory.
40 */
41 static DrivetrainSplineTrajectory Generate(
42 Spline<3>::ControlVector initial,
43 const std::vector<Translation2d>& interiorWaypoints,
44 Spline<3>::ControlVector end, const TrajectoryConfig& config);
45
46 /**
47 * Generates a trajectory from the given waypoints and config. This method
48 * uses clamped cubic splines -- a method in which the initial pose, final
49 * pose, and interior waypoints are provided. The headings are automatically
50 * determined at the interior points to ensure continuous curvature.
51 *
52 * @param start The starting pose.
53 * @param interiorWaypoints The interior waypoints.
54 * @param end The ending pose.
55 * @param config The configuration for the trajectory.
56 * @return The generated spline trajectory.
57 */
58 static DrivetrainSplineTrajectory Generate(
59 const Pose2d& start, const std::vector<Translation2d>& interiorWaypoints,
60 const Pose2d& end, const TrajectoryConfig& config);
61
62 /**
63 * Generates a trajectory from the given quintic control vectors and config.
64 * This method uses quintic hermite splines -- therefore, all points must be
65 * represented by control vectors. Continuous curvature is guaranteed in this
66 * method.
67 *
68 * @param controlVectors List of quintic control vectors.
69 * @param config The configuration for the trajectory.
70 * @return The generated spline trajectory.
71 */
72 static DrivetrainSplineTrajectory Generate(
73 std::vector<Spline<5>::ControlVector> controlVectors,
74 const TrajectoryConfig& config);
75
76 /**
77 * Generates a trajectory from the given waypoints and config. This method
78 * uses quintic hermite splines -- therefore, all points must be represented
79 * by Pose2d objects. Continuous curvature is guaranteed in this method.
80 *
81 * @param waypoints List of waypoints..
82 * @param config The configuration for the trajectory.
83 * @return The generated spline trajectory.
84 */
85 static DrivetrainSplineTrajectory Generate(
86 const std::vector<Pose2d>& waypoints, const TrajectoryConfig& config);
87
88 /**
89 * Generate spline points from a vector of splines by parameterizing the
90 * splines.
91 *
92 * @param splines The splines to parameterize.
93 *
94 * @return The spline points for use in time parameterization of a trajectory.
95 */
96 template <typename Spline>
97 static std::vector<PoseWithCurvature> SplinePointsFromSplines(
98 const std::vector<Spline>& splines) {
99 // Create the vector of spline points.
100 std::vector<PoseWithCurvature> splinePoints;
101
102 // Add the first point to the vector.
103 splinePoints.push_back(splines.front().GetPoint(0.0).value());
104
105 // Iterate through the vector and parameterize each spline, adding the
106 // parameterized points to the final vector.
107 for (auto&& spline : splines) {
108 auto points = SplineParameterizer::Parameterize(spline);
109 // Append the array of poses to the vector. We are removing the first
110 // point because it's a duplicate of the last point from the previous
111 // spline.
112 splinePoints.insert(std::end(splinePoints), std::begin(points) + 1,
113 std::end(points));
114 }
115 return splinePoints;
116 }
117
118 /**
119 * Set error reporting function. By default, it is output to stderr.
120 *
121 * @param func Error reporting function.
122 */
123 static void SetErrorHandler(std::function<void(const char*)> func);
124
125 private:
126 static void ReportError(const char* error);
127
128 static const DrivetrainSplineTrajectory DO_NOTHING_TRAJECTORY;
129 static std::function<void(const char*)> s_errorFunc;
130};
131} // namespace wpi::math
