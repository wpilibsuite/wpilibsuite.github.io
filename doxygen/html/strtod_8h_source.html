53inline Vector<const char> TrimTrailingZeros(Vector<const char> buffer) {
54 for (int i = buffer.length() - 1; i >= 0; --i) {
55 if (buffer[i] != '0') {
56 return buffer.SubVector(0, i + 1);
57 }
58 }
59 return Vector<const char>(buffer.start(), 0);
60}
