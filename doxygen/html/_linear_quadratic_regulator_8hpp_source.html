29/**
30 * Contains the controller coefficients and logic for a linear-quadratic
31 * regulator (LQR).
32 * LQRs use the control law u = K(r - x).
33 *
34 * For more on the underlying math, read
35 * https://file.tavsys.net/control/controls-engineering-in-frc.pdf.
36 *
37 * @tparam States Number of states.
38 * @tparam Inputs Number of inputs.
39 */
40template <int States, int Inputs>
41class LinearQuadraticRegulator {
42 public:
43 using StateVector = Vectord<States>;
44 using InputVector = Vectord<Inputs>;
45
46 using StateArray = wpi::util::array<double, States>;
47 using InputArray = wpi::util::array<double, Inputs>;

49 /**
50 * Constructs a controller with the given coefficients and plant.
51 *
52 * See
53 * https://docs.wpilib.org/en/stable/docs/software/advanced-controls/state-space/state-space-intro.html#lqr-tuning
54 * for how to select the tolerances.
55 *
56 * @tparam Outputs Number of outputs.
57 * @param plant The plant being controlled.
58 * @param Qelems The maximum desired error tolerance for each state.
59 * @param Relems The maximum desired control effort for each input.
60 * @param dt Discretization timestep.
61 * @throws std::invalid_argument If the system is unstabilizable.
62 */
63 template <int Outputs>
64 LinearQuadraticRegulator(const LinearSystem<States, Inputs, Outputs>& plant,
65 const StateArray& Qelems, const InputArray& Relems,
66 wpi::units::second_t dt)

69 /**
70 * Constructs a controller with the given coefficients and plant.
71 *
72 * See
73 * https://docs.wpilib.org/en/stable/docs/software/advanced-controls/state-space/state-space-intro.html#lqr-tuning
74 * for how to select the tolerances.
75 *
76 * @param A Continuous system matrix of the plant being controlled.
77 * @param B Continuous input matrix of the plant being controlled.
78 * @param Qelems The maximum desired error tolerance for each state.
79 * @param Relems The maximum desired control effort for each input.
80 * @param dt Discretization timestep.
81 * @throws std::invalid_argument If the system is unstabilizable.
82 */
83 LinearQuadraticRegulator(const Matrixd<States, States>& A,
84 const Matrixd<States, Inputs>& B,
85 const StateArray& Qelems, const InputArray& Relems,
86 wpi::units::second_t dt)
87 : LinearQuadraticRegulator(A, B, CostMatrix(Qelems), CostMatrix(Relems),
88 dt) {}

90 /**
91 * Constructs a controller with the given coefficients and plant.
92 *
93 * @param A Continuous system matrix of the plant being controlled.
94 * @param B Continuous input matrix of the plant being controlled.
95 * @param Q The state cost matrix.
96 * @param R The input cost matrix.
97 * @param dt Discretization timestep.
98 * @throws std::invalid_argument If the system is unstabilizable.
99 */
100 LinearQuadraticRegulator(const Matrixd<States, States>& A,
101 const Matrixd<States, Inputs>& B,
102 const Matrixd<States, States>& Q,
103 const Matrixd<Inputs, Inputs>& R,
104 wpi::units::second_t dt) {
105 Matrixd<States, States> discA;
106 Matrixd<States, Inputs> discB;
107 DiscretizeAB<States, Inputs>(A, B, dt, &discA, &discB);
108
109 if (auto S = DARE<States, Inputs>(discA, discB, Q, R)) {
110 // K = (BᵀSB + R)⁻¹BᵀSA
111 m_K = (discB.transpose() * S.value() * discB + R)
112 .llt()
113 .solve(discB.transpose() * S.value() * discA);
114 } else if (S.error() == DAREError::QNotSymmetric ||
115 S.error() == DAREError::QNotPositiveSemidefinite) {
116 std::string msg = std::format("{}\n\nQ =\n{}\n", to_string(S.error()), Q);
117
118 wpi::math::MathSharedStore::ReportError(msg);
119 throw std::invalid_argument(msg);
120 } else if (S.error() == DAREError::RNotSymmetric ||
121 S.error() == DAREError::RNotPositiveDefinite) {
122 std::string msg = std::format("{}\n\nR =\n{}\n", to_string(S.error()), R);
123
124 wpi::math::MathSharedStore::ReportError(msg);
125 throw std::invalid_argument(msg);
126 } else if (S.error() == DAREError::ABNotStabilizable) {
127 std::string msg = std::format("{}\n\nA =\n{}\nB =\n{}\n",
128 to_string(S.error()), discA, discB);
129
130 wpi::math::MathSharedStore::ReportError(msg);
131 throw std::invalid_argument(msg);
132 } else if (S.error() == DAREError::ACNotDetectable) {
133 std::string msg = std::format("{}\n\nA =\n{}\nQ =\n{}\n",
134 to_string(S.error()), discA, Q);
135
136 wpi::math::MathSharedStore::ReportError(msg);
137 throw std::invalid_argument(msg);
138 }
139
140 Reset();
141 wpi::util::ReportUsage("LinearQuadraticRegulator", "");
142 }

144 /**
145 * Constructs a controller with the given coefficients and plant.
146 *
147 * @param A Continuous system matrix of the plant being controlled.
148 * @param B Continuous input matrix of the plant being controlled.
149 * @param Q The state cost matrix.
150 * @param R The input cost matrix.
151 * @param N The state-input cross-term cost matrix.
152 * @param dt Discretization timestep.
153 * @throws std::invalid_argument If the system is unstabilizable.
154 */
155 LinearQuadraticRegulator(const Matrixd<States, States>& A,
156 const Matrixd<States, Inputs>& B,
157 const Matrixd<States, States>& Q,
158 const Matrixd<Inputs, Inputs>& R,
159 const Matrixd<States, Inputs>& N,
160 wpi::units::second_t dt) {
161 Matrixd<States, States> discA;
162 Matrixd<States, Inputs> discB;
163 DiscretizeAB<States, Inputs>(A, B, dt, &discA, &discB);
164
165 if (auto S = DARE<States, Inputs>(discA, discB, Q, R, N)) {
166 // K = (BᵀSB + R)⁻¹(BᵀSA + Nᵀ)
167 m_K = (discB.transpose() * S.value() * discB + R)
168 .llt()
169 .solve(discB.transpose() * S.value() * discA + N.transpose());
170 } else if (S.error() == DAREError::QNotSymmetric ||
171 S.error() == DAREError::QNotPositiveSemidefinite) {
172 std::string msg = std::format("{}\n\nQ =\n{}\n", to_string(S.error()), Q);
173
174 wpi::math::MathSharedStore::ReportError(msg);
175 throw std::invalid_argument(msg);
176 } else if (S.error() == DAREError::RNotSymmetric ||
177 S.error() == DAREError::RNotPositiveDefinite) {
178 std::string msg = std::format("{}\n\nR =\n{}\n", to_string(S.error()), R);
179
180 wpi::math::MathSharedStore::ReportError(msg);
181 throw std::invalid_argument(msg);
182 } else if (S.error() == DAREError::ABNotStabilizable) {
183 std::string msg = std::format(
184 "{}\n\nA =\n{}\nB =\n{}\n", to_string(S.error()),
185 (discA - discB * R.llt().solve(N.transpose())).eval(), discB);
186
187 wpi::math::MathSharedStore::ReportError(msg);
188 throw std::invalid_argument(msg);
189 } else if (S.error() == DAREError::ACNotDetectable) {
190 auto R_llt = R.llt();
191 std::string msg =
192 std::format("{}\n\nA =\n{}\nQ =\n{}\n", to_string(S.error()),
193 (discA - discB * R_llt.solve(N.transpose())).eval(),
194 (Q - N * R_llt.solve(N.transpose())).eval());
195
196 wpi::math::MathSharedStore::ReportError(msg);
197 throw std::invalid_argument(msg);
198 }
199
200 Reset();
201 wpi::util::ReportUsage("LinearQuadraticRegulator", "");
202 }

252 /**

255 void Reset() {
256 m_r.setZero();
257 m_u.setZero();
258 m_error.setZero();
259 }
260
261 /**
262 * Returns true if the error is within the tolerance set by SetTolerance()
263 * for every state.
264 */
265 bool AtReference() const {
266 for (int i = 0; i < m_error.rows(); ++i) {
267 if (std::abs(m_error(i)) > m_tolerance(i)) {
268 return false;
269 }
270 }
271 return true;
272 }

274 /**
275 * Sets the error which is considered tolerable for use with AtReference().
276 *
277 * @param tolerance The tolerable error for each state.
278 */
279 void SetTolerance(const StateVector& tolerance) { m_tolerance = tolerance; }
280
281 /**
282 * Returns the next output of the controller.
283 *
284 * @param x The current state x.
285 */
286 InputVector Calculate(const StateVector& x) {
287 m_error = m_r - x;
288 m_u = m_K * m_error;
289 return m_u;
290 }
291
292 /**
293 * Returns the next output of the controller.
294 *
295 * @param x The current state x.
296 * @param nextR The next reference vector r.
297 */
298 InputVector Calculate(const StateVector& x, const StateVector& nextR) {
299 m_r = nextR;
300 return Calculate(x);
301 }

303 /**
304 * Adjusts LQR controller gain to compensate for a pure time delay in the
305 * input.
306 *
307 * Linear-Quadratic regulator controller gains tend to be aggressive. If
308 * sensor measurements are time-delayed too long, the LQR may be unstable.
309 * However, if we know the amount of delay, we can compute the control based
310 * on where the system will be after the time delay.
311 *
312 * See https://file.tavsys.net/control/controls-engineering-in-frc.pdf
313 * appendix C.4 for a derivation.
314 *
315 * @param plant The plant being controlled.
316 * @param dt Discretization timestep.
317 * @param inputDelay Input time delay.
318 */
319 template <int Outputs>
320 void LatencyCompensate(const LinearSystem<States, Inputs, Outputs>& plant,
321 wpi::units::second_t dt,
322 wpi::units::second_t inputDelay) {
323 Matrixd<States, States> discA;
324 Matrixd<States, Inputs> discB;
325 DiscretizeAB<States, Inputs>(plant.A(), plant.B(), dt, &discA, &discB);
326
327 m_K = m_K * (discA - discB * m_K).pow(inputDelay / dt);
328 }
329
330 private:
331 // Current reference
332 StateVector m_r;
333
334 // Computed controller output
335 InputVector m_u;
336
337 // Error at the time of the last controller update
338 StateVector m_error = StateVector::Zero();
339
340 // Error which is considered tolerable for use with AtReference()
341 StateVector m_tolerance = StateVector::Zero();
342
343 // Controller gain
344 Matrixd<Inputs, States> m_K;
345};
