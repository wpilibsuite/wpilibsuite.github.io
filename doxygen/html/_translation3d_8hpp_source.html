28/**
29 * Represents a translation in 3D space.
30 * This object can be used to represent a point or a vector.
31 *
32 * This assumes that you are using conventional mathematical axes. When the
33 * robot is at the origin facing in the positive X direction, forward is
34 * positive X, left is positive Y, and up is positive Z.
35 */
36class WPILIB_DLLEXPORT Translation3d final {
37 public:
38 /**
39 * Constructs a Translation3d with X, Y, and Z components equal to zero.
40 */
41 constexpr Translation3d() = default;
42
43 /**
44 * Constructs a Translation3d with the X, Y, and Z components equal to the
45 * provided values.
46 *
47 * @param x The x component of the translation.
48 * @param y The y component of the translation.
49 * @param z The z component of the translation.
50 */
51 constexpr Translation3d(wpi::units::meter_t x, wpi::units::meter_t y,
52 wpi::units::meter_t z)
53 : m_x{x}, m_y{y}, m_z{z} {}
54
55 /**
56 * Constructs a Translation3d with the provided distance and angle. This is
57 * essentially converting from polar coordinates to Cartesian coordinates.
58 *
59 * @param distance The distance from the origin to the end of the translation.
60 * @param angle The angle between the x-axis and the translation vector.
61 */
62 constexpr Translation3d(wpi::units::meter_t distance,
63 const Rotation3d& angle) {
64 auto rectangular = Translation3d{distance, 0_m, 0_m}.RotateBy(angle);
65 m_x = rectangular.X();
66 m_y = rectangular.Y();
67 m_z = rectangular.Z();
68 }
69
70 /**
71 * Constructs a Translation3d from a 3D translation vector. The values are
72 * assumed to be in meters.
73 *
74 * @param vector The translation vector.
75 */
76 constexpr explicit Translation3d(const Eigen::Vector3d& vector)
77 : m_x{wpi::units::meter_t{vector.x()}},
78 m_y{wpi::units::meter_t{vector.y()}},
79 m_z{wpi::units::meter_t{vector.z()}} {}
80
81 /**
82 * Constructs a 3D translation from a 2D translation in the X-Y plane.
83 *
84 * @param translation The 2D translation.
85 * @see Pose3d(Pose2d)
86 * @see Transform3d(Transform2d)
87 */
88 constexpr explicit Translation3d(const Translation2d& translation)
89 : Translation3d{translation.X(), translation.Y(), 0_m} {}
90
91 /**
92 * Calculates the distance between two translations in 3D space.
93 *
94 * The distance between translations is defined as
95 * √((x₂−x₁)²+(y₂−y₁)²+(z₂−z₁)²).
96 *
97 * @param other The translation to compute the distance to.
98 *
99 * @return The distance between the two translations.
100 */
101 constexpr wpi::units::meter_t Distance(const Translation3d& other) const {
102 return wpi::units::math::sqrt(wpi::units::math::pow<2>(other.m_x - m_x) +
103 wpi::units::math::pow<2>(other.m_y - m_y) +
104 wpi::units::math::pow<2>(other.m_z - m_z));
105 }
106
107 /**
108 * Calculates the squared distance between two translations in 3D space. This
109 * is equivalent to squaring the result of Distance(Translation3d), but avoids
110 * computing a square root.
111 *
112 * The squared distance between translations is defined as
113 * (x₂−x₁)²+(y₂−y₁)²+(z₂−z₁)².
114 *
115 * @param other The translation to compute the squared distance to.
116 * @return The squared distance between the two translations.
117 */
118 constexpr wpi::units::square_meter_t SquaredDistance(
119 const Translation3d& other) const {
120 return wpi::units::math::pow<2>(other.m_x - m_x) +
121 wpi::units::math::pow<2>(other.m_y - m_y) +
122 wpi::units::math::pow<2>(other.m_z - m_z);
123 }
124
125 /**
126 * Returns the X component of the translation.
127 *
128 * @return The X component of the translation.
129 */
130 constexpr wpi::units::meter_t X() const { return m_x; }
131
132 /**
133 * Returns the Y component of the translation.
134 *
135 * @return The Y component of the translation.
136 */
137 constexpr wpi::units::meter_t Y() const { return m_y; }
138
139 /**
140 * Returns the Z component of the translation.
141 *
142 * @return The Z component of the translation.
143 */
144 constexpr wpi::units::meter_t Z() const { return m_z; }
145
146 /**
147 * Returns a 3D translation vector representation of this translation.
148 *
149 * @return A 3D translation vector representation of this translation.
150 */
151 constexpr Eigen::Vector3d ToVector() const {
152 return Eigen::Vector3d{{m_x.value(), m_y.value(), m_z.value()}};
153 }
154
155 /**
156 * Returns the norm, or distance from the origin to the translation.
157 *
158 * @return The norm of the translation.
159 */
160 constexpr wpi::units::meter_t Norm() const {
161 return wpi::units::math::sqrt(m_x * m_x + m_y * m_y + m_z * m_z);
162 }
163
164 /**
165 * Returns the squared norm, or squared distance from the origin to the
166 * translation. This is equivalent to squaring the result of Norm(), but
167 * avoids computing a square root.
168 *
169 * @return The squared norm of the translation.
170 */
171 constexpr wpi::units::square_meter_t SquaredNorm() const {
172 return m_x * m_x + m_y * m_y + m_z * m_z;
173 }
174
175 /**
176 * Applies a rotation to the translation in 3D space.
177 *
178 * For example, rotating a Translation3d of &lt;2, 0, 0&gt; by 90 degrees
179 * around the Z axis will return a Translation3d of &lt;0, 2, 0&gt;.
180 *
181 * @param other The rotation to rotate the translation by.
182 *
183 * @return The new rotated translation.
184 */
185 constexpr Translation3d RotateBy(const Rotation3d& other) const {
186 Quaternion p{0.0, m_x.value(), m_y.value(), m_z.value()};
187 auto qprime = other.GetQuaternion() * p * other.GetQuaternion().Inverse();
188 return Translation3d{wpi::units::meter_t{qprime.X()},
189 wpi::units::meter_t{qprime.Y()},
190 wpi::units::meter_t{qprime.Z()}};
191 }
192
193 /**
194 * Rotates this translation around another translation in 3D space.
195 *
196 * @param other The other translation to rotate around.
197 * @param rot The rotation to rotate the translation by.
198 * @return The new rotated translation.
199 */
200 constexpr Translation3d RotateAround(const Translation3d& other,
201 const Rotation3d& rot) const {
202 return (*this - other).RotateBy(rot) + other;
203 }
204
205 /**
206 * Computes the dot product between this translation and another translation
207 * in 3D space.
208 *
209 * The dot product between two translations is defined as x₁x₂+y₁y₂+z₁z₂.
210 *
211 * @param other The translation to compute the dot product with.
212 * @return The dot product between the two translations.
213 */
214 constexpr wpi::units::square_meter_t Dot(const Translation3d& other) const {
215 return m_x * other.X() + m_y * other.Y() + m_z * other.Z();
216 }
217
218 /**
219 * Computes the cross product between this translation and another
220 * translation in 3D space. The resulting translation will be perpendicular to
221 * both translations.
222 *
223 * The 3D cross product between two translations is defined as <y₁z₂-y₂z₁,
224 * z₁x₂-z₂x₁, x₁y₂-x₂y₁>.
225 *
226 * @param other The translation to compute the cross product with.
227 * @return The cross product between the two translations.
228 */
229 constexpr Eigen::Vector<wpi::units::square_meter_t, 3> Cross(
230 const Translation3d& other) const {
231 return Eigen::Vector<wpi::units::square_meter_t, 3>{
232 {m_y * other.Z() - other.Y() * m_z},
233 {m_z * other.X() - other.Z() * m_x},
234 {m_x * other.Y() - other.X() * m_y}};
235 }
236
237 /**
238 * Returns a Translation2d representing this Translation3d projected into the
239 * X-Y plane.
240 */
241 constexpr Translation2d ToTranslation2d() const {
242 return Translation2d{m_x, m_y};
243 }
244
245 /**
246 * Returns the sum of two translations in 3D space.
247 *
248 * For example, Translation3d{1.0, 2.5, 3.5} + Translation3d{2.0, 5.5, 7.5} =
249 * Translation3d{3.0, 8.0, 11.0}.
250 *
251 * @param other The translation to add.
252 *
253 * @return The sum of the translations.
254 */
255 constexpr Translation3d operator+(const Translation3d& other) const {
256 return {X() + other.X(), Y() + other.Y(), Z() + other.Z()};
257 }
258
259 /**
260 * Returns the difference between two translations.
261 *
262 * For example, Translation3d{5.0, 4.0, 3.0} - Translation3d{1.0, 2.0, 3.0} =
263 * Translation3d{4.0, 2.0, 0.0}.
264 *
265 * @param other The translation to subtract.
266 *
267 * @return The difference between the two translations.
268 */
269 constexpr Translation3d operator-(const Translation3d& other) const {
270 return operator+(-other);
271 }
272
273 /**
274 * Returns the inverse of the current translation. This is equivalent to
275 * negating all components of the translation.
276 *
277 * @return The inverse of the current translation.
278 */
279 constexpr Translation3d operator-() const { return {-m_x, -m_y, -m_z}; }
280
281 /**
282 * Returns the translation multiplied by a scalar.
283 *
284 * For example, Translation3d{2.0, 2.5, 4.5} * 2 = Translation3d{4.0, 5.0,
285 * 9.0}.
286 *
287 * @param scalar The scalar to multiply by.
288 *
289 * @return The scaled translation.
290 */
291 constexpr Translation3d operator*(double scalar) const {
292 return {scalar * m_x, scalar * m_y, scalar * m_z};
293 }
294
295 /**
296 * Returns the translation divided by a scalar.
297 *
298 * For example, Translation3d{2.0, 2.5, 4.5} / 2 = Translation3d{1.0, 1.25,
299 * 2.25}.
300 *
301 * @param scalar The scalar to divide by.
302 *
303 * @return The scaled translation.
304 */
305 constexpr Translation3d operator/(double scalar) const {
306 return operator*(1.0 / scalar);
307 }
308
309 /**
310 * Checks equality between this Translation3d and another object.
311 *
312 * @param other The other object.
313 * @return Whether the two objects are equal.
314 */
315 constexpr bool operator==(const Translation3d& other) const {
316 return wpi::units::math::abs(m_x - other.m_x) < 1E-9_m &&
317 wpi::units::math::abs(m_y - other.m_y) < 1E-9_m &&
318 wpi::units::math::abs(m_z - other.m_z) < 1E-9_m;
319 }
320
321 /**
322 * Returns the nearest Translation3d from a collection of translations
323 * @param translations The collection of translations.
324 * @return The nearest Translation3d from the collection.
325 */
326 constexpr Translation3d Nearest(
327 std::span<const Translation3d> translations) const {
328 return *std::min_element(
329 translations.begin(), translations.end(),
330 [this](const Translation3d& a, const Translation3d& b) {
331 return this->Distance(a) < this->Distance(b);
332 });
333 }
334
335 /**
336 * Returns the nearest Translation3d from a collection of translations
337 * @param translations The collection of translations.
338 * @return The nearest Translation3d from the collection.
339 */
340 constexpr Translation3d Nearest(
341 std::initializer_list<Translation3d> translations) const {
342 return *std::min_element(
343 translations.begin(), translations.end(),
344 [this](const Translation3d& a, const Translation3d& b) {
345 return this->Distance(a) < this->Distance(b);
346 });
347 }
348
349 private:
350 wpi::units::meter_t m_x = 0_m;
351 wpi::units::meter_t m_y = 0_m;
352 wpi::units::meter_t m_z = 0_m;
353};
