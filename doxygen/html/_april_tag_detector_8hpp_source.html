21/**
22 * An AprilTag detector engine. This is expensive to set up and tear down, so
23 * most use cases should only create one of these, add a family to it, set up
24 * any other configuration, and repeatedly call Detect().
25 */
26class WPILIB_DLLEXPORT AprilTagDetector {
27 public:
28 /** Detector configuration. */
29 struct Config {
30 bool operator==(const Config&) const = default;
31
32 /**
33 * How many threads should be used for computation. Default is
34 * single-threaded operation (1 thread).
35 */
36 int numThreads = 1;
37
38 /**
39 * Quad decimation. Detection of quads can be done on a lower-resolution
40 * image, improving speed at a cost of pose accuracy and a slight decrease
41 * in detection rate. Decoding the binary payload is still done at full
42 * resolution. Default is 2.0.
43 */
44 float quadDecimate = 2.0f;
45
46 /**
47 * What Gaussian blur should be applied to the segmented image (used for
48 * quad detection). Very noisy images benefit from non-zero values (e.g.
49 * 0.8). Default is 0.0.
50 */
51 float quadSigma = 0.0f;
52
53 /**
54 * When true, the edges of the each quad are adjusted to "snap to" strong
55 * gradients nearby. This is useful when decimation is employed, as it can
56 * increase the quality of the initial quad estimate substantially.
57 * Generally recommended to be on (true). Default is true.
58 *
59 * Very computationally inexpensive. Option is ignored if
60 * quad_decimate = 1.
61 */
62 bool refineEdges = true;
63
64 /**
65 * How much sharpening should be done to decoded images. This can help
66 * decode small tags but may or may not help in odd lighting conditions or
67 * low light conditions. Default is 0.25.
68 */
69 double decodeSharpening = 0.25;
70
71 /**
72 * Debug mode. When true, the decoder writes a variety of debugging images
73 * to the current working directory at various stages through the detection
74 * process. This is slow and should *not* be used on space-limited systems
75 * such as the Systemcore. Default is disabled (false).
76 */
77 bool debug = false;
78 };
79
80 /** Quad threshold parameters. */
81 struct QuadThresholdParameters {
82 bool operator==(const QuadThresholdParameters&) const = default;
83
84 /**
85 * Threshold used to reject quads containing too few pixels. Default is 300
86 * pixels.
87 */
88 int minClusterPixels = 300;
89
90 /**
91 * How many corner candidates to consider when segmenting a group of pixels
92 * into a quad. Default is 10.
93 */
94 int maxNumMaxima = 10;
95
96 /**
97 * Critical angle. The detector will reject quads where pairs of edges have
98 * angles that are close to straight or close to 180 degrees. Zero means
99 * that no quads are rejected. Default is 45 degrees.
100 */
101 wpi::units::radian_t criticalAngle = 45_deg;
102
103 /**
104 * When fitting lines to the contours, the maximum mean squared error
105 * allowed. This is useful in rejecting contours that are far from being
106 * quad shaped; rejecting these quads "early" saves expensive decoding
107 * processing. Default is 10.0.
108 */
109 float maxLineFitMSE = 10.0f;
110
111 /**
112 * Minimum brightness offset. When we build our model of black & white
113 * pixels, we add an extra check that the white model must be (overall)
114 * brighter than the black model. How much brighter? (in pixel values,
115 * [0,255]). Default is 5.
116 */
117 int minWhiteBlackDiff = 5;
118
119 /**
120 * Whether the thresholded image be should be deglitched. Only useful for
121 * very noisy images. Default is disabled (false).
122 */
123 bool deglitch = false;
124 };
