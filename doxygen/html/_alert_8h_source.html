21/** Represents an alert's level of urgency. */
22enum WPI_AlertLevel {
23 /**
24 * High priority alert - displayed first with a red "X" symbol. Use this type
25 * for problems which will seriously affect the robot's functionality and thus
26 * require immediate attention.
27 */
28 WPI_ALERT_HIGH = 0,
29
30 /**
31 * Medium priority alert - displayed second with a yellow "!" symbol. Use this
32 * type for problems which could affect the robot's functionality but do not
33 * necessarily require immediate attention.
34 */
35 WPI_ALERT_MEDIUM = 1,
36
37 /**
38 * Low priority alert - displayed last with a green "i" symbol. Use this type
39 * for problems which are unlikely to affect the robot's functionality, or any
40 * other alerts which do not fall under the other categories.
41 */
42 WPI_ALERT_LOW = 2,
43};
