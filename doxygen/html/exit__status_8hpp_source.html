12/// Solver exit status. Negative values indicate failure.
13enum class ExitStatus : int8_t {
14 /// Solved the problem to the desired tolerance.
15 SUCCESS = 0,
16 /// The solver returned its solution so far after the user requested a stop.
17 CALLBACK_REQUESTED_STOP = 1,
18 /// The solver determined the problem to be overconstrained and gave up.
19 TOO_FEW_DOFS = -1,
20 /// The solver determined the problem to be locally infeasible and gave up.
21 LOCALLY_INFEASIBLE = -2,
22 /// The problem setup frontend determined the problem to have an empty
23 /// feasible region.
24 GLOBALLY_INFEASIBLE = -3,
25 /// The linear system factorization failed.
26 FACTORIZATION_FAILED = -4,
27 /// The backtracking line search failed, and the problem isn't locally
28 /// infeasible.
29 LINE_SEARCH_FAILED = -5,
30 /// The solver failed to reach the desired tolerance, and feasibility
31 /// restoration failed to converge.
32 FEASIBILITY_RESTORATION_FAILED = -6,
33 /// The solver encountered nonfinite initial cost, constraints, or derivatives
34 /// and gave up.
35 NONFINITE_INITIAL_GUESS = -7,
36 /// The solver encountered diverging primal iterates xₖ and/or sₖ and gave up.
37 DIVERGING_ITERATES = -8,
38 /// The solver returned its solution so far after exceeding the maximum number
39 /// of iterations.
40 MAX_ITERATIONS_EXCEEDED = -9,
41 /// The solver returned its solution so far after exceeding the maximum
42 /// elapsed wall clock time.
43 TIMEOUT = -10,
44};
