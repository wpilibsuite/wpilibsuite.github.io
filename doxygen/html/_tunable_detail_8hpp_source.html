208template <typename T, typename... I>
209 requires wpi::util::StructSerializable<T, I...>
210class TunableStruct : public detail::TunableStructBase {
211 public:
212 template <typename... Args>
213 requires FirstArgNotTunableConfig<Args...>
214 constexpr explicit TunableStruct(I... info, Args&&... args)
215 : m_value{std::forward<Args>(args)...}, m_info{std::move(info)...} {}
216
217 template <typename... Args>
218 explicit TunableStruct(const TunableConfig& config, I... info, Args&&... args)
219 : detail::TunableStructBase{config},
220 m_value{std::forward<Args>(args)...},
221 m_info{std::move(info)...} {}
222
223 constexpr operator const T&() const { // NOLINT(google-explicit-constructor)
224 return m_value;
225 }
226
227 TunableStruct& operator=(T value) {
228 m_value = std::move(value);
229 SetTunableChanged();
230 return *this;
231 }
232
233 constexpr const T& Get() const { return m_value; }
234
235 void Set(T value) {
236 m_value = std::move(value);
237 SetTunableChanged();
238 }
239
240 constexpr T& Mutate() {
241 SetTunableChanged();
242 return m_value;
243 }
244
245 protected:
246 std::string GetStructTypeName() const override {
247 return std::apply(
248 [](const I&... info) {
249 return std::string{wpi::util::GetStructTypeName<T>(info...)};
250 },
251 m_info);
252 }
253
254 size_t GetStructSize() const override {
255 return std::apply(
256 [](const I&... info) { return wpi::util::GetStructSize<T>(info...); },
257 m_info);
258 }
259
260 void ForEachStructSchema(
261 wpi::util::function_ref<void(std::string_view typeName,
262 std::string_view schema)>
263 fn) const override {
264 std::apply(
265 [&](const I&... info) {
266 wpi::util::ForEachStructSchema<T>(fn, info...);
267 },
268 m_info);
269 }
270
271 bool UnpackStruct(std::span<const uint8_t> data) override {
272 if (data.size() != GetStructSize()) {
273 return false;
274 }
275 std::apply(
276 [&](const I&... info) {
277 wpi::util::UnpackStructInto(&m_value, data, info...);
278 },
279 m_info);
280 SetTunableChanged();
281 return true;
282 }
283
284 void PackStruct(std::span<uint8_t> buf) const override {
285 std::apply(
286 [&](const I&... info) { wpi::util::PackStruct(buf, m_value, info...); },
287 m_info);
288 }
289
290 private:
291 T m_value;
292 [[no_unique_address]]
293 std::tuple<I...> m_info;
294};

342 SetTunableChanged();
343 return m_value;
344 }

346 protected:
347 std::string GetStructTypeName() const override {
348 return std::apply(
349 [](const I&... info) {
350 return std::string{
351 wpi::util::MakeStructArrayTypeName<T, std::dynamic_extent>(
352 info...)};
353 },
354 m_info);
355 }
356
357 size_t GetStructSize() const override {
358 return m_value.size() * std::apply(
359 [](const I&... info) {
360 return wpi::util::GetStructSize<T>(info...);
361 },
362 m_info);
363 }
364
365 void ForEachStructSchema(
366 wpi::util::function_ref<void(std::string_view typeName,
367 std::string_view schema)>
368 fn) const override {
369 std::apply(
370 [&](const I&... info) {
371 wpi::util::ForEachStructSchema<T>(fn, info...);
372 },
373 m_info);
374 }
375
376 bool UnpackStruct(std::span<const uint8_t> data) override {
377 bool unpacked = std::apply(
378 [&](const I&... info) {
379 size_t size = wpi::util::GetStructSize<T>(info...);
380 if (size == 0 || data.size() % size != 0) {
381 return false;
382 }
383 m_value.clear();
384 m_value.reserve(data.size() / size);
385 for (size_t pos = 0, end = data.size(); pos < end; pos += size) {
386 m_value.emplace_back(
387 wpi::util::UnpackStruct<T>(data.subspan(pos, size), info...));
388 }
389 return true;
390 },
391 m_info);
392 if (unpacked) {
393 SetTunableChanged();
394 }
395 return unpacked;
396 }
397
398 void PackStruct(std::span<uint8_t> buf) const override {
399 std::apply(
400 [&](const I&... info) {
401 size_t size = wpi::util::GetStructSize<T>(info...);
402 for (auto&& val : m_value) {
403 if (buf.size() < size) {
404 break; // shouldn't happen, but avoid clobbering memory
405 }
406 wpi::util::PackStruct(buf.first(size), val, info...);
407 buf = buf.subspan(size);
408 }
409 },
410 m_info);
411 }
412
413 private:
414 std::vector<T> m_value;
415 [[no_unique_address]]
416 std::tuple<I...> m_info;
417};

496template <typename T, bool Member>
497inline auto CastTunable(TunableBase* tunable, TunableTypeValue type) {
498 if constexpr (std::same_as<T, TunableStructTag>) {
499 if constexpr (Member) {
500 return static_cast<TunableMemberStructBase*>(
501 type == TunableTypeValue::MEMBER_STRUCT ? tunable : nullptr);
502 } else {
503 return static_cast<TunableStructBase*>(
504 type == TunableTypeValue::STRUCT ? tunable : nullptr);
505 }
506 } else if constexpr (std::same_as<T, TunableProtobufTag>) {
507 if constexpr (Member) {
508 return static_cast<TunableMemberProtobufBase*>(
509 type == TunableTypeValue::MEMBER_PROTOBUF ? tunable : nullptr);
510 } else {
511 return static_cast<TunableProtobufBase*>(
512 type == TunableTypeValue::PROTOBUF ? tunable : nullptr);
513 }
514 } else {
515 if constexpr (Member) {
516 return static_cast<TunableMemberValueBase<T>*>(
517 type == GetTunableTypeValue<T, Member>() ? tunable : nullptr);
518 } else {
519 return static_cast<TunableValueBase<T>*>(
520 type == GetTunableTypeValue<T, Member>() ? tunable : nullptr);
521 }
522 }
523}
