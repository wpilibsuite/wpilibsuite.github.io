21/**
22 * Timer handle.
23 * Timer handles are used to schedule signals to be called in the future.
24 */
25class Timer final : public HandleImpl<Timer, uv_timer_t> {
26 struct private_init {};
27
28 public:
29 using Time = std::chrono::duration<uint64_t, std::milli>;
30
31 explicit Timer(const private_init&) {}
32 ~Timer() noexcept override = default;

37 * @param loop Loop object where this handle runs.
38 */
39 static std::shared_ptr<Timer> Create(Loop& loop);

41 /**
42 * Create a timer handle.
43 *
44 * @param loop Loop object where this handle runs.
45 */
46 static std::shared_ptr<Timer> Create(const std::shared_ptr<Loop>& loop) {
47 return Create(*loop);
48 }

50 /**
51 * Create a timer that calls a functor after a given time interval.
52 *
53 * @param loop Loop object where the timer should run.
54 * @param timeout Time interval
55 * @param func Functor
56 */
57 static void SingleShot(Loop& loop, Time timeout, std::function<void()> func);
58
59 /**
60 * Create a timer that calls a functor after a given time interval.
61 *
62 * @param loop Loop object where the timer should run.
63 * @param timeout Time interval
64 * @param func Functor
65 */
66 static void SingleShot(const std::shared_ptr<Loop>& loop, Time timeout,
67 std::function<void()> func) {
68 return SingleShot(*loop, timeout, std::move(func));
69 }
70
71 /**
72 * Start the timer.
73 *
74 * If timeout is zero, an event is emitted on the next event loop
75 * iteration. If repeat is non-zero, an event is emitted first
76 * after timeout milliseconds and then repeatedly after repeat milliseconds.
77 *
78 * @param timeout Milliseconds before to emit an event (use
79 * `std::chrono::duration<uint64_t, std::milli>`).
80 * @param repeat Milliseconds between successive events (use
81 * `std::chrono::duration<uint64_t, std::milli>`).
82 */
83 void Start(Time timeout, Time repeat = Time{0});
84
85 /**
86 * Stop the timer.
87 */
88 void Stop() { Invoke(&uv_timer_stop, GetRaw()); }
89
90 /**
91 * Stop the timer and restart it if it was repeating.
92 *
93 * Stop the timer, and if it is repeating restart it using the repeat value
94 * as the timeout.
95 * If the timer has never been started before it emits sigError.
96 */
97 void Again() { Invoke(&uv_timer_again, GetRaw()); }
98
99 /**
100 * Set the repeat interval value.
101 *
102 * The timer will be scheduled to run on the given interval and will follow
103 * normal timer semantics in the case of a time-slice overrun.
104 * For example, if a 50ms repeating timer first runs for 17ms, it will be
105 * scheduled to run again 33ms later. If other tasks consume more than the
106 * 33ms following the first timer event, then another event will be emitted
107 * as soon as possible.
108 *
109 * If the repeat value is set from a listener bound to an event, it does
110 * not immediately take effect. If the timer was non-repeating before, it
111 * will have been stopped. If it was repeating, then the old repeat value
112 * will have been used to schedule the next timeout.
113 *
114 * @param repeat Repeat interval in milliseconds (use
115 * `std::chrono::duration<uint64_t, std::milli>`).
116 */
117 void SetRepeat(Time repeat) { uv_timer_set_repeat(GetRaw(), repeat.count()); }
118
119 /**
120 * Get the timer repeat value.
121 * @return Timer repeat value in milliseconds (as a
122 * `std::chrono::duration<uint64_t, std::milli>`).
123 */
124 Time GetRepeat() const { return Time{uv_timer_get_repeat(GetRaw())}; }
125
126 /**
127 * Get the timer due value or 0 if it has expired. The time is relative to
128 * uv_now().
129 *
130 * @return Timer due value in milliseconds (as a
131 * `std::chrono::duration<uint64_t, std::milli>`).
132 */
133 Time GetDueIn() const { return Time{uv_timer_get_due_in(GetRaw())}; }
134
135 /**
136 * Signal generated when the timeout event occurs.
137 */
138 wpi::util::sig::Signal<> timeout;
139};
