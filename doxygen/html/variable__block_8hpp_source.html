35 /// Assigns a VariableBlock to the block.
36 ///
37 /// @param values VariableBlock of values.
38 /// @return This VariableBlock.
39 VariableBlock<Mat>& operator=(const VariableBlock<Mat>& values) {
40 if (this == &values) {
41 return *this;
42 }
43
44 if (m_mat == nullptr) {
45 m_mat = values.m_mat;
46 m_row_slice = values.m_row_slice;
47 m_row_slice_length = values.m_row_slice_length;
48 m_col_slice = values.m_col_slice;
49 m_col_slice_length = values.m_col_slice_length;
50 } else {
51 slp_assert(rows() == values.rows() && cols() == values.cols());
52
53 for (int row = 0; row < rows(); ++row) {
54 for (int col = 0; col < cols(); ++col) {
55 (*this)[row, col] = values[row, col];
56 }
57 }
58 }
59
60 return *this;
61 }

432 /// Compound matrix multiplication-assignment operator.
433 ///
434 /// @param rhs Variable to multiply.
435 /// @return Result of multiplication.
436 VariableBlock<Mat>& operator*=(const MatrixLike auto& rhs) {
437 slp_assert(cols() == rhs.rows() && cols() == rhs.cols());
438
439 for (int i = 0; i < rows(); ++i) {
440 Mat lhs_old_row = row(i);
441 for (int j = 0; j < rhs.cols(); ++j) {
442 Variable sum{Scalar(0)};
443 for (int k = 0; k < cols(); ++k) {
444 if constexpr (EigenMatrixLike<decltype(rhs)>) {
445 sum += lhs_old_row[k] * rhs(k, j);
446 } else {
447 sum += lhs_old_row[k] * rhs[k, j];
448 }
449 }
450 (*this)[i, j] = sum;
451 }
452 }
453
454 return *this;
455 }
