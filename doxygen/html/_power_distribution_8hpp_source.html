16/**
17 * Class for getting voltage, current, temperature, power and energy from the
18 * CTRE Power Distribution Panel (PDP) or REV Power Distribution Hub (PDH).
19 */
20class PowerDistribution : public wpi::telemetry::TelemetryLoggable {
21 public:
22 /// Default module number.
23 static constexpr int DEFAULT_MODULE = -1;
24
25 /**
26 * Power distribution module type.
27 */
28 enum class ModuleType {
29 /// CTRE (Cross The Road Electronics) CTRE Power Distribution Panel (PDP).
30 CTRE = 1,
31 /// REV Power Distribution Hub (PDH).
32 REV = 2
33 };
34
35 /**
36 * Constructs a PowerDistribution object.
37 *
38 * Detects the connected PDP/PDH using the default CAN ID (0 for CTRE and 1
39 * for REV).
40 *
41 * @param busId The bus ID.
42 */
43 explicit PowerDistribution(CANPort busId);
44
45 /**
46 * Constructs a PowerDistribution object.
47 *
48 * @param busId The bus ID.
49 * @param module The CAN ID of the PDP/PDH
50 * @param moduleType The type of module
51 */
52 PowerDistribution(CANPort busId, int module, ModuleType moduleType);
53
54 PowerDistribution(PowerDistribution&&) = default;
55 PowerDistribution& operator=(PowerDistribution&&) = default;
56
57 ~PowerDistribution() override = default;
58
59 /**
60 * Gets the number of channels for this power distribution object.
61 *
62 * @return Number of output channels (16 for PDP, 24 for PDH).
63 */
64 int GetNumChannels() const;
65
66 /**
67 * Query the input voltage of the PDP/PDH.
68 *
69 * @return The input voltage in volts
70 * @Common This is one of the commonly used methods for this class
71 */
72 double GetVoltage() const;
73
74 /**
75 * Query the temperature of the PDP.
76 *
77 * Not supported on the Rev PDH and returns 0.
78 *
79 *
80 * @return The temperature in degrees Celsius
81 * @Common This is one of the commonly used methods for this class
82 */
83 double GetTemperature() const;
84
85 /**
86 * Query the current of a single channel of the PDP/PDH.
87 *
88 * @param channel the channel to query (0-15 for PDP, 0-23 for PDH)
89 * @return The current of the channel in Amperes
90 * @Common This is one of the commonly used methods for this class
91 */
92 double GetCurrent(int channel) const;
93
94 /**
95 * Query all currents of the PDP.
96 *
97 * @return The current of each channel in Amperes
98 */
99 std::vector<double> GetAllCurrents() const;
100
101 /**
102 * Query the total current of all monitored PDP/PDH channels.
103 *
104 * @return The total current drawn from all channels in Amperes
105 * @Common This is one of the commonly used methods for this class
106 */
107 double GetTotalCurrent() const;
108
109 /**
110 * Query the total power drawn from all monitored PDP channels.
111 *
112 * Not supported on the Rev PDH and returns 0.
113 *
114 * @return The total power drawn in Watts
115 * @Common This is one of the commonly used methods for this class
116 */
117 double GetTotalPower() const;
118
119 /**
120 * Query the total energy drawn from the monitored PDP channels.
121 *
122 * Not supported on the Rev PDH and returns 0.
123 *
124 * @return The total energy drawn in Joules
125 */
126 double GetTotalEnergy() const;
127
128 /**
129 * Reset the total energy drawn from the PDP.
130 *
131 * Not supported on the Rev PDH and does nothing.
132 *
133 * @see PowerDistribution#GetTotalEnergy
134 */
135 void ResetTotalEnergy();
136
137 /**
138 * Remove all of the fault flags on the PDP/PDH.
139 */
140 void ClearStickyFaults();
141
142 /**
143 * Gets module number (CAN ID).
144 */
145 int GetModule() const;
146
147 /**
148 * Gets module type.
149 */
150 ModuleType GetType() const;
151
152 /**
153 * Gets whether the PDH switchable channel is turned on or off. Returns false
154 * with the CTRE PDP.
155 *
156 * @return The output state of the PDH switchable channel
157 */
158 bool GetSwitchableChannel() const;
159
160 /**
161 * Sets the PDH switchable channel on or off. Does nothing with the CTRE PDP.
162 *
163 * @param enabled Whether to turn the PDH switchable channel on or off
164 */
165 void SetSwitchableChannel(bool enabled);
