12/**
13 * The overall robot mode (not including enabled state).
14 */
15enum class RobotMode {
16 /// Unknown.
17 UNKNOWN = HAL_ROBOT_MODE_UNKNOWN,
18 /// Autonomous.
19 AUTONOMOUS = HAL_ROBOT_MODE_AUTONOMOUS,
20 /// Teleoperated.
21 TELEOPERATED = HAL_ROBOT_MODE_TELEOPERATED,
22 /// Utility.
23 UTILITY = HAL_ROBOT_MODE_UTILITY
24};

26/**
27 * A wrapper around Driver Station control word.
28 */
29class ControlWord {
30 public:
31 /**
32 * Default constructor.
33 */
34 ControlWord() = default;
35
36 /**
37 * Constructs from state values.
38 *
39 * @param opModeHash opmode hash
40 * @param robotMode robot mode
41 * @param enabled enabled
42 * @param eStop emergency stopped
43 * @param fmsAttached FMS attached
44 * @param dsAttached DS attached
45 */
46 ControlWord(int64_t opModeHash, RobotMode robotMode, bool enabled, bool eStop,
47 bool fmsAttached, bool dsAttached)
48 : m_word{HAL_MakeControlWord(opModeHash,
49 static_cast<HAL_RobotMode>(robotMode),
50 enabled, eStop, fmsAttached, dsAttached)} {}
51
52 /**
53 * Constructs from the native HAL value.
54 *
55 * @param word value
56 */
57 explicit ControlWord(HAL_ControlWord word) : m_word{word} {}
58
59 ControlWord(const ControlWord&) = default;
60 ControlWord& operator=(const ControlWord&) = default;
61
62 /**
63 * Updates from state values.
64 *
65 * @param opModeHash opmode hash
66 * @param robotMode robot mode
67 * @param enabled enabled
68 * @param eStop emergency stopped
69 * @param fmsAttached FMS attached
70 * @param dsAttached DS attached
71 */
72 void Update(int64_t opModeHash, RobotMode robotMode, bool enabled, bool eStop,
73 bool fmsAttached, bool dsAttached) {
74 m_word =
75 HAL_MakeControlWord(opModeHash, static_cast<HAL_RobotMode>(robotMode),
76 enabled, eStop, fmsAttached, dsAttached);
77 }
78
79 /**
80 * Updates from the native HAL value.
81 *
82 * @param word value
83 */
84 void Update(HAL_ControlWord word) { m_word = word; }
85
86 /**
87 * Check if the DS has enabled the robot.
88 *
89 * @return True if the robot is enabled and the DS is connected
90 */
91 bool IsEnabled() const { return HAL_ControlWord_IsEnabled(m_word); }
92
93 /**
94 * Gets the current robot mode.
95 *
96 * <p>Note that this does not indicate whether the robot is enabled or
97 * disabled.
98 *
99 * @return robot mode
100 */
101 RobotMode GetRobotMode() const {
102 return static_cast<RobotMode>(HAL_ControlWord_GetRobotMode(m_word));
103 }
104
105 /**
106 * Gets the current opmode ID.
107 *
108 * @return opmode
109 */
110 int64_t GetOpModeId() const { return HAL_ControlWord_GetOpModeId(m_word); }
111
112 /**
113 * Sets the opmode ID.
114 *
115 * @param id opmode ID
116 */
117 void SetOpModeId(int64_t id) { HAL_ControlWord_SetOpModeId(&m_word, id); }
118
119 /**
120 * Check if the robot is e-stopped.
121 *
122 * @return True if the robot is e-stopped
123 */
124 bool IsEStopped() const { return HAL_ControlWord_IsEStopped(m_word); }
125
126 /**
127 * Is the driver station attached to a Field Management System?
128 *
129 * @return True if the robot is competing on a field being controlled by a
130 * Field Management System
131 */
132 bool IsFMSAttached() const { return HAL_ControlWord_IsFMSAttached(m_word); }
133
134 /**
135 * Check if the DS is attached.
136 *
137 * @return True if the DS is connected to the robot
138 */
139 bool IsDSAttached() const { return HAL_ControlWord_IsDSAttached(m_word); }
140
141 /**
142 * Check if the DS is commanding autonomous mode.
143 *
144 * @return True if the robot is being commanded to be in autonomous mode
145 */
146 bool IsAutonomous() const { return GetRobotMode() == RobotMode::AUTONOMOUS; }
147
148 /**
149 * Check if the DS is commanding autonomous mode and if it has enabled the
150 * robot.
151 *
152 * @return True if the robot is being commanded to be in autonomous mode and
153 * enabled.
154 */
155 bool IsAutonomousEnabled() const {
156 return IsAutonomous() && IsEnabled() && IsDSAttached();
157 }
158
159 /**
160 * Check if the DS is commanding teleop mode.
161 *
162 * @return True if the robot is being commanded to be in teleop mode
163 */
164 bool IsTeleop() const { return GetRobotMode() == RobotMode::TELEOPERATED; }
165
166 /**
167 * Check if the DS is commanding teleop mode and if it has enabled the robot.
168 *
169 * @return True if the robot is being commanded to be in teleop mode and
170 * enabled.
171 */
172 bool IsTeleopEnabled() const {
173 return IsTeleop() && IsEnabled() && IsDSAttached();
174 }
175
176 /**
177 * Check if the DS is commanding utility mode.
178 *
179 * @return True if the robot is being commanded to be in utility mode
180 */
181 bool IsUtility() const { return GetRobotMode() == RobotMode::UTILITY; }
182
183 /**
184 * Check if the DS is commanding utility mode and if it has enabled the robot.
185 *
186 * @return True if the robot is being commanded to be in utility mode and
187 * enabled.
188 */
189 bool IsUtilityEnabled() const {
190 return IsUtility() && IsEnabled() && IsDSAttached();
191 }
192
193 /**
194 * Get the HAL raw value.
195 *
196 * @return Control word value
197 */
198 HAL_ControlWord GetValue() const { return m_word; }
199
200 private:
201 HAL_ControlWord m_word{.value = 0};
202};
203
204inline bool operator==(const ControlWord& lhs, const ControlWord& rhs) {
205 return lhs.GetValue().value == rhs.GetValue().value;
206}
207
208inline bool operator!=(const ControlWord& lhs, const ControlWord& rhs) {
209 return !(lhs == rhs);
210}
