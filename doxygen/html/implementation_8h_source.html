46/**
47 * Parses a URL string according to the WHATWG URL Standard.
48 *
49 * This is the main entry point for URL parsing in Ada. The function takes
50 * a string input and optionally a base URL for resolving relative URLs.
51 *
52 * @tparam result_type The URL type to return. Can be either `ada::url` or
53 * `ada::url_aggregator` (default). The `url_aggregator` type is more
54 * memory-efficient as it stores components as offsets into a single
55 * buffer.
56 *
57 * @param input The URL string to parse. Must be valid ASCII or UTF-8 encoded.
58 * Leading and trailing whitespace is automatically trimmed.
59 * @param base_url Optional pointer to a base URL for resolving relative URLs.
60 * If nullptr (default), only absolute URLs can be parsed successfully.
61 *
62 * @return A `result<result_type>` containing either the parsed URL on success,
63 * or an error code on failure. Use the boolean conversion or
64 * `has_value()` to check for success.
65 *
66 * @note The parser is fully compliant with the WHATWG URL Standard.
67 *
68 * Parsing fails if the input or the resulting normalized URL exceeds
69 * `get_max_input_length()` bytes (default ~4 GB, configurable via
70 * `set_max_input_length()`). This accounts for percent-encoding expansion:
71 * a short input that normalizes into a long URL is still rejected.
72 *
73 * @example
74 * ```cpp
75 * // Parse an absolute URL
76 * auto url = ada::parse("https://user:pass@example.com:8080/path?query#hash");
77 * if (url) {
78 * std::cout << url->get_hostname(); // "example.com"
79 * std::cout << url->get_pathname(); // "/path"
80 * }
81 *
82 * // Parse a relative URL with a base
83 * auto base = ada::parse("https://example.com/dir/");
84 * if (base) {
85 * auto relative = ada::parse("../other/page", &*base);
86 * if (relative) {
87 * std::cout << relative->get_href(); //
88 * "https://example.com/other/page"
89 * }
90 * }
91 * ```
92 *
93 * @see https://url.spec.whatwg.org/#url-parsing
94 */
95template <class result_type = ada::url_aggregator>
96ada_warn_unused ada::result<result_type> parse(
97 std::string_view input, const result_type* base_url = nullptr);

162/**
163 * Sets the maximum allowed length for URLs.
164 *
165 * Both the raw input and the resulting normalized URL (the href) are checked
166 * against this limit. Parsing or setter calls that would produce a URL
167 * exceeding this length are rejected. The same limit also applies to
168 * `href_from_file` and to query strings passed to `url_search_params`
169 * construction / `reset`. The value must fit in a uint32_t.
170 * The default is std::numeric_limits<uint32_t>::max() (approximately 4 GB).
171 *
172 * @param length The new maximum URL length in bytes.
173 */
174void set_max_input_length(uint32_t length);
