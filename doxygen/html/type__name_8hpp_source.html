24struct RawTypeNameTraits {
25 static constexpr auto name = GetTypeNameHelper<double>();
26 static constexpr size_t len = std::string_view("double").size();
27 static constexpr size_t prefixLen = name.find("double");
28 static_assert(prefixLen != std::string_view::npos,
29 "cannot extract typename from function signature");
30 static constexpr size_t suffixLen = name.size() - prefixLen - len;
31};
32} // namespace detail

34/**
35 * Gets the name of a type as a string view.
36 *
37 * @tparam T the type
38 * @return the name of the type
39 */
40template <typename T>
41constexpr std::string_view GetTypeName() {
42 std::string_view name = detail::GetTypeNameHelper<T>();
43 return std::string_view(name.data() + detail::RawTypeNameTraits::prefixLen,
44 name.size() - detail::RawTypeNameTraits::prefixLen -
45 detail::RawTypeNameTraits::suffixLen);
46}
