31 /**
32 * Allocates a PIDController with the given constants for Kp, Ki, and Kd.
33 *
34 * @param Kp The proportional coefficient. Must be >= 0.
35 * @param Ki The integral coefficient. Must be >= 0.
36 * @param Kd The derivative coefficient. Must be >= 0.
37 * @param period The period between controller updates in seconds. The
38 * default is 20 milliseconds. Must be positive.
39 */
40 constexpr PIDController(double Kp, double Ki, double Kd,
41 wpi::units::second_t period = 20_ms)
42 : m_Kp(Kp), m_Ki(Ki), m_Kd(Kd), m_period(period) {
43 bool invalidGains = false;
44 if (Kp < 0.0) {
45 wpi::math::MathSharedStore::ReportError(
46 "Kp must be a non-negative number, got {}!", Kp);
47 invalidGains = true;
48 }
49 if (Ki < 0.0) {
50 wpi::math::MathSharedStore::ReportError(
51 "Ki must be a non-negative number, got {}!", Ki);
52 invalidGains = true;
53 }
54 if (Kd < 0.0) {
55 wpi::math::MathSharedStore::ReportError(
56 "Kd must be a non-negative number, got {}!", Kd);
57 invalidGains = true;
58 }
59 if (invalidGains) {
60 m_Kp = 0.0;
61 m_Ki = 0.0;
62 m_Kd = 0.0;
63 wpi::math::MathSharedStore::ReportWarning("PID gains defaulted to 0.");
64 }
65
66 if (period <= 0_s) {
67 wpi::math::MathSharedStore::ReportError(
68 "Controller period must be a positive number, got {}!",
69 period.value());
70 m_period = 20_ms;
71 wpi::math::MathSharedStore::ReportWarning(
72 "Controller period defaulted to 20ms.");
73 }
74 if !consteval {
75 ++instances;
76
77 wpi::util::ReportUsage("PIDController", std::to_string(instances));
78 }
79 }

132 /**
133 * Sets the IZone range. When the absolute value of the position error is
134 * greater than IZone, the total accumulated error will reset to zero,
135 * disabling integral gain until the absolute value of the position error is
136 * less than IZone. This is used to prevent integral windup. Must be
137 * non-negative. Passing a value of zero will effectively disable integral
138 * gain. Passing a value of infinity disables IZone functionality.
139 *
140 * @param iZone Maximum magnitude of error to allow integral control. Must be
141 * >= 0.
142 */
143 constexpr void SetIZone(double iZone) {
144 if (std::is_constant_evaluated() && iZone < 0) {
145 wpi::math::MathSharedStore::ReportError(
146 "IZone must be a non-negative number, got {}!", iZone);
147 }
148 m_iZone = iZone;
149 }

210 /**
211 * Sets the setpoint for the PIDController.
212 *
213 * @param setpoint The desired setpoint.
214 */
215 constexpr void SetSetpoint(double setpoint) {
216 m_setpoint = setpoint;
217 m_haveSetpoint = true;
218
219 if (m_continuous) {
220 double errorBound = (m_maximumInput - m_minimumInput) / 2.0;
221 m_error =
222 InputModulus(m_setpoint - m_measurement, -errorBound, errorBound);
223 } else {
224 m_error = m_setpoint - m_measurement;
225 }
226
227 m_errorDerivative = (m_error - m_prevError) / m_period.value();
228 }

237 /**
238 * Returns true if the error is within the tolerance of the setpoint.
239 * The error tolerance defaults to 0.05, and the error derivative tolerance
240 * defaults to ∞.
241 *
242 * This will return false until at least one input value has been computed.
243 */
244 constexpr bool AtSetpoint() const {
245 return m_haveMeasurement && m_haveSetpoint &&
246 gcem::abs(m_error) < m_errorTolerance &&
247 gcem::abs(m_errorDerivative) < m_errorDerivativeTolerance;
248 }

316 /**
317 * Returns the next output of the PID controller.
318 *
319 * @param measurement The current measurement of the process variable.
320 */
321 constexpr double Calculate(double measurement) {
322 m_measurement = measurement;
323 m_prevError = m_error;
324 m_haveMeasurement = true;
325
326 if (m_continuous) {
327 double errorBound = (m_maximumInput - m_minimumInput) / 2.0;
328 m_error =
329 InputModulus(m_setpoint - m_measurement, -errorBound, errorBound);
330 } else {
331 m_error = m_setpoint - m_measurement;
332 }
333
334 m_errorDerivative = (m_error - m_prevError) / m_period.value();
335
336 // If the absolute value of the position error is outside of IZone, reset
337 // the total error
338 if (gcem::abs(m_error) > m_iZone) {
339 m_totalError = 0;
340 } else if (m_Ki != 0) {
341 m_totalError =
342 std::clamp(m_totalError + m_error * m_period.value(),
343 m_minimumIntegral / m_Ki, m_maximumIntegral / m_Ki);
344 }
345
346 return m_Kp * m_error + m_Ki * m_totalError + m_Kd * m_errorDerivative;
347 }
348
349 /**
350 * Returns the next output of the PID controller.
351 *
352 * @param measurement The current measurement of the process variable.
353 * @param setpoint The new setpoint of the controller.
354 */
355 constexpr double Calculate(double measurement, double setpoint) {
356 m_setpoint = setpoint;
357 m_haveSetpoint = true;
358 return Calculate(measurement);
359 }
