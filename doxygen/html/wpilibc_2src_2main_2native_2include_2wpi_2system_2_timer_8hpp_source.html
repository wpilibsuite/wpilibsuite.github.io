33/**
34 * A timer class.
35 *
36 * Note that if the user calls wpi::sim::RestartTiming(), they should also reset
37 * the timer so Get() won't return a negative duration.
38 */
39class Timer {
40 public:
41 /**
42 * Create a new timer object.
43 *
44 * Create a new timer object and reset the time to zero. The timer is
45 * initially not running and must be started. Consider using
46 * CreateStarted() if the timer is used immediately after creation.
47 */
48 Timer();
49
50 virtual ~Timer() = default;
51
52 Timer(const Timer&) = default;
53 Timer& operator=(const Timer&) = default;
54 Timer(Timer&&) = default;
55 Timer& operator=(Timer&&) = default;

57 /**
58 * Get the current time from the timer. If the clock is running it is derived
59 * from the current system clock the start time stored in the timer class. If
60 * the clock is not running, then return the time when it was last stopped.
61 *
62 * @return Current time value for this timer in seconds
63 * @Common This is one of the commonly used methods for this class
64 */
65 wpi::units::second_t Get() const;
66
67 /**
68 * Reset the timer by setting the time to 0.
69 *
70 * Make the timer startTime the current time so new requests will be relative
71 * to now.
72 *
73 * @Common This is one of the commonly used methods for this class
74 */
75 void Reset();
76
77 /**
78 * Start the timer running.
79 *
80 * Just set the running flag to true indicating that all time requests should
81 * be relative to the system clock. Note that this method is a no-op if the
82 * timer is already running.
83 *
84 * @Common This is one of the commonly used methods for this class
85 */
86 void Start();
87
88 /**
89 * Restart the timer by stopping the timer, if it is not already stopped,
90 * resetting the accumulated time, then starting the timer again. If you
91 * want an event to periodically reoccur at some time interval from the
92 * start time, consider using AdvanceIfElapsed() instead.
93 *
94 * @Common This is one of the commonly used methods for this class
95 */
96 void Restart();
97
98 /**
99 * Stop the timer.
100 *
101 * This computes the time as of now and clears the running flag, causing all
102 * subsequent time requests to be read from the accumulated time rather than
103 * looking at the system clock.
104 *
105 * @Common This is one of the commonly used methods for this class
106 */
107 void Stop();
108
109 /**
110 * Check if the period specified has passed.
111 *
112 * @param period The period to check.
113 * @return True if the period has passed.
114 */
115 bool HasElapsed(wpi::units::second_t period) const;
116
117 /**
118 * Check if the period specified has passed and if it has, advance the start
119 * time by that period. This is useful to decide if it's time to do periodic
120 * work without drifting later by the time it took to get around to checking.
121 *
122 * @param period The period to check for.
123 * @return True if the period has passed.
124 */
125 bool AdvanceIfElapsed(wpi::units::second_t period);
126
127 /**
128 * Whether the timer is currently running.
129 *
130 * @return true if running.
131 */
132 bool IsRunning() const;
133
134 /**
135 * Creates a new timer that begins started.
136 *
137 * <p>This is equivalent to
138 * <pre>
139 * wpi::Timer timer;
140 * timer.Start();
141 * </pre>
142 *
143 * @return A new started timer.
144 */
145 static wpi::Timer CreateStarted();
146
147 /**
148 * Return the clock time in seconds. By default, the time is the time returned
149 * by GetMonotonicTimestamp(). However, the return value of this method may be
150 * modified to use any time base, including non-monotonic time bases.
151 *
152 * @returns Robot running time in seconds.
153 */
154 static wpi::units::second_t GetTimestamp();
155
156 /**
157 * Return the monotonic clock time in seconds.
158 *
159 * Return the time from the monotonic clock in seconds.
160 *
161 * @returns Monotonic time in seconds.
162 */
163 static wpi::units::second_t GetMonotonicTimestamp();
164
165 /**
166 * Return the approximate match time. The FMS does not send an official match
167 * time to the robots, but does send an approximate match time. The value will
168 * count down the time remaining in the current period (auto or teleop).
169 * Warning: This is not an official time (so it cannot be used to dispute ref
170 * calls or guarantee that a function will trigger before the match ends).
171 *
172 * <p>When connected to the real field, this number only changes in full
173 * integer increments, and always counts down.
174 *
175 * <p>When the DS is in practice mode, this number is a floating point number,
176 * and counts down.
177 *
178 * <p>When the DS is in teleop or autonomous mode, this number returns -1.0.
179 *
180 * <p>Simulation matches DS behavior without an FMS connected.
181 *
182 * @return Time remaining in current match period (auto or teleop) in seconds
183 */
184 static wpi::units::second_t GetMatchTime();
185
186 private:
187 double GetNanoseconds() const;
188
189 std::chrono::nanoseconds m_startTime{0};
190 double m_startTimeRemainderNs = 0.0;
191 double m_accumulatedTimeNs = 0.0;
192 bool m_running = false;
193};
