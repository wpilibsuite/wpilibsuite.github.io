15/**
16 * A move-only C++ wrapper around a WPI handle.
17 * Will free the handle if FreeFunction is provided
18 */
19template <typename CType, auto FreeFunction = nullptr,
20 auto CInvalid = WPI_INVALID_HANDLE>
21 requires std::same_as<decltype(FreeFunction), std::nullptr_t> ||
22 std::invocable<decltype(FreeFunction), CType>
23class Handle {
24 public:
25 Handle() = default;
26 /*implicit*/ Handle(CType val) : m_handle(val) {} // NOLINT
27
28 Handle(const Handle&) = delete;
29 Handle& operator=(const Handle&) = delete;
30
31 Handle(Handle&& rhs) : m_handle(rhs.m_handle) {
32 rhs.m_handle = static_cast<CType>(CInvalid);
33 }
34
35 Handle& operator=(Handle&& rhs) {
36 if (this != &rhs) {
37 if constexpr (std::invocable<decltype(FreeFunction), CType>) {
38 if (m_handle != static_cast<CType>(CInvalid)) {
39 FreeFunction(m_handle);
40 }
41 }
42 }
43 m_handle = rhs.m_handle;
44 rhs.m_handle = static_cast<CType>(CInvalid);
45 return *this;
46 }
47
48 /**
49 * Releases ownership of the handle without freeing it.
50 *
51 * @return The released handle.
52 */
53 CType Release() {
54 CType handle = m_handle;
55 m_handle = static_cast<CType>(CInvalid);
56 return handle;
57 }
58
59 ~Handle() {
60 if constexpr (std::invocable<decltype(FreeFunction), CType>) {
61 if (m_handle != static_cast<CType>(CInvalid)) {
62 FreeFunction(m_handle);
63 }
64 }
65 }
66
67 operator CType() const { return m_handle; } // NOLINT
68
69 private:
70 CType m_handle = static_cast<CType>(CInvalid);
71};
