15/**
16 * Handle input from Gamepad controllers connected to the Driver Station.
17 *
18 * This class handles Gamepad input that comes from the Driver Station. Each
19 * time a value is requested the most recent value is returned. There is a
20 * single class instance for each controller and the mapping of ports to
21 * hardware buttons depends on the code in the Driver Station.
22 *
23 * Only first party controllers from Generic are guaranteed to have the
24 * correct mapping, and only through the official NI DS. Sim is not guaranteed
25 * to have the same mapping, as well as any 3rd party controllers.
26 */
27class Gamepad : public HIDDevice, public wpi::telemetry::TelemetryLoggable {
28 public:

233 /**
234 * Constructs an event instance around the axis value of the left trigger.
235 * The returned trigger will be true when the axis value is greater than
236 * {@code threshold}.
237 * @param threshold the minimum axis value for the returned event to be true.
238 * This value should be in the range [0, 1] where 0 is the unpressed state of
239 * the axis.
240 * @param loop the event loop instance to attach the event to.
241 * @return an event instance that is true when the left trigger's axis
242 * exceeds the provided threshold, attached to the given event loop
243 */
244 BooleanEvent LeftTrigger(double threshold, EventLoop* loop) const;
