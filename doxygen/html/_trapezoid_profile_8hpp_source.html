61/**
62 * A trapezoid-shaped velocity profile.
63 *
64 * While this class can be used for a profiled movement from start to finish,
65 * the intended usage is to filter a reference's dynamics based on trapezoidal
66 * velocity constraints. To compute the reference obeying this constraint, do
67 * the following.
68 *
69 * Initialization:
70 * @code{.cpp}
71 * TrapezoidProfile::Constraints constraints{kMaxV, kMaxA};
72 * TrapezoidProfile::State previousProfiledReference = initialReference;
73 * TrapezoidProfile profile{constraints};
74 * @endcode
75 *
76 * Run on update:
77 * @code{.cpp}
78 * previousProfiledReference = profile.Calculate(timeSincePreviousUpdate,
79 * previousProfiledReference,
80 * unprofiledReference);
81 * @endcode
82 *
83 * where `unprofiledReference` is free to change between calls. Note that when
84 * the unprofiled reference is within the constraints, `Calculate()` returns the
85 * unprofiled reference unchanged.
86 *
87 * Otherwise, a timer can be started to provide monotonic values for
88 * `Calculate()` and to determine when the profile has completed via
89 * `IsFinished()`.
90 */
91template <class Distance>
92class TrapezoidProfile {
93 public:
94 using Distance_t = wpi::units::unit_t<Distance>;
95 using Velocity =
96 wpi::units::compound_unit<Distance,
97 wpi::units::inverse<wpi::units::seconds>>;
98 using Velocity_t = wpi::units::unit_t<Velocity>;
99 using Acceleration =
100 wpi::units::compound_unit<Velocity,
101 wpi::units::inverse<wpi::units::seconds>>;
102 using Acceleration_t = wpi::units::unit_t<Acceleration>;
103
104 using Constraints = detail::TrapezoidProfileConstraints<Distance>;
105
106 /**
107 * Profile state.
108 */
109 class State {
110 public:
111 /// The position at this state.
112 Distance_t position{0};
113
114 /// The velocity at this state.
115 Velocity_t velocity{0};
116
117 constexpr bool operator==(const State&) const = default;
118 };

120 /**
121 * Profile timings.
122 */
123 class ProfileTiming {
124 public:
125 /// The time the profile spends in the first leg.
126 wpi::units::second_t t_1;
127 /// The time the profile spends at the velocity limit.
128 wpi::units::second_t t_2;
129 /// The time the profile spends in the last leg.
130 wpi::units::second_t t_3;
131
132 constexpr bool operator==(const ProfileTiming&) const = default;
133 };

136 * Constructs a TrapezoidProfile.

148 /**
149 * Calculates the position and velocity for the profile at a time t where the
150 * current state is at time t = 0.
151 *
152 * @param t The amount of time to advance from the current state toward the
153 * desired state.
154 * @param current The current state.
155 * @param goal The desired state when the profile is complete.
156 * @return The position and velocity of the profile at time t.
157 */
158 constexpr State Calculate(wpi::units::second_t t, State current, State goal) {
159 // Sampled trajectory should start at the current state, regardless of
160 // validity.
161 State sample{current};
162
163 // Adjust states so that they are within the constraints and get the time
164 // required for the current state to return to a valid state.
165 wpi::units::second_t recoveryTime = AdjustStates(current, goal);
166 double sign = GetSign(current, goal);
167 m_profile = GenerateProfile(sign, current, goal);
168
169 // In the case that the sign of the profile and the sign of the acceleration
170 // are identical, the recovery can be treated as an extension of the first
171 // segment. In the case that they differ, the recovered state will have a
172 // velocity of v_l and the above calculated t_1 will be zero. To handle
173 // this, a check is added to the sampling of the first segment to ensure
174 // proper recovery.
175 m_profile.t_1 += recoveryTime;
176
177 auto advance = [](wpi::units::second_t time, Acceleration_t acceleration,
178 State& state) {
179 // x = x_i + v_i t + at² / 2 (2)
180 state.position +=
181 state.velocity * time + acceleration / 2.0 * time * time;
182 // v = v_i + at (1)
183 state.velocity += acceleration * time;
184 };
185
186 Acceleration_t acceleration = sign * m_constraints.maxAcceleration;
187 advance(wpi::units::math::min(t, m_profile.t_1),
188 // Handle recovery to a feasible state if necessary.
189 recoveryTime > 0.0_s && sample.velocity * sign > Velocity_t{0.0}
190 ? -acceleration
191 : acceleration,
192 sample);
193
194 if (t > m_profile.t_1) {
195 t -= m_profile.t_1;
196 advance(wpi::units::math::min(t, m_profile.t_2), Acceleration_t{0.0},
197 sample);
198
199 if (t > m_profile.t_2) {
200 t -= m_profile.t_2;
201 advance(wpi::units::math::min(t, m_profile.t_3), -acceleration, sample);
202
203 if (t > m_profile.t_3) {
204 sample = goal;
205 }
206 }
207 }
208
209 return sample;
210 }
211
212 /**
213 * Returns the time to get between two states. This does not affect the
214 * internal variables, and as a result, may be used for states not on the
215 * active trajectory.
216 *
217 * @param current The current state.
218 * @param goal The goal state.
219 * @return The time left until the target state.
220 */
221 constexpr wpi::units::second_t TimeLeftUntil(State current,
222 State goal) const {
223 // Adjust states so that they are within the constraints and get the time
224 // required for the current state to return to a valid state.
225 wpi::units::second_t recoveryTime = AdjustStates(current, goal);
226 double sign = GetSign(current, goal);
227 ProfileTiming profile = GenerateProfile(sign, current, goal);
228
229 profile.t_1 += recoveryTime;
230
231 return profile.t_1 + profile.t_2 + profile.t_3;
232 }
233
234 /**
235 * Returns the duration of the profile.
236 *
237 * @return The duration of the profile, or zero if no goal was set.
238 */
239 constexpr wpi::units::second_t Duration() const {
240 return m_profile.t_1 + m_profile.t_2 + m_profile.t_3;
241 }
242
243 /**
244 * Returns true if the profile has reached the goal.
245 *
246 * The profile has reached the goal if the time since the profile started has
247 * exceeded the profile's total time.
248 *
249 * @param t The time since the beginning of the profile.
250 * @return True if the profile has reached the goal.
251 */
252 constexpr bool IsFinished(wpi::units::second_t t) const {
253 return t >= Duration();
254 }
255
256 private:
257 /**
258 * Adjusts the profile states to be within the constraints and returns the
259 * time needed to bring the current state back within the constraints.
260 *
261 * In order to smoothly return to a state within the constraints, the current
262 * state is modified to be the result of accelerating towards a valid
263 * velocity at the maximum acceleration. This method returns the time this
264 * transition takes. By contrast, the goal velocity is simply clamped
265 * to the valid region.
266 *
267 * @param current The current state to be adjusted.
268 * @param goal The goal state state to be adjusted.
269 * @return The time taken to make the current state valid.
270 */
271 constexpr wpi::units::second_t AdjustStates(State& current,
272 State& goal) const {
273 if (wpi::units::math::abs(goal.velocity) > m_constraints.maxVelocity) {
274 goal.velocity =
275 wpi::units::math::copysign(m_constraints.maxVelocity, goal.velocity);
276 }
277
278 wpi::units::second_t recoveryTime{0.0};
279 Velocity_t violationAmount =
280 wpi::units::math::abs(current.velocity) - m_constraints.maxVelocity;
281
282 if (violationAmount > Velocity_t{0.0}) {
283 recoveryTime = violationAmount / m_constraints.maxAcceleration;
284 // x = x_i + v_i t + at² / 2 (2)
285 current.position +=
286 current.velocity * recoveryTime +
287 wpi::units::math::copysign(m_constraints.maxAcceleration,
288 -current.velocity) *
289 recoveryTime * recoveryTime / 2.0;
290 // The closest valid velocity will have the magnitude of the max velocity.
291 current.velocity = wpi::units::math::copysign(m_constraints.maxVelocity,
292 current.velocity);
293 }
294
295 return recoveryTime;
296 }
297
298 /**
299 * Returns the sign of the profile.
300 *
301 * The current and goal states must be within the profile constraints for a
302 * valid sign.
303 *
304 * @param current The initial state, adjusted not to violate the constraints.
305 * @param goal The goal state of the profile.
306 * @return 1.0 if the profile direction is positive, -1.0 if it is not.
307 */
308 constexpr double GetSign(const State& current, const State& goal) const {
309 Distance_t dx = goal.position - current.position;
310
311 // Calculate threshold displacement
312 // d = |v_t - v_i|(v_t + v_i) / (2 a_m) (9)
313 Distance_t d = wpi::units::math::abs(goal.velocity - current.velocity) *
314 (goal.velocity + current.velocity) /
315 (2.0 * m_constraints.maxAcceleration);
316
317 // As discussed in TrapezoidProfile.md, the correct sign must be chosen when
318 // dx == d because following a suboptimal profile may lead to "chattering".
319 // Additionally, if numerical precision errors cause the calculated optimal
320 // sign to change throughout the profile, that may lead to suboptimal states
321 // being calculated. To fix this, we add a tolerance such that if |dx - d| <
322 // epsilon, we return the sign that would lead to the minimum profile being
323 // calculated. We do not have control over the floating point precision
324 // error from previous calculations, and as such, it is difficult to bound
325 // the possible error. 1e-12 should be good enough for FRC though.
326 if (wpi::units::math::abs(dx - d) < Distance_t{1e-12}) {
327 return std::copysign(1.0, goal.velocity.value());
328 } else {
329 if (dx > d) {
330 return 1.0;
331 } else {
332 return -1.0;
333 }
334 }
335 }

337 /**
338 * Generates profile timings from valid current and goal states.
339 *
340 * Returns the time for each section of the profile from current
341 * and goal states with valid velocities.
342 *
343 * @param sign The sign of the profile to generate.
344 * @param current The valid current state.
345 * @param goal The valid goal state.
346 * @return The time for each section of the profile.
347 */
348 constexpr ProfileTiming GenerateProfile(double sign, const State& current,
349 const State& goal) const {
350 ProfileTiming profile{};
351
352 Acceleration_t acceleration = sign * m_constraints.maxAcceleration;
353 Velocity_t velocityLimit = sign * m_constraints.maxVelocity;
354 Distance_t dx = goal.position - current.position;
355
356 // Calculate the peak velocity to compare to velocity constraint.
357 // v_p = √(aΔx + (v_t² + v_i²) / 2) (8)
358 Velocity_t peakVelocity =
359 sign * wpi::units::math::sqrt(wpi::units::math::max(
360 acceleration * dx + (goal.velocity * goal.velocity +
361 current.velocity * current.velocity) /
362 2,
363 wpi::units::math::pow<2>(Velocity_t{0.0})));
364
365 // Handle the case where we hit maximum velocity.
366 if (sign * peakVelocity > m_constraints.maxVelocity) {
367 // t_1 = (v_l - v_i) / a (13)
368 profile.t_1 = (velocityLimit - current.velocity) / acceleration;
369 // t_3 = (v_l - v_t) / a (15)
370 profile.t_3 = (velocityLimit - goal.velocity) / acceleration;
371
372 // x_1 = (v_p² - v_i²) / (2a) (6)
373 // Substitute v_p for v_l because this is the velocity constrained case.
374 // x_1 = (v_l² - v_i²) / (2a)
375 Distance_t x_1 = (velocityLimit * velocityLimit -
376 current.velocity * current.velocity) /
377 (2 * acceleration);
378
379 // x_3 = (v_p² - v_t²) / (2a) (7)
380 // Substitute v_p for v_l because this is the velocity constrained case.
381 // x_3 = (v_l² - v_t²) / (2a)
382 Distance_t x_3 =
383 (velocityLimit * velocityLimit - goal.velocity * goal.velocity) /
384 (2 * acceleration);
385
386 // x_2 = Δx - x_1 - x_3 (12)
387 Distance_t x_2 = dx - x_1 - x_3;
388
389 // t_2 = x_2 / v_l (14)
390 profile.t_2 = x_2 / velocityLimit;
391 } else {
392 // t_1 = (v_p - v_i) / a (13)
393 profile.t_1 = (peakVelocity - current.velocity) / acceleration;
394 // t_3 = (v_p - v_t) / a (15)
395 profile.t_3 = (peakVelocity - goal.velocity) / acceleration;
396 }
397
398 return profile;
399 }

402 ProfileTiming m_profile;
403};
