17/**
18 * Visual 2D representation of arms, elevators, and general mechanisms through
19 * a node-based API.
20 *
21 * A Mechanism2d object is published and contains at least one root node. A root
22 * is the anchor point of other nodes (such as ligaments). Other nodes are
23 * recursively appended based on other nodes.
24 *
25 * Except for the Mechanism2d container object, none of the objects should be
26 * passed or interacted with by value! Obtain pointers from factory methods such
27 * as Mechanism2d.GetRoot() and MechanismObject2d.Append<>(). The Mechanism2d
28 * container object owns the root nodes, and each node internally owns the nodes
29 * based on it. Beware not to let the Mechanism2d object out of scope - all
30 * nodes will be recursively destructed!
31 *
32 * @see MechanismObject2d
33 * @see MechanismLigament2d
34 * @see MechanismRoot2d
35 */
36class Mechanism2d : public wpi::telemetry::TelemetryLoggable {
37 public:
38 /**
39 * Create a new Mechanism2d with the given dimensions and background color.
40 *
41 * The dimensions represent the canvas that all the nodes are drawn on. The
42 * default color is dark blue.
43 *
44 * @param width the width
45 * @param height the height
46 * @param backgroundColor the background color
47 */
48 Mechanism2d(double width, double height,
49 const wpi::util::Color8Bit& backgroundColor = {0, 0, 32});
50
51 /**
52 * Get or create a root in this Mechanism2d with the given name and
53 * position.
54 *
55 * <p>If a root with the given name already exists, the given x and y
56 * coordinates are not used.
57 *
58 * @param name the root name
59 * @param x the root x coordinate
60 * @param y the root y coordinate
61 * @return a new root object, or the existing one with the given name.
62 */
63 MechanismRoot2d* GetRoot(std::string_view name, double x, double y);
64
65 /**
66 * Set the Mechanism2d background color.
67 *
68 * @param color the new background color
69 */
70 void SetBackgroundColor(const wpi::util::Color8Bit& color);
71
72 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
73
74 std::string_view GetTelemetryType() const override;
75
76 private:
77 double m_width;
78 double m_height;
79 std::string m_color;
80 mutable wpi::util::mutex m_mutex;
81 wpi::util::StringMap<MechanismRoot2d> m_roots;
82};
