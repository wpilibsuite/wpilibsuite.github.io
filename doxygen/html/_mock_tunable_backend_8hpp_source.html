186 /**
187 * Gets the value of a struct-serialized tunable.
188 *
189 * @tparam T value type
190 * @tparam I optional info parameters for the struct serializer
191 * @param path normalized path
192 * @param info optional info parameters for the struct serializer
193 * @return value
194 */
195 template <typename T, typename... I>
196 requires wpi::util::StructSerializable<T, I...>
197 T GetStruct(std::string_view path, I... info) const {
198 std::string expected{wpi::util::GetStructTypeName<T>(info...)};
199 std::string actual = GetStructTypeName(path);
200 if (actual != expected) {
201 throw std::invalid_argument{std::format(
202 "Struct tunable at {} is a {}, not a {}", path, actual, expected)};
203 }
204 return wpi::util::UnpackStruct<T>(GetStructData(path), info...);
205 }
206
207 /**
208 * Gets the value of a struct-serialized array tunable.
209 *
210 * @tparam T value type
211 * @tparam I optional info parameters for the struct serializer
212 * @param path normalized path
213 * @param info optional info parameters for the struct serializer
214 * @return value
215 */
216 template <typename T, typename... I>
217 requires wpi::util::StructSerializable<T, I...>
218 std::vector<T> GetStructVector(std::string_view path, I... info) const {
219 std::string expected{
220 wpi::util::MakeStructArrayTypeName<T, std::dynamic_extent>(info...)};
221 std::string actual = GetStructTypeName(path);
222 if (actual != expected) {
223 throw std::invalid_argument{std::format(
224 "Struct tunable at {} is a {}, not a {}", path, actual, expected)};
225 }
226
227 std::vector<uint8_t> data = GetStructData(path);
228 size_t structSize = wpi::util::GetStructSize<T>(info...);
229 if (structSize == 0 || data.size() % structSize != 0) {
230 throw std::runtime_error{
231 std::format("Struct array tunable has invalid size: {}", path)};
232 }
233
234 std::vector<T> values;
235 values.reserve(data.size() / structSize);
236 for (size_t pos = 0, end = data.size(); pos < end; pos += structSize) {
237 values.emplace_back(wpi::util::UnpackStruct<T>(
238 std::span{data}.subspan(pos, structSize), info...));
239 }
240 return values;
241 }

259 /**
260 * Gets the value of a protobuf-serialized tunable.
261 *
262 * @tparam T value type
263 * @param path normalized path
264 * @return value
265 */
266 template <wpi::util::ProtobufSerializable T>
267 T GetProtobuf(std::string_view path) const {
268 wpi::util::ProtobufMessage<T> msg;
269 std::string expected = msg.GetTypeString();
270 std::string actual = GetProtobufTypeString(path);
271 if (actual != expected) {
272 throw std::invalid_argument{std::format(
273 "Protobuf tunable at {} is a {}, not a {}", path, actual, expected)};
274 }
275
276 auto value = msg.Unpack(GetProtobufData(path));
277 if (!value) {
278 throw std::runtime_error{
279 std::format("Failed to unpack protobuf tunable: {}", path)};
280 }
281 return *value;
282 }

425 /**
426 * Sets the value of a struct-serialized tunable.
427 *
428 * @tparam T value type
429 * @tparam I optional info parameters for the struct serializer
430 * @param path normalized path
431 * @param value value
432 * @param info optional info parameters for the struct serializer
433 */
434 template <typename T, typename... I>
435 requires wpi::util::StructSerializable<T, I...>
436 void SetStruct(std::string_view path, const T& value, I... info) {
437 std::vector<uint8_t> buf(wpi::util::GetStructSize<T>(info...));
438 wpi::util::PackStruct(buf, value, info...);
439 SetStructData(path, wpi::util::GetStructTypeName<T>(info...), buf);
440 }

442 /**
443 * Sets the value of a struct-serialized array tunable.
444 *
445 * @tparam T value type
446 * @tparam I optional info parameters for the struct serializer
447 * @param path normalized path
448 * @param value value
449 * @param info optional info parameters for the struct serializer
450 */
451 template <typename T, typename... I>
452 requires wpi::util::StructSerializable<T, I...>
453 void SetStructVector(std::string_view path, std::span<const T> value,
454 I... info) {
455 size_t structSize = wpi::util::GetStructSize<T>(info...);
456 std::vector<uint8_t> buf(structSize * value.size());
457 for (size_t i = 0; i < value.size(); ++i) {
458 wpi::util::PackStruct(
459 std::span<uint8_t>{buf}.subspan(i * structSize, structSize), value[i],
460 info...);
461 }
462 SetStructData(path, wpi::util::GetStructTypeName<T>(info...), buf);
463 }
