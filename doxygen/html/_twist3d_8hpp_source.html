22/**
23 * A change in distance along a 3D arc since the last pose update. We can use
24 * ideas from differential calculus to create new Pose3ds from a Twist3d and
25 * vice versa.
26 *
27 * A Twist can be used to represent a difference between two poses.
28 */
29struct WPILIB_DLLEXPORT Twist3d final {
30 /**
31 * Linear "dx" component
32 */
33 wpi::units::meter_t dx = 0_m;
34
35 /**
36 * Linear "dy" component
37 */
38 wpi::units::meter_t dy = 0_m;
39
40 /**
41 * Linear "dz" component
42 */
43 wpi::units::meter_t dz = 0_m;
44
45 /**
46 * Rotation vector x component.
47 */
48 wpi::units::radian_t rx = 0_rad;
49
50 /**
51 * Rotation vector y component.
52 */
53 wpi::units::radian_t ry = 0_rad;
54
55 /**
56 * Rotation vector z component.
57 */
58 wpi::units::radian_t rz = 0_rad;
59
60 /**
61 * Obtain a new Transform3d from a (constant curvature) velocity.
62 *
63 * See "https://file.tavsys.net/control/controls-engineering-in-frc.pdf"
64 * Controls Engineering in the FIRST Robotics Competition section 10.2
65 * "Pose exponential" for a derivation.
66 *
67 * The twist is a change in pose in the robot's coordinate frame since the
68 * previous pose update. When the user runs Exp() on the twist, the user will
69 * receive the pose delta.
70 *
71 * "Exp" represents the pose exponential, which is solving a differential
72 * equation moving the pose forward in time.
73 *
74 * @return The pose delta of the robot.
75 */
76 constexpr Transform3d Exp() const;
77
78 /**
79 * Checks equality between this Twist3d and another object.
80 *
81 * @param other The other object.
82 * @return Whether the two objects are equal.
83 */
84 constexpr bool operator==(const Twist3d& other) const {
85 return wpi::units::math::abs(dx - other.dx) < 1E-9_m &&
86 wpi::units::math::abs(dy - other.dy) < 1E-9_m &&
87 wpi::units::math::abs(dz - other.dz) < 1E-9_m &&
88 wpi::units::math::abs(rx - other.rx) < 1E-9_rad &&
89 wpi::units::math::abs(ry - other.ry) < 1E-9_rad &&
90 wpi::units::math::abs(rz - other.rz) < 1E-9_rad;
91 }
92
93 /**
94 * Scale this by a given factor.
95 *
96 * @param factor The factor by which to scale.
97 * @return The scaled Twist3d.
98 */
99 constexpr Twist3d operator*(double factor) const {
100 return Twist3d{dx * factor, dy * factor, dz * factor,
101 rx * factor, ry * factor, rz * factor};
102 }
103};

112constexpr Transform3d Twist3d::Exp() const {
113 // Implementation from Section 3.2 of https://ethaneade.org/lie.pdf
114
115 auto impl = [this]<typename Matrix3d, typename Vector3d>() -> Transform3d {
116 Vector3d u{{dx.value(), dy.value(), dz.value()}};
117 Vector3d rvec{{rx.value(), ry.value(), rz.value()}};
118 Matrix3d omega = detail::RotationVectorToMatrix(rvec);
119 Matrix3d omegaSq = omega * omega;
120 double theta = rvec.norm();
121 double thetaSq = theta * theta;
122
123 double A;
124 double B;
125 double C;
126 if (gcem::abs(theta) < 1E-7) {
127 // Taylor Expansions around θ = 0
128 // A = 1/1! - θ²/3! + θ⁴/5!
129 // B = 1/2! - θ²/4! + θ⁴/6!
130 // C = 1/3! - θ²/5! + θ⁴/7!
131 // sources:
132 // A:
133 // https://www.wolframalpha.com/input?i2d=true&i=series+expansion+of+Divide%5Bsin%5C%2840%29x%5C%2841%29%2Cx%5D+at+x%3D0
134 // B:
135 // https://www.wolframalpha.com/input?i2d=true&i=series+expansion+of+Divide%5B1-cos%5C%2840%29x%5C%2841%29%2CPower%5Bx%2C2%5D%5D+at+x%3D0
136 // C:
137 // https://www.wolframalpha.com/input?i2d=true&i=series+expansion+of+Divide%5B1-Divide%5Bsin%5C%2840%29x%5C%2841%29%2Cx%5D%2CPower%5Bx%2C2%5D%5D+at+x%3D0
138 A = 1 - thetaSq / 6 + thetaSq * thetaSq / 120;
139 B = 1 / 2.0 - thetaSq / 24 + thetaSq * thetaSq / 720;
140 C = 1 / 6.0 - thetaSq / 120 + thetaSq * thetaSq / 5040;
141 } else {
142 // A = sinθ/θ
143 // B = (1 - cosθ)/θ²
144 // C = (1 - A)/θ²
145 A = gcem::sin(theta) / theta;
146 B = (1 - gcem::cos(theta)) / thetaSq;
147 C = (1 - A) / thetaSq;
148 }
149
150 Matrix3d R = Matrix3d::Identity() + A * omega + B * omegaSq;
151 Matrix3d V = Matrix3d::Identity() + B * omega + C * omegaSq;
152
153 Vector3d translation_component = V * u;
154
155 const Transform3d transform{
156 Translation3d{wpi::units::meter_t{translation_component(0)},
157 wpi::units::meter_t{translation_component(1)},
158 wpi::units::meter_t{translation_component(2)}},
159 Rotation3d{R}};
160
161 return transform;
162 };
163
164 if consteval {
165 return impl.template operator()<ct_matrix3d, ct_vector3d>();
166 }
167 return impl.template operator()<Eigen::Matrix3d, Eigen::Vector3d>();
168}
169
170} // namespace wpi::math
