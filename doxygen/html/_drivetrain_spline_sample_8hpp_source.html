25/**
26 * Represents a single sample in a spline-based drivetrain trajectory.
27 */
28class DrivetrainSplineSample : public HolonomicSample {
29 public:
30 /** The curvature of the path at this sample. */
31 wpi::units::curvature_t curvature{0.0};
32
33 /** Constructs a default DrivetrainSplineSample with all zero values. */
34 constexpr DrivetrainSplineSample() = default;
35
36 /**
37 * Constructs a SplineSample from path-relative scalars.
38 *
39 * The robot follows the path along its heading, so the scalar forward
40 * velocity and acceleration are rotated into the field frame using the
41 * sample's pose. The resulting velocity and acceleration are field-relative.
42 *
43 * @param time The time of the sample relative to the trajectory start.
44 * @param pose The robot pose at this sample (in the field reference frame).
45 * @param velocity The signed forward (path-tangential) velocity.
46 * @param acceleration The signed forward (path-tangential) acceleration.
47 * @param curvature The curvature of the path at this sample.
48 */
49 constexpr DrivetrainSplineSample(
50 wpi::units::second_t time, const Pose2d& pose,
51 wpi::units::meters_per_second_t velocity,
52 wpi::units::meters_per_second_squared_t acceleration,
53 wpi::units::curvature_t curvature)
54 : HolonomicSample{time, pose,
55 ChassisVelocities{velocity, 0_mps, velocity * curvature}
56 .ToFieldRelative(pose.Rotation()),
57 ChassisAccelerations{acceleration, 0_mps_sq,
58 acceleration * curvature}
59 .ToFieldRelative(pose.Rotation())},
60 curvature{curvature} {}
61
62 /**
63 * Constructs a SplineSample.
64 *
65 * @param time The time of the sample relative to the trajectory start.
66 * @param pose The robot pose at this sample (in the field reference frame).
67 * @param velocity The robot velocity at this sample (in the field reference
68 * frame).
69 * @param acceleration The robot acceleration at this sample (in the field
70 * reference frame).
71 * @param curvature The curvature of the path at this sample.
72 */
73 constexpr DrivetrainSplineSample(wpi::units::second_t time,
74 const Pose2d& pose,
75 const ChassisVelocities& velocity,
76 const ChassisAccelerations& acceleration,
77 wpi::units::curvature_t curvature)
78 : HolonomicSample{time, pose, velocity, acceleration},
79 curvature{curvature} {}
80
81 /**
82 * Constructs a SplineSample from a generic sample.
83 * Curvature is calculated as omega / forward velocity.
84 *
85 * @param sample The HolonomicSample to convert.
86 */
87 explicit constexpr DrivetrainSplineSample(const HolonomicSample& sample)
88 : HolonomicSample{sample.time, sample.pose, sample.velocity,
89 sample.acceleration} {
90 auto vx = sample.velocity.ToRobotRelative(sample.pose.Rotation()).vx;
91 curvature = sample.velocity.omega / (vx == 0_mps ? 1e-9_mps : vx);
92 }
93
94 /**
95 * Returns the signed forward (robot-relative tangential) velocity at this
96 * sample. This is the field-relative velocity projected onto the sample's
97 * heading.
98 *
99 * @return The forward velocity.
100 */
101 constexpr wpi::units::meters_per_second_t ForwardVelocity() const {
102 return velocity.ToRobotRelative(pose.Rotation()).vx;
103 }
104
105 /**
106 * Returns the signed forward (robot-relative tangential) acceleration at this
107 * sample. This is the field-relative acceleration projected onto the sample's
108 * heading.
109 *
110 * @return The forward acceleration.
111 */
112 constexpr wpi::units::meters_per_second_squared_t ForwardAcceleration()
113 const {
114 return acceleration.ToRobotRelative(pose.Rotation()).ax;
115 }
116
117 /**
118 * Transforms the pose of this sample by the given transform.
119 *
120 * @param transform The transform to apply to the pose.
121 * @return A new sample with the transformed pose.
122 */
123 constexpr DrivetrainSplineSample Transform(
124 const Transform2d& transform) const {
125 return DrivetrainSplineSample{
126 time, pose.TransformBy(transform),
127 velocity.ToFieldRelative(transform.Rotation()),
128 acceleration.ToFieldRelative(transform.Rotation()), curvature};
129 }
130
131 /**
132 * Transforms this sample to be relative to the given pose.
133 *
134 * @param other The pose to make this sample relative to.
135 * @return A new sample with the relative pose.
136 */
137 constexpr DrivetrainSplineSample RelativeTo(const Pose2d& other) const {
138 return DrivetrainSplineSample{
139 time, pose.RelativeTo(other),
140 velocity.ToRobotRelative(other.Rotation()),
141 acceleration.ToRobotRelative(other.Rotation()), curvature};
142 }
143
144 /**
145 * Checks equality between this SplineSample and another object.
146 *
147 * @return True if the samples are equal.
148 */
149 constexpr bool operator==(const DrivetrainSplineSample& other) const =
150 default;
151};
