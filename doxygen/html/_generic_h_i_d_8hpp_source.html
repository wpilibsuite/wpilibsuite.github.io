21/**
22 * Handle input from standard HID devices connected to the Driver Station.
23 *
24 * <p>This class handles standard input that comes from the Driver Station. Each
25 * time a value is requested the most recent value is returned. There is a
26 * single class instance for each device and the mapping of ports to hardware
27 * buttons depends on the code in the Driver Station.
28 */
29class GenericHID final : public HIDDevice {
30 public:
31 friend class DriverStation;
32 friend class internal::DriverStationBackend;

39 /// motor.
40 LEFT_RUMBLE,
41 /// Right rumble motor. On most controllers, this is the high-frequency
42 /// motor.
43 RIGHT_RUMBLE,
44 /// Left trigger rumble motor, on controllers that have one.
45 LEFT_TRIGGER_RUMBLE,
46 /// Right trigger rumble motor, on controllers that have one.
47 RIGHT_TRIGGER_RUMBLE,
48 };

123 /**
124 * Get the button value (starting at button 1).
125 *
126 * The buttons are returned in a single 16 bit value with one bit representing
127 * the state of each button. The appropriate button is returned as a boolean
128 * value.
129 *
130 * This method returns true if the button is being held down at the time
131 * that this method is being called.
132 *
133 * @param button The button number to be read (starting at 1)
134 * @return The state of the button.
135 */

138 /**
139 * Whether the button was pressed since the last check. %Button indexes begin
140 * at 1.
141 *
142 * This method returns true if the button went from not pressed to held down
143 * since the last time this method was called. This is useful if you only
144 * want to call a function once when you press the button.
145 *
146 * @param button The button index, beginning at 0.
147 * @return Whether the button was pressed since the last check.
148 */
