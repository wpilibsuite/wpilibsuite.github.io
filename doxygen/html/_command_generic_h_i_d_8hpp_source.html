236 * @param threshold The value below which this trigger should return true.

238 * {@link CommandScheduler::GetDefaultButtonLoop() the default command
239 * scheduler button loop}.
240 * @return a Trigger instance that is true when the axis value is greater than
241 * the provided threshold.
242 */
243 Trigger AxisGreaterThan(
244 int axis, double threshold,
245 wpi::EventLoop* loop =
246 CommandScheduler::GetInstance().GetDefaultButtonLoop()) const;

248 /**
249 * Constructs a Trigger instance that is true when the axis magnitude value is

251 *
252 * @param axis The axis to read, starting at 0
253 * @param threshold The value above which this trigger should return true.
254 * @param loop the event loop instance to attach the trigger to.
255 * @return a Trigger instance that is true when the axis magnitude value is
256 * greater than the provided threshold.
257 */
258 Trigger AxisMagnitudeGreaterThan(
259 int axis, double threshold,
260 wpi::EventLoop* loop =
261 CommandScheduler::GetInstance().GetDefaultButtonLoop()) const;
262
263 /**
264 * Set the rumble output for the HID.
265 *
266 * The DS currently supports 2 rumble values, left rumble and right rumble.
267 *
268 * @param type Which rumble value to set
269 * @param value The normalized value (0 to 1) to set the rumble to
270 */
271 void SetRumble(wpi::GenericHID::RumbleType type, double value);
272
273 /**
274 * Get if the HID is connected.
275 *
276 * @return true if the HID is connected
277 */
278 bool IsConnected() const;
279
280 private:
281 wpi::GenericHID* m_hid;
282};
