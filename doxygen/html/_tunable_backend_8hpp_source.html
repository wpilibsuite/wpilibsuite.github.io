23/** Backend interface for publishing and updating tunables. */
24class TunableBackend {
25 public:
26 /**
27 * A tunable removed from a backend.
28 *
29 * @param path normalized path
30 * @param uid tunable uid
31 */
32 struct PublishedTunable {
33 std::string path;
34 uint32_t uid;
35 };
36
37 /** Destroys this tunable backend. */
38 virtual ~TunableBackend() = default;
39
40 /**
41 * Retires this backend after it is removed from TunableRegistry.
42 *
43 * This stops external backend effects while the backend object may still be
44 * alive. Destructors remain the final RAII cleanup path and may call this
45 * method.
46 *
47 * Implementations must not throw from this method.
48 */
49 virtual void Retire() {}
50
51 /**
52 * Publishes a tunable.
53 *
54 * Backend implementations must not throw from this method. Recoverable
55 * publication failures should be reported through
56 * TunableRegistry::ReportWarning() and skipped.
57 *
58 * @param path normalized path
59 * @param uid the tunable uid
60 * @param tunable the tunable
61 * @param config the tunable config
62 * @param type the tunable type
63 * @return true if the tunable was published, false otherwise
64 */
65 virtual bool Publish(std::string_view path, uint32_t uid,
66 detail::TunableBase& tunable,
67 const TunableConfig* config,
68 detail::TunableTypeValue type) = 0;
69
70 /**
71 * Marks a published tunable dirty after its local value changed.
72 *
73 * Backend implementations must not throw from this method.
74 *
75 * @param uid tunable uid
76 */
77 virtual void MarkDirty(uint32_t uid) = 0;
78
79 /**
80 * Removes a tunable.
81 *
82 * Backend implementations must not throw from this method.
83 *
84 * @param path normalized path
85 */
86 virtual void Remove(std::string_view path) = 0;
87
88 /**
89 * Removes all tunables under a prefix.
90 *
91 * Backend implementations must not throw from this method. The returned list
92 * must describe the tunables actually removed from the backend.
93 *
94 * @param prefix normalized path prefix
95 * @return removed tunables
96 */
97 virtual std::vector<PublishedTunable> RemovePrefix(
98 std::string_view prefix) = 0;
99
100 /**
101 * Unregisters a tunable.
102 *
103 * Backend implementations must not throw from this method.
104 *
105 * @param uid tunable uid
106 */
107 virtual void UnregisterTunable(uint32_t uid) = 0;
108
109 /**
110 * Updates all tunable values and calls callbacks where appropriate.
111 *
112 * Backend implementations must not throw from this method. Recoverable update
113 * failures should be reported through TunableRegistry::ReportWarning() and
114 * skipped so the robot loop and registry state can continue updating.
115 */
116 virtual void Update() = 0;
117};
118
119} // namespace wpi::tunables
