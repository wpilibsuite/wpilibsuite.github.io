32/**

34 * profile.

52 /**
53 * Allocates a ProfiledPIDController with the given constants for Kp, Ki, and
54 * Kd. Users should call reset() when they first start running the controller
55 * to avoid unwanted behavior.
56 *
57 * @param Kp The proportional coefficient. Must be >= 0.
58 * @param Ki The integral coefficient. Must be >= 0.
59 * @param Kd The derivative coefficient. Must be >= 0.
60 * @param constraints Velocity and acceleration constraints for goal.
61 * @param period The period between controller updates in seconds. The
62 * default is 20 milliseconds. Must be positive.
63 */
64 constexpr ProfiledPIDController(double Kp, double Ki, double Kd,
65 Constraints constraints,
66 wpi::units::second_t period = 20_ms)
67 : m_controller{Kp, Ki, Kd, period},
68 m_constraints{constraints},
69 m_profile{m_constraints} {
70 if !consteval {
71 int instances = detail::IncrementAndGetProfiledPIDControllerInstances();
72 wpi::util::ReportUsage("ProfiledPIDController",
73 std::to_string(instances));
74 }
75 }

350 /**
351 * Returns the next output of the PID controller.
352 *
353 * @param measurement The current measurement of the process variable.
354 */
355 constexpr double Calculate(Distance_t measurement) {
356 if (m_controller.IsContinuousInputEnabled()) {
357 // Get error which is smallest distance between goal and measurement
358 auto errorBound = (m_maximumInput - m_minimumInput) / 2.0;
359 auto goalMinDistance = wpi::math::InputModulus<Distance_t>(
360 m_goal.position - measurement, -errorBound, errorBound);
361 auto setpointMinDistance = wpi::math::InputModulus<Distance_t>(
362 m_setpoint.position - measurement, -errorBound, errorBound);
363
364 // Recompute the profile goal with the smallest error, thus giving the
365 // shortest path. The goal may be outside the input range after this
366 // operation, but that's OK because the controller will still go there and
367 // report an error of zero. In other words, the setpoint only needs to be
368 // offset from the measurement by the input range modulus; they don't need
369 // to be equal.
370 m_goal.position = goalMinDistance + measurement;
371 double goalPosition = ToBaseGoalPosition(m_goal.position);
372 bool goalPositionChanged = m_goalPosition != goalPosition;
373 m_goalPosition = goalPosition;
374 if !consteval {
375 if (goalPositionChanged) {
376 SetChildTunableChanged("goal");
377 }
378 }
379 m_setpoint.position = setpointMinDistance + measurement;
380 }
381
382 m_setpoint = m_profile.Calculate(GetPeriod(), m_setpoint, m_goal);
383 return m_controller.Calculate(measurement.value(),
384 m_setpoint.position.value());
385 }

476 goalConfig.onTune = [](wpi::tunables::detail::TunableBase&,

479 controller->SetGoal(FromBaseGoalPosition(controller->m_goalPosition));

481 };
482 goalConfig.parent = this;
483 table.Publish("goal", this, &ProfiledPIDController::m_goalPosition,
484 goalConfig);
