97 /**
98 * Constexpr version of Eigen's scalar multiplication operator.
99 *
100 * @param lhs LHS scalar.
101 * @param rhs RHS matrix.
102 * @return Result of multiplication.
103 */
104 friend constexpr ct_matrix<Scalar, Rows, Cols> operator*(
105 Scalar lhs, const ct_matrix<Scalar, Rows, Cols>& rhs) {
106 if consteval {
107 ct_matrix<Scalar, Rows, Cols> result;
108
109 for (int i = 0; i < rhs.rows(); ++i) {
110 for (int j = 0; j < rhs.cols(); ++j) {
111 result(i, j) = lhs * rhs(i, j);
112 }
113 }
114
115 return result;
116 } else {
117 return lhs * rhs.m_storage;
118 }
119 }

121 /**
122 * Constexpr version of Eigen's matrix multiplication operator.
123 *
124 * @tparam Cols2 Columns of RHS matrix.
125 * @param lhs LHS matrix.
126 * @param rhs RHS matrix.
127 * @return Result of multiplication.
128 */
129 template <int Cols2>
130 friend constexpr ct_matrix<Scalar, Rows, Cols2> operator*(
131 const ct_matrix<Scalar, Rows, Cols>& lhs,
132 const ct_matrix<Scalar, Rows, Cols2>& rhs) {
133 if consteval {
134 ct_matrix<Scalar, Rows, Cols2> result;
135
136 for (int i = 0; i < lhs.rows(); ++i) {
137 for (int j = 0; j < rhs.cols(); ++j) {
138 Scalar sum = 0.0;
139 for (int k = 0; k < lhs.cols(); ++k) {
140 sum += lhs(i, k) * rhs(k, j);
141 }
142 result(i, j) = sum;
143 }
144 }
145
146 return result;
147 } else {
148 return lhs.m_storage * rhs.storage();
149 }
150 }

202 /**
203 * Constexpr version of Eigen's transpose member function.
204 *
205 * @return Transpose of matrix.

207 constexpr ct_matrix<Scalar, Cols, Rows> transpose() const {
208 if consteval {
209 ct_matrix<Scalar, Cols, Rows> result;
210
211 for (int row = 0; row < rows(); ++row) {
212 for (int col = 0; col < cols(); ++col) {
213 result(col, row) = (*this)(row, col);
214 }
215 }
216
217 return result;
218 } else {
219 return m_storage.transpose().eval();

223 /**
224 * Constexpr version of Eigen's identity function.
225 *
226 * @return Identity matrix of the specified size.
227 */
228 static constexpr ct_matrix<Scalar, Rows, Cols> Identity()
229 requires(Rows != Eigen::Dynamic && Cols != Eigen::Dynamic)
230 {
231 if consteval {
232 ct_matrix<Scalar, Rows, Cols> result;
233
234 for (int row = 0; row < Rows; ++row) {
235 for (int col = 0; col < Cols; ++col) {
236 if (row == col) {
237 result(row, row) = 1.0;
238 } else {
239 result(row, col) = 0.0;
240 }
241 }
242 }
243
244 return result;
245 } else {
246 return Eigen::Matrix<Scalar, Rows, Cols>::Identity();
247 }
248 }

250 /**
251 * Constexpr version of Eigen's vector dot member function.
252 *
253 * @tparam RhsRows Rows of RHS vector.
254 * @tparam RhsCols Columns of RHS vector.
255 * @param rhs RHS vector.
256 * @return Dot product of two vectors.
257 */
258 template <int RhsRows, int RhsCols>
259 requires(Rows == 1 || Cols == 1) && (RhsRows == 1 || RhsCols == 1) &&
260 (Rows * Cols == RhsRows * RhsCols)
261 constexpr Scalar dot(const ct_matrix<Scalar, RhsRows, RhsCols>& rhs) const {
262 if consteval {
263 Scalar sum = 0.0;
264
265 for (int index = 0; index < rows() * rhs.cols(); ++index) {
266 sum += (*this)(index)*rhs(index);
267 }
268
269 return sum;
270 } else {
271 return m_storage.dot(rhs.storage());
272 }
273 }

275 /**
276 * Constexpr version of Eigen's norm member function.
277 *
278 * @return Norm of matrix.
279 */
280 constexpr Scalar norm() const {
281 if consteval {
282 Scalar sum = 0.0;
283
284 for (int row = 0; row < rows(); ++row) {
285 for (int col = 0; col < cols(); ++col) {
286 sum += (*this)(row, col) * (*this)(row, col);
287 }
288 }
289
290 return gcem::sqrt(sum);
291 } else {
292 return m_storage.norm();
293 }
294 }

296 /**
297 * Constexpr version of Eigen's 3D vector cross member function.
298 *
299 * @param rhs RHS vector.
300 * @return Cross product of two vectors.
301 */
302 constexpr ct_matrix<Scalar, 3, 1> cross(const ct_matrix<Scalar, 3, 1>& rhs)
303 requires(Rows == 3 && Cols == 1)
304 {
305 return Eigen::Vector3d{{(*this)(1) * rhs(2) - rhs(1) * (*this)(2),
306 rhs(0) * (*this)(2) - (*this)(0) * rhs(2),
307 (*this)(0) * rhs(1) - rhs(0) * (*this)(1)}};
308 }

310 /**
311 * Constexpr version of Eigen's 2x2 matrix determinant member function.
312 *
313 * @return Determinant of matrix.
314 */
315 constexpr Scalar determinant() const
316 requires(Rows == 2 && Cols == 2)
317 {
318 // |a b|
319 // |c d| = ad - bc
320 Scalar a = (*this)(0, 0);
321 Scalar b = (*this)(0, 1);
322 Scalar c = (*this)(1, 0);
323 Scalar d = (*this)(1, 1);
324 return a * d - b * c;
325 }
326
327 /**
328 * Constexpr version of Eigen's 3x3 matrix determinant member function.
329 *
330 * @return Determinant of matrix.
331 */
332 constexpr Scalar determinant() const
333 requires(Rows == 3 && Cols == 3)
334 {
335 // |a b c|
336 // |d e f| = aei + bfg + cdh - ceg - bdi - afh
337 // |g h i|
338 Scalar a = (*this)(0, 0);
339 Scalar b = (*this)(0, 1);
340 Scalar c = (*this)(0, 2);
341 Scalar d = (*this)(1, 0);
342 Scalar e = (*this)(1, 1);
343 Scalar f = (*this)(1, 2);
344 Scalar g = (*this)(2, 0);
345 Scalar h = (*this)(2, 1);
346 Scalar i = (*this)(2, 2);
347 return a * e * i + b * f * g + c * d * h - c * e * g - b * d * i -
348 a * f * h;
349 }
