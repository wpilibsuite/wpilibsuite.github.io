43/**
44 * The Selectable class is a useful tool for presenting a selection of
45 * options on a dashboard.
46 *
47 * @tparam T The type of values to be stored
48 */
49template <class T>
50 requires std::copy_constructible<T> && std::default_initializable<T>
51class Selectable final : public detail::SelectableBase {
52 wpi::util::StringMap<T> m_map;
53 std::function<void(T)> m_listener;
54 template <class U>
55 static U _unwrap_smart_ptr(const U& value) {
56 return value;
57 }
58
59 template <class U>
60 static std::weak_ptr<U> _unwrap_smart_ptr(const std::shared_ptr<U>& value) {
61 return value;
62 }
63
64 auto FindSelectedOrDefault(std::string_view selected) const {
65 if (!selected.empty()) {
66 auto selectedIt = m_map.find(selected);
67 if (selectedIt != m_map.end()) {
68 return selectedIt;
69 }
70 }
71 if (m_defaultChoice.empty()) {
72 return m_map.end();
73 }
74 return m_map.find(m_defaultChoice);
75 }
76
77 public:
78 using CopyType = decltype(_unwrap_smart_ptr(m_map.find("")->second));
79
80 /** Constructs a selectable tunable. */
81 Selectable() = default;
82
83 /**
84 * Move constructs a selectable tunable.
85 *
86 * @param rhs selectable tunable to move from
87 */
88 Selectable(Selectable&& rhs) = default;
89
90 /**
91 * Move assigns this selectable tunable.
92 *
93 * @param rhs selectable tunable to move from
94 * @return this selectable tunable
95 */
96 Selectable& operator=(Selectable&& rhs) = default;
97
98 /**
99 * Adds the given object to the list of options.
100 *
101 * On the dashboard, the object will appear as the given name.
102 *
103 * @param name the name of the option
104 * @param object the option
105 */
106 void Add(std::string_view name, T object) {
107 if (!m_map.insert_or_assign(name, std::move(object)).second) {
108 std::erase_if(m_options, [name](const std::string& option) {
109 return option == name;
110 });
111 }
112 m_options.emplace_back(name);
113 SetChildTunableChanged("options");
114 }
115
116 /**
117 * Adds the given object to the list of options and marks it as the default.
118 *
119 * @param name the name of the option
120 * @param object the option
121 */
122 void AddDefault(std::string_view name, T object) {
123 Add(name, std::move(object));
124 SetDefault(name);
125 }
126
127 /**
128 * Removes the option with the given name. If the removed option is the
129 * default, the default is reset. Does not change the selected option;
130 * GetSelected() will fall back to the default after this is called if the
131 * selected option name no longer matches an option.
132 *
133 * @param name the name of the option
134 */
135 void Remove(std::string_view name) {
136 if (m_map.erase(name) == 0) {
137 return;
138 }
139 std::erase_if(m_options,
140 [name](const std::string& option) { return option == name; });
141 if (m_defaultChoice == name) {
142 m_defaultChoice = "";
143 SetChildTunableChanged("default");
144 }
145 SetChildTunableChanged("options");
146 }
147
148 /**
149 * Marks the given option as the default.
150 *
151 * @param name the name of the option
152 */
153 void SetDefault(std::string_view name) {
154 m_defaultChoice = name;
155 SetChildTunableChanged("default");
156 }
157
158 /**
159 * Clears the list of options and resets the default. Does not change the
160 * selected option; GetSelected() will return a value-initialized instance
161 * after this is called until a new matching option or default is added.
162 */
163 void Clear() {
164 m_map.clear();
165 m_options.clear();
166 m_defaultChoice = "";
167 SetChildTunableChanged("default");
168 SetChildTunableChanged("options");
169 }
170
171 /**
172 * Returns a copy of the selected option (a std::weak_ptr&lt;U&gt; if T =
173 * std::shared_ptr&lt;U&gt;).
174 *
175 * If there is none or no valid option selected, it will return the default.
176 * If there is no valid option selected and no default, then it will return a
177 * value-initialized instance. For integer types, this is 0. For container
178 * types like std::string, this is an empty string.
179 *
180 * @return The option selected
181 */
182 CopyType GetSelected() const {
183 auto it = FindSelectedOrDefault(m_selected);
184 if (it == m_map.end()) {
185 return CopyType{};
186 }
187 return _unwrap_smart_ptr(it->second);
188 }
189
190 /**
191 * Bind a listener that's called when the selected value changes.
192 * Only one listener can be bound. Calling this function will replace the
193 * previous listener.
194 * @param listener The function to call that accepts the new value
195 */
196 void OnChange(std::function<void(T)> listener) { m_listener = listener; }
197
198 protected:
199 void Changed(std::string_view val) override {
200 if (m_listener) {
201 auto it = FindSelectedOrDefault(val);
202 if (it != m_map.end()) {
203 m_listener(it->second);
204 }
205 }
206 }
207};
