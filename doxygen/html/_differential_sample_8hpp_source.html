23/**
24 * Represents a single sample in a differential drive trajectory.
25 */
26class DifferentialSample : public HolonomicSample {
27 public:
28 /**
29 * The left wheel velocity at this sample.
30 */
31 wpi::units::meters_per_second_t leftVelocity{0_mps};
32
33 /**
34 * The right wheel velocity at this sample.
35 */
36 wpi::units::meters_per_second_t rightVelocity{0_mps};
37
38 /** Constructs a default DifferentialSample with all zero values. */
39 constexpr DifferentialSample() = default;
40
41 /**
42 * Constructs a DifferentialSample.
43 *
44 * @param time The time of the sample relative to the trajectory start.
45 * @param pose The robot pose at this sample (in the field reference frame).
46 * @param velocity The robot velocity at this sample (in the field reference
47 * frame).
48 * @param acceleration The robot acceleration at this sample (in the field
49 * reference frame).
50 * @param leftVelocity The left wheel velocity at this sample.
51 * @param rightVelocity The right wheel velocity at this sample.
52 */
53 constexpr DifferentialSample(wpi::units::second_t time, const Pose2d& pose,
54 const ChassisVelocities& velocity,
55 const ChassisAccelerations& acceleration,
56 wpi::units::meters_per_second_t leftVelocity,
57 wpi::units::meters_per_second_t rightVelocity)
58 : HolonomicSample{time, pose, velocity, acceleration},
59 leftVelocity{leftVelocity},
60 rightVelocity{rightVelocity} {}
61
62 /**
63 * Constructs a DifferentialSample.
64 *
65 * @param time The time of the sample relative to the trajectory start.
66 * @param pose The robot pose at this sample (in the field reference frame).
67 * @param velocity The robot velocity at this sample (in the field reference
68 * frame).
69 * @param acceleration The robot acceleration at this sample (in the field
70 * reference frame).
71 * @param kinematics The kinematics of the drivetrain.
72 */
73 constexpr DifferentialSample(wpi::units::second_t time, const Pose2d& pose,
74 const ChassisVelocities& velocity,
75 const ChassisAccelerations& acceleration,
76 const DifferentialDriveKinematics& kinematics)
77 : HolonomicSample{time, pose, velocity, acceleration} {
78 // Wheel velocitys are derived from the robot-relative velocity.
79 auto wheelVelocitys =
80 kinematics.ToWheelVelocities(velocity.ToRobotRelative(pose.Rotation()));
81 leftVelocity = wheelVelocitys.left;
82 rightVelocity = wheelVelocitys.right;
83 }
84
85 /**
86 * Constructs a DifferentialSample from a TrajectorySample.
87 *
88 * @param sample The TrajectorySample to copy.
89 * @param leftVelocity The left wheel velocity at this sample.
90 * @param rightVelocity The right wheel velocity at this sample.
91 */
92 constexpr DifferentialSample(const HolonomicSample& sample,
93 wpi::units::meters_per_second_t leftVelocity,
94 wpi::units::meters_per_second_t rightVelocity)
95 : DifferentialSample{sample.time, sample.pose, sample.velocity,
96 sample.acceleration, leftVelocity, rightVelocity} {}
97
98 /**
99 * Constructs a DifferentialSample from a TrajectorySample.
100 *
101 * @param sample The TrajectorySample to copy.
102 * @param kinematics The kinematics of the drivetrain.
103 */
104 constexpr DifferentialSample(const HolonomicSample& sample,
105 const DifferentialDriveKinematics& kinematics)
106 : DifferentialSample{sample.time, sample.pose, sample.velocity,
107 sample.acceleration, kinematics} {}
108
109 /**
110 * Constructs a DifferentialSample from a SplineSample.
111 *
112 * @param sample The SplineSample to copy.
113 * @param kinematics The kinematics of the drivetrain.
114 */
115 constexpr DifferentialSample(const DrivetrainSplineSample& sample,
116 const DifferentialDriveKinematics& kinematics)
117 : HolonomicSample{sample.time, sample.pose, sample.velocity,
118 sample.acceleration},
119 leftVelocity{kinematics
120 .ToWheelVelocities(sample.velocity.ToRobotRelative(
121 sample.pose.Rotation()))
122 .left},
123 rightVelocity{kinematics
124 .ToWheelVelocities(sample.velocity.ToRobotRelative(
125 sample.pose.Rotation()))
126 .right} {}
127
128 /**
129 * Transforms the pose of this sample by the given transform.
130 *
131 * @param transform The transform to apply to the pose.
132 * @return A new sample with the transformed pose.
133 */
134 constexpr DifferentialSample Transform(const Transform2d& transform) const {
135 return DifferentialSample{
136 time,
137 pose.TransformBy(transform),
138 velocity.ToFieldRelative(transform.Rotation()),
139 acceleration.ToFieldRelative(transform.Rotation()),
140 leftVelocity,
141 rightVelocity};
142 }
143
144 /**
145 * Transforms this sample to be relative to the given pose.
146 *
147 * @param other The pose to make this sample relative to.
148 * @return A new sample with the relative pose.
149 */
150 constexpr DifferentialSample RelativeTo(const Pose2d& other) const {
151 return DifferentialSample{time,
152 pose.RelativeTo(other),
153 velocity.ToRobotRelative(other.Rotation()),
154 acceleration.ToRobotRelative(other.Rotation()),
155 leftVelocity,
156 rightVelocity};
157 }
158
159 /**
160 * Checks equality between this DifferentialSample and another object.
161 *
162 * @return True if the samples are equal.
163 */
164 constexpr bool operator==(const DifferentialSample& other) const = default;
165};
