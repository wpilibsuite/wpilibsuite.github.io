19/**
20 * A class for driving Mecanum drive platforms.
21 *
22 * Mecanum drives are rectangular with one wheel on each corner. Each wheel has
23 * rollers toed in 45 degrees toward the front or back. When looking at the
24 * wheels from the top, the roller axles should form an X across the robot.
25 *
26 * Drive base diagram:
27 * <pre>
28 * \\_______/
29 * \\ | | /
30 * | |
31 * /_|___|_\\
32 * / \\
33 * </pre>
34 *
35 * Each Drive() function provides different inverse kinematic relations for a
36 * Mecanum drive robot.
37 *
38 * This library uses the NWU axes convention (North-West-Up as external
39 * reference in the world frame). The positive X axis points ahead, the positive
40 * Y axis points to the left, and the positive Z axis points up. Rotations
41 * follow the right-hand rule, so counterclockwise rotation around the Z axis is
42 * positive.
43 *
44 * Inputs smaller then 0.02 will be set to 0, and larger values will be scaled
45 * so that the full range is still used. This deadband value can be changed
46 * with SetDeadband().
47 *
48 * MotorSafety is enabled by default. The DriveCartesian or DrivePolar
49 * methods should be called periodically to avoid Motor Safety timeouts.
50 */
51class MecanumDrive : public RobotDriveBase,
52 public wpi::telemetry::TelemetryLoggable {
53 public:
54 /**
55 * Wheel velocities for a mecanum drive.
56 *
57 * Uses normalized voltage [-1.0..1.0].
58 */
59 struct WheelVelocities {

62 /// Front-right wheel velocity.
63 double frontRight = 0.0;
64 /// Rear-left wheel velocity.
65 double rearLeft = 0.0;
66 /// Rear-right wheel velocity.
67 double rearRight = 0.0;
68 };

104 /**
105 * Drive method for Mecanum platform.
106 *
107 * Angles are measured counterclockwise from the positive X axis. The robot's
108 * velocity is independent from its angle or rotation rate.
109 *
110 * @param xVelocity The robot's velocity along the X axis [-1.0..1.0]. Forward
111 * is positive.
112 * @param yVelocity The robot's velocity along the Y axis [-1.0..1.0]. Left is
113 * positive.
114 * @param zRotation The robot's rotation rate around the Z axis [-1.0..1.0].
115 * Counterclockwise is positive.
116 * @param gyroAngle The gyro heading around the Z axis. Use this to implement
117 * field-oriented controls.
118 * @Common This is one of the commonly used methods for this class
119 */
120 void DriveCartesian(double xVelocity, double yVelocity, double zRotation,
121 wpi::math::Rotation2d gyroAngle = 0_rad);
122
123 /**
124 * Drive method for Mecanum platform.
125 *
126 * Angles are measured counterclockwise from the positive X axis. The robot's
127 * velocity is independent from its angle or rotation rate.
128 *
129 * @param magnitude The robot's velocity at a given angle [-1.0..1.0]. Forward
130 * is positive.
131 * @param angle The angle around the Z axis at which the robot drives.
132 * @param zRotation The robot's rotation rate around the Z axis [-1.0..1.0].
133 * Counterclockwise is positive.
134 * @Common This is one of the commonly used methods for this class
135 */
136 void DrivePolar(double magnitude, wpi::math::Rotation2d angle,
137 double zRotation);
138
139 /**
140 * Cartesian inverse kinematics for Mecanum platform.
141 *
142 * Angles are measured counterclockwise from the positive X axis. The robot's
143 * velocity is independent from its angle or rotation rate.
144 *
145 * @param xVelocity The robot's velocity along the X axis [-1.0..1.0]. Forward
146 * is positive.
147 * @param yVelocity The robot's velocity along the Y axis [-1.0..1.0]. Left is
148 * positive.
149 * @param zRotation The robot's rotation rate around the Z axis [-1.0..1.0].
150 * Counterclockwise is positive.
151 * @param gyroAngle The gyro heading around the Z axis. Use this to implement
152 * field-oriented controls.
153 * @return Wheel velocities [-1.0..1.0].
154 */
155 static WheelVelocities DriveCartesianIK(
156 double xVelocity, double yVelocity, double zRotation,
157 wpi::math::Rotation2d gyroAngle = 0_rad);
158
159 /**
160 * @Common This is one of the commonly used methods for this class
161 */
162 void StopMotor() override;
163 std::string GetDescription() const override;
164
165 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
166
167 std::string_view GetTelemetryType() const override;
168
169 private:
170 std::function<void(double)> m_frontLeftMotor;
171 std::function<void(double)> m_rearLeftMotor;
172 std::function<void(double)> m_frontRightMotor;
173 std::function<void(double)> m_rearRightMotor;
174
175 // Used for telemetry property getters
176 double m_frontLeftOutput = 0.0;
177 double m_rearLeftOutput = 0.0;
178 double m_frontRightOutput = 0.0;
179 double m_rearRightOutput = 0.0;
180
181 bool reported = false;
182};
