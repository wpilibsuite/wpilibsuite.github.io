21/**
22 * Implements a bang-bang controller, which outputs either 0 or 1 depending on
23 * whether the measurement is less than the setpoint. This maximally-aggressive
24 * control approach works very well for velocity control of high-inertia
25 * mechanisms, and poorly on most other things.
26 *
27 * <p>Note that this is an *asymmetric* bang-bang controller - it will not exert
28 * any control effort in the reverse direction (e.g. it won't try to slow down
29 * an overspeeding shooter wheel). This asymmetry is *extremely important*.
30 * Bang-bang control is extremely simple, but also potentially hazardous. Always
31 * ensure that your motor controllers are set to "coast" before attempting to
32 * control them with a bang-bang controller.
33 */
34class WPILIB_DLLEXPORT BangBangController
35 : public wpi::telemetry::TelemetryLoggable,
36 public wpi::tunables::ComplexTunable {
37 public:
38 /**
39 * Creates a new bang-bang controller.
40 *
41 * <p>Always ensure that your motor controllers are set to "coast" before
42 * attempting to control them with a bang-bang controller.
43 *
44 * @param tolerance Tolerance for atSetpoint.
45 */
46 constexpr explicit BangBangController(
47 double tolerance = std::numeric_limits<double>::infinity())
48 : m_tolerance(tolerance) {
49 if !consteval {
50 ++instances;
51 wpi::util::ReportUsage("BangBangController", std::to_string(instances));
52 }
53 }
54
55 /**
56 * Sets the setpoint for the bang-bang controller.
57 *
58 * @param setpoint The desired setpoint.
59 */
60 constexpr void SetSetpoint(double setpoint) { m_setpoint = setpoint; }
61
62 /**
63 * Returns the current setpoint of the bang-bang controller.
64 *
65 * @return The current setpoint.
66 */
67 constexpr double GetSetpoint() const { return m_setpoint; }
68
69 /**
70 * Returns true if the error is within the tolerance of the setpoint.
71 *
72 * @return Whether the error is within the acceptable bounds.
73 */
74 constexpr bool AtSetpoint() const {
75 return gcem::abs(m_setpoint - m_measurement) < m_tolerance;
76 }
77
78 /**
79 * Sets the error within which AtSetpoint will return true.
80 *
81 * @param tolerance Position error which is tolerable.
82 */
83 constexpr void SetTolerance(double tolerance) { m_tolerance = tolerance; }
84
85 /**
86 * Returns the current tolerance of the controller.
87 *
88 * @return The current tolerance.
89 */
90 constexpr double GetTolerance() const { return m_tolerance; }
91
92 /**
93 * Returns the current measurement of the process variable.
94 *
95 * @return The current measurement of the process variable.
96 */
97 constexpr double GetMeasurement() const { return m_measurement; }
98
99 /**
100 * Returns the current error.
101 *
102 * @return The current error.
103 */
104 constexpr double GetError() const { return m_setpoint - m_measurement; }
105
106 /**
107 * Returns the calculated control output.
108 *
109 * <p>Always ensure that your motor controllers are set to "coast" before
110 * attempting to control them with a bang-bang controller.
111 *
112 * @param measurement The most recent measurement of the process variable.
113 * @param setpoint The setpoint for the process variable.
114 * @return The calculated motor output (0 or 1).
115 */
116 constexpr double Calculate(double measurement, double setpoint) {
117 m_measurement = measurement;
118 m_setpoint = setpoint;
119
120 return measurement < setpoint ? 1 : 0;
121 }
122
123 /**
124 * Returns the calculated control output.
125 *
126 * @param measurement The most recent measurement of the process variable.
127 * @return The calculated motor output (0 or 1).
128 */
129 constexpr double Calculate(double measurement) {
130 return Calculate(measurement, m_setpoint);
131 }
132
133 void LogTo(wpi::telemetry::TelemetryTable& table) const override;
134
135 std::string_view GetTelemetryType() const override;
136
137 void PublishTunable(wpi::tunables::TunableTable& table) override;
138
139 std::string_view GetTunableType() const override;
140
141 private:
142 wpi::tunables::Tunable<double> m_tolerance;
143
144 wpi::tunables::Tunable<double> m_setpoint;
145 double m_measurement = 0;
146
147 // Usage reporting instances
148 inline static int instances = 0;
149};
150
151} // namespace wpi::math
