24template <typename T, bool Member = false>
25constexpr TunableTypeValue GetTunableTypeValue() {
26 if constexpr (std::same_as<T, bool>) {
27 return Member ? TunableTypeValue::MEMBER_BOOLEAN
28 : TunableTypeValue::BOOLEAN;
29 } else if constexpr (std::same_as<T, int32_t>) {
30 return Member ? TunableTypeValue::MEMBER_INT32 : TunableTypeValue::INT32;
31 } else if constexpr (std::same_as<T, int64_t>) {
32 return Member ? TunableTypeValue::MEMBER_INT64 : TunableTypeValue::INT64;
33 } else if constexpr (std::same_as<T, float>) {
34 return Member ? TunableTypeValue::MEMBER_FLOAT : TunableTypeValue::FLOAT;
35 } else if constexpr (std::same_as<T, double>) {
36 return Member ? TunableTypeValue::MEMBER_DOUBLE : TunableTypeValue::DOUBLE;
37 } else if constexpr (std::same_as<T, std::string>) {
38 return Member ? TunableTypeValue::MEMBER_STRING : TunableTypeValue::STRING;
39 } else if constexpr (std::same_as<T, std::vector<uint8_t>>) {
40 return Member ? TunableTypeValue::MEMBER_RAW : TunableTypeValue::RAW;
41 } else if constexpr (std::same_as<T, std::vector<bool>>) {
42 return Member ? TunableTypeValue::MEMBER_BOOLEAN_ARRAY
43 : TunableTypeValue::BOOLEAN_ARRAY;
44 } else if constexpr (std::same_as<T, std::vector<int32_t>>) {
45 return Member ? TunableTypeValue::MEMBER_INT32_ARRAY
46 : TunableTypeValue::INT32_ARRAY;
47 } else if constexpr (std::same_as<T, std::vector<int64_t>>) {
48 return Member ? TunableTypeValue::MEMBER_INT64_ARRAY
49 : TunableTypeValue::INT64_ARRAY;
50 } else if constexpr (std::same_as<T, std::vector<float>>) {
51 return Member ? TunableTypeValue::MEMBER_FLOAT_ARRAY
52 : TunableTypeValue::FLOAT_ARRAY;
53 } else if constexpr (std::same_as<T, std::vector<double>>) {
54 return Member ? TunableTypeValue::MEMBER_DOUBLE_ARRAY
55 : TunableTypeValue::DOUBLE_ARRAY;
56 } else if constexpr (std::same_as<T, std::vector<std::string>>) {
57 return Member ? TunableTypeValue::MEMBER_STRING_ARRAY
58 : TunableTypeValue::STRING_ARRAY;
59 } else {
60 static_assert(always_false<T>::value, "Don't know how to serialize type");
61 }
62}
