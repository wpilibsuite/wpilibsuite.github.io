29/**
30 * Struct serialization template. Unspecialized class has no members; only
31 * specializations of this class are useful, and only if they meet the
32 * StructSerializable concept.
33 *
34 * @tparam T type to serialize/deserialize
35 * @tparam I optional struct type info
36 */
37template <typename T, typename... I>
38struct Struct {};
39
40/**
41 * Specifies that a type is capable of raw struct serialization and
42 * deserialization.
43 *
44 * This is designed for serializing small fixed-size data structures in the
45 * fastest and most compact means possible. Serialization consists of writing
46 * values into a mutable std::span and deserialization consists of reading
47 * values from an immutable std::span.
48 *
49 * Implementations must define a template specialization for wpi::util::Struct
50 * with T being the type that is being serialized/deserialized, with the
51 * following static members (as enforced by this concept):
52 * - std::string_view GetTypeName(): function that returns the type name
53 * - size_t GetSize(): function that returns the structure size in bytes
54 * - std::string_view GetSchema(): function that returns the struct schema
55 * - T Unpack(std::span<const uint8_t>): function for deserialization
56 * - void Pack(std::span<uint8_t>, T&& value): function for
57 * serialization
58 *
59 * If possible, the GetTypeName(), GetSize(), and GetSchema() functions should
60 * be marked constexpr. GetTypeName() and GetSchema() may return types other
61 * than std::string_view, as long as the return value is convertible to
62 * std::string_view.
63 *
64 * If the struct has nested structs, implementations should also meet the
65 * requirements of HasNestedStruct<T>.
66 */
67template <typename T, typename... I>
68concept StructSerializable = requires(std::span<const uint8_t> in,
69 std::span<uint8_t> out, T&& value,
70 const I&... info) {
71 typename Struct<typename std::remove_cvref_t<T>,
72 typename std::remove_cvref_t<I>...>;
73 {
74 Struct<typename std::remove_cvref_t<T>,
75 typename std::remove_cvref_t<I>...>::GetTypeName(info...)
76 } -> std::convertible_to<std::string_view>;
77 {
78 Struct<typename std::remove_cvref_t<T>,
79 typename std::remove_cvref_t<I>...>::GetSize(info...)
80 } -> std::convertible_to<size_t>;
81 {
82 Struct<typename std::remove_cvref_t<T>,
83 typename std::remove_cvref_t<I>...>::GetSchema(info...)
84 } -> std::convertible_to<std::string_view>;
85 {
86 Struct<typename std::remove_cvref_t<T>,
87 typename std::remove_cvref_t<I>...>::Unpack(in, info...)
88 } -> std::same_as<typename std::remove_cvref_t<T>>;
89 Struct<typename std::remove_cvref_t<T>,
90 typename std::remove_cvref_t<I>...>::Pack(out, std::forward<T>(value),
91 info...);
92};

94/**
95 * Specifies that a type is capable of in-place raw struct deserialization.
96 *
97 * In addition to meeting StructSerializable, implementations must define a
98 * wpi::util::Struct<T> static member `void UnpackInto(T*, std::span<const
99 * uint8_t>)` to update the pointed-to T with the contents of the span.
100 */
101template <typename T, typename... I>
102concept MutableStructSerializable =
103 StructSerializable<T, I...> &&
104 requires(T* out, std::span<const uint8_t> in, const I&... info) {
105 Struct<typename std::remove_cvref_t<T>,
106 typename std::remove_cvref_t<I>...>::UnpackInto(out, in, info...);
107 };

109/**
110 * Specifies that a struct type has nested struct declarations.
111 *
112 * In addition to meeting StructSerializable, implementations must define a
113 * wpi::util::Struct<T> static member
114 * `void ForEachNested(std::invocable<std::string_view, std::string_view) auto
115 * fn)` (or equivalent) and call ForEachStructSchema<Type> on each nested struct
116 * type.
117 */
118template <typename T, typename... I>
119concept HasNestedStruct =
120 StructSerializable<T, I...> &&
121 requires(function_ref<void(std::string_view, std::string_view)> fn,
122 const I&... info) {
123 Struct<typename std::remove_cvref_t<T>,
124 typename std::remove_cvref_t<I>...>::ForEachNested(fn, info...);
125 };

159/**
160 * Unpack a serialized struct array starting at a given offset within the data.
161 * This is primarily useful in unpack implementations to unpack nested struct
162 * arrays.
163 *
164 * @tparam T object type
165 * @tparam Offset starting offset
166 * @tparam N number of objects
167 * @param data raw struct data
168 * @return Deserialized array
169 */
170template <StructSerializable T, size_t Offset, size_t N>
171inline wpi::util::array<T, N> UnpackStructArray(std::span<const uint8_t> data) {
172 if (is_constexpr([] { Struct<std::remove_cvref_t<T>>::GetSize(); })) {
173 constexpr auto StructSize = Struct<std::remove_cvref_t<T>>::GetSize();
174 wpi::util::array<T, N> arr(wpi::util::empty_array);
175 [&]<size_t... Is>(std::index_sequence<Is...>) {
176 ((arr[Is] = UnpackStruct<T, Offset + Is * StructSize>(data)), ...);
177 }(std::make_index_sequence<N>{});
178 return arr;
179 } else {
180 auto size = Struct<std::remove_cvref_t<T>>::GetSize();
181 wpi::util::array<T, N> arr(wpi::util::empty_array);
182 for (size_t i = 0; i < N; i++) {
183 arr[i] = UnpackStruct<T>(data);
184 data = data.subspan(size);
185 }
186 return arr;
187 }
188}

222/**
223 * Pack a serialized struct array starting at a given offset within the data.
224 * This is primarily useful in pack implementations to pack nested struct
225 * arrays.
226 *
227 * @tparam Offset starting offset
228 * @tparam N number of objects
229 * @param data struct storage (mutable, output)
230 * @param arr array of object
231 */
232template <size_t Offset, size_t N, StructSerializable T>
233inline void PackStructArray(std::span<uint8_t> data,
234 const wpi::util::array<T, N>& arr) {
235 if (is_constexpr([] { Struct<std::remove_cvref_t<T>>::GetSize(); })) {
236 constexpr auto StructSize = Struct<std::remove_cvref_t<T>>::GetSize();
237 [&]<size_t... Is>(std::index_sequence<Is...>) {
238 (PackStruct<Offset + Is * StructSize>(data, arr[Is]), ...);
239 }(std::make_index_sequence<N>{});
240 } else {
241 auto size = Struct<std::remove_cvref_t<T>>::GetSize();
242 for (auto&& val : arr) {
243 PackStruct(data, val);
244 data = data.subspan(size);
245 }
246 }
247}

341template <typename T, size_t N, typename... I>
342 requires StructSerializable<T, I...>
343constexpr auto MakeStructArrayTypeName(const I&... info) {
344 using S = Struct<T, typename std::remove_cvref_t<I>...>;
345 if constexpr (sizeof...(I) == 0 &&
346 is_constexpr([&info...] { S::GetTypeName(info...); })) {
347 constexpr auto typeName = S::GetTypeName(info...);
348 using namespace literals;
349 if constexpr (N == std::dynamic_extent) {
350 return Concat(
351 ct_string<char, std::char_traits<char>, typeName.size()>{typeName},
352 "[]"_ct_string);
353 } else {
354 return Concat(
355 ct_string<char, std::char_traits<char>, typeName.size()>{typeName},
356 "["_ct_string, NumToCtString<N>(), "]"_ct_string);
357 }
358 } else {
359 if constexpr (N == std::dynamic_extent) {
360 return std::format("{}[]", S::GetTypeName(info...));
361 } else {
362 return std::format("{}[{}]", S::GetTypeName(info...), N);
363 }
364 }
365}

421template <typename T, typename... I>
422 requires StructSerializable<T, I...>
423void ForEachStructSchema(
424 std::invocable<std::string_view, std::string_view> auto fn,
425 const I&... info) {

434template <typename T, typename... I>
435 requires StructSerializable<T, I...>
436class StructArrayBuffer {
437 using S = Struct<T, I...>;
438
439 public:
440 StructArrayBuffer() = default;
441 StructArrayBuffer(const StructArrayBuffer&) = delete;
442 StructArrayBuffer& operator=(const StructArrayBuffer&) = delete;
443 StructArrayBuffer(StructArrayBuffer&& rhs) : m_buf{std::move(rhs.m_buf)} {}
444 StructArrayBuffer& operator=(StructArrayBuffer&& rhs) {
445 m_buf = std::move(rhs.m_buf);
446 return *this;
447 }
448
449 template <typename U, typename F>
450 requires
451#if __cpp_lib_ranges >= 201911L
452 std::ranges::range<U> &&
453 std::convertible_to<std::ranges::range_value_t<U>, T> &&
454#endif
455 std::invocable<F, std::span<const uint8_t>>
456 void Write(U&& data, F&& func, const I&... info) {
457 auto size = S::GetSize(info...);
458 if ((std::size(data) * size) < 256) {
459 // use the stack
460 uint8_t buf[256];
461 auto out = buf;
462 for (auto&& val : data) {
463 S::Pack(std::span<uint8_t>{std::to_address(out), size},
464 std::forward<decltype(val)>(val), info...);
465 out += size;
466 }
467 func(std::span<uint8_t>{buf, out});
468 } else {
469 std::scoped_lock lock{m_mutex};
470 m_buf.resize(std::size(data) * size);
471 auto out = m_buf.begin();
472 for (auto&& val : data) {
473 S::Pack(std::span<uint8_t>{std::to_address(out), size},
474 std::forward<decltype(val)>(val), info...);
475 out += size;
476 }
477 func(m_buf);
478 }
479 }
480
481 private:
482 wpi::util::mutex m_mutex;
483 std::vector<uint8_t> m_buf;
484};

486/**
487 * Raw struct support for fixed-size arrays of other structs.
488 */
489template <typename T, size_t N, typename... I>
490 requires StructSerializable<T, I...>
491struct Struct<std::array<T, N>, I...> {
492 static constexpr auto GetTypeName(const I&... info) {
493 return MakeStructArrayTypeName<T, N>(info...);
494 }
495 static constexpr size_t GetSize(const I&... info) {
496 return N * GetStructSize<T>(info...);
497 }
498 static constexpr auto GetSchema(const I&... info) {
499 return MakeStructArraySchema<T, N>(info...);
500 }
501 static std::array<T, N> Unpack(std::span<const uint8_t> data,
502 const I&... info) {
503 auto size = GetStructSize<T>(info...);
504 std::array<T, N> result;
505 for (size_t i = 0; i < N; ++i) {
506 result[i] = UnpackStruct<T, 0>(data, info...);
507 data = data.subspan(size);
508 }
509 return result;
510 }
511 static void Pack(std::span<uint8_t> data, std::span<const T, N> values,
512 const I&... info) {
513 auto size = GetStructSize<T>(info...);
514 std::span<uint8_t> unsizedData = data;
515 for (auto&& val : values) {
516 PackStruct(unsizedData, val, info...);
517 unsizedData = unsizedData.subspan(size);
518 }
519 }
520 static void UnpackInto(std::array<T, N>* out, std::span<const uint8_t> data,
521 const I&... info) {
522 UnpackInto(std::span{*out}, data, info...);
523 }
524 // alternate span-based function
525 static void UnpackInto(std::span<T, N> out, std::span<const uint8_t> data,
526 const I&... info) {
527 auto size = GetStructSize<T>(info...);
528 std::span<const uint8_t> unsizedData = data;
529 for (size_t i = 0; i < N; ++i) {
530 UnpackStructInto(&out[i], unsizedData, info...);
531 unsizedData = unsizedData.subspan(size);
532 }
533 }
534};
