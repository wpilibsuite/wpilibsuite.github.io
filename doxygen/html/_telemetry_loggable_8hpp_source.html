16/**
17 * Specifies that a type is capable of telemetry logging to an individual value.
18 *
19 * Implementations must define a non-member function (can be via ADL lookup) for
20 * logging to a single name in a TelemetryTable with the following signature (as
21 * enforced by this concept):
22 * - void LogValueTo(TelemetryTable& table, std::string_view name, T&& value):
23 * function for logging
24 */
25template <typename T, typename... I>
26concept SupportsTelemetryValue = requires(
27 TelemetryTable& table, std::string_view name, T&& value, I... info) {
28 { LogValueTo(table, name, std::forward<T>(value), info...) };
29};
30
31/**
32 * Specifies that a type is capable of telemetry logging to a table.
33 *
34 * Implementations must define a non-member function (can be via ADL lookup) for
35 * logging to a TelemetryTable with the following signature (as enforced by this
36 * concept):
37 * - void LogTo(TelemetryTable& table, T&& value): function for logging
38 */
39template <typename T, typename... I>
40concept SupportsTelemetry =
41 requires(TelemetryTable& table, T&& value, I... info) {
42 { LogTo(table, std::forward<T>(value), info...) };
43 };
44
45/**
46 * Specifies that a type is capable of telemetry logging with a specified type.
47 *
48 * Implementations must define two non-member functions (can be via ADL lookup)
49 * for logging to a TelemetryTable with the following signatures (as enforced by
50 * this concept):
51 * - std::string_view GetTelemetryTypeName(T&& value): function that returns the
52 * type name
53 * - void LogTo(TelemetryTable& table, T&& value): function for logging
54 *
55 * GetTelemetryTypeName() may return types other than std::string_view, as long
56 * as the return value is convertible to std::string_view.
57 */
58template <typename T, typename... I>
59concept SupportsTelemetryWithTypeName =
60 SupportsTelemetry<T, I...> && requires(T&& value, I... info) {
61 {
62 GetTelemetryTypeName(std::forward<T>(value), info...)
63 } -> std::convertible_to<std::string_view>;
64 };

66/**
67 * A base interface for complex telemetry objects. Should be used instead of
68 * standalone functions when defining telemetry support for non-final classes.
69 */
70class WPILIB_DLLEXPORT TelemetryLoggable {
71 void anchor(); // provide a place for the vtable
72
73 public:
74 /**
75 * Logs the object to a @ref wpi::telemetry::TelemetryTable.
76 *
77 * @param table telemetry table
78 */
79 virtual void LogTo(TelemetryTable& table) const = 0;
80
81 /**
82 * Gets the telemetry table type. Default is no specified table type (empty).
83 * A mismatch in type during logging results in a reported warning, so this
84 * should always return the same value.
85 *
86 * @return Table type
87 */
88 virtual std::string_view GetTelemetryType() const { return {}; }
89
90 protected:
91 virtual ~TelemetryLoggable() = default;
92};
