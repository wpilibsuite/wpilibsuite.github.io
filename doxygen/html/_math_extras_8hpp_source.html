30// We can't verify signedness, since callers rely on implicit coercions to
31// signed/unsigned.
32template <typename T, typename U>
33using enableif_int =
34 std::enable_if_t<std::is_integral_v<T> && std::is_integral_v<U>>;

36// Use std::common_type_t to widen only up to the widest argument.
37template <typename T, typename U, typename = enableif_int<T, U>>
38using common_uint =
39 std::common_type_t<std::make_unsigned_t<T>, std::make_unsigned_t<U>>;
40template <typename T, typename U, typename = enableif_int<T, U>>
41using common_sint =
42 std::common_type_t<std::make_signed_t<T>, std::make_signed_t<U>>;

44/// Create a bitmask with the N right-most bits set to 1, and all other
45/// bits set to 0. Only unsigned types are allowed.
46template <typename T> constexpr T maskTrailingOnes(unsigned N) {
47 static_assert(std::is_unsigned_v<T>, "Invalid type!");
48 const unsigned Bits = CHAR_BIT * sizeof(T);
49 assert(N <= Bits && "Invalid bit index");
50 if (N == 0)
51 return 0;
52 return T(-1) >> (Bits - N);
53}
54
55/// Create a bitmask with the N left-most bits set to 1, and all other
56/// bits set to 0. Only unsigned types are allowed.
57template <typename T> constexpr T maskLeadingOnes(unsigned N) {
58 return ~maskTrailingOnes<T>(CHAR_BIT * sizeof(T) - N);
59}
60
61/// Create a bitmask with the N right-most bits set to 0, and all other
62/// bits set to 1. Only unsigned types are allowed.
63template <typename T> constexpr T maskTrailingZeros(unsigned N) {
64 return maskLeadingOnes<T>(CHAR_BIT * sizeof(T) - N);
65}

67/// Create a bitmask with the N left-most bits set to 0, and all other
68/// bits set to 1. Only unsigned types are allowed.
69template <typename T> constexpr T maskLeadingZeros(unsigned N) {
70 return maskTrailingOnes<T>(CHAR_BIT * sizeof(T) - N);
71}

73/// Macro compressed bit reversal table for 256 bits.
74///
75/// http://graphics.stanford.edu/~seander/bithacks.html#BitReverseTable
76static const unsigned char BitReverseTable256[256] = {
77#define R2(n) n, n + 2 * 64, n + 1 * 64, n + 3 * 64
78#define R4(n) R2(n), R2(n + 2 * 16), R2(n + 1 * 16), R2(n + 3 * 16)
79#define R6(n) R4(n), R4(n + 2 * 4), R4(n + 1 * 4), R4(n + 3 * 4)
80 R6(0), R6(2), R6(1), R6(3)
81#undef R2
82#undef R4
83#undef R6
84};

86/// Reverse the bits in \p Val.
87template <typename T> constexpr T reverseBits(T Val) {
88#if __has_builtin(__builtin_bitreverse8)
89 if constexpr (std::is_same_v<T, uint8_t>)
90 return __builtin_bitreverse8(Val);
91#endif
92#if __has_builtin(__builtin_bitreverse16)
93 if constexpr (std::is_same_v<T, uint16_t>)
94 return __builtin_bitreverse16(Val);
95#endif
96#if __has_builtin(__builtin_bitreverse32)
97 if constexpr (std::is_same_v<T, uint32_t>)
98 return __builtin_bitreverse32(Val);
99#endif
100#if __has_builtin(__builtin_bitreverse64)
101 if constexpr (std::is_same_v<T, uint64_t>)
102 return __builtin_bitreverse64(Val);
103#endif
104
105 unsigned char in[sizeof(Val)];
106 unsigned char out[sizeof(Val)];
107 std::memcpy(in, &Val, sizeof(Val));
108 for (unsigned i = 0; i < sizeof(Val); ++i)
109 out[(sizeof(Val) - i) - 1] = BitReverseTable256[in[i]];
110 std::memcpy(&Val, out, sizeof(Val));
111 return Val;
112}

133/// Checks if an integer fits into the given bit width.
134template <unsigned N> constexpr bool isInt(int64_t x) {
135 if constexpr (N == 0)
136 return 0 == x;
137 if constexpr (N == 8)
138 return static_cast<int8_t>(x) == x;
139 if constexpr (N == 16)
140 return static_cast<int16_t>(x) == x;
141 if constexpr (N == 32)
142 return static_cast<int32_t>(x) == x;
143 if constexpr (N < 64)
144 return -(INT64_C(1) << (N - 1)) <= x && x < (INT64_C(1) << (N - 1));
145 (void)x; // MSVC v19.25 warns that x is unused.
146 return true;
147}

149/// Checks if a signed integer is an N bit number shifted left by S.
150template <unsigned N, unsigned S>
151constexpr bool isShiftedInt(int64_t x) {
152 static_assert(S < 64, "isShiftedInt<N, S> with S >= 64 is too much.");
153 static_assert(N + S <= 64, "isShiftedInt<N, S> with N + S > 64 is too wide.");
154 return isInt<N + S>(x) && (x % (UINT64_C(1) << S) == 0);
155}
156
157/// Checks if an unsigned integer fits into the given bit width.
158template <unsigned N> constexpr bool isUInt(uint64_t x) {
159 if constexpr (N < 64)
160 return (x >> N) == 0;
161 (void)x; // MSVC v19.25 warns that x is unused.
162 return true;
163}

165/// Checks if a unsigned integer is an N bit number shifted left by S.
166template <unsigned N, unsigned S>
167constexpr bool isShiftedUInt(uint64_t x) {
168 static_assert(S < 64, "isShiftedUInt<N, S> with S >= 64 is too much.");
169 static_assert(N + S <= 64,
170 "isShiftedUInt<N, S> with N + S > 64 is too wide.");
171 // S must be strictly less than 64. So 1 << S is not undefined behavior.
172 return isUInt<N + S>(x) && (x % (UINT64_C(1) << S) == 0);
173}

175/// Gets the maximum value for a N-bit unsigned integer.
176inline constexpr uint64_t maxUIntN(uint64_t N) {
177 assert(N <= 64 && "integer width out of range");
178
179 // uint64_t(1) << 64 is undefined behavior, so we can't do
180 // (uint64_t(1) << N) - 1
181 // without checking first that N != 64. But this works and doesn't have a
182 // branch for N != 0.
183 // Unfortunately, shifting a uint64_t right by 64 bit is undefined
184 // behavior, so the condition on N == 0 is necessary. Fortunately, most
185 // optimizers do not emit branches for this check.
186 if (N == 0)
187 return 0;
188 return UINT64_MAX >> (64 - N);
189}

196/// Gets the minimum value for a N-bit signed integer.
197inline constexpr int64_t minIntN(int64_t N) {
198 assert(N >= 0 && N <= 64 && "integer width out of range");
199
200 if (N == 0)
201 return 0;
202 return UINT64_MAX << (N - 1);
203}

209/// Gets the maximum value for a N-bit signed integer.
210inline constexpr int64_t maxIntN(int64_t N) {
211 assert(N >= 0 && N <= 64 && "integer width out of range");
212
213 // This relies on two's complement wraparound when N == 64, so we convert to
214 // int64_t only at the very end to avoid UB.
215 if (N == 0)
216 return 0;
217 return (UINT64_C(1) << (N - 1)) - 1;
218}

230/// Return true if the argument is a non-empty sequence of ones starting at the
231/// least significant bit with the remainder zero (32 bit version).
232/// Ex. isMask_32(0x0000FFFFU) == true.
233constexpr bool isMask_32(uint32_t Value) {
234 return Value && ((Value + 1) & Value) == 0;
235}

237/// Return true if the argument is a non-empty sequence of ones starting at the
238/// least significant bit with the remainder zero (64 bit version).
239constexpr bool isMask_64(uint64_t Value) {
240 return Value && ((Value + 1) & Value) == 0;
241}

243/// Return true if the argument contains a non-empty sequence of ones with the
244/// remainder zero (32 bit version.) Ex. isShiftedMask_32(0x0000FF00U) == true.
245constexpr bool isShiftedMask_32(uint32_t Value) {
246 return Value && isMask_32((Value - 1) | Value);
247}

251constexpr bool isShiftedMask_64(uint64_t Value) {
252 return Value && isMask_64((Value - 1) | Value);

266/// Return true if the argument contains a non-empty sequence of ones with the
267/// remainder zero (32 bit version.) Ex. isShiftedMask_32(0x0000FF00U) == true.
268/// If true, \p MaskIdx will specify the index of the lowest set bit and \p
269/// MaskLen is updated to specify the length of the mask, else neither are
270/// updated.
271inline bool isShiftedMask_32(uint32_t Value, unsigned &MaskIdx,
272 unsigned &MaskLen) {
273 if (!isShiftedMask_32(Value))
274 return false;
275 MaskIdx = std::countr_zero(Value);
276 MaskLen = std::popcount(Value);
277 return true;
278}

289 MaskLen = std::popcount(Value);

332/// A and B are either alignments or offsets. Return the minimum alignment that
333/// may be assumed after adding the two together.
334template <typename U, typename V, typename T = common_uint<U, V>>
335constexpr T MinAlign(U A, V B) {
336 // The largest power of 2 that divides both A and B.
337 //
338 // Replace "-Value" by "1+~Value" in the following commented code to avoid
339 // MSVC warning C4146
340 // return (A | B) & -(A | B);
341 return (A | B) & (1 + ~(A | B));
342}
343
344/// Fallback when arguments aren't integral.
345constexpr uint64_t MinAlign(uint64_t A, uint64_t B) {
346 return (A | B) & (1 + ~(A | B));
347}

349/// Returns the next power of two (in 64-bits) that is strictly greater than A.
350/// Returns zero on overflow.
351constexpr uint64_t NextPowerOf2(uint64_t A) {
352 A |= (A >> 1);
353 A |= (A >> 2);
354 A |= (A >> 4);
355 A |= (A >> 8);
356 A |= (A >> 16);
357 A |= (A >> 32);
358 return A + 1;
359}

362/// Essentially, it is a ceil operation across the domain of powers of two.

369/// Returns the integer ceil(Numerator / Denominator). Unsigned version.
370/// Guaranteed to never overflow.
371template <typename U, typename V, typename T = common_uint<U, V>>
372constexpr T divideCeil(U Numerator, V Denominator) {
373 assert(Denominator && "Division by zero");
374 T Bias = (Numerator != 0);
375 return (Numerator - Bias) / Denominator + Bias;
376}

378/// Fallback when arguments aren't integral.
379constexpr uint64_t divideCeil(uint64_t Numerator, uint64_t Denominator) {
380 assert(Denominator && "Division by zero");
381 uint64_t Bias = (Numerator != 0);
382 return (Numerator - Bias) / Denominator + Bias;
383}

385// Check whether divideCeilSigned or divideFloorSigned would overflow. This
386// happens only when Numerator = INT_MIN and Denominator = -1.
387template <typename U, typename V>
388constexpr bool divideSignedWouldOverflow(U Numerator, V Denominator) {
389 return Numerator == (std::numeric_limits<U>::min)() && Denominator == -1;
390}
391
392/// Returns the integer ceil(Numerator / Denominator). Signed version.
393/// Overflow is explicitly forbidden with an assert.
394template <typename U, typename V, typename T = common_sint<U, V>>
395constexpr T divideCeilSigned(U Numerator, V Denominator) {
396 assert(Denominator && "Division by zero");
397 assert(!divideSignedWouldOverflow(Numerator, Denominator) &&
398 "Divide would overflow");
399 if (!Numerator)
400 return 0;
401 // C's integer division rounds towards 0.
402 T Bias = Denominator >= 0 ? 1 : -1;
403 bool SameSign = (Numerator >= 0) == (Denominator >= 0);
404 return SameSign ? (Numerator - Bias) / Denominator + 1
405 : Numerator / Denominator;
406}
407
408/// Returns the integer floor(Numerator / Denominator). Signed version.
409/// Overflow is explicitly forbidden with an assert.
410template <typename U, typename V, typename T = common_sint<U, V>>
411constexpr T divideFloorSigned(U Numerator, V Denominator) {
412 assert(Denominator && "Division by zero");
413 assert(!divideSignedWouldOverflow(Numerator, Denominator) &&
414 "Divide would overflow");
415 if (!Numerator)
416 return 0;
417 // C's integer division rounds towards 0.
418 T Bias = Denominator >= 0 ? -1 : 1;
419 bool SameSign = (Numerator >= 0) == (Denominator >= 0);
420 return SameSign ? Numerator / Denominator
421 : (Numerator - Bias) / Denominator - 1;
422}

424/// Returns the remainder of the Euclidean division of LHS by RHS. Result is
425/// always non-negative.
426template <typename U, typename V, typename T = common_sint<U, V>>
427constexpr T mod(U Numerator, V Denominator) {
428 assert(Denominator >= 1 && "Mod by non-positive number");
429 T Mod = Numerator % Denominator;
430 return Mod < 0 ? Mod + Denominator : Mod;
431}

433/// Returns (Numerator / Denominator) rounded by round-half-up. Guaranteed to
434/// never overflow.
435template <typename U, typename V, typename T = common_uint<U, V>>
436constexpr T divideNearest(U Numerator, V Denominator) {
437 assert(Denominator && "Division by zero");
438 T Mod = Numerator % Denominator;
439 return (Numerator / Denominator) +
440 (Mod > (static_cast<T>(Denominator) - 1) / 2);
441}

443/// Returns the next integer (mod 2**nbits) that is greater than or equal to
444/// \p Value and is a multiple of \p Align. \p Align must be non-zero.
445///
446/// Examples:
447/// \code
448/// alignTo(5, 8) = 8
449/// alignTo(17, 8) = 24
450/// alignTo(~0LL, 8) = 0
451/// alignTo(321, 255) = 510
452/// \endcode
453///
454/// Will overflow only if result is not representable in T.
455template <typename U, typename V, typename T = common_uint<U, V>>
456constexpr T alignTo(U Value, V Align) {
457 assert(Align != 0u && "Align can't be 0.");
458 T CeilDiv = divideCeil(Value, Align);
459 return CeilDiv * Align;
460}

469/// Will overflow only if result is not representable in T.
470template <typename U, typename V, typename T = common_uint<U, V>>
471constexpr T alignToPowerOf2(U Value, V Align) {
472 assert(Align != 0 && (Align & (Align - 1)) == 0 &&
473 "Align must be a power of 2");
474 T NegAlign = static_cast<T>(0) - Align;
475 return (Value + (Align - 1)) & NegAlign;
476}

486/// If non-zero \p Skew is specified, the return value will be a minimal integer
487/// that is greater than or equal to \p Size and equal to \p A * N + \p Skew for
488/// some integer N. If \p Skew is larger than \p A, its value is adjusted to '\p
489/// Skew mod \p A'. \p Align must be non-zero.
490///
491/// Examples:
492/// \code
493/// alignTo(5, 8, 7) = 7
494/// alignTo(17, 8, 1) = 17
495/// alignTo(~0LL, 8, 3) = 3
496/// alignTo(321, 255, 42) = 552
497/// \endcode
498///
499/// May overflow.
500template <typename U, typename V, typename W,
501 typename T = common_uint<common_uint<U, V>, W>>
502constexpr T alignTo(U Value, V Align, W Skew) {
503 assert(Align != 0u && "Align can't be 0.");
504 Skew %= Align;
505 return alignTo(Value - Skew, Align) + Skew;
506}

519/// Returns the largest unsigned integer less than or equal to \p Value and is
520/// \p Skew mod \p Align. \p Align must be non-zero. Guaranteed to never
521/// overflow.
522template <typename U, typename V, typename W = uint8_t,

566/// Return the absolute value of a signed integer, converted to the

582/// Add two unsigned integers, X and Y, of type T. Clamp the result to the
583/// maximum representable value of T on overflow. ResultOverflowed indicates if
584/// the result is larger than the maximum representable value of type T.
585template <typename T>
586std::enable_if_t<std::is_unsigned_v<T>, T>
587SaturatingAdd(T X, T Y, bool *ResultOverflowed = nullptr) {
588 bool Dummy;
589 bool &Overflowed = ResultOverflowed ? *ResultOverflowed : Dummy;
590 // Hacker's Delight, p. 29
591 T Z = X + Y;
592 Overflowed = (Z < X || Z < Y);
593 if (Overflowed)
594 return (std::numeric_limits<T>::max)();
595 else
596 return Z;

600/// maximum representable value of T on overflow.
601template <class T, class... Ts>
602std::enable_if_t<std::is_unsigned_v<T>, T> SaturatingAdd(T X, T Y, T Z,
603 Ts... Args) {
604 bool Overflowed = false;
605 T XY = SaturatingAdd(X, Y, &Overflowed);

607 return SaturatingAdd((std::numeric_limits<T>::max)(), T(1), Args...);

611/// Multiply two unsigned integers, X and Y, of type T. Clamp the result to the
612/// maximum representable value of T on overflow. ResultOverflowed indicates if
613/// the result is larger than the maximum representable value of type T.
614template <typename T>
615std::enable_if_t<std::is_unsigned_v<T>, T>
616SaturatingMultiply(T X, T Y, bool *ResultOverflowed = nullptr) {
617 bool Dummy;
618 bool &Overflowed = ResultOverflowed ? *ResultOverflowed : Dummy;
619
620 // Hacker's Delight, p. 30 has a different algorithm, but we don't use that
621 // because it fails for uint16_t (where multiplication can have undefined
622 // behavior due to promotion to int), and requires a division in addition
623 // to the multiplication.
624
625 Overflowed = false;
626
627 // Log2(Z) would be either Log2Z or Log2Z + 1.
628 // Special case: if X or Y is 0, Log2_64 gives -1, and Log2Z
629 // will necessarily be less than Log2Max as desired.
630 int Log2Z = Log2_64(X) + Log2_64(Y);
631 const T Max = (std::numeric_limits<T>::max)();
632 int Log2Max = Log2_64(Max);
633 if (Log2Z < Log2Max) {
634 return X * Y;
635 }
636 if (Log2Z > Log2Max) {
637 Overflowed = true;
638 return Max;
639 }
640
641 // We're going to use the top bit, and maybe overflow one
642 // bit past it. Multiply all but the bottom bit then add
643 // that on at the end.
644 T Z = (X >> 1) * Y;
645 if (Z & ~(Max >> 1)) {
646 Overflowed = true;
647 return Max;
648 }
649 Z <<= 1;
650 if (X & 1)
651 return SaturatingAdd(Z, Y, ResultOverflowed);
652
653 return Z;
654}

656/// Multiply two unsigned integers, X and Y, and add the unsigned integer, A to
657/// the product. Clamp the result to the maximum representable value of T on
658/// overflow. ResultOverflowed indicates if the result is larger than the
659/// maximum representable value of type T.
660template <typename T>
661std::enable_if_t<std::is_unsigned_v<T>, T>
662SaturatingMultiplyAdd(T X, T Y, T A, bool *ResultOverflowed = nullptr) {
663 bool Dummy;
664 bool &Overflowed = ResultOverflowed ? *ResultOverflowed : Dummy;
665
666 T Product = SaturatingMultiply(X, Y, &Overflowed);
667 if (Overflowed)
668 return Product;
669
670 return SaturatingAdd(A, Product, &Overflowed);
671}

676/// Add two signed integers, computing the two's complement truncated result,
677/// returning true if overflow occurred.
678template <typename T>
679std::enable_if_t<std::is_signed_v<T>, T> AddOverflow(T X, T Y, T &Result) {
680#if __has_builtin(__builtin_add_overflow)
681 return __builtin_add_overflow(X, Y, &Result);
682#else
683 // Perform the unsigned addition.
684 using U = std::make_unsigned_t<T>;
685 const U UX = static_cast<U>(X);
686 const U UY = static_cast<U>(Y);
687 const U UResult = UX + UY;
688
689 // Convert to signed.
690 Result = static_cast<T>(UResult);
691
692 // Adding two positive numbers should result in a positive number.
693 if (X > 0 && Y > 0)
694 return Result <= 0;
695 // Adding two negatives should result in a negative number.
696 if (X < 0 && Y < 0)
697 return Result >= 0;
698 return false;
699#endif
700}

702/// Subtract two signed integers, computing the two's complement truncated
703/// result, returning true if an overflow occurred.
704template <typename T>
705std::enable_if_t<std::is_signed_v<T>, T> SubOverflow(T X, T Y, T &Result) {
706#if __has_builtin(__builtin_sub_overflow)
707 return __builtin_sub_overflow(X, Y, &Result);
708#else
709 // Perform the unsigned addition.
710 using U = std::make_unsigned_t<T>;
711 const U UX = static_cast<U>(X);
712 const U UY = static_cast<U>(Y);
713 const U UResult = UX - UY;
714
715 // Convert to signed.
716 Result = static_cast<T>(UResult);
717
718 // Subtracting a positive number from a negative results in a negative number.
719 if (X <= 0 && Y > 0)
720 return Result >= 0;
721 // Subtracting a negative number from a positive results in a positive number.
722 if (X >= 0 && Y < 0)
723 return Result <= 0;
724 return false;
725#endif
726}

728/// Multiply two signed integers, computing the two's complement truncated
729/// result, returning true if an overflow occurred.
730template <typename T>
731std::enable_if_t<std::is_signed_v<T>, T> MulOverflow(T X, T Y, T &Result) {
732#if __has_builtin(__builtin_mul_overflow)
733 return __builtin_mul_overflow(X, Y, &Result);
734#else
735 // Perform the unsigned multiplication on absolute values.
736 using U = std::make_unsigned_t<T>;
737 const U UX = X < 0 ? (0 - static_cast<U>(X)) : static_cast<U>(X);
738 const U UY = Y < 0 ? (0 - static_cast<U>(Y)) : static_cast<U>(Y);
739 const U UResult = UX * UY;
740
741 // Convert to signed.
742 const bool IsNegative = (X < 0) ^ (Y < 0);
743 Result = IsNegative ? (0 - UResult) : UResult;
744
745 // If any of the args was 0, result is 0 and no overflow occurs.
746 if (UX == 0 || UY == 0)
747 return false;
748
749 // UX and UY are in [1, 2^n], where n is the number of digits.
750 // Check how the max allowed absolute value (2^n for negative, 2^(n-1) for
751 // positive) divided by an argument compares to the other.
752 if (IsNegative)
753 return UX > (static_cast<U>((std::numeric_limits<T>::max)()) + U(1)) / UY;
754 else
755 return UX > (static_cast<U>((std::numeric_limits<T>::max)())) / UY;
756#endif
757}
