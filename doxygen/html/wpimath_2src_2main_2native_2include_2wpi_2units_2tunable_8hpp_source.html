35template <class Units, typename T, template <typename> class NonLinearScale>
36class TunableUnit {
37 using BaseUnits =
38 unit<std::ratio<1>, typename traits::unit_traits<Units>::base_unit_type>;
39
40 public:
41 using ValueType = unit_t<Units, T, NonLinearScale>;
42
43 TunableUnit() : m_tunable{MakeUnitTunableConfig<Units>()} {}
44 explicit TunableUnit(ValueType val)
45 : m_tunable{convert<Units, BaseUnits>(val).template to<T>(),
46 MakeUnitTunableConfig<Units>()} {}
47 explicit TunableUnit(const wpi::tunables::TunableConfig& config)
48 : m_tunable{MakeUnitTunableConfig<Units>(config)} {}
49 TunableUnit(ValueType val, const wpi::tunables::TunableConfig& config)
50 : m_tunable{convert<Units, BaseUnits>(val).template to<T>(),
51 MakeUnitTunableConfig<Units>(config)} {}
52
53 ValueType Get() const {
54 return unit_t<BaseUnits, T, NonLinearScale>{m_tunable.Get()};
55 }
56 void Set(ValueType value) {
57 m_tunable = convert<Units, BaseUnits>(value).template to<T>();
58 }
59
60 wpi::tunables::Tunable<T>& GetInnerTunable() { return m_tunable; }
61
62 template <typename U>
63 requires traits::is_convertible_unit_t<ValueType, U>::value
64 operator U() const { // NOLINT(google-explicit-constructor)
65 return Get();
66 }
67
68 private:
69 wpi::tunables::Tunable<T> m_tunable;
70};

72template <class Units, typename T, template <typename> class NonLinearScale>
73class TunableMemberValue
74 : public wpi::tunables::detail::TunableMemberValueBase<double> {
75 using BaseUnits =
76 unit<std::ratio<1>, typename traits::unit_traits<Units>::base_unit_type>;
77
78 public:

81 template <std::derived_from<wpi::tunables::ComplexTunable> Class>
82 explicit TunableMemberValue(ValueType Class::* member)
83 : TunableMemberValueBase<double>{MakeUnitTunableConfig<Units>()},
84 m_ptr{member} {}
85
86 template <std::derived_from<wpi::tunables::ComplexTunable> Class>
87 explicit TunableMemberValue(ValueType Class::* member,
88 const wpi::tunables::TunableConfig& config)
89 : TunableMemberValueBase<double>{MakeUnitTunableConfig<Units>(config)},
90 m_ptr{member} {}
91
92 const double& Get(const wpi::tunables::ComplexTunable* obj) const override {
93 m_value = convert<Units, BaseUnits>(m_ptr.Get(obj)).template to<T>();
94 return m_value;
95 }
96
97 void Set(wpi::tunables::ComplexTunable* obj, double value) override {
98 m_ptr.Get(obj) = unit_t<BaseUnits, T, NonLinearScale>{value};
99 this->SetTunableChanged();

101
102 private:
103 mutable double m_value;
104 wpi::tunables::detail::TunableMemberPointer<ValueType> m_ptr;
105};
