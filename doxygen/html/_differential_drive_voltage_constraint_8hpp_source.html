20namespace wpi::math {
21/**
22 * A class that enforces constraints on differential drive voltage expenditure
23 * based on the motor dynamics and the drive kinematics. Ensures that the
24 * acceleration of any wheel of the robot while following the trajectory is
25 * never higher than what can be achieved with the given maximum voltage.
26 */
27class WPILIB_DLLEXPORT DifferentialDriveVoltageConstraint
28 : public TrajectoryConstraint {
29 public:
30 /**
31 * Creates a new DifferentialDriveVoltageConstraint.
32 *
33 * @param feedforward A feedforward component describing the behavior of the
34 * drive.
35 * @param kinematics A kinematics component describing the drive geometry.
36 * @param maxVoltage The maximum voltage available to the motors while
37 * following the path. Should be somewhat less than the nominal battery
38 * voltage (12V) to account for "voltage sag" due to current draw.
39 */
40 constexpr DifferentialDriveVoltageConstraint(
41 const SimpleMotorFeedforward<wpi::units::meter>& feedforward,
42 DifferentialDriveKinematics kinematics, wpi::units::volt_t maxVoltage)
43 : m_feedforward(feedforward),
44 m_kinematics(std::move(kinematics)),
45 m_maxVoltage(maxVoltage) {}
46
47 constexpr wpi::units::meters_per_second_t MaxVelocity(
48 const Pose2d& pose, wpi::units::curvature_t curvature,
49 wpi::units::meters_per_second_t velocity) const override {
50 return wpi::units::meters_per_second_t{std::numeric_limits<double>::max()};
51 }
52
53 constexpr MinMax MinMaxAcceleration(
54 const Pose2d& pose, wpi::units::curvature_t curvature,
55 wpi::units::meters_per_second_t velocity) const override {
56 auto wheelVelocities =
57 m_kinematics.ToWheelVelocities({velocity, 0_mps, velocity * curvature});
58
59 auto maxWheelVelocity =
60 (std::max)(wheelVelocities.left, wheelVelocities.right);
61 auto minWheelVelocity =
62 (std::min)(wheelVelocities.left, wheelVelocities.right);
63
64 // Calculate maximum/minimum possible accelerations from motor dynamics
65 // and max/min wheel velocities
66 auto maxWheelAcceleration =
67 m_feedforward.MaxAchievableAcceleration(m_maxVoltage, maxWheelVelocity);
68 auto minWheelAcceleration =
69 m_feedforward.MinAchievableAcceleration(m_maxVoltage, minWheelVelocity);
70
71 // Robot chassis turning on radius = 1/|curvature|. Outer wheel has radius
72 // increased by half of the trackwidth T. Inner wheel has radius decreased
73 // by half of the trackwidth. Achassis / radius = Aouter / (radius + T/2),
74 // so Achassis = Aouter * radius / (radius + T/2) = Aouter / (1 +
75 // |curvature|T/2). Inner wheel is similar.
76
77 // sgn(velocity) term added to correctly account for which wheel is on
78 // outside of turn:
79 // If moving forward, max acceleration constraint corresponds to wheel on
80 // outside of turn If moving backward, max acceleration constraint
81 // corresponds to wheel on inside of turn
82
83 // When velocity is zero, then wheel velocities are uniformly zero (robot
84 // cannot be turning on its center) - we have to treat this as a special
85 // case, as it breaks the signum function. Both max and min acceleration
86 // are *reduced in magnitude* in this case.
87
88 wpi::units::meters_per_second_squared_t maxChassisAcceleration;
89 wpi::units::meters_per_second_squared_t minChassisAcceleration;
90
91 if (velocity == 0_mps) {
92 maxChassisAcceleration =
93 maxWheelAcceleration /
94 (1 + m_kinematics.trackwidth * wpi::units::math::abs(curvature) /
95 (2_rad));
96 minChassisAcceleration =
97 minWheelAcceleration /
98 (1 + m_kinematics.trackwidth * wpi::units::math::abs(curvature) /
99 (2_rad));
100 } else {
101 maxChassisAcceleration =
102 maxWheelAcceleration /
103 (1 + m_kinematics.trackwidth * wpi::units::math::abs(curvature) *
104 wpi::util::sgn(velocity) / (2_rad));
105 minChassisAcceleration =
106 minWheelAcceleration /
107 (1 - m_kinematics.trackwidth * wpi::units::math::abs(curvature) *
108 wpi::util::sgn(velocity) / (2_rad));
109 }
110
111 // When turning about a point inside of the wheelbase (i.e. radius less than
112 // half the trackwidth), the inner wheel's direction changes, but the
113 // magnitude remains the same. The formula above changes sign for the inner
114 // wheel when this happens. We can accurately account for this by simply
115 // negating the inner wheel.
116
117 if ((m_kinematics.trackwidth / 2) >
118 1_rad / wpi::units::math::abs(curvature)) {
119 if (velocity > 0_mps) {
120 minChassisAcceleration = -minChassisAcceleration;
121 } else if (velocity < 0_mps) {
122 maxChassisAcceleration = -maxChassisAcceleration;
123 }
124 }
125
126 return {minChassisAcceleration, maxChassisAcceleration};
127 }
128
129 private:
130 SimpleMotorFeedforward<wpi::units::meter> m_feedforward;
131 DifferentialDriveKinematics m_kinematics;
132 wpi::units::volt_t m_maxVoltage;
133};
134} // namespace wpi::math
